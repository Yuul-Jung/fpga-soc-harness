AXI4-Lite BLDC 6스텝 홀센서 IP + 데드타임 PWM

홀 3입력 → 6스텝 해독 FSM(양방향) → 보완 PWM 게이트 출력 + 상·하 암 데드타임 삽입. ADC 샘플 트리거를 데드타임 종료 시점에 출력한다.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • 홀 신호 3비트 → 6스텝 해독(순방향·역방향 모두 지원)
  • 10비트 PWM 카운터로 상위 암(HIN) 듀티 제어, 하위 암(LIN)은 DC ON
  • 정류 전환 시 상·하 암 데드타임(클럭 수 단위) 삽입으로 슛스루 방지
  • 데드타임 종료 시점에 1사이클 ADC 샘플 트리거 펄스 출력
  • AXI4-Lite 레지스터 6개(ID/CTRL/DUTY/DEADTIME/STATUS/IRQ) 로 CPU 연동
  • 자체 점검형 테스트벤치: 홀 순서, 슛스루 없음, 데드타임 길이 검증

범위 밖

  • FOC(Field-Oriented Control) — 다음 사다리 단계
  • 전류 센싱·ADC 처리 — 샘플 트리거 출력만 제공
  • 게이트 드라이버 IC 제어 신호(ENABLE, FAULT 입력) — 외부에서 처리
  • 모터 보호(과전류 차단) — 외부 하드웨어 또는 상위 IP 담당

rtl/bldc_hall_sixstep_ip.v 320줄
// bldc_hall_sixstep_ip.v
// AXI4-Lite BLDC 6-step hall decoder + deadtime PWM IP
// Hall(3) → step decode → complementary PWM + commutation deadtime
// Outputs: HIN[2:0] (high-side gates), LIN[2:0] (low-side gates)
// Hall sequence (forward): 101→001→011→010→110→100→101
// Board: Basys3 (100 MHz), Verilog-2001
// Part of capstone mmio-bldc-hall-sixstep-ip (I-023)

`timescale 1ns/1ps

module bldc_hall_sixstep_ip #(
    parameter ADDR_W          = 8,
    parameter DATA_W          = 32,
    parameter DEFAULT_DUTY    = 10'd512,  // 50% default
    parameter DEFAULT_DEADTIME= 8'd50     // 50 cycles = 500 ns @ 100 MHz
) (
    input  wire        clk,
    input  wire        rst_n,   // active-low synchronous reset

    // AXI4-Lite write address channel
    input  wire [ADDR_W-1:0] s_awaddr,
    input  wire              s_awvalid,
    output reg               s_awready,

    // AXI4-Lite write data channel
    input  wire [DATA_W-1:0] s_wdata,
    input  wire [3:0]        s_wstrb,
    input  wire              s_wvalid,
    output reg               s_wready,

    // AXI4-Lite write response channel
    output reg  [1:0]        s_bresp,
    output reg               s_bvalid,
    input  wire              s_bready,

    // AXI4-Lite read address channel
    input  wire [ADDR_W-1:0] s_araddr,
    input  wire              s_arvalid,
    output reg               s_arready,

    // AXI4-Lite read data channel
    output reg  [DATA_W-1:0] s_rdata,
    output reg  [1:0]        s_rresp,
    output reg               s_rvalid,
    input  wire              s_rready,

    // Hall sensor inputs (raw, synchronized internally)
    input  wire [2:0]        hall_in,     // {H_C, H_B, H_A}

    // Gate drive outputs (3-phase bridge)
    output reg  [2:0]        hin,         // high-side: {H_C, H_B, H_A}
    output reg  [2:0]        lin,         // low-side:  {L_C, L_B, L_A}

    // ADC sample trigger (1-cycle pulse at end of commutation deadtime)
    output reg               sample_trig,

    // Level interrupt (active-high)
    output wire              irq
);

    // -----------------------------------------------------------------------
    // Register addresses
    // -----------------------------------------------------------------------
    localparam REG_ID        = 8'h00;  // 0x424C4443 'BLDC' (R)
    localparam REG_CTRL      = 8'h04;  // [0]=EN, [1]=IE, [2]=DIR (RW)
    localparam REG_DUTY      = 8'h08;  // [9:0] 10-bit duty 0..1023 (RW)
    localparam REG_DEADTIME  = 8'h0C;  // [7:0] deadtime clock cycles (RW)
    localparam REG_STATUS    = 8'h10;  // [2:0]=HALL, [5:3]=STEP, [6]=FAULT (R)
    localparam REG_IRQ_STAT  = 8'h14;  // [0]=STEP_CHANGE W1C (RW)

    localparam ID_VAL = 32'h424C4443; // 'BLDC'

    // -----------------------------------------------------------------------
    // 2-stage synchronizer for hall inputs
    // -----------------------------------------------------------------------
    reg [2:0] hall_s0, hall_s1, hall_s2;
    always @(posedge clk) begin
        if (!rst_n) begin
            hall_s0 <= 3'b101; hall_s1 <= 3'b101; hall_s2 <= 3'b101;
        end else begin
            hall_s0 <= hall_in;
            hall_s1 <= hall_s0;
            hall_s2 <= hall_s1;
        end
    end
    wire [2:0] hall = hall_s2; // synchronized

    // -----------------------------------------------------------------------
    // Hall → step decode function
    // Forward sequence: 101(0)→001(1)→011(2)→010(3)→110(4)→100(5)
    // Returns {fault, step[2:0]}
    // -----------------------------------------------------------------------
    function [3:0] hall2step;
        input [2:0] h;
        case (h)
            3'b101: hall2step = 4'b0_000; // step 0
            3'b001: hall2step = 4'b0_001; // step 1
            3'b011: hall2step = 4'b0_010; // step 2
            3'b010: hall2step = 4'b0_011; // step 3
            3'b110: hall2step = 4'b0_100; // step 4
            3'b100: hall2step = 4'b0_101; // step 5
            default: hall2step = 4'b1_000; // 000/111 = fault
        endcase
    endfunction

    // Step → {HIN[2:0], LIN[2:0]} gate pattern
    // Step 0: A+ B-   HIN=001  LIN=010
    // Step 1: A+ C-   HIN=001  LIN=100
    // Step 2: B+ C-   HIN=010  LIN=100
    // Step 3: B+ A-   HIN=010  LIN=001
    // Step 4: C+ A-   HIN=100  LIN=001
    // Step 5: C+ B-   HIN=100  LIN=010
    function [5:0] step2gate;
        input [2:0] s;
        case (s)
            3'd0: step2gate = 6'b001_010;
            3'd1: step2gate = 6'b001_100;
            3'd2: step2gate = 6'b010_100;
            3'd3: step2gate = 6'b010_001;
            3'd4: step2gate = 6'b100_001;
            3'd5: step2gate = 6'b100_010;
            default: step2gate = 6'b000_000;
        endcase
    endfunction

    // -----------------------------------------------------------------------
    // Control registers
    // -----------------------------------------------------------------------
    reg         ctrl_en;
    reg         ctrl_ie;
    reg         ctrl_dir;       // 1 = reverse commutation
    reg [9:0]   r_duty;
    reg [7:0]   r_deadtime;
    reg [31:0]  irq_status;     // [0] = STEP_CHANGE (W1C)

    // Write address capture
    reg [ADDR_W-1:0] waddr_lat;
    reg              waddr_valid;

    // -----------------------------------------------------------------------
    // Hall decode and step tracking
    // -----------------------------------------------------------------------
    wire [3:0] decode     = hall2step(hall);
    wire       fault_raw  = decode[3];
    wire [2:0] step_fwd   = decode[2:0];

    // Apply direction: reverse maps step 0↔5, 1↔4, 2↔3
    // Equivalent to (5 - step_fwd) for 3-bit wrap
    wire [2:0] step = ctrl_dir ? (3'd5 - step_fwd) : step_fwd;

    reg [2:0] step_prev;
    reg       fault_lat;      // sticky fault (single driver below)
    reg       fault_raw_d;    // previous cycle fault_raw for edge detection

    wire fault_rise   = fault_raw & ~fault_raw_d;  // first cycle of invalid hall
    // Combinatorial: CTRL register write in progress this cycle
    wire ctrl_writing = s_wvalid & waddr_valid & (waddr_lat == REG_CTRL);

    wire step_change = (step != step_prev) && !fault_raw;

    // -----------------------------------------------------------------------
    // Single-driver fault_lat block (rising-edge trigger; CTRL write clears)
    // Setting fault_lat on rising edge only means a sustained invalid-hall
    // signal (fault_raw stays high) does NOT re-assert fault_lat after a
    // software clear via CTRL register write.
    // -----------------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n) begin
            fault_lat   <= 1'b0;
            fault_raw_d <= 1'b0;
        end else begin
            fault_raw_d <= fault_raw;
            if (ctrl_writing)       // software clear wins
                fault_lat <= 1'b0;
            else if (fault_rise)    // set on first cycle of invalid hall only
                fault_lat <= 1'b1;
        end
    end

    // -----------------------------------------------------------------------
    // Commutation deadtime counter
    // -----------------------------------------------------------------------
    reg [7:0] dt_cnt;
    wire      deadtime_active = (dt_cnt != 8'd0);

    always @(posedge clk) begin
        if (!rst_n) begin
            dt_cnt    <= 8'd0;
            step_prev <= 3'd0;
            sample_trig <= 1'b0;
        end else begin
            sample_trig <= 1'b0;

            step_prev <= step;

            if (step_change && ctrl_en) begin
                dt_cnt <= r_deadtime;  // reload deadtime on commutation step
            end else if (dt_cnt != 8'd0) begin
                dt_cnt <= dt_cnt - 1'b1;
                if (dt_cnt == 8'd1)
                    sample_trig <= 1'b1;  // pulse at end of deadtime
            end
        end
    end

    // -----------------------------------------------------------------------
    // 10-bit PWM counter
    // -----------------------------------------------------------------------
    reg [9:0] pwm_cnt;
    always @(posedge clk) begin
        if (!rst_n || !ctrl_en) pwm_cnt <= 10'd0;
        else                     pwm_cnt <= pwm_cnt + 1'b1;
    end
    wire raw_pwm = (pwm_cnt < r_duty);

    // -----------------------------------------------------------------------
    // Gate output logic
    // High-side is PWM'd; low-side is DC ON in active step.
    // All outputs forced to 0 during deadtime or when disabled.
    // -----------------------------------------------------------------------
    wire [5:0] gate_pattern = step2gate(step);
    wire [2:0] hin_step = gate_pattern[5:3];
    wire [2:0] lin_step = gate_pattern[2:0];

    always @(posedge clk) begin
        if (!rst_n || !ctrl_en || fault_lat || deadtime_active) begin
            hin <= 3'd0;
            lin <= 3'd0;
        end else begin
            hin <= hin_step & {3{raw_pwm}};  // PWM on high-side
            lin <= lin_step;                   // DC on low-side (complementary phase)
        end
    end

    // -----------------------------------------------------------------------
    // IRQ
    // -----------------------------------------------------------------------
    assign irq = ctrl_ie & (|irq_status);

    // -----------------------------------------------------------------------
    // AXI4-Lite write channel
    // -----------------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n) begin
            s_awready  <= 1'b0; s_wready  <= 1'b0;
            s_bvalid   <= 1'b0; s_bresp   <= 2'b00;
            waddr_lat  <= {ADDR_W{1'b0}}; waddr_valid <= 1'b0;
            ctrl_en    <= 1'b0; ctrl_ie   <= 1'b0; ctrl_dir <= 1'b0;
            r_duty     <= DEFAULT_DUTY;
            r_deadtime <= DEFAULT_DEADTIME;
            irq_status <= 32'd0;
        end else begin
            // latch write address
            s_awready <= 1'b0;
            if (s_awvalid && !waddr_valid) begin
                s_awready  <= 1'b1;
                waddr_lat  <= s_awaddr;
                waddr_valid<= 1'b1;
            end

            // latch write data + perform write
            s_wready <= 1'b0;
            if (s_wvalid && waddr_valid) begin
                s_wready    <= 1'b1;
                waddr_valid <= 1'b0;
                s_bvalid    <= 1'b1;
                s_bresp     <= 2'b00;
                case (waddr_lat)
                    REG_CTRL: begin
                        ctrl_en  <= s_wdata[0];
                        ctrl_ie  <= s_wdata[1];
                        ctrl_dir <= s_wdata[2];
                        // fault_lat cleared via ctrl_writing signal in its own always block
                    end
                    REG_DUTY:     r_duty     <= s_wdata[9:0];
                    REG_DEADTIME: r_deadtime <= s_wdata[7:0];
                    REG_IRQ_STAT: irq_status <= irq_status & ~s_wdata; // W1C
                    default: ;
                endcase
            end

            if (s_bvalid && s_bready) s_bvalid <= 1'b0;

            // hardware-set sticky IRQ bits
            if (step_change) irq_status[0] <= 1'b1;
        end
    end

    // -----------------------------------------------------------------------
    // AXI4-Lite read channel
    // -----------------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n) begin
            s_arready <= 1'b0;
            s_rvalid  <= 1'b0;
            s_rdata   <= 32'd0;
            s_rresp   <= 2'b00;
        end else begin
            s_arready <= 1'b0;

            if (s_arvalid && !s_rvalid) begin
                s_arready <= 1'b1;
                s_rvalid  <= 1'b1;
                s_rresp   <= 2'b00;
                case (s_araddr)
                    REG_ID:       s_rdata <= ID_VAL;
                    REG_CTRL:     s_rdata <= {29'd0, ctrl_dir, ctrl_ie, ctrl_en};
                    REG_DUTY:     s_rdata <= {22'd0, r_duty};
                    REG_DEADTIME: s_rdata <= {24'd0, r_deadtime};
                    REG_STATUS:   s_rdata <= {25'd0, fault_lat, step, hall};
                    REG_IRQ_STAT: s_rdata <= irq_status;
                    default:      s_rdata <= 32'hDEAD_BEEF;
                endcase
            end

            if (s_rvalid && s_rready) s_rvalid <= 1'b0;
        end
    end

endmodule

// tb_bldc_hall_sixstep_ip.v
// Self-checking testbench for bldc_hall_sixstep_ip
// Tests:
//  T1: ID register read = 0x424C4443
//  T2: Enable, set duty/deadtime, check gate assertion after deadtime
//  T3: Forward hall sequence, check step changes + IRQ
//  T4: Shoot-through check (hin[i] & lin[i] must never both be 1)
//  T5: Deadtime measurement on commutation event
//  T6: Fault injection (hall=000 → hin/lin must be 0, STATUS[6]=1)
// Output: PASS or FAIL (first token)
// Verilog-2001

`timescale 1ns/1ps

module tb_bldc_hall_sixstep_ip;

    // -----------------------------------------------------------------------
    // Simulation parameters
    // -----------------------------------------------------------------------
    localparam SIM_DUTY     = 10'd256;   // 25% duty for fast test
    localparam SIM_DEADTIME = 8'd8;      // 8 cycles deadtime for fast test

    // -----------------------------------------------------------------------
    // Clock & reset
    // -----------------------------------------------------------------------
    reg clk;
    initial clk = 1'b0;
    always #5 clk = ~clk; // 100 MHz

    reg rst_n;

    // -----------------------------------------------------------------------
    // DUT wires
    // -----------------------------------------------------------------------
    reg  [7:0]  awaddr;
    reg         awvalid;
    wire        awready;
    reg  [31:0] wdata;
    reg  [3:0]  wstrb;
    reg         wvalid;
    wire        wready;
    wire [1:0]  bresp;
    wire        bvalid;
    reg         bready;
    reg  [7:0]  araddr;
    reg         arvalid;
    wire        arready;
    wire [31:0] rdata;
    wire [1:0]  rresp;
    wire        rvalid;
    reg         rready;

    reg  [2:0]  hall_in;
    wire [2:0]  hin;
    wire [2:0]  lin;
    wire        sample_trig;
    wire        irq;

    bldc_hall_sixstep_ip #(
        .DEFAULT_DUTY    (SIM_DUTY),
        .DEFAULT_DEADTIME(SIM_DEADTIME)
    ) dut (
        .clk        (clk),
        .rst_n      (rst_n),
        .s_awaddr   (awaddr),
        .s_awvalid  (awvalid),
        .s_awready  (awready),
        .s_wdata    (wdata),
        .s_wstrb    (wstrb),
        .s_wvalid   (wvalid),
        .s_wready   (wready),
        .s_bresp    (bresp),
        .s_bvalid   (bvalid),
        .s_bready   (bready),
        .s_araddr   (araddr),
        .s_arvalid  (arvalid),
        .s_arready  (arready),
        .s_rdata    (rdata),
        .s_rresp    (rresp),
        .s_rvalid   (rvalid),
        .s_rready   (rready),
        .hall_in    (hall_in),
        .hin        (hin),
        .lin        (lin),
        .sample_trig(sample_trig),
        .irq        (irq)
    );

    // -----------------------------------------------------------------------
    // AXI4-Lite helper tasks
    // -----------------------------------------------------------------------
    task axi_write(input [7:0] addr, input [31:0] data);
        begin
            @(posedge clk);
            awaddr  = addr;
            awvalid = 1'b1;
            wdata   = data;
            wstrb   = 4'hF;
            wvalid  = 1'b1;
            bready  = 1'b1;

            @(posedge clk);
            while (!awready) @(posedge clk);
            @(posedge clk);
            awvalid = 1'b0;

            while (!wready) @(posedge clk);
            wvalid = 1'b0;

            while (!bvalid) @(posedge clk);
            @(posedge clk);
            bready = 1'b0;
        end
    endtask

    task axi_read(input [7:0] addr, output [31:0] data);
        begin
            @(posedge clk);
            araddr  = addr;
            arvalid = 1'b1;
            rready  = 1'b1;

            @(posedge clk);
            while (!arready) @(posedge clk);
            arvalid = 1'b0;

            while (!rvalid) @(posedge clk);
            data = rdata;
            @(posedge clk);
            rready = 1'b0;
        end
    endtask

    // -----------------------------------------------------------------------
    // Test variables
    // -----------------------------------------------------------------------
    reg [31:0] rd_val;
    integer    fail_count;
    integer    i;
    integer    dt_cycles;

    // Shoot-through monitor: combinational, asserts immediately
    reg shoot_through_seen;

    task check(input pass_cond, input [7:0] test_id, input [31:0] got, input [31:0] exp);
        begin
            if (!pass_cond) begin
                $display("FAIL T%0d: got=0x%08X exp=0x%08X", test_id, got, exp);
                fail_count = fail_count + 1;
            end
        end
    endtask

    // -----------------------------------------------------------------------
    // Shoot-through monitor (clocked check)
    // -----------------------------------------------------------------------
    always @(posedge clk) begin
        // If any high-side AND low-side of same phase are both driven high
        // that is a shoot-through condition
        if (rst_n && ((hin[0] & lin[0]) || (hin[1] & lin[1]) || (hin[2] & lin[2]))) begin
            shoot_through_seen = 1'b1;
        end
    end

    // -----------------------------------------------------------------------
    // Forward hall sequence: 101→001→011→010→110→100→(101)
    // -----------------------------------------------------------------------
    reg [2:0] hall_fwd [0:5];
    initial begin
        hall_fwd[0] = 3'b101;
        hall_fwd[1] = 3'b001;
        hall_fwd[2] = 3'b011;
        hall_fwd[3] = 3'b010;
        hall_fwd[4] = 3'b110;
        hall_fwd[5] = 3'b100;
    end

    // -----------------------------------------------------------------------
    // Stimulus
    // -----------------------------------------------------------------------
    initial begin
        // Init
        fail_count         = 0;
        shoot_through_seen = 1'b0;
        rst_n    = 1'b0;
        hall_in  = 3'b101;    // step 0 valid
        awvalid  = 1'b0; awaddr = 8'd0;
        wvalid   = 1'b0; wdata  = 32'd0; wstrb = 4'hF;
        bready   = 1'b0;
        arvalid  = 1'b0; araddr = 8'd0;
        rready   = 1'b0;

        repeat (4) @(posedge clk);
        rst_n = 1'b1;
        repeat (2) @(posedge clk);

        // ---- T1: ID register ----
        axi_read(8'h00, rd_val);
        check(rd_val === 32'h424C4443, 1, rd_val, 32'h424C4443);

        // ---- T2: Enable, set duty and deadtime, check gate assertion ----
        // Write duty = SIM_DUTY
        axi_write(8'h08, {22'd0, SIM_DUTY});
        // Write deadtime = SIM_DEADTIME
        axi_write(8'h0C, {24'd0, SIM_DEADTIME});
        // Enable: ctrl_en=1, ie=1, dir=0
        axi_write(8'h04, 32'h3);

        // Wait for deadtime to expire (SIM_DEADTIME + sync latency + margin)
        repeat (SIM_DEADTIME + 10) @(posedge clk);

        // With hall_in=101 (step 0), after deadtime: hin[0] should PWM, lin[1] should be 1
        // Check lin[1] (B-) is asserted (DC on low-side)
        // Allow a full PWM period (SIM_DUTY < 1024) to see at least some hin activity
        // We just check lin is nonzero
        begin : wait_lin
            integer cnt;
            cnt = 0;
            while (lin === 3'b000 && cnt < 100) begin
                @(posedge clk);
                cnt = cnt + 1;
            end
            check(lin !== 3'b000, 2, {29'd0, lin}, 32'h2);
        end

        // ---- T3: Forward hall sequence, check IRQ fires ----
        // Clear IRQ status first
        axi_write(8'h14, 32'h1);  // W1C bit [0]

        for (i = 1; i < 6; i = i + 1) begin
            // Advance hall to next step
            hall_in = hall_fwd[i];
            // Wait for 3 sync stages + deadtime + margin
            repeat (SIM_DEADTIME + 20) @(posedge clk);
        end

        // IRQ_STAT[0] should be set (at least one step change happened)
        axi_read(8'h14, rd_val);
        check(rd_val[0] === 1'b1, 3, rd_val, 32'h1);

        // ---- T4: Shoot-through check (monitor runs throughout) ----
        // No explicit steps needed; the monitor caught any during T2/T3
        check(shoot_through_seen === 1'b0, 4, {31'd0, shoot_through_seen}, 32'h0);

        // ---- T5: Deadtime measurement ----
        // Trigger a fresh step change and count cycles before hin goes non-zero
        // First clear deadtime by staying on current step for a while
        hall_in = 3'b100; // step 5
        repeat (30) @(posedge clk);

        // Clear IRQ
        axi_write(8'h14, 32'h1);

        // Now step to 101 (step 0) — commutation event
        hall_in = 3'b101;

        // Count cycles until hin goes non-zero (high side asserts)
        begin : dt_measure
            integer cnt;
            reg hin_seen;
            cnt      = 0;
            hin_seen = 1'b0;
            repeat (SIM_DEADTIME + 50) @(posedge clk) begin
                cnt = cnt + 1;
                if (hin !== 3'b000) begin
                    hin_seen = 1'b1;
                    disable dt_measure;
                end
            end
            // hin must have appeared and after at least SIM_DEADTIME-3 cycles
            // (3 sync stages shift the step_change, then deadtime counts down)
            check(hin_seen === 1'b1,  51, {31'd0, hin_seen}, 32'h1);
            check(cnt >= SIM_DEADTIME, 52, cnt, {24'd0, SIM_DEADTIME});
        end

        // ---- T6: Fault injection ----
        // 000 and 111 are invalid hall codes → fault
        hall_in = 3'b000;
        repeat (10) @(posedge clk); // let sync stages propagate

        // hin and lin must both be 0 (outputs suppressed on fault)
        check(hin === 3'b000, 61, {29'd0, hin}, 32'h0);
        check(lin === 3'b000, 62, {29'd0, lin}, 32'h0);

        // STATUS[6] = fault_lat should be set
        axi_read(8'h10, rd_val);
        check(rd_val[6] === 1'b1, 63, rd_val, 32'h40);

        // Restore valid hall FIRST so fault_raw goes low before CTRL write.
        // (3 sync stages must propagate; writing CTRL while fault_raw=1 would
        //  re-latch the fault on the very next cycle.)
        hall_in = 3'b101;
        repeat (6) @(posedge clk); // 3 sync + 3 margin → fault_raw=0

        // Now clear fault by writing CTRL (re-enable).
        // Correct usage order: fix hardware problem, then acknowledge fault.
        axi_write(8'h04, 32'h3);  // ctrl_en=1, ctrl_ie=1, fault_lat cleared

        repeat (5) @(posedge clk);

        // STATUS[6] should now be clear
        axi_read(8'h10, rd_val);
        check(rd_val[6] === 1'b0, 64, rd_val, 32'h0);

        // ---- Final result ----
        repeat (10) @(posedge clk);
        if (fail_count === 0)
            $display("PASS: all tests passed");
        else
            $display("FAIL: %0d test(s) failed", fail_count);
        $finish;
    end

    // Timeout watchdog
    initial begin
        #500_000;
        $display("FAIL: timeout");
        $finish;
    end

endmodule

# Makefile — mmio-bldc-hall-sixstep-ip
# Targets: lint, sim, synth, all, clean
# Tools: iverilog (sim), verilator (lint), yosys (synth)

SLUG    := mmio-bldc-hall-sixstep-ip
TOP     := bldc_hall_sixstep_ip
TB      := tb_bldc_hall_sixstep_ip

RTL_SRC := rtl/$(TOP).v
TB_SRC  := tb/$(TB).v

BUILD   := build
VLOG    := iverilog
VVP     := vvp
VLOG_FLAGS := -g2001 -Wall -Wno-timescale

VERILATOR := verilator
VER_FLAGS := --lint-only -Wall --Wno-UNUSEDSIGNAL --Wno-MULTITOP -sv

YOSYS   := yosys
SYNTH_SCRIPT := $(BUILD)/synth.ys

.PHONY: all lint sim synth clean

all: lint sim synth

# -----------------------------------------------------------------------
# Lint (verilator)
# -----------------------------------------------------------------------
lint: $(RTL_SRC)
	@mkdir -p $(BUILD)
	$(VERILATOR) $(VER_FLAGS) $(RTL_SRC) 2>&1 | tee $(BUILD)/lint.log
	@echo "--- LINT DONE ---"

# -----------------------------------------------------------------------
# Simulation (iverilog + vvp)
# -----------------------------------------------------------------------
sim: $(RTL_SRC) $(TB_SRC)
	@mkdir -p $(BUILD)
	$(VLOG) $(VLOG_FLAGS) -o $(BUILD)/$(TB).vvp $(TB_SRC) $(RTL_SRC)
	$(VVP) $(BUILD)/$(TB).vvp | tee $(BUILD)/sim.log
	@grep -q "^PASS" $(BUILD)/sim.log && echo "--- SIM PASS ---" || (echo "--- SIM FAIL ---"; exit 1)

# -----------------------------------------------------------------------
# Synthesis (yosys, resource estimate only)
# -----------------------------------------------------------------------
synth: $(RTL_SRC)
	@mkdir -p $(BUILD)
	@printf 'read_verilog -sv %s\nsynth -top %s\nstat\n' $(RTL_SRC) $(TOP) > $(SYNTH_SCRIPT)
	$(YOSYS) -q -s $(SYNTH_SCRIPT) 2>&1 | tee $(BUILD)/synth.log
	@echo "--- SYNTH DONE ---"

# -----------------------------------------------------------------------
# Clean
# -----------------------------------------------------------------------
clean:
	rm -rf $(BUILD)

위험·한계

  • 게이트 드라이버 로직 레벨이 5V이면 Pmod 출력(3.3V) → 레벨 변환 없이 연결 금지
  • 모터 전원과 Basys3 전원을 절대 공통 연결하지 않음(GND만 공통, 별도 공급)
  • DEFAULT_DEADTIME=50(500 ns)는 대부분의 소형 게이트 드라이버에 충분하나, 사용하는 드라이버의 turn-off 지연을 반드시 확인
  • 홀 센서 신호를 3단 플립플롭 동기화(hall_s0→s1→s2)로 처리하므로 최소 3사이클 지연 발생. 고속 모터에서 단계 오검출 가능성 낮음
  • fault(000/111 홀 코드) 발생 시 CTRL 레지스터 쓰기로만 해제됨. 소프트웨어가 이를 처리하지 않으면 모터가 멈춘 채 유지됨
