UART→AXI4-Lite 메모리 브리지

PC UART 명령(쓰기/읽기)을 AXI4-Lite 마스터 트랜잭션으로 변환하는 FSM IP. CPU 없이도 호스트가 AXI4-Lite 슬레이브 레지스터를 직접 읽고 쓸 수 있다. 테스트벤치에서 UART 바이트열을 주입해 슬레이브 레지스터 값이 맞는지 PASS/FAIL로 검사한다.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: pending

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • 9바이트 쓰기 프레임(0x57+ADDR 4B+DATA 4B) 및 5바이트 읽기 프레임(0x52+ADDR 4B)을 AXI4-Lite 마스터 트랜잭션으로 변환
  • CPU 없이 호스트 PC에서 Basys3 위 AXI4-Lite 슬레이브 레지스터를 읽고 쓰기
  • iverilog 시뮬레이션으로 프로토콜 디코더와 AXI4-Lite 마스터 FSM을 PASS/FAIL 검증

범위 밖

  • MicroBlaze V CPU 연동(다음 단계 과제)
  • 다중 슬레이브 주소 디코더
  • DMA·버스트 전송
  • 타임아웃 복구 메커니즘
  • 보드 동작 검증(S3는 사용자 확인)

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-012 기반. 외부 부품 없음(USB-UART 내장).

클럭: 100 MHz (W5), 비동기 리셋 btnC 액티브 하이

UART 프레임: 쓰기 9바이트(0x57+ADDR 4B BE+DATA 4B BE)→ACK 0x06. 읽기 5바이트(0x52+ADDR 4B BE)→0x05+DATA 4B BE. 모든 주소는 32비트 정렬.

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스AXI4-Lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 부분 적합
이식 계획현재는 CPU 없는 독립 데모. MicroBlaze V 연동 시: (1) 슬롯 [16] 0x4016_0000에 브리지 상태 레지스터 AXI4-Lite 슬레이브 추가, (2) intc_free[13]을 완료 인터럽트로 배선. 사용자 확인 후 동결.

6. 기술 트리 (이 캡스톤의 확장 경로)

  • T1MMIO 뼈대 (GPIO + UART) 참고: mmio-uart-gpio-skeleton
  • T2UART→AXI4-Lite 브리지 FSM (CPU-less 독립 데모)
  • T3MicroBlaze V 연동: 슬롯 [16]에 브리지 상태 레지스터, intc=13 인터럽트

rtl/uart_rx.v 65줄
// uart_rx.v — UART 8N1 수신기
// 115200 baud @ 100 MHz → BAUD_DIV = 868
// 출력: data[7:0], valid (1클럭 펄스)
//
// 출처: 하네스 직접 작성 원본. 외부 코드 없음.

`timescale 1ns/1ps
module uart_rx #(
    parameter BAUD_DIV = 868
) (
    input  wire       clk,
    input  wire       rst,
    input  wire       rx,
    output reg  [7:0] data,
    output reg        valid
);
    localparam HALF     = BAUD_DIV / 2;
    // 스타트 비트 감지 후 첫 샘플까지 대기: BAUD_DIV + HALF - 2
    // = 1비트(스타트 완료) + 반비트(데이터 비트 0 중간) - 2클럭(감지+1사이클)
    localparam INIT_CNT = BAUD_DIV + HALF - 2;

    reg [10:0] cnt;      // 보율 카운터 (1300 < 2^11)
    reg [3:0]  bit_idx;  // 수신 비트 인덱스 (0..7 = 데이터, 8 = 스톱)
    reg [7:0]  shift;
    reg        active;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            cnt    <= 0;
            bit_idx<= 0;
            active <= 1'b0;
            valid  <= 1'b0;
            data   <= 8'h00;
            shift  <= 8'h00;
        end else begin
            valid <= 1'b0;
            if (!active) begin
                // 스타트 비트 하강 에지 감지
                if (!rx) begin
                    active  <= 1'b1;
                    cnt     <= INIT_CNT[10:0]; // 데이터 비트 0 중간 샘플 위치까지 대기
                    bit_idx <= 4'd0;
                end
            end else begin
                if (cnt == 0) begin
                    cnt <= BAUD_DIV[10:0] - 1;
                    if (bit_idx < 4'd8) begin
                        // 데이터 비트 샘플 (LSB 먼저)
                        shift   <= {rx, shift[7:1]};
                        bit_idx <= bit_idx + 1;
                    end else begin
                        // 스톱 비트 샘플
                        active <= 1'b0;
                        if (rx) begin   // 유효한 스톱 비트
                            data  <= shift;
                            valid <= 1'b1;
                        end
                    end
                end else begin
                    cnt <= cnt - 1;
                end
            end
        end
    end
endmodule
rtl/uart_tx.v 53줄
// uart_tx.v — UART 8N1 송신기
// 115200 baud @ 100 MHz → BAUD_DIV = 868
// 핸드셰이크: valid↑ + ready=1 → 전송 시작. ready=0이면 대기.
//
// 출처: 하네스 직접 작성 원본. 외부 코드 없음.

`timescale 1ns/1ps
module uart_tx #(
    parameter BAUD_DIV = 868
) (
    input  wire       clk,
    input  wire       rst,
    input  wire [7:0] data,
    input  wire       valid,
    output reg        ready,
    output reg        tx
);
    reg [10:0] cnt;
    reg [3:0]  bit_cnt;  // 0 = 스타트, 1-8 = 데이터, 9+ = 스톱 후 완료
    reg [7:0]  tx_sreg;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            tx      <= 1'b1;
            ready   <= 1'b1;
            cnt     <= 0;
            bit_cnt <= 0;
            tx_sreg <= 8'h00;
        end else if (ready) begin
            if (valid) begin
                tx_sreg <= data;
                ready   <= 1'b0;
                cnt     <= BAUD_DIV[10:0] - 1;
                bit_cnt <= 4'd0;
                tx      <= 1'b0;   // 스타트 비트
            end
        end else begin
            if (cnt == 0) begin
                cnt <= BAUD_DIV[10:0] - 1;
                if (bit_cnt <= 4'd7) begin
                    tx      <= tx_sreg[bit_cnt];
                    bit_cnt <= bit_cnt + 1;
                end else begin
                    // 스톱 비트 출력 후 완료
                    tx    <= 1'b1;
                    ready <= 1'b1;
                end
            end else begin
                cnt <= cnt - 1;
            end
        end
    end
endmodule
rtl/uart_axilite_bridge.v 237줄
// uart_axilite_bridge.v — UART→AXI4-Lite 메모리 브리지
//
// UART 프레임 프로토콜:
//   쓰기: 0x57('W') + ADDR[31:0]BE + DATA[31:0]BE  (9바이트) → 응답 0x06
//   읽기: 0x52('R') + ADDR[31:0]BE                  (5바이트) → 응답 0x05 + DATA[31:0]BE
// 모든 주소는 32비트 정렬 가정. 오류 복구 없음(리셋으로 복구).
//
// 출처: 하네스 직접 작성 원본. 외부 코드 없음.

`timescale 1ns/1ps
module uart_axilite_bridge #(
    parameter BAUD_DIV = 868
) (
    input  wire        clk,
    input  wire        rst,

    // UART 핀
    input  wire        uart_rx_pin,
    output wire        uart_tx_pin,

    // AXI4-Lite 마스터 포트
    output reg  [31:0] m_awaddr,
    output reg         m_awvalid,
    input  wire        m_awready,

    output reg  [31:0] m_wdata,
    output reg  [3:0]  m_wstrb,
    output reg         m_wvalid,
    input  wire        m_wready,

    input  wire [1:0]  m_bresp,
    input  wire        m_bvalid,
    output reg         m_bready,

    output reg  [31:0] m_araddr,
    output reg         m_arvalid,
    input  wire        m_arready,

    input  wire [31:0] m_rdata,
    input  wire [1:0]  m_rresp,
    input  wire        m_rvalid,
    output reg         m_rready,

    // 상태 출력
    output reg         busy,
    output reg         err_flag
);

    // UART RX 인터페이스
    wire [7:0] rx_data;
    wire       rx_valid;

    uart_rx #(.BAUD_DIV(BAUD_DIV)) u_rx (
        .clk(clk), .rst(rst),
        .rx(uart_rx_pin),
        .data(rx_data), .valid(rx_valid)
    );

    // UART TX 인터페이스
    reg  [7:0] tx_data;
    reg        tx_valid;
    wire       tx_ready;

    uart_tx #(.BAUD_DIV(BAUD_DIV)) u_tx (
        .clk(clk), .rst(rst),
        .data(tx_data), .valid(tx_valid),
        .ready(tx_ready), .tx(uart_tx_pin)
    );

    // FSM 상태 인코딩
    localparam [3:0]
        S_IDLE   = 4'd0,
        S_ADDR0  = 4'd1,
        S_ADDR1  = 4'd2,
        S_ADDR2  = 4'd3,
        S_ADDR3  = 4'd4,
        S_WDAT0  = 4'd5,
        S_WDAT1  = 4'd6,
        S_WDAT2  = 4'd7,
        S_WDAT3  = 4'd8,
        S_AXI_AW = 4'd9,
        S_AXI_B  = 4'd10,
        S_AXI_AR = 4'd11,
        S_AXI_R  = 4'd12,
        S_TX     = 4'd13;

    reg [3:0]  state;
    reg        is_write;
    reg [31:0] addr_buf;
    reg [23:0] wdata_hi;   // wdata[31:8] (상위 3바이트 임시 저장)
    reg        aw_done, w_done;

    // TX 응답 버퍼: 최대 5바이트 (인덱스 0이 먼저 송신)
    reg [7:0]  tx_buf [0:4];
    reg [2:0]  tx_len;
    reg [2:0]  tx_idx;

    // --- 메인 FSM ---
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state     <= S_IDLE;
            is_write  <= 1'b0;
            addr_buf  <= 32'h0;
            wdata_hi  <= 24'h0;
            aw_done   <= 1'b0;
            w_done    <= 1'b0;
            m_awvalid <= 1'b0;  m_awaddr <= 32'h0;
            m_wvalid  <= 1'b0;  m_wdata  <= 32'h0;  m_wstrb <= 4'hF;
            m_bready  <= 1'b0;
            m_arvalid <= 1'b0;  m_araddr <= 32'h0;
            m_rready  <= 1'b0;
            tx_valid  <= 1'b0;  tx_data  <= 8'h00;
            tx_len    <= 3'd0;  tx_idx   <= 3'd0;
            busy      <= 1'b0;  err_flag <= 1'b0;
        end else begin
            tx_valid <= 1'b0;  // 기본: 비활성 (S_TX에서 덮어씀)

            case (state)
                S_IDLE: begin
                    busy <= 1'b0;
                    if (rx_valid) begin
                        busy <= 1'b1;
                        if (rx_data == 8'h57) begin        // 'W'
                            is_write <= 1'b1;
                            state    <= S_ADDR0;
                        end else if (rx_data == 8'h52) begin  // 'R'
                            is_write <= 1'b0;
                            state    <= S_ADDR0;
                        end else begin
                            err_flag <= 1'b1;
                        end
                    end
                end

                // 주소 수신 (MSB first)
                S_ADDR0: if (rx_valid) begin addr_buf[31:24] <= rx_data; state <= S_ADDR1; end
                S_ADDR1: if (rx_valid) begin addr_buf[23:16] <= rx_data; state <= S_ADDR2; end
                S_ADDR2: if (rx_valid) begin addr_buf[15: 8] <= rx_data; state <= S_ADDR3; end
                S_ADDR3: if (rx_valid) begin
                    addr_buf[7:0] <= rx_data;
                    state <= is_write ? S_WDAT0 : S_AXI_AR;
                end

                // 쓰기 데이터 수신 (MSB first, 상위 3바이트 임시 보관)
                S_WDAT0: if (rx_valid) begin wdata_hi[23:16] <= rx_data; state <= S_WDAT1; end
                S_WDAT1: if (rx_valid) begin wdata_hi[15: 8] <= rx_data; state <= S_WDAT2; end
                S_WDAT2: if (rx_valid) begin wdata_hi[ 7: 0] <= rx_data; state <= S_WDAT3; end
                S_WDAT3: if (rx_valid) begin
                    // addr_buf 완전히 채워진 상태, rx_data = wdata[7:0]
                    m_awaddr  <= addr_buf;
                    m_awvalid <= 1'b1;
                    m_wdata   <= {wdata_hi, rx_data};
                    m_wstrb   <= 4'hF;
                    m_wvalid  <= 1'b1;
                    aw_done   <= 1'b0;
                    w_done    <= 1'b0;
                    state     <= S_AXI_AW;
                end

                // AXI4-Lite 쓰기 AW+W 채널 (동시 구동)
                S_AXI_AW: begin
                    if (m_awvalid && m_awready) begin
                        m_awvalid <= 1'b0;
                        aw_done   <= 1'b1;
                    end
                    if (m_wvalid && m_wready) begin
                        m_wvalid <= 1'b0;
                        w_done   <= 1'b1;
                    end
                    // 양쪽 채널 모두 완료 시 B 채널 대기
                    if ((aw_done || (m_awvalid && m_awready)) &&
                        (w_done  || (m_wvalid  && m_wready))) begin
                        m_awvalid <= 1'b0;
                        m_wvalid  <= 1'b0;
                        m_bready  <= 1'b1;
                        state     <= S_AXI_B;
                    end
                end

                // AXI4-Lite B 채널 대기
                S_AXI_B: begin
                    if (m_bvalid) begin
                        m_bready  <= 1'b0;
                        err_flag  <= (m_bresp != 2'b00) ? 1'b1 : err_flag;
                        tx_buf[0] <= 8'h06;   // 쓰기 ACK
                        tx_len    <= 3'd1;
                        tx_idx    <= 3'd0;
                        state     <= S_TX;
                    end
                end

                // AXI4-Lite AR 채널
                S_AXI_AR: begin
                    m_araddr  <= addr_buf;
                    m_arvalid <= 1'b1;
                    if (m_arvalid && m_arready) begin
                        m_arvalid <= 1'b0;
                        m_rready  <= 1'b1;
                        state     <= S_AXI_R;
                    end
                end

                // AXI4-Lite R 채널 대기
                S_AXI_R: begin
                    if (m_rvalid) begin
                        m_rready  <= 1'b0;
                        err_flag  <= (m_rresp != 2'b00) ? 1'b1 : err_flag;
                        tx_buf[0] <= 8'h05;             // 읽기 ACK
                        tx_buf[1] <= m_rdata[31:24];
                        tx_buf[2] <= m_rdata[23:16];
                        tx_buf[3] <= m_rdata[15: 8];
                        tx_buf[4] <= m_rdata[ 7: 0];
                        tx_len    <= 3'd5;
                        tx_idx    <= 3'd0;
                        state     <= S_TX;
                    end
                end

                // TX 응답 바이트 순차 송신
                S_TX: begin
                    if (tx_ready && !tx_valid) begin
                        if (tx_idx < tx_len) begin
                            tx_data  <= tx_buf[tx_idx];
                            tx_valid <= 1'b1;
                            tx_idx   <= tx_idx + 1;
                        end else begin
                            busy  <= 1'b0;
                            state <= S_IDLE;
                        end
                    end
                end

                default: state <= S_IDLE;
            endcase
        end
    end
endmodule

// ─────────────────────────────────────────────
// demo_gpio_slave — AXI4-Lite 슬레이브 (데모 전용)
//
// 오프셋  이름        접근  설명
//  0x0000  GPIO_DATA   RW   쓰기→led_reg, 읽기→sw
//  0x0004  GPIO_DIR    RO   항상 0xFFFF0000
// ─────────────────────────────────────────────
module demo_gpio_slave (
    input  wire        clk,
    input  wire        rst,

    // AXI4-Lite 슬레이브 포트
    input  wire [31:0] s_awaddr,
    input  wire        s_awvalid,
    output reg         s_awready,

    input  wire [31:0] s_wdata,
    input  wire [3:0]  s_wstrb,
    input  wire        s_wvalid,
    output reg         s_wready,

    output reg  [1:0]  s_bresp,
    output reg         s_bvalid,
    input  wire        s_bready,

    input  wire [31:0] s_araddr,
    input  wire        s_arvalid,
    output reg         s_arready,

    output reg  [31:0] s_rdata,
    output reg  [1:0]  s_rresp,
    output reg         s_rvalid,
    input  wire        s_rready,

    // GPIO
    input  wire [15:0] sw,
    output reg  [15:0] led_reg
);
    // 쓰기 주소 래치
    reg [31:0] aw_addr_lat;
    reg        aw_latched;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            s_awready  <= 1'b1;
            s_wready   <= 1'b1;
            s_bvalid   <= 1'b0;
            s_bresp    <= 2'b00;
            s_arready  <= 1'b0;
            s_rvalid   <= 1'b0;
            s_rdata    <= 32'h0;
            s_rresp    <= 2'b00;
            led_reg    <= 16'h0;
            aw_addr_lat<= 32'h0;
            aw_latched <= 1'b0;
        end else begin
            // ── 쓰기 ──────────────────────────────────────
            // AW 채널
            if (s_awvalid && s_awready) begin
                aw_addr_lat <= s_awaddr;
                aw_latched  <= 1'b1;
                s_awready   <= 1'b0;
            end
            // W 채널
            if (s_wvalid && s_wready) begin
                if (aw_latched || (s_awvalid && s_awready)) begin
                    // GPIO_DATA (오프셋 0x0000)
                    if ((aw_latched ? aw_addr_lat[3:0] : s_awaddr[3:0]) == 4'h0) begin
                        if (s_wstrb[0]) led_reg[ 7:0] <= s_wdata[ 7:0];
                        if (s_wstrb[1]) led_reg[15:8] <= s_wdata[15:8];
                    end
                    // GPIO_DIR (오프셋 0x0004)는 RO, 무시
                end
                s_wready   <= 1'b0;
                s_bvalid   <= 1'b1;
                s_bresp    <= 2'b00;
                aw_latched <= 1'b0;
            end
            // B 채널
            if (s_bvalid && s_bready) begin
                s_bvalid  <= 1'b0;
                s_awready <= 1'b1;
                s_wready  <= 1'b1;
            end

            // ── 읽기 ──────────────────────────────────────
            if (!s_arready) begin
                s_arready <= 1'b1;
            end
            if (s_arvalid && s_arready) begin
                s_arready <= 1'b0;
                s_rvalid  <= 1'b1;
                s_rresp   <= 2'b00;
                case (s_araddr[3:0])
                    4'h0:    s_rdata <= {16'h0, sw};         // GPIO_DATA: 읽기→SW
                    4'h4:    s_rdata <= 32'hFFFF0000;         // GPIO_DIR
                    default: s_rdata <= 32'h0;
                endcase
            end
            if (s_rvalid && s_rready) begin
                s_rvalid  <= 1'b0;
                s_arready <= 1'b1;
            end
        end
    end
endmodule

// tb_uart_axilite_bridge.v — UART→AXI4-Lite 브리지 자동 검증 테스트벤치
//
// 테스트 항목:
//   TC1: 쓰기 (0x57 + ADDR 0x0000_0000 + DATA 0xABCD_1234) → ACK 0x06 확인
//       GPIO_DATA 읽기 → 슬레이브 led_reg 확인
//   TC2: 읽기 (0x52 + ADDR 0x0004) GPIO_DIR → 0x05+0xFFFF0000 확인
//   TC3: SW→GPIO 읽기 (sw=0x5A5A)
//
// 통과: PASS, 실패: FAIL (각 줄 첫 글자)
// 비교: !== (X/Z도 실패로 처리)
//
// 출처: 하네스 직접 작성 원본. 외부 코드 없음.

`timescale 1ns/1ps

module tb_uart_axilite_bridge;

    // ─── 파라미터 ───────────────────────────────────
    localparam BAUD_DIV  = 20;    // 시뮬레이션 속도: 실제 868 대신 20클럭/비트
    localparam CLK_HALF  = 5;     // 100 MHz 클럭 반주기 (ns)
    localparam BIT_TICKS = BAUD_DIV; // 비트당 클럭 수

    // ─── 클럭 / 리셋 ────────────────────────────────
    reg clk = 0;
    reg rst = 1;
    always #CLK_HALF clk = ~clk;

    // ─── DUT 신호 ───────────────────────────────────
    reg  uart_rxd_drv = 1'b1;   // PC→FPGA (테스트벤치가 구동)
    wire uart_txd;               // FPGA→PC
    reg  [15:0] sw = 16'h0000;
    wire [15:0] led;

    // ─── DUT ────────────────────────────────────────
    basys3_top #() dut (   // BAUD_DIV는 내부 로컬 — 직접 override 불가
        .clk     (clk),
        .rst     (rst),
        .uart_rxd(uart_rxd_drv),
        .uart_txd(uart_txd),
        .sw      (sw),
        .led     (led)
    );

    // BAUD_DIV override: DUT 내부 u_bridge의 파라미터를 시뮬레이션 속도로 변경하기 위해
    // 하위 인스턴스 직접 접근 (iverilog 허용)
    // → 대신 별도 모듈 계층으로 직접 테스트한다 (아래 참조)

    // ─── UART 전송 태스크 (테스트벤치→DUT) ──────────
    // BIT_TICKS 클럭씩 각 비트를 구동
    task send_byte;
        input [7:0] data;
        integer i;
        begin
            // 스타트 비트
            @(posedge clk); #1;
            uart_rxd_drv = 1'b0;
            repeat (BIT_TICKS) @(posedge clk);
            // 데이터 비트 (LSB first)
            for (i = 0; i < 8; i = i + 1) begin
                #1; uart_rxd_drv = data[i];
                repeat (BIT_TICKS) @(posedge clk);
            end
            // 스톱 비트
            #1; uart_rxd_drv = 1'b1;
            repeat (BIT_TICKS) @(posedge clk);
        end
    endtask

    // ─── UART 수신 태스크 (DUT→테스트벤치) ──────────

        output [7:0] data;
        integer i;
        begin
            // 스타트 비트 하강 에지 대기
            @(negedge uart_txd);
            // 반주기 대기 (중간 샘플)
            repeat (BIT_TICKS/2) @(posedge clk);
            // 데이터 비트 샘플 (LSB first)
            for (i = 0; i < 8; i = i + 1) begin
                repeat (BIT_TICKS) @(posedge clk);
                data[i] = uart_txd;
            end
            // 스톱 비트 대기
            repeat (BIT_TICKS) @(posedge clk);
        end
    endtask

    // ─── 직접 테스트 모듈 (파라미터화) ─────────────
    // basys3_top은 파라미터를 외부로 내보내지 않으므로
    // uart_axilite_bridge + demo_gpio_slave를 직접 인스턴스화

    wire [31:0] m_awaddr;
    wire        m_awvalid, m_awready;
    wire [31:0] m_wdata;
    wire [3:0]  m_wstrb;
    wire        m_wvalid, m_wready;

    wire [31:0] m_araddr;
    wire        m_arvalid, m_arready;
    wire [31:0] m_rdata;
    wire [1:0]  m_rresp;
    wire        m_rvalid, m_rready;
    wire        busy, err_flag;
    wire [15:0] gpio_led;
    wire        uart_txd2;

    reg  uart_rxd2 = 1'b1;
    reg  [15:0] sw2 = 16'h0000;

    uart_axilite_bridge #(.BAUD_DIV(BAUD_DIV)) u_brg (
        .clk        (clk),  .rst     (rst),
        .uart_rx_pin(uart_rxd2), .uart_tx_pin(uart_txd2),
        .m_awaddr(m_awaddr), .m_awvalid(m_awvalid), .m_awready(m_awready),
        .m_wdata (m_wdata),  .m_wstrb  (m_wstrb),
        .m_wvalid(m_wvalid), .m_wready (m_wready),
        .m_bresp (m_bresp),  .m_bvalid (m_bvalid),  .m_bready(m_bready),
        .m_araddr(m_araddr), .m_arvalid(m_arvalid), .m_arready(m_arready),
        .m_rdata (m_rdata),  .m_rresp  (m_rresp),
        .m_rvalid(m_rvalid), .m_rready (m_rready),
        .busy(busy), .err_flag(err_flag)
    );

    demo_gpio_slave u_gpio (
        .clk(clk), .rst(rst),
        .s_awaddr(m_awaddr), .s_awvalid(m_awvalid), .s_awready(m_awready),
        .s_wdata (m_wdata),  .s_wstrb  (m_wstrb),
        .s_wvalid(m_wvalid), .s_wready (m_wready),
        .s_bresp (m_bresp),  .s_bvalid (m_bvalid),  .s_bready(m_bready),
        .s_araddr(m_araddr), .s_arvalid(m_arvalid), .s_arready(m_arready),
        .s_rdata (m_rdata),  .s_rresp  (m_rresp),
        .s_rvalid(m_rvalid), .s_rready (m_rready),
        .sw(sw2), .led_reg(gpio_led)
    );

    // ─── 수신 태스크 (uart_txd2 기준) ───────────────
    task recv_byte2;
        output [7:0] data;
        integer i;
        begin
            @(negedge uart_txd2);
            repeat (BIT_TICKS/2) @(posedge clk);
            for (i = 0; i < 8; i = i + 1) begin
                repeat (BIT_TICKS) @(posedge clk);
                data[i] = uart_txd2;
            end
            repeat (BIT_TICKS) @(posedge clk);
        end
    endtask

    // ─── 전송 태스크 (uart_rxd2 기준) ───────────────
    task send_byte2;
        input [7:0] data;
        integer i;
        begin
            @(posedge clk); #1;
            uart_rxd2 = 1'b0;
            repeat (BIT_TICKS) @(posedge clk);
            for (i = 0; i < 8; i = i + 1) begin
                #1; uart_rxd2 = data[i];
                repeat (BIT_TICKS) @(posedge clk);
            end
            #1; uart_rxd2 = 1'b1;
            repeat (BIT_TICKS) @(posedge clk);
        end
    endtask

    // ─── 검사 변수 ───────────────────────────────────
    integer fail_count;
    reg [7:0] rx_byte;
    reg [31:0] rx_word;

    // ─── 메인 ────────────────────────────────────────
    initial begin
        fail_count = 0;

        // 리셋
        rst = 1;
        repeat (10) @(posedge clk);
        @(posedge clk); #1; rst = 0;
        repeat (5) @(posedge clk);

        // ════════════════════════════════════════════
        // TC1: 쓰기 — 0xABCD1234를 주소 0x00000000에 씀
        // ════════════════════════════════════════════
        // 프레임: 0x57 0x00 0x00 0x00 0x00 0xAB 0xCD 0x12 0x34
        send_byte2(8'h57);       // 'W'
        send_byte2(8'h00);       // ADDR[31:24]
        send_byte2(8'h00);       // ADDR[23:16]
        send_byte2(8'h00);       // ADDR[15:8]
        send_byte2(8'h00);       // ADDR[7:0]
        send_byte2(8'hAB);       // DATA[31:24]
        send_byte2(8'hCD);       // DATA[23:16]
        send_byte2(8'h12);       // DATA[15:8]
        send_byte2(8'h34);       // DATA[7:0]

        // ACK 수신 (0x06)
        recv_byte2(rx_byte);
        if (rx_byte !== 8'h06) begin
            $display("FAIL TC1a: write ACK got 0x%02X, expected 0x06", rx_byte);
            fail_count = fail_count + 1;
        end else
            $display("PASS TC1a: write ACK = 0x06");

        // GPIO 슬레이브 led_reg 확인
        repeat (5) @(posedge clk);
        if (gpio_led !== 16'h1234) begin
            $display("FAIL TC1b: led_reg got 0x%04X, expected 0x1234", gpio_led);
            fail_count = fail_count + 1;
        end else
            $display("PASS TC1b: led_reg = 0x1234");

        // ════════════════════════════════════════════
        // TC2: GPIO_DIR 읽기 — 주소 0x00000004 → 0xFFFF0000
        // ════════════════════════════════════════════
        send_byte2(8'h52);       // 'R'
        send_byte2(8'h00);       // ADDR[31:24]
        send_byte2(8'h00);       // ADDR[23:16]
        send_byte2(8'h00);       // ADDR[15:8]
        send_byte2(8'h04);       // ADDR[7:0]

        recv_byte2(rx_byte);
        if (rx_byte !== 8'h05) begin
            $display("FAIL TC2a: read ACK got 0x%02X, expected 0x05", rx_byte);
            fail_count = fail_count + 1;
        end else
            $display("PASS TC2a: read ACK = 0x05");

        recv_byte2(rx_word[31:24]);
        recv_byte2(rx_word[23:16]);
        recv_byte2(rx_word[15:8]);
        recv_byte2(rx_word[7:0]);

        if (rx_word !== 32'hFFFF0000) begin
            $display("FAIL TC2b: GPIO_DIR got 0x%08X, expected 0xFFFF0000", rx_word);
            fail_count = fail_count + 1;
        end else
            $display("PASS TC2b: GPIO_DIR = 0xFFFF0000");

        // ════════════════════════════════════════════
        // TC3: SW→GPIO 읽기 (sw2=0x5A5A)
        // ════════════════════════════════════════════
        sw2 = 16'h5A5A;
        repeat (3) @(posedge clk);

        send_byte2(8'h52);       // 'R'
        send_byte2(8'h00);
        send_byte2(8'h00);
        send_byte2(8'h00);
        send_byte2(8'h00);       // GPIO_DATA 주소

        recv_byte2(rx_byte);
        if (rx_byte !== 8'h05) begin
            $display("FAIL TC3a: read ACK got 0x%02X, expected 0x05", rx_byte);
            fail_count = fail_count + 1;
        end else
            $display("PASS TC3a: read ACK = 0x05");

        recv_byte2(rx_word[31:24]);
        recv_byte2(rx_word[23:16]);
        recv_byte2(rx_word[15:8]);
        recv_byte2(rx_word[7:0]);

        if (rx_word !== 32'h00005A5A) begin
            $display("FAIL TC3b: GPIO_DATA (SW) got 0x%08X, expected 0x00005A5A", rx_word);
            fail_count = fail_count + 1;
        end else
            $display("PASS TC3b: GPIO_DATA (SW=0x5A5A) = 0x00005A5A");

        // ════════════════════════════════════════════
        // TC4: err_flag가 쓰기/읽기 성공 후 클리어 상태 확인
        // ════════════════════════════════════════════
        repeat (5) @(posedge clk);
        if (err_flag !== 1'b0) begin
            $display("FAIL TC4: err_flag should be 0 after clean transactions");
            fail_count = fail_count + 1;
        end else
            $display("PASS TC4: err_flag = 0");

        // ════════════════════════════════════════════
        // 최종 판정
        // ════════════════════════════════════════════
        repeat (10) @(posedge clk);
        if (fail_count == 0)
            $display("PASS ALL: all %0d checks passed", 6);
        else
            $display("FAIL: %0d check(s) failed", fail_count);

        $finish;
    end

    // 타임아웃 가드
    initial begin
        #2000000;
        $display("FAIL TIMEOUT: simulation exceeded limit");
        $finish;
    end

# Makefile — UART→AXI4-Lite 브리지 캡스톤
# 타깃: lint sim synth all clean
# 도구: verilator, iverilog/vvp, yosys

RTL_FILES  := rtl/uart_rx.v rtl/uart_tx.v rtl/uart_axilite_bridge.v rtl/basys3_top.v
TB_FILES   := tb/tb_uart_axilite_bridge.v $(RTL_FILES)
BUILD_DIR  := build
SIM_VVP    := $(BUILD_DIR)/sim.vvp
SIM_LOG    := $(BUILD_DIR)/sim.log

.PHONY: all lint sim synth clean

all: lint sim synth

$(BUILD_DIR):
	mkdir -p $(BUILD_DIR)

# ── 린트 (verilator) ────────────────────────────────────────────────────────
lint: $(BUILD_DIR)
	verilator --lint-only -Wall -Wno-fatal \
	    --top-module basys3_top $(RTL_FILES) \
	    2>&1 | tee $(BUILD_DIR)/lint.log
	@echo "Lint done."

# ── 시뮬레이션 (iverilog + vvp) ─────────────────────────────────────────────
sim: $(BUILD_DIR)
	iverilog -g2005 -o $(SIM_VVP) $(TB_FILES)
	vvp $(SIM_VVP) | tee $(SIM_LOG)
	@grep -E '^(PASS|FAIL)' $(SIM_LOG) || true

# ── 합성 (yosys) ────────────────────────────────────────────────────────────
synth: $(BUILD_DIR)
	yosys -p " \
	    read_verilog $(RTL_FILES); \
	    synth_xilinx -top basys3_top -family xc7; \
	    stat \
	" 2>&1 | tee $(BUILD_DIR)/synth.log
	@echo "Synth done."

# ── 정리 ────────────────────────────────────────────────────────────────────
clean:
	rm -rf $(BUILD_DIR)
