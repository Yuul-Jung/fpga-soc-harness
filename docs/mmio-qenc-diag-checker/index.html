캡스톤 패키지 · Basys3 · 단계 T2

직교 엔코더 진단 IP: 4체배 디코더 + 불가능 전이 카운터 + AXI4-Lite 레지스터 자체 점검

A/B 2상 엔코더 신호를 4체배(x4) 해독해 위치·속도·방향·오류 전이 수를 AXI4-Lite 슬레이브 레지스터로 노출하는 IP. 테스트벤치가 슬라이드 스위치 모사 AXI 쓰기/읽기로 정방향·역방향·글리치 주입·창 속도를 PASS/FAIL로 검사한다.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • A/B 엔코더 신호 4체배 해독 (방향·위치 32비트 부호 있는 카운터)
  • 10 ms 고정 창 속도 카운터 (100 MHz 기준 10^6 사이클)
  • 불가능 전이 오류 카운터 (글리치·단선 감지용)
  • AXI4-Lite 슬레이브 5개 레지스터 (CTRL/POSITION/VELOCITY/STATUS/ERROR_CNT)
  • 테스트벤치: 정방향 4펄스 → 위치 +4, 역방향 4펄스 → 위치 0, 글리치 주입 → 오류 카운터 ≥1 PASS/FAIL 검사

범위 밖

  • 인덱스(Z상) 처리
  • 고속 엔코더 보간 (보간 없이 4체배까지만)
  • 속도 PID 폐루프 (I-013 패키지 담당)
  • 보드 실물 검증 (S3)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-016 (ready, 차단조건재심사 9점)으로 생성됨 (2026-11-04). 직교 엔코더 디코더 + 불가능 전이 카운터 + AXI4-Lite 슬레이브.

// qenc_diag.v — 4체배 직교 엔코더 디코더 + 불가능 전이 카운터
// I-016 엔코더 난이도 사다리 (mmio-qenc-diag-checker, T2)
// Verilog-2001. iverilog/verilator/yosys 호환.
//
// 포트:
//   clk        — 100 MHz 시스템 클럭
//   rst        — 동기 active-high 리셋
//   en         — 1이면 디코드 활성
//   srst       — 소프트 리셋 (카운터 0으로 초기화, en 무관)
//   window_en  — 1이면 10 ms 창 속도 계산 활성
//   a_in, b_in — 엔코더 A/B 신호 (Basys3 Pmod, 3.3V)
//
//   position   — 32비트 부호 있는 위치 카운터
//   velocity   — 16비트 최근 10 ms 창 카운트
//   dir        — 0=정방향(A 먼저 올라감), 1=역방향
//   ovfl       — 위치 오버플로 래치 (부호 비트 토글)
//   err_latch  — 불가능 전이 발생 래치
//   error_cnt  — 16비트 불가능 전이 누산 카운터

`default_nettype none

module qenc_diag #(
    parameter WINDOW_CYCLES = 1_000_000  // 10 ms @ 100 MHz
) (
    input  wire        clk,
    input  wire        rst,
    input  wire        en,
    input  wire        srst,
    input  wire        window_en,

    // 엔코더 입력 (외부에서 메타스테이블 방지 2단 FF 적용 후 연결)
    input  wire        a_in,
    input  wire        b_in,

    // 출력
    output reg  [31:0] position,
    output reg  [15:0] velocity,
    output reg         dir,
    output reg         ovfl,
    output reg         err_latch,
    output reg  [15:0] error_cnt
);

    // ── 2단 메타스테이블 방지 ──────────────────────────────
    reg a0, a1, b0, b1;
    always @(posedge clk) begin
        a0 <= a_in; a1 <= a0;
        b0 <= b_in; b1 <= b0;
    end

    // ── 직전/현재 상태 레지스터 ───────────────────────────
    reg [1:0] prev_ab;
    wire [1:0] curr_ab = {a1, b1};

    // 4체배 Gray 전이 테이블
    // {prev[1],prev[0],curr[1],curr[0]} → {valid, dir_now}
    // dir_now: 0=정방향(+1), 1=역방향(-1)
    // 불가능 전이: 2비트 동시 변화 (00→11, 11→00, 01→10, 10→01)
    reg       trans_valid;
    reg       trans_dir;    // 0=CW(+), 1=CCW(-)

    always @(*) begin
        case ({prev_ab, curr_ab})
            // 정방향 (+1): 00→10, 10→11, 11→01, 01→00
            4'b00_10: begin trans_valid = en; trans_dir = 1'b0; end
            4'b10_11: begin trans_valid = en; trans_dir = 1'b0; end
            4'b11_01: begin trans_valid = en; trans_dir = 1'b0; end
            4'b01_00: begin trans_valid = en; trans_dir = 1'b0; end
            // 역방향 (-1): 00→01, 01→11, 11→10, 10→00
            4'b00_01: begin trans_valid = en; trans_dir = 1'b1; end
            4'b01_11: begin trans_valid = en; trans_dir = 1'b1; end
            4'b11_10: begin trans_valid = en; trans_dir = 1'b1; end
            4'b10_00: begin trans_valid = en; trans_dir = 1'b1; end
            // 정지(변화 없음): valid이지만 카운트하지 않음
            4'b00_00,
            4'b01_01,
            4'b10_10,
            4'b11_11: begin trans_valid = 1'b1; trans_dir = dir; end
            // 그 외: 불가능 전이 (2비트 동시 변화)
            default:  begin trans_valid = 1'b0; trans_dir = dir; end
        endcase
    end

    // ── 위치 카운터 ───────────────────────────────────────
    reg [31:0] pos_next;
    always @(*) begin
        if (trans_valid && en && (curr_ab !== prev_ab)) begin
            if (trans_dir == 1'b0)
                pos_next = position + 1;
            else
                pos_next = position - 1;
        end else begin
            pos_next = position;
        end
    end

    // 오버플로 감지: 부호 비트가 바뀌면서 절대값이 작아지는 경우
    wire ovfl_now = (pos_next[31] !== position[31]) &&
                    (pos_next[30:0] < position[30:0]);

    // ── 창 속도 카운터 ────────────────────────────────────
    reg [26:0] win_timer;    // log2(1_000_000) ≈ 20비트 → 27비트 여유
    reg [15:0] win_acc;      // 현재 창 누산
    reg        win_tick;     // 창 완료 플래그

    always @(posedge clk) begin
        if (rst || srst) begin
            win_timer <= 0;
            win_acc   <= 0;
            win_tick  <= 0;
            velocity  <= 0;
        end else if (!window_en) begin
            win_timer <= 0;
            win_acc   <= 0;
            win_tick  <= 0;
        end else begin
            win_tick <= 0;
            if (win_timer >= WINDOW_CYCLES - 1) begin
                win_timer <= 0;
                win_tick  <= 1;
            end else begin
                win_timer <= win_timer + 1;
            end
            // 창 내 카운트 누산 (정방향·역방향 모두 절댓값으로)
            if (trans_valid && en && (curr_ab !== prev_ab))
                win_acc <= win_acc + 1;
            // 창 완료 시 velocity 갱신, 누산 초기화
            // win_tick=1일 때 velocity는 현재 win_acc를 캡처하고, 다음 클럭에 win_acc를 초기화
            if (win_tick) begin
                velocity <= win_acc;
                win_acc  <= 0;
            end
        end
    end

    // ── 메인 레지스터 클록 ────────────────────────────────
    always @(posedge clk) begin
        if (rst || srst) begin
            position  <= 0;
            dir       <= 0;
            ovfl      <= 0;
            err_latch <= 0;
            error_cnt <= 0;
            prev_ab   <= 0;
        end else begin
            prev_ab <= curr_ab;

            // 위치 갱신
            if (en && trans_valid && (curr_ab !== prev_ab)) begin
                position <= pos_next;
                dir      <= trans_dir;
                if (ovfl_now)
                    ovfl <= 1'b1;
            end

            // 오류 카운터
            if (en && !trans_valid && (curr_ab !== prev_ab)) begin
                err_latch <= 1'b1;
                if (error_cnt != 16'hFFFF)
                    error_cnt <= error_cnt + 1;
            end
        end
    end

endmodule

`default_nettype wire

// axil_qenc_wrap.v — AXI4-Lite 슬레이브 래퍼 (qenc_diag 인스턴스)
// I-016 엔코더 난이도 사다리 (mmio-qenc-diag-checker, T2)
// Verilog-2001. iverilog/verilator/yosys 호환.
//
// 레지스터 맵 (AXI araddr[7:0] 기준):
//   0x00  CTRL      [RW] bit0=EN, bit1=SRST, bit2=WINDOW_EN
//   0x04  POSITION  [RO] 32비트 부호 있는 위치
//   0x08  VELOCITY  [RO] 16비트 최근 10ms 창 속도 (상위 16비트 0)
//   0x0C  STATUS    [RO] bit0=DIR, bit1=OVFL, bit2=ERR_LATCH
//   0x10  ERROR_CNT [RO] 16비트 오류 누산 (상위 16비트 0)

`default_nettype none

module axil_qenc_wrap #(
    parameter WINDOW_CYCLES = 1_000_000  // 10 ms @ 100 MHz
) (
    // AXI 클럭/리셋
    input  wire        aclk,
    input  wire        aresetn,  // active-low

    // AXI4-Lite Write Channel
    input  wire [7:0]  s_awaddr,
    input  wire        s_awvalid,
    output reg         s_awready,
    input  wire [31:0] s_wdata,
    input  wire [3:0]  s_wstrb,
    input  wire        s_wvalid,
    output reg         s_wready,
    output reg  [1:0]  s_bresp,
    output reg         s_bvalid,
    input  wire        s_bready,

    // AXI4-Lite Read Channel
    input  wire [7:0]  s_araddr,
    input  wire        s_arvalid,
    output reg         s_arready,
    output reg  [31:0] s_rdata,
    output reg  [1:0]  s_rresp,
    output reg         s_rvalid,
    input  wire        s_rready,

    // 엔코더 외부 핀 (메타스테이블 방지는 qenc_diag 내부)
    input  wire        enc_a,
    input  wire        enc_b
);

    // ── 내부 리셋 (active-high) ───────────────────────────
    wire rst = !aresetn;

    // ── CTRL 레지스터 ────────────────────────────────────
    reg        ctrl_en;
    reg        ctrl_srst;
    reg        ctrl_window_en;

    // ── qenc_diag 출력 ───────────────────────────────────
    wire [31:0] position;
    wire [15:0] velocity;
    wire        dir;
    wire        ovfl;
    wire        err_latch;
    wire [15:0] error_cnt;

    // ── qenc_diag 인스턴스 ───────────────────────────────
    qenc_diag #(
        .WINDOW_CYCLES(WINDOW_CYCLES)
    ) u_core (
        .clk       (aclk),
        .rst       (rst),
        .en        (ctrl_en),
        .srst      (ctrl_srst),
        .window_en (ctrl_window_en),
        .a_in      (enc_a),
        .b_in      (enc_b),
        .position  (position),
        .velocity  (velocity),
        .dir       (dir),
        .ovfl      (ovfl),
        .err_latch (err_latch),
        .error_cnt (error_cnt)
    );

    // ── Write FSM ─────────────────────────────────────────
    // 한 사이클에 awready+wready 동시 assert, 다음 사이클에 bvalid
    reg aw_pend;
    reg [7:0] aw_addr_r;
    reg w_pend;
    reg [31:0] w_data_r;

    always @(posedge aclk) begin
        if (rst) begin
            s_awready  <= 1'b0;
            s_wready   <= 1'b0;
            s_bvalid   <= 1'b0;
            s_bresp    <= 2'b00;
            aw_pend    <= 1'b0;
            w_pend     <= 1'b0;
            ctrl_en        <= 1'b0;
            ctrl_srst      <= 1'b0;
            ctrl_window_en <= 1'b0;
        end else begin
            // SRST는 1사이클 펄스
            ctrl_srst <= 1'b0;

            // awready 핸드셰이크
            if (s_awvalid && !aw_pend) begin
                s_awready <= 1'b1;
                aw_pend   <= 1'b1;
                aw_addr_r <= s_awaddr;
            end else begin
                s_awready <= 1'b0;
            end

            // wready 핸드셰이크
            if (s_wvalid && !w_pend) begin
                s_wready <= 1'b1;
                w_pend   <= 1'b1;
                w_data_r <= s_wdata;
            end else begin
                s_wready <= 1'b0;
            end

            // 주소와 데이터 모두 수신 → 레지스터 쓰기 + bresp
            if (aw_pend && w_pend && !s_bvalid) begin
                // CTRL(0x00) 만 쓰기 가능, 나머지는 무시(OKAY 반환)
                if (aw_addr_r[7:2] == 6'h00) begin
                    ctrl_en        <= w_data_r[0];
                    ctrl_srst      <= w_data_r[1];
                    ctrl_window_en <= w_data_r[2];
                end
                s_bvalid <= 1'b1;
                s_bresp  <= 2'b00;  // OKAY
                aw_pend  <= 1'b0;
                w_pend   <= 1'b0;
            end

            if (s_bvalid && s_bready)
                s_bvalid <= 1'b0;
        end
    end

    // ── Read FSM ──────────────────────────────────────────
    reg [7:0] ar_addr_r;

    always @(posedge aclk) begin
        if (rst) begin
            s_arready <= 1'b0;
            s_rvalid  <= 1'b0;
            s_rdata   <= 32'h0;
            s_rresp   <= 2'b00;
        end else begin
            if (s_arvalid && !s_rvalid) begin
                s_arready <= 1'b1;
                ar_addr_r <= s_araddr;
            end else begin
                s_arready <= 1'b0;
            end

            if (s_arready && s_arvalid) begin
                s_rvalid <= 1'b1;
                s_rresp  <= 2'b00;
                case (ar_addr_r[7:2])
                    6'h00: s_rdata <= {29'b0, ctrl_window_en, ctrl_srst, ctrl_en};
                    6'h01: s_rdata <= position;
                    6'h02: s_rdata <= {16'b0, velocity};
                    6'h03: s_rdata <= {29'b0, err_latch, ovfl, dir};
                    6'h04: s_rdata <= {16'b0, error_cnt};
                    default: s_rdata <= 32'hDEAD_BEEF;
                endcase
            end

            if (s_rvalid && s_rready)
                s_rvalid <= 1'b0;
        end
    end

endmodule

// tb_axil_qenc_wrap.v — 자체 점검형 테스트벤치
// I-016 엔코더 난이도 사다리 (mmio-qenc-diag-checker, T2)
// Verilog-2001. iverilog 호환.
//
// 통과 → "PASS" 로 시작하는 줄, 실패 → "FAIL" 로 시작하는 줄 출력.
// 비교는 !== 로 X/Z도 실패로 센다.
//
// 테스트 케이스:
//   TC1: 정방향 4스텝 → POSITION == 4, DIR == 0
//   TC2: 역방향 4스텝 → POSITION == 0, DIR == 1
//   TC3: 불가능 전이(글리치 주입 00→11) → ERROR_CNT >= 1, ERR_LATCH == 1
//   TC4: 소프트 리셋 → POSITION == 0, ERROR_CNT == 0
//   TC5: 창 속도 (WINDOW_EN=1, 2펄스 → VELOCITY >= 1)
//        테스트용 WINDOW_CYCLES=100으로 설정
//
// 설계 메모:
//   enc_fwd_step / enc_rev_step = 1전이(1카운트). Gray 위상을 enc_state로 추적.
//   4번 호출 = 1 기계 사이클 (00→10→11→01→00).
//   2단 FF 전파를 위해 상태마다 클럭 4개를 대기.
//
//   TC5 타이밍 (WINDOW_CYCLES=100):
//     1) WINDOW_EN 쓰기 후 105클럭 대기 → 첫 창 완료, 새 창 시작 보장
//     2) enc_fwd_step × 2 (8클럭, << 100클럭 창)
//     3) 105클럭 대기 → 해당 창 완료 → VELOCITY = 2

`timescale 1ns/1ps
`default_nettype none

module tb_axil_qenc_wrap;

    // ── 클럭 ──────────────────────────────────────────────
    reg aclk;
    initial aclk = 0;
    always #5 aclk = ~aclk;  // 100 MHz (10 ns 주기)

    // ── DUT 포트 ──────────────────────────────────────────
    reg        aresetn;
    reg  [7:0] s_awaddr;
    reg        s_awvalid;
    wire       s_awready;
    reg  [31:0] s_wdata;
    reg  [3:0]  s_wstrb;
    reg         s_wvalid;
    wire        s_wready;
    wire [1:0]  s_bresp;
    wire        s_bvalid;
    reg         s_bready;
    reg  [7:0]  s_araddr;
    reg         s_arvalid;
    wire        s_arready;
    wire [31:0] s_rdata;
    wire [1:0]  s_rresp;
    wire        s_rvalid;
    reg         s_rready;
    reg         enc_a;
    reg         enc_b;

    // ── 엔코더 위상 추적 (enc_fwd_step / enc_rev_step 공유) ─
    // {enc_a, enc_b}의 현재 Gray 위상
    reg [1:0] enc_state;

    // ── DUT 인스턴스 (WINDOW_CYCLES=100 짧게) ────────────
    axil_qenc_wrap #(
        .WINDOW_CYCLES(100)
    ) dut (
        .aclk      (aclk),
        .aresetn   (aresetn),
        .s_awaddr  (s_awaddr),
        .s_awvalid (s_awvalid),
        .s_awready (s_awready),
        .s_wdata   (s_wdata),
        .s_wstrb   (s_wstrb),
        .s_wvalid  (s_wvalid),
        .s_wready  (s_wready),
        .s_bresp   (s_bresp),
        .s_bvalid  (s_bvalid),
        .s_bready  (s_bready),
        .s_araddr  (s_araddr),
        .s_arvalid (s_arvalid),
        .s_arready (s_arready),
        .s_rdata   (s_rdata),
        .s_rresp   (s_rresp),
        .s_rvalid  (s_rvalid),
        .s_rready  (s_rready),
        .enc_a     (enc_a),
        .enc_b     (enc_b)
    );

    // ── 실패 카운터 ───────────────────────────────────────
    integer fail_cnt;

    // ── AXI4-Lite 헬퍼 태스크 ────────────────────────────
    task axil_write;
        input [7:0]  addr;
        input [31:0] data;
        begin
            @(posedge aclk); #1;
            s_awaddr = addr; s_awvalid = 1;
            s_wdata  = data; s_wvalid  = 1; s_wstrb = 4'hF;
            // awready 대기
            wait(s_awready); @(posedge aclk); #1;
            s_awvalid = 0;
            // wready 대기
            wait(s_wready);  @(posedge aclk); #1;
            s_wvalid  = 0;
            // bresp 대기
            s_bready = 1;
            wait(s_bvalid);  @(posedge aclk); #1;
            s_bready = 0;
        end
    endtask

    task axil_read;
        input  [7:0]  addr;
        output [31:0] data;
        begin
            @(posedge aclk); #1;
            s_araddr = addr; s_arvalid = 1;
            wait(s_arready); @(posedge aclk); #1;
            s_arvalid = 0;
            s_rready  = 1;
            wait(s_rvalid);
            data = s_rdata;
            @(posedge aclk); #1;
            s_rready = 0;
        end
    endtask

    // ── 엔코더 정방향 1전이 ──────────────────────────────
    // Gray 정방향: 00→10→11→01→00. 한 번 호출 = 카운트 +1.
    // 4번 호출하면 1 기계 사이클 완성 (enc_state가 00으로 복귀).
    // 2단 FF 전파를 위해 클럭 4개 대기.
    task enc_fwd_step;
        begin
            case (enc_state)
                2'b00: enc_state = 2'b10;
                2'b10: enc_state = 2'b11;
                2'b11: enc_state = 2'b01;
                2'b01: enc_state = 2'b00;
            endcase
            enc_a = enc_state[1];
            enc_b = enc_state[0];
            repeat(4) @(posedge aclk);
        end
    endtask

    // ── 엔코더 역방향 1전이 ──────────────────────────────
    // Gray 역방향: 00→01→11→10→00. 한 번 호출 = 카운트 -1.
    task enc_rev_step;
        begin
            case (enc_state)
                2'b00: enc_state = 2'b01;
                2'b01: enc_state = 2'b11;
                2'b11: enc_state = 2'b10;
                2'b10: enc_state = 2'b00;
            endcase
            enc_a = enc_state[1];
            enc_b = enc_state[0];
            repeat(4) @(posedge aclk);
        end
    endtask

    // ── 체크 헬퍼 ────────────────────────────────────────
    task check_eq;
        input [255:0] name;
        input [31:0]  got;
        input [31:0]  exp;
        begin
            if (got !== exp) begin
                $display("FAIL %0s: got=%0h exp=%0h", name, got, exp);
                fail_cnt = fail_cnt + 1;
            end else begin
                $display("pass %0s: %0h", name, got);
            end
        end
    endtask

    task check_ge;
        input [255:0] name;
        input [31:0]  got;
        input [31:0]  threshold;
        begin
            if (got < threshold) begin
                if (got < threshold) begin
                    $display("FAIL %0s: got=%0h < threshold=%0h", name, got, threshold);
                    fail_cnt = fail_cnt + 1;
                end else begin
                    $display("pass %0s: got=%0h >= %0h", name, got, threshold);
                end
            end else begin
                $display("pass %0s: got=%0h >= %0h", name, got, threshold);
            end
        end
    endtask

    // ── 메인 ─────────────────────────────────────────────
    reg [31:0] rd;
    integer    i;

    initial begin
        // 초기화
        fail_cnt  = 0;
        enc_state = 2'b00;
        aresetn   = 0;
        s_awvalid = 0; s_wvalid = 0; s_bready = 0;
        s_arvalid = 0; s_rready = 0;
        s_awaddr  = 0; s_wdata = 0; s_wstrb = 0;
        s_araddr  = 0;
        enc_a = 0; enc_b = 0;

        repeat(5) @(posedge aclk);
        aresetn = 1;
        repeat(2) @(posedge aclk);

        // ── CTRL: EN=1, WINDOW_EN=0 먼저 ────────────────
        axil_write(8'h00, 32'h1);  // EN=1

        // ────────────────────────────────────────────────
        // TC1: 정방향 4스텝 → POSITION == 4, DIR == 0
        // 각 enc_fwd_step = 1전이 = 카운트 +1
        // 4회 호출: 00→10→11→01→00, POSITION = 4
        // ────────────────────────────────────────────────
        $display("=== TC1: 정방향 4스텝 ===");
        for (i = 0; i < 4; i = i + 1)
            enc_fwd_step;
        repeat(5) @(posedge aclk);

        axil_read(8'h04, rd); check_eq("TC1:POSITION", rd, 32'd4);
        axil_read(8'h0C, rd); check_eq("TC1:DIR",      rd & 32'h1, 32'h0);

        // ────────────────────────────────────────────────
        // TC2: 역방향 4스텝 → POSITION == 0, DIR == 1
        // TC1 종료 후 enc_state=00. 역방향 4전이로 복귀.
        // ────────────────────────────────────────────────
        $display("=== TC2: 역방향 4스텝 ===");
        for (i = 0; i < 4; i = i + 1)
            enc_rev_step;
        repeat(5) @(posedge aclk);

        axil_read(8'h04, rd); check_eq("TC2:POSITION", rd, 32'd0);
        axil_read(8'h0C, rd); check_eq("TC2:DIR",      rd & 32'h1, 32'h1);

        // ────────────────────────────────────────────────
        // TC3: 글리치 주입 (불가능 전이 00→11)
        // TC2 종료 후 enc_state=00. 직접 신호 조작.
        // ────────────────────────────────────────────────
        $display("=== TC3: 글리치 주입 ===");
        enc_a = 0; enc_b = 0; repeat(4) @(posedge aclk);
        enc_a = 1; enc_b = 1; repeat(4) @(posedge aclk);  // 00→11: 불가능
        enc_a = 0; enc_b = 0; repeat(4) @(posedge aclk);  // 11→00: 불가능
        repeat(5) @(posedge aclk);

        axil_read(8'h10, rd); check_ge("TC3:ERROR_CNT",  rd, 32'd1);
        axil_read(8'h0C, rd); check_eq("TC3:ERR_LATCH",  (rd >> 2) & 1, 32'h1);

        // ────────────────────────────────────────────────
        // TC4: 소프트 리셋 → POSITION==0, ERROR_CNT==0
        // ────────────────────────────────────────────────
        $display("=== TC4: 소프트 리셋 ===");
        // CTRL: EN=1, SRST=1
        axil_write(8'h00, 32'h3);
        repeat(3) @(posedge aclk);
        // SRST 해제
        axil_write(8'h00, 32'h1);
        repeat(3) @(posedge aclk);

        axil_read(8'h04, rd); check_eq("TC4:POSITION",  rd, 32'd0);
        axil_read(8'h10, rd); check_eq("TC4:ERROR_CNT", rd, 32'd0);

        // ────────────────────────────────────────────────
        // TC5: 창 속도 (WINDOW_EN=1, 2펄스 발생, 창 완료 후 VELOCITY>=1)
        //
        // WINDOW_CYCLES=100 타이밍:
        //   1) WINDOW_EN 쓰기 후 105클럭 대기 → 첫 창 완료, 새 창 시작 보장
        //   2) enc_fwd_step × 2 (8클럭, << 100클럭 창)
        //   3) 105클럭 대기 → 해당 창 완료 → VELOCITY = 2 (안전 여유 92클럭)
        // ────────────────────────────────────────────────
        $display("=== TC5: 창 속도 ===");
        axil_write(8'h00, 32'h5);  // EN=1, WINDOW_EN=1
        // 창 경계 동기화: 첫 창이 완료되고 새 창이 시작될 때까지 기다린다
        repeat(105) @(posedge aclk);
        // 새 창 안에 2펄스를 넣는다 (8클럭 << 100클럭 창)
        enc_fwd_step;
        enc_fwd_step;
        // 이 창이 완료될 때까지 기다린다
        repeat(105) @(posedge aclk);
        axil_read(8'h08, rd); check_ge("TC5:VELOCITY", rd & 32'hFFFF, 32'd1);

        // ────────────────────────────────────────────────
        // 최종 판정
        // ────────────────────────────────────────────────
        if (fail_cnt === 0)
            $display("PASS all %0d checks", 7);
        else
            $display("FAIL %0d check(s) failed", fail_cnt);

        $finish;
    end

    // ── 타임아웃 ─────────────────────────────────────────
    initial begin
        #500000;
        $display("FAIL timeout");
        $finish;
    end

endmodule
