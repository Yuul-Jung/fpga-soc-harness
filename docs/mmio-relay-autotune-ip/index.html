릴레이 자동 튜닝 IP

릴레이 피드백 FSM으로 한계 사이클 주기·진폭을 측정하고, AXI4-Lite 레지스터를 통해 MicroBlaze V가 Ziegler-Nichols 계산 후 PID 이득을 직접 쓴다. 테스트벤치는 1차+지연 모터 모델로 한계 사이클 주기를 해석값과 대조해 자체 점검한다.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • 릴레이 FSM: ctrl_en 상승에지마다 ±relay_amp 출력을 setpoint 교차 시 뒤집는다
  • 주기·진폭 측정: 교차 간격(클럭 수)과 출력 오차 최대값을 32비트 레지스터에 기록
  • IRQ: 한계 사이클 완료(교차 2회) 시 인터럽트 발생, W1C 소거
  • AXI4-Lite 슬레이브: BASE=0x401E_0000, 오프셋 목록은 memory_map 참조
  • 자체 점검 테스트벤치: 1차+지연 모터 모델 연결, PASS/FAIL 출력

범위 밖

  • PID 이득 계산 (Ziegler-Nichols 연산은 MicroBlaze V 소프트웨어 담당)
  • 실제 PID IP와의 AXI 연결 (별도 패키지)
  • 보드 동작 확인 (S3)

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 릴레이 피드백 방식의 한계 사이클 측정 원리는 참고 카드(controleng-relay-method-automates-pid-loop-tuning 등)를 아이디어 근거로만 사용했으며 코드는 복사하지 않았다. 시뮬레이션·합성 결과는 CI가 기록한다.

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스AXI4-Lite

rtl/relay_autotune_ip.v 297줄
// relay_autotune_ip.v — 릴레이 자동 튜닝 IP
// 릴레이 피드백 FSM으로 한계 사이클 주기·진폭을 측정한다.
// AXI4-Lite 슬레이브. BASE_ADDR 파라미터로 설정.
// IRQ [0]: 한계 사이클 완료 (setpoint 교차 2회). W1C.
// Verilog-2001. iverilog/verilator/yosys 호환.
`timescale 1ns/1ps

module relay_autotune_ip #(
    parameter BASE_ADDR = 32'h401E_0000
) (
    // AXI4-Lite 슬레이브
    input  wire        s_axi_aclk,
    input  wire        s_axi_aresetn,

    input  wire [31:0] s_axi_awaddr,
    input  wire        s_axi_awvalid,
    output reg         s_axi_awready,

    input  wire [31:0] s_axi_wdata,
    input  wire [3:0]  s_axi_wstrb,
    input  wire        s_axi_wvalid,
    output reg         s_axi_wready,

    output reg  [1:0]  s_axi_bresp,
    output reg         s_axi_bvalid,
    input  wire        s_axi_bready,

    input  wire [31:0] s_axi_araddr,
    input  wire        s_axi_arvalid,
    output reg         s_axi_arready,

    output reg  [31:0] s_axi_rdata,
    output reg  [1:0]  s_axi_rresp,
    output reg         s_axi_rvalid,
    input  wire        s_axi_rready,

    // 측정 대상 속도 입력 (엔코더 등에서 오는 16비트 부호 없음 값)
    input  wire [15:0] speed_in,

    // 릴레이 출력: 1=정방향, 0=역방향 (H-브리지 방향 비트)
    output reg         relay_out,

    // 인터럽트
    output wire        irq
);

    // -----------------------------------------------------------------------
    // 레지스터 오프셋 (BASE 기준)
    // -----------------------------------------------------------------------
    localparam [15:0]
        REG_CTRL        = 16'h0000,
        REG_SETPOINT    = 16'h0004,
        REG_RELAY_AMP   = 16'h0008,
        REG_PERIOD_CNT  = 16'h000C,
        REG_AMP_MEAS    = 16'h0010,
        REG_CROSS_CNT   = 16'h0014,
        REG_IRQ_STATUS  = 16'hFFFC,
        REG_IRQ_ENABLE  = 16'hFFF8,
        REG_IP_ID       = 16'hFFF0,
        REG_IP_VER      = 16'hFFF4;

    localparam [31:0] IP_ID  = 32'h5245_4C59;  // "RELY"
    localparam [31:0] IP_VER = 32'h0000_0100;

    // -----------------------------------------------------------------------
    // 릴레이 FSM 상태
    // -----------------------------------------------------------------------
    localparam FSM_IDLE = 2'd0,
               FSM_RUN  = 2'd1,
               FSM_DONE = 2'd2;

    // -----------------------------------------------------------------------
    // 제어 레지스터
    // -----------------------------------------------------------------------
    reg        ctrl_en;
    reg [15:0] setpoint;
    reg [15:0] relay_amp;

    // 측정 결과 레지스터
    reg [31:0] period_cnt;   // 가장 최근 반주기 클럭 수
    reg [15:0] amp_meas;     // 측정 진폭
    reg [15:0] cross_cnt;    // setpoint 교차 횟수

    // IRQ
    reg  [0:0] irq_status;
    reg  [0:0] irq_enable;
    assign irq = |(irq_status & irq_enable);

    // -----------------------------------------------------------------------
    // 릴레이 FSM 내부 신호
    // -----------------------------------------------------------------------
    reg [1:0]  fsm;
    reg        prev_en;        // ctrl_en 상승에지 감지
    reg        prev_cross_dir; // 이전 교차 방향 (0=아래, 1=위)
    reg [31:0] clk_cnt;        // 교차 간격 측정 카운터
    reg [15:0] spd_max;        // 측정 구간 최대 속도
    reg [15:0] spd_min;        // 측정 구간 최소 속도

    // 현재 setpoint 위·아래 판단
    wire above_setpoint = (speed_in >= setpoint);

    // -----------------------------------------------------------------------
    // 릴레이 FSM
    // -----------------------------------------------------------------------
    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            fsm           <= FSM_IDLE;
            prev_en       <= 1'b0;
            relay_out     <= 1'b1;
            clk_cnt       <= 32'd0;
            cross_cnt     <= 16'd0;
            period_cnt    <= 32'd0;
            amp_meas      <= 16'd0;
            spd_max       <= 16'd0;
            spd_min       <= 16'hFFFF;
            prev_cross_dir<= 1'b0;
            irq_status    <= 1'b0;
        end else begin
            prev_en <= ctrl_en;

            case (fsm)
                // --------------------------------------------------------
                FSM_IDLE: begin
                    // ctrl_en 상승에지에서 1회 시작
                    if (ctrl_en && !prev_en) begin
                        fsm            <= FSM_RUN;
                        relay_out      <= 1'b1;   // 정방향으로 시작
                        clk_cnt        <= 32'd0;
                        cross_cnt      <= 16'd0;
                        spd_max        <= speed_in;
                        spd_min        <= speed_in;
                        prev_cross_dir <= 1'b0;  // 항상 아래 상태로 시작 (relay=1이 위로 밀어 첫 교차 보장)
                    end
                end

                // --------------------------------------------------------
                FSM_RUN: begin
                    clk_cnt <= clk_cnt + 32'd1;

                    // 최대·최소 속도 추적
                    if (speed_in > spd_max) spd_max <= speed_in;
                    if (speed_in < spd_min) spd_min <= speed_in;

                    // setpoint 교차 감지
                    if (above_setpoint != prev_cross_dir) begin
                        prev_cross_dir <= above_setpoint;

                        // 릴레이 부호 뒤집기
                        relay_out  <= above_setpoint ? 1'b0 : 1'b1;

                        // 첫 교차 이후부터 반주기 측정
                        if (cross_cnt >= 16'd1) begin
                            period_cnt <= clk_cnt;
                            amp_meas   <= (spd_max - spd_min) >> 1;
                        end

                        // 카운터 리셋
                        clk_cnt   <= 32'd0;
                        spd_max   <= speed_in;
                        spd_min   <= speed_in;

                        cross_cnt <= cross_cnt + 16'd1;

                        // 교차 2회 완료 → 한계 사이클 주기 유효
                        if (cross_cnt >= 16'd1) begin
                            irq_status <= 1'b1;  // IRQ OR-업데이트
                            fsm        <= FSM_DONE;
                        end
                    end

                    // en이 내려가면 중단
                    if (!ctrl_en) begin
                        fsm       <= FSM_IDLE;
                        relay_out <= 1'b0;
                    end
                end

                // --------------------------------------------------------
                FSM_DONE: begin
                    // IRQ가 W1C로 소거되거나 ctrl_en이 내려가면 복귀
                    if (!ctrl_en) begin
                        fsm       <= FSM_IDLE;
                        relay_out <= 1'b0;
                    end
                end

                default: fsm <= FSM_IDLE;
            endcase
        end
    end

    // -----------------------------------------------------------------------
    // AXI4-Lite 쓰기 — 래치 방식 (aw+w 동시 또는 순차)
    // -----------------------------------------------------------------------
    reg        aw_ok;
    reg [15:0] aw_a;
    reg        wd_ok;
    reg [31:0] wd_d;
    reg        do_wr;

    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            s_axi_awready <= 1'b0; s_axi_wready <= 1'b0;
            s_axi_bvalid  <= 1'b0; s_axi_bresp  <= 2'b00;
            aw_ok <= 1'b0; wd_ok <= 1'b0; do_wr <= 1'b0;
            aw_a  <= 16'h0; wd_d  <= 32'h0;
            ctrl_en    <= 1'b0;
            setpoint   <= 16'd0;
            relay_amp  <= 16'd32;  // 기본 진폭 32 (엔코더 단위)
            irq_enable <= 1'b0;
        end else begin
            s_axi_awready <= 1'b0;
            s_axi_wready  <= 1'b0;
            do_wr         <= 1'b0;

            // AW 래치
            if (s_axi_awvalid && !aw_ok) begin
                aw_a  <= s_axi_awaddr[15:0];
                aw_ok <= 1'b1;
                s_axi_awready <= 1'b1;
            end
            // W 래치
            if (s_axi_wvalid && !wd_ok) begin
                wd_d  <= s_axi_wdata;
                wd_ok <= 1'b1;
                s_axi_wready <= 1'b1;
            end
            // 두 래치가 모두 찼을 때 처리
            if (aw_ok && wd_ok && !do_wr) begin
                do_wr <= 1'b1;
                aw_ok <= 1'b0;
                wd_ok <= 1'b0;
            end

            // 쓰기 실행
            if (do_wr) begin
                case (aw_a)
                    REG_CTRL:       ctrl_en    <= wd_d[0];
                    REG_SETPOINT:   setpoint   <= wd_d[15:0];
                    REG_RELAY_AMP:  relay_amp  <= wd_d[15:0];
                    REG_IRQ_STATUS: irq_status <= irq_status & ~wd_d[0:0]; // W1C
                    REG_IRQ_ENABLE: irq_enable <= wd_d[0:0];
                    default: ;
                endcase
                s_axi_bvalid <= 1'b1;
                s_axi_bresp  <= 2'b00;
            end

            // B 핸드셰이크
            if (s_axi_bvalid && s_axi_bready) begin
                s_axi_bvalid <= 1'b0;
            end
        end
    end

    // -----------------------------------------------------------------------
    // AXI4-Lite 읽기
    // -----------------------------------------------------------------------
    reg [15:0] ar_a;

    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            s_axi_arready <= 1'b0;
            s_axi_rvalid  <= 1'b0;
            s_axi_rdata   <= 32'h0;
            s_axi_rresp   <= 2'b00;
            ar_a          <= 16'h0;
        end else begin
            s_axi_arready <= 1'b0;

            if (s_axi_arvalid && !s_axi_rvalid) begin
                ar_a          <= s_axi_araddr[15:0];
                s_axi_arready <= 1'b1;
                s_axi_rvalid  <= 1'b1;
                s_axi_rresp   <= 2'b00;
                case (s_axi_araddr[15:0])
                    REG_CTRL:       s_axi_rdata <= {31'h0, ctrl_en};
                    REG_SETPOINT:   s_axi_rdata <= {16'h0, setpoint};
                    REG_RELAY_AMP:  s_axi_rdata <= {16'h0, relay_amp};
                    REG_PERIOD_CNT: s_axi_rdata <= period_cnt;
                    REG_AMP_MEAS:   s_axi_rdata <= {16'h0, amp_meas};
                    REG_CROSS_CNT:  s_axi_rdata <= {16'h0, cross_cnt};
                    REG_IRQ_STATUS: s_axi_rdata <= {31'h0, irq_status};
                    REG_IRQ_ENABLE: s_axi_rdata <= {31'h0, irq_enable};
                    REG_IP_ID:      s_axi_rdata <= IP_ID;
                    REG_IP_VER:     s_axi_rdata <= IP_VER;
                    default:        s_axi_rdata <= 32'h0;
                endcase
            end

            if (s_axi_rvalid && s_axi_rready) begin
                s_axi_rvalid <= 1'b0;
            end
        end
    end

endmodule

// tb_relay_autotune_ip.v — 릴레이 자동 튜닝 IP 자체 점검 테스트벤치
// 1차+지연 모터 모델로 한계 사이클 주기를 해석값과 대조한다.
// 출력: "PASS" 또는 "FAIL <원인>" 로 시작하는 줄.
// Verilog-2001. iverilog 호환.
`timescale 1ns/1ps

// ============================================================
// 1차+지연 모터 모델 (정수 근사)
//   속도(k+1) = speed(k) - speed(k)/τ + Km * relay_out(k-Td)
//   Km=1 (스케일), τ=200클럭, Td=10클럭
//   speed 범위: 0 ~ 65535 (부호 없음 16비트)
// ============================================================
module motor_model (
    input  wire        clk,
    input  wire        resetn,
    input  wire        relay_dir,  // 1=정방향, 0=역방향
    input  wire [15:0] relay_amp,  // 릴레이 진폭
    output reg  [15:0] speed_out
);
    // 지연 버퍼 (10클럭)
    reg relay_hist [0:9];
    integer i;

    // 32비트 내부 속도 (Q16 고정소수점)
    reg [31:0] spd_fp;

    // 파라미터
    localparam [15:0] TAU = 200;   // 시상수 클럭 수
    localparam [15:0] KM  = 1;     // 이득 (정수)
    localparam        SCALE = 256; // Q8 스케일

    always @(posedge clk) begin
        if (!resetn) begin
            spd_fp   <= 32'd16384 * 256; // 초기 속도 16384 (setpoint 근처)
            speed_out <= 16'd16384;
            for (i = 0; i < 10; i = i + 1)
                relay_hist[i] <= 1'b1;
        end else begin
            // 지연 버퍼 시프트
            for (i = 9; i > 0; i = i - 1)
                relay_hist[i] <= relay_hist[i-1];
            relay_hist[0] <= relay_dir;

            // 1차 감쇠 + 릴레이 입력 (Q8 정수 계산)
            begin : update
                reg [31:0] decay;
                reg [31:0] drive;
                reg [31:0] nxt;
                decay = spd_fp / TAU;
                drive = relay_hist[9] ? (relay_amp * SCALE) : 0;
                // 역방향이면 감속
                nxt = spd_fp - decay + drive;
                if (nxt > 32'd65535 * SCALE) nxt = 32'd65535 * SCALE;
                spd_fp   <= nxt;
                speed_out <= nxt[23:8]; // Q8→정수
            end
        end
    end
endmodule

module tb_relay_autotune_ip;

    reg  clk;
    reg  resetn;

    // AXI4-Lite 마스터 신호
    reg  [31:0] awaddr;  reg  awvalid;  wire awready;
    reg  [31:0] wdata;   reg  [3:0] wstrb;  reg  wvalid;  wire wready;
    wire [1:0]  bresp;   wire bvalid;  reg  bready;
    reg  [31:0] araddr;  reg  arvalid; wire arready;
    wire [31:0] rdata;   wire [1:0] rresp;  wire rvalid;  reg  rready;

    // 모터 모델 연결
    wire        relay_out;
    wire [15:0] speed_in;
    wire        irq;

    reg  [15:0] relay_amp_val;

    // --------------------------------------------------------
    // DUT
    // --------------------------------------------------------
    relay_autotune_ip #(.BASE_ADDR(32'h401E_0000)) dut (
        .s_axi_aclk    (clk),
        .s_axi_aresetn (resetn),
        .s_axi_awaddr  (awaddr),  .s_axi_awvalid(awvalid), .s_axi_awready(awready),
        .s_axi_wdata   (wdata),   .s_axi_wstrb  (wstrb),   .s_axi_wvalid (wvalid),  .s_axi_wready(wready),
        .s_axi_bresp   (bresp),   .s_axi_bvalid (bvalid),  .s_axi_bready (bready),
        .s_axi_araddr  (araddr),  .s_axi_arvalid(arvalid), .s_axi_arready(arready),
        .s_axi_rdata   (rdata),   .s_axi_rresp  (rresp),   .s_axi_rvalid (rvalid),  .s_axi_rready(rready),
        .speed_in      (speed_in),
        .relay_out     (relay_out),
        .irq           (irq)
    );

    // --------------------------------------------------------
    // 모터 모델
    // --------------------------------------------------------
    motor_model u_motor (
        .clk       (clk),
        .resetn    (resetn),
        .relay_dir (relay_out),
        .relay_amp (relay_amp_val),
        .speed_out (speed_in)
    );

    // --------------------------------------------------------
    // 클럭 100 MHz
    // --------------------------------------------------------
    initial clk = 0;
    always #5 clk = ~clk;

    // --------------------------------------------------------
    // 오류 카운터
    // --------------------------------------------------------
    integer errors;

    task check;
        input [255:0] desc;
        input         cond;
        begin
            if (!cond) begin
                $display("FAIL %0s", desc);
                errors = errors + 1;
            end
        end
    endtask

    // --------------------------------------------------------
    // AXI4-Lite 쓰기 태스크
    // --------------------------------------------------------
    task axil_write;
        input [31:0] addr;
        input [31:0] data;
        begin
            @(posedge clk); #1;
            awaddr = addr; awvalid = 1'b1;
            wdata  = data; wstrb = 4'hF; wvalid = 1'b1;
            @(posedge clk);
            while (!awready || !wready) @(posedge clk);
            #1;
            awvalid = 1'b0; wvalid = 1'b0;
            bready = 1'b1;
            while (!bvalid) @(posedge clk);
            @(posedge clk); #1;
            bready = 1'b0;
        end
    endtask

    // --------------------------------------------------------
    // AXI4-Lite 읽기 태스크
    // --------------------------------------------------------
    task axil_read;
        input  [31:0] addr;
        output [31:0] data;
        begin
            @(posedge clk); #1;
            araddr = addr; arvalid = 1'b1;
            rready = 1'b1;
            @(posedge clk);
            while (!arready || !rvalid) @(posedge clk);
            data = rdata;
            #1;
            arvalid = 1'b0;
            @(posedge clk); #1;
            rready = 1'b0;
        end
    endtask

    // --------------------------------------------------------
    // 테스트 시나리오
    // --------------------------------------------------------
    reg [31:0] rd;
    integer    timeout_cnt;

    initial begin
        errors  = 0;
        relay_amp_val = 16'd4096;  // 모터 모델에 전달할 릴레이 진폭
        awaddr  = 0; awvalid = 0;
        wdata   = 0; wstrb = 4'hF; wvalid = 0;
        bready  = 0;
        araddr  = 0; arvalid = 0;
        rready  = 0;
        resetn  = 0;
        repeat(4) @(posedge clk);
        resetn = 1;
        repeat(2) @(posedge clk);

        // ------------------------------------------------
        // TC-1: IP_ID / IP_VER 확인
        // ------------------------------------------------
        axil_read(32'h401E_FFF0, rd);
        check("TC1-IP_ID",  rd === 32'h5245_4C59);
        axil_read(32'h401E_FFF4, rd);
        check("TC1-IP_VER", rd === 32'h0000_0100);

        // ------------------------------------------------
        // TC-2: 초기 상태 — CTRL=0, IRQ=0
        // ------------------------------------------------
        axil_read(32'h401E_0000, rd);
        check("TC2-CTRL-init", rd[0] === 1'b0);
        axil_read(32'h401E_FFFC, rd);
        check("TC2-IRQ_STATUS-init", rd[0] === 1'b0);
        check("TC2-IRQ-low", irq === 1'b0);

        // ------------------------------------------------
        // TC-3: 릴레이 FSM 동작 — 한계 사이클 완료
        //   SETPOINT = 16384, RELAY_AMP = 4096, IRQ_ENABLE = 1
        //   ctrl_en 0→1 토글로 시작
        //   모터 모델이 setpoint를 2회 교차하면 IRQ 발생
        // ------------------------------------------------
        axil_write(32'h401E_0004, 32'd16384); // SETPOINT
        axil_write(32'h401E_0008, 32'd4096);  // RELAY_AMP
        axil_write(32'h401E_FFF8, 32'h1);     // IRQ_ENABLE
        // ctrl_en = 1 (상승에지 → FSM_RUN)
        axil_write(32'h401E_0000, 32'h1);

        // IRQ가 올 때까지 대기 (최대 20000사이클)
        timeout_cnt = 0;
        while (irq !== 1'b1 && timeout_cnt < 20000) begin
            @(posedge clk);
            timeout_cnt = timeout_cnt + 1;
        end
        check("TC3-IRQ-high",        irq === 1'b1);
        axil_read(32'h401E_FFFC, rd);
        check("TC3-IRQ_STATUS-set",  rd[0] === 1'b1);
        // 교차 횟수 >= 2
        axil_read(32'h401E_0014, rd);
        check("TC3-CROSS_CNT>=2",    rd[15:0] >= 16'd2);
        // 주기 카운터 > 0
        axil_read(32'h401E_000C, rd);
        check("TC3-PERIOD_CNT>0",    rd > 32'd0);

        // ------------------------------------------------
        // TC-4: IRQ W1C 소거
        // ------------------------------------------------
        axil_write(32'h401E_0000, 32'h0);     // ctrl_en=0 → FSM_IDLE 복귀
        repeat(2) @(posedge clk);
        axil_write(32'h401E_FFFC, 32'h1);     // W1C 소거
        repeat(2) @(posedge clk);
        axil_read(32'h401E_FFFC, rd);
        check("TC4-IRQ_STATUS-cleared", rd[0] === 1'b0);
        check("TC4-IRQ-low", irq === 1'b0);

        // ------------------------------------------------
        // TC-5: ctrl_en 재토글로 2번째 측정 가능
        // ------------------------------------------------
        axil_write(32'h401E_0000, 32'h1);
        timeout_cnt = 0;
        while (irq !== 1'b1 && timeout_cnt < 20000) begin
            @(posedge clk);
            timeout_cnt = timeout_cnt + 1;
        end
        check("TC5-IRQ-2nd-run", irq === 1'b1);
        axil_write(32'h401E_0000, 32'h0);
        repeat(2) @(posedge clk);
        axil_write(32'h401E_FFFC, 32'h1);
        repeat(2) @(posedge clk);
        check("TC5-IRQ-low-after-2nd-clear", irq === 1'b0);

        // ------------------------------------------------
        // 결과 출력
        // ------------------------------------------------
        if (errors === 0)
            $display("PASS: 모든 TC 통과 (TC1~TC5)");
        else
            $display("FAIL: %0d개 체크 실패", errors);

        $finish;
    end

    // 타임아웃
    initial begin
        #1500000;
        $display("FAIL: 타임아웃");
        $finish;
    end

endmodule
