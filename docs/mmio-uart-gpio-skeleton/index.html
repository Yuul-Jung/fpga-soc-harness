MMIO 뼈대 SoC: GPIO + UART (Basys3)

CPU 슬롯이 비어 있는 최소 SoC. 단일 마스터 버스(mmio-lite)에 GPIO와 UART 송신기를 붙이고, 마스터 자리에는 데모 FSM을 둔다. 이후 이 슬롯을 PicoRV32 같은 코어로 교체하는 것이 목표다.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: 미실행

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-04. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • 스위치 값을 버스로 읽어 LED에 쓴다.
  • 버튼을 누르면 스위치 하위 8비트를 UART(115200 8N1)로 송신한다.
  • 마스터 포트를 분리해 CPU로 교체할 수 있게 한다.

범위 밖

  • CPU 구현 (다음 단계)
  • UART 수신, 인터럽트, DMA
  • 외부 메모리 (Basys3에는 DRAM이 없다)

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 시뮬레이션·합성 결과는 CI가 기록한다.

2. 계통도

클럭: 100 MHz 단일 클럭 (clk, W5), 동기 리셋 (btnC, 2단 동기화)

버스 규약: mmio-lite

신호방향설명

m_wemaster→slave1이면 쓰기

미매핑 주소 읽기는 0을 돌려주고 오류를 알리지 않는다. 대기 상태·오류 응답이 필요한 CPU로 바꿀 때 확장한다.

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획이 패키지는 CPU 없이 시뮬레이션하려고 자체 버스 mmio-lite(0x1000_xxxx)를 쓴 파일럿이다. 동결 맵(mbv-frozen-v2)으로 옮기는 것은 주소만 바꾸는 일이 아니다. (1) gpio·uart는 동결 맵에 이미 AMD IP가 있다([00] AXI GPIO, [02] UART Lite). 레지스터 배치가 달라(AXI GPIO는 DATA/TRI, UART Lite는 RX/TX FIFO·STAT·CTRL) 우리 모듈을 그대로 꽂을 수 없고, 이식 때는 AMD IP를 쓰고 드라이버를 맞춘다. (2) 자체 주변장치는 프로젝트의 IP 표준(64KB 창, 0xFFF0 ID·VER·IRQ_STATUS·IRQ_ENABLE)에 맞춰 새로 감싼다. (3) mmio-lite→AXI4-Lite 마스터 다리는 아직 없다(gap).

5. 외부 부품과 전기 회로도

2절의 계통도는 디지털 모듈 연결도다. 전압, 풀업, 커넥터 배선 같은 전기 사항은 이 절에서 따로 다룬다. 기계가 읽는 사본: bom.csv · connections.json

전기 회로도: 미작성

외부 부품의 전기 회로도(커넥터 핀 배선, 풀업, 전원 분리)는 아직 작성하지 않았다. 이 패키지의 외부 부품은 Basys3 내장 장치와 Pmod 모듈이라 핀 배정은 constr/basys3.xdc를 따른다. 새 외부 모듈을 쓰는 캡스톤은 회로도를 별도 산출물로 둔다.

Basys3 FPGA 핀은 3.3V 신호만 직접 받는다. 5V·12V·모터·릴레이는 직결하지 않는다.

6. 기술 트리 (이 캡스톤의 확장 경로)

  • T1MMIO 버스 + GPIO + UART, 마스터는 FSM
  • T2마스터를 PicoRV32로 교체, 부트 ROM(BRAM)과 SRAM 추가 참고: fanguoc2len-risc-v-computer, tmahlburg-picosoc-basys3
  • T2버스를 AXI4-Lite로 교체 (CPU 인터페이스 맞춤) 참고: suoglu-axi-lite-slave, pkt-gif-custom-vending-machine
  • T3VGA·PS/2 등 주변장치 추가, 펌웨어(C) 작성 참고: fanguoc2len-risc-v-computer

rtl/mmio_bus.v 21줄
// mmio-lite 버스: 단일 마스터, 주소 [31:12]로 슬레이브 선택, 읽기 데이터 조합 경로, 항상 ready.
module mmio_bus (
    input  wire        valid,
    input  wire [31:0] addr,
    output wire        sel_gpio,
    output wire        sel_uart,
    input  wire [31:0] rdata_gpio,
    input  wire [31:0] rdata_uart,
    output reg  [31:0] rdata,
    output wire        ready
);
    assign sel_gpio = valid && (addr[31:12] == 20'h10000);
    assign sel_uart = valid && (addr[31:12] == 20'h10001);
    assign ready    = 1'b1;

    always @* begin
        if (addr[31:12] == 20'h10000)      rdata = rdata_gpio;
        else if (addr[31:12] == 20'h10001) rdata = rdata_uart;
        else                               rdata = 32'h0000_0000;
    end
endmodule
rtl/gpio.v 33줄
// GPIO: 0x00 LED(RW, 16bit), 0x04 SW(RO, 2단 동기화된 스위치 16bit)
module gpio (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [7:0]  off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    input  wire [15:0] sw,
    output wire [15:0] led
);
    reg [15:0] led_r;
    reg [15:0] sw_s1, sw_s2;
    assign led = led_r;

    always @(posedge clk) begin
        sw_s1 <= sw;
        sw_s2 <= sw_s1;
        if (rst)
            led_r <= 16'h0000;
        else if (sel && we && off == 8'h00)
            led_r <= wdata[15:0];
    end

    always @* begin
        case (off)
            8'h00:   rdata = {16'h0000, led_r};
            8'h04:   rdata = {16'h0000, sw_s2};
            default: rdata = 32'h0000_0000;
        endcase
    end
endmodule
rtl/uart_tx.v 64줄
// UART 송신(8N1): 0x00 DATA(WO, busy일 때 쓰기는 무시), 0x04 STATUS(RO, bit0=busy)
module uart_tx #(
    parameter CLK_HZ = 100_000_000,
    parameter BAUD   = 115200
) (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [7:0]  off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    output wire        txd
);
    localparam DIV = CLK_HZ / BAUD;

    reg [9:0]  sh;
    reg [3:0]  nbit;
    reg [15:0] cnt;
    reg        busy;
    reg        tx;
    assign txd = tx;

    wire wr_data = sel && we && (off == 8'h00);

    always @(posedge clk) begin
        if (rst) begin
            busy <= 1'b0;
            tx   <= 1'b1;
            cnt  <= 16'd0;
            nbit <= 4'd0;
            sh   <= 10'h3FF;
        end else if (!busy) begin
            tx <= 1'b1;
            if (wr_data) begin
                sh   <= {1'b1, wdata[7:0], 1'b0};
                busy <= 1'b1;
                cnt  <= DIV - 1;
                nbit <= 4'd0;
                tx   <= 1'b0;
            end
        end else begin
            if (cnt == 16'd0) begin
                cnt <= DIV - 1;
                if (nbit == 4'd9) begin
                    busy <= 1'b0;
                    tx   <= 1'b1;
                end else begin
                    nbit <= nbit + 4'd1;
                    tx   <= sh[nbit + 4'd1];
                end
            end else begin
                cnt <= cnt - 16'd1;
            end
        end
    end

    always @* begin
        case (off)
            8'h04:   rdata = {31'd0, busy};
            default: rdata = 32'h0000_0000;
        endcase
    end
endmodule
rtl/soc_top.v 39줄
// SoC 코어: 마스터 포트(mmio-lite)를 밖으로 노출한다. 마스터는 데모 FSM, 테스트벤치, 나중에 CPU가 맡는다.
module soc_top #(
    parameter CLK_HZ = 100_000_000,
    parameter BAUD   = 115200
) (
    input  wire        clk,
    input  wire        rst,
    input  wire        m_valid,
    input  wire        m_we,
    input  wire [31:0] m_addr,
    input  wire [31:0] m_wdata,
    output wire [31:0] m_rdata,
    output wire        m_ready,
    input  wire [15:0] sw,
    output wire [15:0] led,
    output wire        uart_txd
);
    wire        sel_gpio, sel_uart;
    wire [31:0] rd_gpio, rd_uart;

    mmio_bus u_bus (
        .valid(m_valid), .addr(m_addr),
        .sel_gpio(sel_gpio), .sel_uart(sel_uart),
        .rdata_gpio(rd_gpio), .rdata_uart(rd_uart),
        .rdata(m_rdata), .ready(m_ready)
    );

    gpio u_gpio (
        .clk(clk), .rst(rst), .sel(sel_gpio), .we(m_we),
        .off(m_addr[7:0]), .wdata(m_wdata), .rdata(rd_gpio),
        .sw(sw), .led(led)
    );

    uart_tx #(.CLK_HZ(CLK_HZ), .BAUD(BAUD)) u_uart (
        .clk(clk), .rst(rst), .sel(sel_uart), .we(m_we),
        .off(m_addr[7:0]), .wdata(m_wdata), .rdata(rd_uart),
        .txd(uart_txd)
    );
endmodule
rtl/demo_master.v 53줄
// 데모 마스터(CPU 슬롯 대체용 FSM): SW를 읽어 LED에 쓰고, 버튼 상승 에지마다 SW[7:0]을 UART로 보낸다.
module demo_master (
    input  wire        clk,
    input  wire        rst,
    input  wire        btn,
    output reg         m_valid,
    output reg         m_we,
    output reg  [31:0] m_addr,
    output reg  [31:0] m_wdata,
    input  wire [31:0] m_rdata,
    input  wire        m_ready
);
    localparam [2:0] RD_SW = 3'd0, WR_LED = 3'd1, CHK = 3'd2, RD_ST = 3'd3, WR_TX = 3'd4;

    reg [2:0]  st;
    reg [15:0] sw_q;
    reg        pend;
    reg        b1, b2, b3;

    always @(posedge clk) begin
        if (rst) begin
            st   <= RD_SW;
            sw_q <= 16'h0000;
            pend <= 1'b0;
            b1 <= 1'b0; b2 <= 1'b0; b3 <= 1'b0;
        end else begin
            b1 <= btn; b2 <= b1; b3 <= b2;
            if (b2 && !b3) pend <= 1'b1;
            case (st)
                RD_SW:   if (m_ready) begin sw_q <= m_rdata[15:0]; st <= WR_LED; end
                WR_LED:  if (m_ready) st <= CHK;
                CHK:     st <= pend ? RD_ST : RD_SW;
                RD_ST:   if (m_ready) st <= m_rdata[0] ? RD_SW : WR_TX;
                WR_TX:   if (m_ready) begin pend <= 1'b0; st <= RD_SW; end
                default: st <= RD_SW;
            endcase
        end
    end

    always @* begin
        m_valid = 1'b0;
        m_we    = 1'b0;
        m_addr  = 32'h0000_0000;
        m_wdata = 32'h0000_0000;
        case (st)
            RD_SW:  begin m_valid = 1'b1; m_addr = 32'h1000_0004; end
            WR_LED: begin m_valid = 1'b1; m_we = 1'b1; m_addr = 32'h1000_0000; m_wdata = {16'h0000, sw_q}; end
            RD_ST:  begin m_valid = 1'b1; m_addr = 32'h1000_1004; end
            WR_TX:  begin m_valid = 1'b1; m_we = 1'b1; m_addr = 32'h1000_1000; m_wdata = {24'h000000, sw_q[7:0]}; end
            default: ;
        endcase
    end
endmodule

tb/tb_soc.v 117줄
`timescale 1ns/1ps
// 자체 점검 테스트벤치: LED/SW 레지스터, 미매핑 주소, UART 비트열(8N1, LSB 먼저)을 확인한다.
module tb_soc;
    localparam CLK_HZ = 1_000_000;
    localparam BAUD   = 100_000;
    localparam BIT_NS = (CLK_HZ / BAUD) * 10;   // 클럭 주기 10ns × 분주비

    reg         clk = 1'b0;
    reg         rst = 1'b1;
    reg         m_valid = 1'b0;
    reg         m_we = 1'b0;
    reg  [31:0] m_addr = 32'd0;
    reg  [31:0] m_wdata = 32'd0;
    wire [31:0] m_rdata;
    wire        m_ready;
    reg  [15:0] sw = 16'hA5C3;
    wire [15:0] led;
    wire        txd;
    integer     errors = 0;

    always #5 clk = ~clk;

    soc_top #(.CLK_HZ(CLK_HZ), .BAUD(BAUD)) dut (
        .clk(clk), .rst(rst),
        .m_valid(m_valid), .m_we(m_we), .m_addr(m_addr), .m_wdata(m_wdata),
        .m_rdata(m_rdata), .m_ready(m_ready),
        .sw(sw), .led(led), .uart_txd(txd)
    );

    task bus_write(input [31:0] a, input [31:0] d);
        begin
            @(posedge clk);
            m_valid <= 1'b1; m_we <= 1'b1; m_addr <= a; m_wdata <= d;
            @(posedge clk);
            m_valid <= 1'b0; m_we <= 1'b0;
        end
    endtask

    task bus_read(input [31:0] a, output [31:0] d);
        begin
            @(posedge clk);
            m_valid <= 1'b1; m_we <= 1'b0; m_addr <= a;
            @(posedge clk);
            d = m_rdata;
            m_valid <= 1'b0;
        end
    endtask

    task check32(input [255:0] name, input [31:0] got, input [31:0] exp);
        begin
            if (got !== exp) begin
                errors = errors + 1;
                $display("FAIL %0s: got %h expected %h", name, got, exp);
            end
        end
    endtask

    task uart_expect(input [7:0] data);
        integer i;
        begin
            @(negedge txd);
            #(BIT_NS / 2);
            if (txd !== 1'b0) begin errors = errors + 1; $display("FAIL start bit"); end
            for (i = 0; i < 8; i = i + 1) begin
                #(BIT_NS);
                if (txd !== data[i]) begin
                    errors = errors + 1;
                    $display("FAIL data bit %0d: got %b expected %b", i, txd, data[i]);
                end
            end
            #(BIT_NS);
            if (txd !== 1'b1) begin errors = errors + 1; $display("FAIL stop bit"); end
        end
    endtask

    reg [31:0] rd;

    initial begin
        #200000;
        $display("FAIL timeout");
        $finish;
    end

    initial begin
        repeat (5) @(posedge clk);
        rst <= 1'b0;
        repeat (3) @(posedge clk);

        // GPIO
        bus_write(32'h1000_0000, 32'h0000_1234);
        @(posedge clk);
        check32("led pin", {16'h0000, led}, 32'h0000_1234);
        bus_read(32'h1000_0000, rd);
        check32("led reg readback", rd, 32'h0000_1234);
        bus_read(32'h1000_0004, rd);
        check32("sw reg", rd, 32'h0000_A5C3);

        // 미매핑 주소
        bus_read(32'h2000_0000, rd);
        check32("unmapped read", rd, 32'h0000_0000);

        // UART
        bus_read(32'h1000_1004, rd);
        check32("uart idle status", rd, 32'h0000_0000);
        fork
            bus_write(32'h1000_1000, 32'h0000_00B2);
            uart_expect(8'hB2);
        join
        repeat (20) @(posedge clk);                 // 정지 비트가 끝나 busy가 내려갈 때까지 대기
        bus_read(32'h1000_1004, rd);
        check32("uart idle after tx", rd, 32'h0000_0000);

        if (errors == 0) $display("PASS");
        else             $display("FAIL: %0d errors", errors);
        $finish;
    end
endmodule

# 단계: lint(verilator) → sim(iverilog) → synth(yosys, 자원 추정). Vivado 빌드는 README의 절차를 따른다.
RTL_ALL = rtl/mmio_bus.v rtl/gpio.v rtl/uart_tx.v rtl/soc_top.v rtl/demo_master.v rtl/basys3_top.v
RTL_SOC = rtl/mmio_bus.v rtl/gpio.v rtl/uart_tx.v rtl/soc_top.v
TB      = tb/tb_soc.v

.PHONY: lint sim synth clean
build:
	mkdir -p build

lint: build
	verilator --lint-only -Wall -Wno-fatal --top-module basys3_top $(RTL_ALL) 2>&1 | tee build/lint.log
	@! grep -q "%Error" build/lint.log

sim: build
	iverilog -g2005 -o build/sim.vvp $(TB) $(RTL_SOC)
	vvp build/sim.vvp | tee build/sim.log
	@grep -q "^PASS" build/sim.log && ! grep -q "^FAIL" build/sim.log

synth: build
	yosys -q -l build/synth.log -p "read_verilog $(RTL_ALL); synth_xilinx -top basys3_top -family xc7; stat"
	@grep -E "LUT|FDRE|FDCE|FDPE|Number of cells" build/synth.log | tail -12

clean:
	rm -rf build
