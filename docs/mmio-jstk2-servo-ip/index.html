조이스틱 서보 로봇팔 IP

PmodJSTK2 SPI 판독 IP(X/Y 10비트, 버튼 2개)와 2채널 서보 PWM IP를 단일 AXI4-Lite 슬레이브로 묶는다. 소프트웨어가 조이스틱 값을 서보 위치로 선형 매핑하고, 기울기 제한 레지스터로 급가속을 막는다. 테스트벤치는 JSTK2 SPI 응답 모델과 서보 펄스 폭 측정기로 자체 점검한다.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • JSTK2 SPI 마스터: 1 MHz, 5바이트 전이중 교환, 100 ms마다 자동 샘플링, X/Y/BTN 레지스터 갱신
  • 서보 PWM 2채널: 50 Hz 주기(2,000,000 클럭 @ 100 MHz), 펄스 폭 1,000,000~2,000,000 클럭(1~2 ms)
  • AXI4-Lite 슬레이브: BASE=0x401F_0000, IP 표준 rev1 준수(관리 오프셋 FFF0~FFFC 포함)
  • 레지스터: JSTK2_X, JSTK2_Y, JSTK2_BTN(읽기), SERVO_A_POS, SERVO_B_POS, CTRL, 기울기 제한 SLEW
  • 자체 점검 테스트벤치: JSTK2 SPI 응답 모델 + 서보 펄스 폭 !==비교로 PASS/FAIL 출력

범위 밖

  • JSTK2 가속도계(오프셋 보정 없음)
  • 서보 보드 동작 확인(S3)
  • 소프트웨어 선형 매핑 로직(MicroBlaze V 담당)
  • 3개 이상 서보 채널

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. PmodJSTK2 SPI 프로토콜(5바이트 교환)과 서보 PWM(50 Hz, 1~2 ms 펄스) 원리는 참고 카드(digikey-forum-jstk2-pmod-controller-vhdl, vhdlwhiz-rc-servo-controller, shashankr-basys3-fpga-robotic-arm)를 아이디어 근거로만 사용했으며 코드는 복사하지 않았다. 시뮬레이션·합성 결과는 CI가 기록한다.

rtl/jstk2_servo_ip.v 206줄
// jstk2_servo_ip.v — AXI4-Lite 슬레이브: JSTK2 SPI + 2채널 서보 PWM
// 하네스 원본 (외부 코드 포함 없음)
// IP 표준 rev1 준수: 관리 오프셋 FFF0~FFFC, ID=0x4A535356, VER=0x01010000
// BASE = 0x401F_0000, 창 크기 0x10000
`timescale 1ns/1ps
`default_nettype none

module jstk2_servo_ip #(
    parameter CLK_FREQ_HZ    = 100_000_000,
    parameter SPI_CLK_HZ     =   1_000_000,
    parameter SAMPLE_PERIOD_MS = 100
) (
    input  wire        s_axi_aclk,
    input  wire        s_axi_aresetn,
    // AXI4-Lite 슬레이브 채널
    input  wire [31:0] s_axi_awaddr,
    input  wire        s_axi_awvalid,
    output reg         s_axi_awready,
    input  wire [31:0] s_axi_wdata,
    input  wire [3:0]  s_axi_wstrb,
    input  wire        s_axi_wvalid,
    output reg         s_axi_wready,
    output reg  [1:0]  s_axi_bresp,
    output reg         s_axi_bvalid,
    input  wire        s_axi_bready,
    input  wire [31:0] s_axi_araddr,
    input  wire        s_axi_arvalid,
    output reg         s_axi_arready,
    output reg  [31:0] s_axi_rdata,
    output reg  [1:0]  s_axi_rresp,
    output reg         s_axi_rvalid,
    input  wire        s_axi_rready,
    // IRQ
    output wire        irq,
    // SPI 포트 (JSTK2)
    output wire        spi_cs_n,
    output wire        spi_sclk,
    output wire        spi_mosi,
    input  wire        spi_miso,
    // 서보 PWM 출력
    output wire        pwm_a,
    output wire        pwm_b
);

// ─── 레지스터 맵 오프셋 ───────────────────────────────────────────
localparam ADDR_JSTK_X     = 16'h0000;
localparam ADDR_JSTK_Y     = 16'h0004;
localparam ADDR_JSTK_BTN   = 16'h0008;
localparam ADDR_SERVO_A    = 16'h000C;
localparam ADDR_SERVO_B    = 16'h0010;
localparam ADDR_CTRL       = 16'h0014;
localparam ADDR_SLEW       = 16'h0018;
localparam ADDR_ID         = 16'hFFF0;
localparam ADDR_VER        = 16'hFFF4;
localparam ADDR_IRQ_STATUS = 16'hFFF8;
localparam ADDR_IRQ_ENABLE = 16'hFFFC;

// IP 고정 상수
localparam IP_ID  = 32'h4A535356; // 'JSSV'
localparam IP_VER = 32'h01010000; // std_rev=1, major=1, minor=0

// ─── 내부 레지스터 ────────────────────────────────────────────────
reg [31:0] reg_servo_a;   // 서보 A 위치
reg [31:0] reg_servo_b;   // 서보 B 위치
reg [31:0] reg_ctrl;      // [0]=JSTK en, [1]=PWM en
reg [31:0] reg_slew;      // 기울기 제한
reg        reg_irq_status; // W1C
reg        reg_irq_enable;

// JSTK2 SPI 서브모듈 출력
wire [9:0]  jstk_x;
wire [9:0]  jstk_y;
wire [1:0]  jstk_btn;
wire        sample_done;

// ─── 서브모듈 인스턴스 ────────────────────────────────────────────
jstk2_spi #(
    .CLK_FREQ_HZ    (CLK_FREQ_HZ),
    .SPI_CLK_HZ     (SPI_CLK_HZ),
    .SAMPLE_PERIOD_MS(SAMPLE_PERIOD_MS)
) u_jstk2_spi (
    .clk        (s_axi_aclk),
    .rst_n      (s_axi_aresetn),
    .en         (reg_ctrl[0]),
    .jstk_x     (jstk_x),
    .jstk_y     (jstk_y),
    .jstk_btn   (jstk_btn),
    .sample_done(sample_done),
    .spi_cs_n   (spi_cs_n),
    .spi_sclk   (spi_sclk),
    .spi_mosi   (spi_mosi),
    .spi_miso   (spi_miso)
);

servo_pwm #(
    .CLK_FREQ_HZ (CLK_FREQ_HZ),
    .SERVO_PERIOD(2_000_000),
    .PULSE_MIN   (1_000_000),
    .PULSE_MAX   (2_000_000)
) u_servo_pwm (
    .clk   (s_axi_aclk),
    .rst_n (s_axi_aresetn),
    .en    (reg_ctrl[1]),
    .pos_a (reg_servo_a),
    .pos_b (reg_servo_b),
    .slew  (reg_slew),
    .pwm_a (pwm_a),
    .pwm_b (pwm_b)
);

assign irq = reg_irq_status & reg_irq_enable;

// ─── AXI4-Lite 쓰기 채널 ─────────────────────────────────────────
// IRQ status는 여기에서만 구동 (single driver): sample_done에서 set, W1C에서 clear
reg [15:0] aw_addr_lat;

always @(posedge s_axi_aclk or negedge s_axi_aresetn) begin
    if (!s_axi_aresetn) begin
        s_axi_awready  <= 1'b0;
        s_axi_wready   <= 1'b0;
        s_axi_bvalid   <= 1'b0;
        s_axi_bresp    <= 2'b00;
        aw_addr_lat    <= 16'd0;
        reg_servo_a    <= 32'd1_500_000; // 중앙 1.5 ms
        reg_servo_b    <= 32'd1_500_000;
        reg_ctrl       <= 32'd0;
        reg_slew       <= 32'd0;
        reg_irq_enable <= 1'b0;
        reg_irq_status <= 1'b0;
    end else begin
        s_axi_awready <= 1'b0;
        s_axi_wready  <= 1'b0;

        // IRQ status: sample_done에서 set (single driver)
        if (sample_done) reg_irq_status <= 1'b1;

        // AW 핸드셰이크
        if (s_axi_awvalid && !s_axi_awready) begin
            s_axi_awready <= 1'b1;
            aw_addr_lat   <= s_axi_awaddr[15:0];
        end

        // W 핸드셰이크 + 레지스터 쓰기 + B 응답 세트
        if (s_axi_wvalid && !s_axi_wready) begin
            s_axi_wready <= 1'b1;
            s_axi_bvalid <= 1'b1;
            s_axi_bresp  <= 2'b00;
            // awvalid와 wvalid가 같은 클럭에 도착하면 NBA로 aw_addr_lat이 아직 갱신되지 않으므로
            // awvalid가 살아있는 동안은 awaddr를 직접 사용한다
            case (s_axi_awvalid ? s_axi_awaddr[15:0] : aw_addr_lat)
                ADDR_SERVO_A: reg_servo_a   <= s_axi_wdata;
                ADDR_SERVO_B: reg_servo_b   <= s_axi_wdata;
                ADDR_CTRL:    reg_ctrl      <= s_axi_wdata;
                ADDR_SLEW:    reg_slew      <= s_axi_wdata;
                ADDR_IRQ_STATUS: begin
                    // W1C: 1이 쓰인 비트를 소거
                    if (s_axi_wdata[0]) reg_irq_status <= 1'b0;
                end
                ADDR_IRQ_ENABLE: reg_irq_enable <= s_axi_wdata[0];
                default: ; // 없는 주소: 무시
            endcase
        end

        // B 채널 소거
        if (s_axi_bready && s_axi_bvalid)
            s_axi_bvalid <= 1'b0;
    end
end

// ─── AXI4-Lite 읽기 채널 ─────────────────────────────────────────
always @(posedge s_axi_aclk or negedge s_axi_aresetn) begin
    if (!s_axi_aresetn) begin
        s_axi_arready <= 1'b0;
        s_axi_rvalid  <= 1'b0;
        s_axi_rdata   <= 32'd0;
        s_axi_rresp   <= 2'b00;
    end else begin
        if (s_axi_arvalid && !s_axi_arready) begin
            s_axi_arready <= 1'b1;
            s_axi_rvalid  <= 1'b1;
            s_axi_rresp   <= 2'b00;
            case (s_axi_araddr[15:0])
                ADDR_JSTK_X:     s_axi_rdata <= {22'd0, jstk_x};
                ADDR_JSTK_Y:     s_axi_rdata <= {22'd0, jstk_y};
                ADDR_JSTK_BTN:   s_axi_rdata <= {30'd0, jstk_btn};
                ADDR_SERVO_A:    s_axi_rdata <= reg_servo_a;
                ADDR_SERVO_B:    s_axi_rdata <= reg_servo_b;
                ADDR_CTRL:       s_axi_rdata <= reg_ctrl;
                ADDR_SLEW:       s_axi_rdata <= reg_slew;
                ADDR_ID:         s_axi_rdata <= IP_ID;
                ADDR_VER:        s_axi_rdata <= IP_VER;
                ADDR_IRQ_STATUS: s_axi_rdata <= {31'd0, reg_irq_status};
                ADDR_IRQ_ENABLE: s_axi_rdata <= {31'd0, reg_irq_enable};
                default:         s_axi_rdata <= 32'd0;
            endcase
        end else begin
            s_axi_arready <= 1'b0;
        end

        if (s_axi_rready && s_axi_rvalid)
            s_axi_rvalid <= 1'b0;
    end
end

endmodule
`default_nettype wire

// jstk2_spi.v — PmodJSTK2 SPI 마스터 (5바이트 전이중 교환)
// 하네스 원본 (외부 코드 포함 없음)
// SPI Mode 0 (CPOL=0, CPHA=0), 클럭 100 MHz → 50 클럭 토글 → 1 MHz SCLK
// 5바이트 TX: {0x01, 0x00, 0x00, 0x00, 0x00}  RX: {X_L, X_H, Y_L, Y_H, BTN}
// sample_period: 100 ms 샘플링 주기 (파라미터)
`timescale 1ns/1ps
`default_nettype none

module jstk2_spi #(
    parameter CLK_FREQ_HZ    = 100_000_000,
    parameter SPI_CLK_HZ     =   1_000_000,
    parameter SAMPLE_PERIOD_MS = 100       // 샘플링 주기 (ms)
) (
    input  wire       clk,
    input  wire       rst_n,
    // 활성화
    input  wire       en,          // 1=샘플링 활성
    // 결과 레지스터
    output reg [9:0]  jstk_x,
    output reg [9:0]  jstk_y,
    output reg [1:0]  jstk_btn,
    output reg        sample_done, // 1클럭 펄스, 새 샘플 도착
    // SPI 포트
    output reg        spi_cs_n,
    output reg        spi_sclk,
    output reg        spi_mosi,
    input  wire       spi_miso
);

localparam SCLK_DIV    = CLK_FREQ_HZ / SPI_CLK_HZ / 2; // 50 클럭 per 반주기
localparam SAMPLE_CNT  = CLK_FREQ_HZ / 1000 * SAMPLE_PERIOD_MS; // 샘플 주기 클럭 수
localparam N_BYTES     = 5;
localparam N_BITS      = N_BYTES * 8; // 40

// 상태 머신
localparam S_IDLE    = 2'd0;
localparam S_ASSERT  = 2'd1; // CS# 하강 후 1 클럭 대기
localparam S_XFER    = 2'd2; // 비트 전송
localparam S_DONE    = 2'd3; // CS# 해제

reg [1:0]  state;
reg [26:0] period_cnt;    // 샘플 주기 카운터
reg [5:0]  div_cnt;       // SCLK 분주 카운터 (0~49)
reg [5:0]  bit_cnt;       // 전송 비트 카운터 (0~39)
reg        sclk_phase;    // 0=low, 1=high
reg [39:0] tx_shift;      // TX 쉬프트 레지스터
reg [39:0] rx_shift;      // RX 쉬프트 레지스터

// TX 데이터 고정: {0x01, 0x00, 0x00, 0x00, 0x00}
localparam [39:0] TX_DATA = 40'h01_00_00_00_00;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state       <= S_IDLE;
        period_cnt  <= 27'd0;
        div_cnt     <= 6'd0;
        bit_cnt     <= 6'd0;
        sclk_phase  <= 1'b0;
        tx_shift    <= 40'd0;
        rx_shift    <= 40'd0;
        spi_cs_n    <= 1'b1;
        spi_sclk    <= 1'b0;
        spi_mosi    <= 1'b0;
        jstk_x      <= 10'd512;
        jstk_y      <= 10'd512;
        jstk_btn    <= 2'b00;
        sample_done <= 1'b0;
    end else begin
        sample_done <= 1'b0; // 기본 소거

        case (state)
            S_IDLE: begin
                spi_cs_n   <= 1'b1;
                spi_sclk   <= 1'b0;
                if (!en) begin
                    period_cnt <= 27'd0;
                end else if (period_cnt >= SAMPLE_CNT - 1) begin
                    period_cnt <= 27'd0;
                    state      <= S_ASSERT;
                    tx_shift   <= TX_DATA;
                    rx_shift   <= 40'd0;
                    bit_cnt    <= 6'd0;
                    div_cnt    <= 6'd0;
                    sclk_phase <= 1'b0;
                end else begin
                    period_cnt <= period_cnt + 1'b1;
                end
            end

            S_ASSERT: begin
                // CS# 활성(LOW), 최소 1 클럭 대기
                spi_cs_n <= 1'b0;
                spi_mosi <= tx_shift[39]; // MSB first
                state    <= S_XFER;
                div_cnt  <= 6'd0;
            end

            S_XFER: begin
                if (div_cnt < SCLK_DIV - 1) begin
                    div_cnt <= div_cnt + 1'b1;
                end else begin
                    div_cnt    <= 6'd0;
                    sclk_phase <= ~sclk_phase;
                    spi_sclk   <= ~sclk_phase; // 현재 반주기 반전

                    if (!sclk_phase) begin
                        // 하강에지(SCLK 1→0): 다음 MOSI 준비
                        if (bit_cnt < N_BITS - 1) begin
                            bit_cnt  <= bit_cnt + 1'b1;
                            tx_shift <= {tx_shift[38:0], 1'b0};
                            spi_mosi <= tx_shift[38];
                        end
                    end else begin
                        // 상승에지(SCLK 0→1): MISO 샘플
                        rx_shift <= {rx_shift[38:0], spi_miso};
                        if (bit_cnt == N_BITS - 1) begin
                            state <= S_DONE;
                        end
                    end
                end
            end

            S_DONE: begin
                spi_cs_n <= 1'b1;
                spi_sclk <= 1'b0;
                // 수신 데이터 파싱: RX = {X_L, X_H, Y_L, Y_H, BTN}
                // X = {X_H[1:0], X_L[7:0]} (10비트)
                // Y = {Y_H[1:0], Y_L[7:0]} (10비트)
                // BTN = BTN_BYTE[1:0]
                jstk_x   <= {rx_shift[25:24], rx_shift[39:32]}; // {X_H[1:0], X_L}
                jstk_y   <= {rx_shift[9:8],   rx_shift[23:16]}; // {Y_H[1:0], Y_L}
                jstk_btn <= rx_shift[1:0];                       // BTN 바이트 [1:0]
                sample_done <= 1'b1;
                state    <= S_IDLE;
            end

            default: state <= S_IDLE;
        endcase
    end
end

module servo_pwm #(
    parameter CLK_FREQ_HZ  = 100_000_000,
    parameter SERVO_PERIOD  = 2_000_000, // 20 ms @ 100 MHz
    parameter PULSE_MIN     = 1_000_000, // 1 ms @ 100 MHz
    parameter PULSE_MAX     = 2_000_000  // 2 ms @ 100 MHz
) (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        en,            // 1=PWM 출력 활성
    input  wire [31:0] pos_a,         // 채널 A 펄스 폭 (클럭 수)
    input  wire [31:0] pos_b,         // 채널 B 펄스 폭 (클럭 수)
    input  wire [31:0] slew,          // 기울기 제한 (0=제한 없음)
    output reg         pwm_a,         // 서보 A 신호
    output reg         pwm_b          // 서보 B 신호
);

// 현재 출력 위치 (기울기 제한 적용)
reg [31:0] cur_a;
reg [31:0] cur_b;

// 주기 카운터
reg [20:0] period_cnt;

// 목표 위치 클램프 함수 (combinational)
function [31:0] clamp;
    input [31:0] val;
    begin
        if (val < PULSE_MIN) clamp = PULSE_MIN;
        else if (val > PULSE_MAX) clamp = PULSE_MAX;
        else clamp = val;
    end
endfunction

// 기울기 제한 적용 (슬루율 제어)
// slew != 0일 때 주기마다 최대 slew 클럭만큼 이동
// 주기 경계에서 목표 갱신
wire [31:0] target_a = clamp(pos_a);
wire [31:0] target_b = clamp(pos_b);

reg period_tick; // 주기 경계 1클럭 펄스

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        period_cnt <= 21'd0;
        period_tick <= 1'b0;
        cur_a <= PULSE_MIN + (PULSE_MAX - PULSE_MIN) / 2; // 중앙
        cur_b <= PULSE_MIN + (PULSE_MAX - PULSE_MIN) / 2;
        pwm_a <= 1'b0;
        pwm_b <= 1'b0;
    end else begin
        period_tick <= 1'b0;

        if (period_cnt >= SERVO_PERIOD - 1) begin
            period_cnt  <= 21'd0;
            period_tick <= 1'b1;
        end else begin
            period_cnt <= period_cnt + 1'b1;
        end

        // 주기 경계에서 목표 위치 갱신 (기울기 제한)
        if (period_tick) begin
            // 채널 A 기울기 제한
            if (slew == 32'd0) begin
                cur_a <= target_a;
            end else if (target_a > cur_a) begin
                cur_a <= (target_a - cur_a > slew) ? cur_a + slew : target_a;
            end else begin
                cur_a <= (cur_a - target_a > slew) ? cur_a - slew : target_a;
            end
            // 채널 B 기울기 제한
            if (slew == 32'd0) begin
                cur_b <= target_b;
            end else if (target_b > cur_b) begin
                cur_b <= (target_b - cur_b > slew) ? cur_b + slew : target_b;
            end else begin
                cur_b <= (cur_b - target_b > slew) ? cur_b - slew : target_b;
            end
        end

        // PWM 출력: period_cnt < cur_x 동안 HIGH
        if (!en) begin
            pwm_a <= 1'b0;
            pwm_b <= 1'b0;
        end else begin
            pwm_a <= (period_cnt < cur_a) ? 1'b1 : 1'b0;
            pwm_b <= (period_cnt < cur_b) ? 1'b1 : 1'b0;
        end
    end
end

// MISO에 고정 패킷 반환: X_L=0x00, X_H=0x08(=512>>2?→10bit 표현),

task axi_write;
    input [31:0] addr;
    input [31:0] data;
    begin
        @(posedge clk); #1;
        s_awaddr  <= addr;
        s_awvalid <= 1'b1;
        s_wdata   <= data;
        s_wstrb   <= 4'hF;
        s_wvalid  <= 1'b1;
        @(posedge clk); #1;

        // while exits at posedge+1ns 시점: wready=1 && awready=1 && bvalid=1 (동일 클럭에 세트됨)
        s_awvalid <= 1'b0;
        s_wvalid  <= 1'b0;
        s_bready  <= 1'b1;
        @(posedge clk); #1;  // DUT가 bready&&bvalid를 보고 bvalid=0으로 소거

        // while (!s_bvalid) 는 제거: bvalid는 bready 첫 클럭에서 소거되어 바로 0이 됨
    end
endtask

task axi_read;
    input  [31:0] addr;
    output [31:0] data;
    begin
        @(posedge clk); #1;
        s_araddr  <= addr;
        s_arvalid <= 1'b1;
        s_rready  <= 1'b1;
        @(posedge clk); #1;
        while (!s_rvalid) @(posedge clk); #1;
        data      = s_rdata;   // blocking: 같은 timestep에 호출자가 읽어야 하므로
        s_arvalid <= 1'b0;
        s_rready  <= 1'b0;
    end
endtask

        fail_count = fail_count + 1;
    end else begin
        $display("T3 IRQ received OK (timeout_left=%0d)", timeout);
        irq_got = 1;
    end

    // ── 테스트 4: IRQ 소거 (W1C) ─────────────────────────────────
    axi_write(32'h401F_FFF8, 32'h1); // IRQ_STATUS W1C
    @(posedge clk); @(posedge clk);
    axi_read(32'h401F_FFF8, rd_data);
    check_eq(rd_data[0], 1'b0, "IRQ_STATUS cleared");
    $display("T4 IRQ W1C OK");

    // ── 테스트 5: JSTK2 레지스터 값 확인 ────────────────────────
    // SPI 모델: X=512=0x200, Y=511=0x1FF, BTN=0x01
    axi_read(32'h401F_0000, rd_data);
    check_eq(rd_data[9:0], 10'd512, "JSTK2_X=512");
    axi_read(32'h401F_0004, rd_data);
    check_eq(rd_data[9:0], 10'd511, "JSTK2_Y=511");
    axi_read(32'h401F_0008, rd_data);
    check_eq(rd_data[1:0], 2'b01, "JSTK2_BTN=0x01");
    $display("T5 JSTK2 regs OK");

    // ── 테스트 6: 서보 A PWM 펄스 폭 측정 ───────────────────────
    // 현재 SERVO_A=1,500,000. 이를 1,200,000(1.2ms)로 변경해 펄스 폭 측정
    axi_write(32'h401F_000C, 32'd1_200_000); // 1.2 ms
    // 서보 PWM은 20ms 주기. 한 주기 내 펄스 측정
    // 최대 대기: 6,000,000 클럭 (약 60ms, 3주기 여유)
    // 하강(현재 HIGH일 수 있음)→상승→하강 순서로 측정
    timeout = 6_000_000;
    while (pwm_a && timeout > 0) begin @(posedge clk); timeout = timeout - 1; end // 하강 대기
    while (!pwm_a && timeout > 0) begin @(posedge clk); timeout = timeout - 1; end // 상승 대기
    if (timeout <= 0) begin
        $display("FAIL: PWM_A no rising edge");
        fail_count = fail_count + 1;
    end else begin
        pwm_rise_time = $realtime;
        while (pwm_a && timeout > 0) begin @(posedge clk); timeout = timeout - 1; end
        pwm_fall_time = $realtime;
        pulse_width_clk = (pwm_fall_time - pwm_rise_time) / 10.0; // ns → 클럭
        $display("T6 PWM_A pulse_width=%.0f clk (exp=1200000, tol±5)", pulse_width_clk);
        if (pulse_width_clk < 1_199_995 || pulse_width_clk > 1_200_005) begin
            $display("FAIL: PWM_A pulse width out of tolerance");
            fail_count = fail_count + 1;
        end else begin
            $display("T6 PWM_A pulse width OK");
        end
    end

    // ── 최종 판정 ────────────────────────────────────────────────
    if (fail_count === 0)
        $display("PASS: all tests passed");
    else
        $display("FAIL: %0d test(s) failed", fail_count);

    $finish;
end

    $display("FAIL: global simulation timeout");
    $finish;
end
