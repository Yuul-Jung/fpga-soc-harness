XADC 온다이 전원 감시 AXI4-Lite IP

Artix-7 내장 XADC의 온다이 공급 전압 채널(Vccint·Vccaux·Vccbram)을 DRP로 순차 읽어, 채널별 임계값(하한·상한) 위반 시 AXI4-Lite 인터럽트를 생성하는 T2 전원 감시 IP. 외부 부품 없음 — XADC 온칩 채널만 사용.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • XADC DRP 주소 0x01(Vccint)·0x02(Vccaux)·0x06(Vccbram)를 FSM으로 순차 읽기
  • 채널별 12비트 임계값(하한·상한) 레지스터와 위반 시 IRQ 발생 (W1C 소거 지원)
  • mbv-frozen-v2 free_slot [16] = 0x4016_0000 제안, intc 13 제안
  • IP 표준 Rev.1 관리 레지스터(IP_ID·IP_VER·IRQ_STATUS·IRQ_ENABLE) 준수
  • iverilog 시뮬레이션에서 모의 DRP 모델로 TC1~TC5 PASS 확인

범위 밖

  • 외부 아날로그 채널(JXADC) 사용 — mmio-xadc-hub 패키지가 담당
  • 온도 센서(DRP 0x00) 판독 — 단순화를 위해 제외
  • 연속 자동 스캔(auto 모드 레지스터 비트는 선언하나 CI 점검 범위 외)
  • MicroBlaze V AXI4-Lite 버스 완전 통합 (래퍼 기준 단독 데모)
  • 보드 동작 확인 S3 — 사용자가 직접 한다

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-075(§13.7)로 생성됨 (2026-11-04). DRP 모의 모델을 테스트벤치에 포함해 iverilog 시뮬레이션으로 Vccint/Vccaux/Vccbram 판독·임계값 위반·IRQ W1C 소거를 자체 점검한다. 기존 mmio-xadc-hub(외부 JXADC 채널)와 구별: 이 IP는 온다이 공급 전압 채널(DRP 주소 0x01·0x02·0x06)만 사용한다.

7. 실행 소스

저장소의 capstones/mmio-xadc-pwr-mon/에 있는 파일 그대로다. 이 문서는 파일에서 읽어 넣은 것이므로 항상 일치한다. 소스 전체 내려받기 (.zip) · GitHub에서 보기

rtl/xadc_pwr_mon_ip.v 329줄
// xadc_pwr_mon_ip.v
// Artix-7 XADC 전원 감시 AXI4-Lite IP
// 내부 공급 전압 채널(Vccint/Vccaux/Vccbram)을 DRP로 읽어 임계값 위반 시 IRQ 발생.
// 외부 부품 없음 — XADC 온다이 채널만 사용.
// Verilog-2001. iverilog/verilator/yosys 호환.
// 출처: capstones/mmio-xadc-pwr-mon/capstone.json
`default_nettype none
`timescale 1ns/1ps

module xadc_pwr_mon_ip #(
    parameter BASE_ADDR = 32'h4016_0000
) (
    input  wire        s_axi_aclk,
    input  wire        s_axi_aresetn,
    // AW
    input  wire [31:0] s_axi_awaddr,
    input  wire        s_axi_awvalid,
    output reg         s_axi_awready,
    // W
    input  wire [31:0] s_axi_wdata,
    input  wire [3:0]  s_axi_wstrb,
    input  wire        s_axi_wvalid,
    output reg         s_axi_wready,
    // B
    output reg  [1:0]  s_axi_bresp,
    output reg         s_axi_bvalid,
    input  wire        s_axi_bready,
    // AR
    input  wire [31:0] s_axi_araddr,
    input  wire        s_axi_arvalid,
    output reg         s_axi_arready,
    // R
    output reg  [31:0] s_axi_rdata,
    output reg  [1:0]  s_axi_rresp,
    output reg         s_axi_rvalid,
    input  wire        s_axi_rready,
    // DRP interface (to XADC primitive or simulation model)
    output reg         drp_den,
    output wire        drp_dwe,
    output reg  [6:0]  drp_daddr,
    output wire [15:0] drp_di,
    input  wire [15:0] drp_do,
    input  wire        drp_drdy,
    // Interrupt output
    output wire        irq
);

    // -----------------------------------------------------------------------
    // XADC 내부 공급 전압 DRP 상태 레지스터 주소 (Artix-7 UG480 Table 2-6)
    // -----------------------------------------------------------------------
    localparam [6:0] DRP_VCCINT  = 7'h01;  // Vccint  결과 (~1.0V 공칭)
    localparam [6:0] DRP_VCCAUX  = 7'h02;  // Vccaux  결과 (~1.8V 공칭)
    localparam [6:0] DRP_VCCBRAM = 7'h06;  // Vccbram 결과 (~1.0V 공칭)

    // DRP는 읽기 전용으로만 사용; 쓰기 없음
    assign drp_dwe = 1'b0;
    assign drp_di  = 16'h0;

    // -----------------------------------------------------------------------
    // 레지스터 오프셋 (기저 주소 BASE_ADDR 기준, [15:0] 사용)
    // -----------------------------------------------------------------------
    localparam [15:0]
        REG_CTRL        = 16'h0000,  // [0]=en [1]=auto (RW)
        REG_STATUS      = 16'h0004,  // [0]=busy [5:3]=viol_live (RO)
        REG_VCCINT      = 16'h0008,  // [15:4]=12비트 원시값 (RO)
        REG_VCCAUX      = 16'h000C,  // [15:4]=12비트 원시값 (RO)
        REG_VCCBRAM     = 16'h0010,  // [15:4]=12비트 원시값 (RO)
        REG_THR_INT_LO  = 16'h0014,  // [11:0]=Vccint 하한 (RW)
        REG_THR_INT_HI  = 16'h0018,  // [11:0]=Vccint 상한 (RW)
        REG_THR_AUX_LO  = 16'h001C,  // [11:0]=Vccaux 하한 (RW)
        REG_THR_AUX_HI  = 16'h0020,  // [11:0]=Vccaux 상한 (RW)
        REG_THR_BRAM_LO = 16'h0024,  // [11:0]=Vccbram 하한 (RW)
        REG_THR_BRAM_HI = 16'h0028,  // [11:0]=Vccbram 상한 (RW)
        REG_IP_ID       = 16'hFFF0,  // 0x5041_584D "PAXM" (RO)
        REG_IP_VER      = 16'hFFF4,  // 0x0000_0100 Rev1.0 (RO)
        REG_IRQ_STATUS  = 16'hFFF8,  // [2:0]=viol (W1C)
        REG_IRQ_ENABLE  = 16'hFFFC;  // [2:0]=마스크 (RW)

    // IP_ID 아스키: "XAPM" = 0x5841_504D
    localparam [31:0] IP_ID  = 32'h5841_504D;
    localparam [31:0] IP_VER = 32'h0000_0100;

    // -----------------------------------------------------------------------
    // DRP FSM 상태
    // -----------------------------------------------------------------------
    localparam [2:0]
        F_IDLE  = 3'd0,
        F_REQ1  = 3'd1,  // Vccint 읽기 요청
        F_WAI1  = 3'd2,  // Vccint drdy 대기
        F_REQ2  = 3'd3,  // Vccaux 읽기 요청
        F_WAI2  = 3'd4,  // Vccaux drdy 대기
        F_REQ3  = 3'd5,  // Vccbram 읽기 요청
        F_WAI3  = 3'd6,  // Vccbram drdy 대기
        F_CMP   = 3'd7;  // 임계값 비교 → IDLE

    reg [2:0]  fsm;
    reg        ctrl_en;
    reg        ctrl_auto;

    // 판독값 (12비트, raw[15:4])
    reg [11:0] rd_vccint;
    reg [11:0] rd_vccaux;
    reg [11:0] rd_vccbram;

    // 임계값 레지스터
    reg [11:0] thr_int_lo;  reg [11:0] thr_int_hi;
    reg [11:0] thr_aux_lo;  reg [11:0] thr_aux_hi;
    reg [11:0] thr_bram_lo; reg [11:0] thr_bram_hi;

    // IRQ
    reg  [2:0] irq_status;   // W1C
    reg  [2:0] irq_enable;
    wire [2:0] viol;         // 현재 임계값 위반 조합

    // 자동 스캔 카운터 (1024 클럭마다 한 사이클)
    reg [9:0] auto_cnt;
    reg       scan_trig;
    reg       prev_en;    // ctrl_en 상승에지 감지용 (non-auto one-shot)

    // -----------------------------------------------------------------------
    // AXI4-Lite 쓰기 래치
    // -----------------------------------------------------------------------
    reg        aw_ok;
    reg [15:0] aw_a;
    reg        wd_ok;
    reg [31:0] wd_d;
    reg        do_wr;

    // -----------------------------------------------------------------------
    // 임계값 위반 로직
    // -----------------------------------------------------------------------
    assign viol[0] = (rd_vccint  < thr_int_lo)  | (rd_vccint  > thr_int_hi);
    assign viol[1] = (rd_vccaux  < thr_aux_lo)  | (rd_vccaux  > thr_aux_hi);
    assign viol[2] = (rd_vccbram < thr_bram_lo) | (rd_vccbram > thr_bram_hi);
    assign irq     = |(irq_status & irq_enable);

    // -----------------------------------------------------------------------
    // DRP FSM — 세 채널 순차 읽기
    // -----------------------------------------------------------------------
    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            fsm       <= F_IDLE;
            drp_den   <= 1'b0;
            drp_daddr <= 7'h0;
            scan_trig <= 1'b0;
            auto_cnt  <= 10'd0;
            prev_en   <= 1'b0;
        end else begin
            drp_den   <= 1'b0;  // 기본값 (1사이클 펄스)
            scan_trig <= 1'b0;
            prev_en   <= ctrl_en;  // 상승에지 감지를 위해 이전 값 저장

            // 자동 스캔 트리거
            if (ctrl_auto && ctrl_en) begin
                if (auto_cnt == 10'd1023) begin
                    auto_cnt  <= 10'd0;
                    scan_trig <= (fsm == F_IDLE);
                end else
                    auto_cnt <= auto_cnt + 10'd1;
            end else
                auto_cnt <= 10'd0;

            case (fsm)
                F_IDLE: begin
                    // non-auto 모드: ctrl_en 상승에지에서만 1회 트리거
                    if (scan_trig || (!ctrl_auto && ctrl_en && !prev_en))
                        fsm <= F_REQ1;
                end
                F_REQ1: begin
                    drp_den   <= 1'b1;
                    drp_daddr <= DRP_VCCINT;
                    fsm       <= F_WAI1;
                end
                F_WAI1: begin
                    if (drp_drdy) begin
                        rd_vccint <= drp_do[15:4];
                        fsm       <= F_REQ2;
                    end
                end
                F_REQ2: begin
                    drp_den   <= 1'b1;
                    drp_daddr <= DRP_VCCAUX;
                    fsm       <= F_WAI2;
                end
                F_WAI2: begin
                    if (drp_drdy) begin
                        rd_vccaux <= drp_do[15:4];
                        fsm       <= F_REQ3;
                    end
                end
                F_REQ3: begin
                    drp_den   <= 1'b1;
                    drp_daddr <= DRP_VCCBRAM;
                    fsm       <= F_WAI3;
                end
                F_WAI3: begin
                    if (drp_drdy) begin
                        rd_vccbram <= drp_do[15:4];
                        fsm        <= F_CMP;
                    end
                end
                F_CMP: begin
                    // 임계값 비교 후 IRQ_STATUS OR-업데이트
                    irq_status <= irq_status | viol;
                    fsm        <= F_IDLE;
                end
                default: fsm <= F_IDLE;
            endcase
        end
    end

    // -----------------------------------------------------------------------
    // AXI4-Lite 슬레이브 — 쓰기 채널
    // -----------------------------------------------------------------------
    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            s_axi_awready <= 1'b0;
            s_axi_wready  <= 1'b0;
            s_axi_bvalid  <= 1'b0;
            s_axi_bresp   <= 2'b00;
            aw_ok <= 1'b0; wd_ok <= 1'b0; do_wr <= 1'b0;
            aw_a  <= 16'h0; wd_d  <= 32'h0;
        end else begin
            do_wr <= 1'b0;
            if (!aw_ok) begin
                if (s_axi_awvalid) begin
                    aw_a          <= s_axi_awaddr[15:0];
                    aw_ok         <= 1'b1;
                    s_axi_awready <= 1'b1;
                end
            end else
                s_axi_awready <= 1'b0;

            if (!wd_ok) begin
                if (s_axi_wvalid) begin
                    wd_d          <= s_axi_wdata;
                    wd_ok         <= 1'b1;
                    s_axi_wready  <= 1'b1;
                end
            end else
                s_axi_wready <= 1'b0;

            if (aw_ok && wd_ok && !s_axi_bvalid) begin
                do_wr  <= 1'b1;
                aw_ok  <= 1'b0;
                wd_ok  <= 1'b0;
                s_axi_bvalid <= 1'b1;
                s_axi_bresp  <= 2'b00;
            end

            if (s_axi_bvalid && s_axi_bready)
                s_axi_bvalid <= 1'b0;
        end
    end

    // -----------------------------------------------------------------------
    // AXI4-Lite 슬레이브 — 읽기 채널
    // -----------------------------------------------------------------------
    reg [15:0] ar_a;
    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            s_axi_arready <= 1'b0;
            s_axi_rvalid  <= 1'b0;
            s_axi_rdata   <= 32'h0;
            s_axi_rresp   <= 2'b00;
            ar_a          <= 16'h0;
        end else begin
            if (s_axi_arvalid && !s_axi_rvalid) begin
                ar_a          <= s_axi_araddr[15:0];
                s_axi_arready <= 1'b1;
                s_axi_rvalid  <= 1'b1;
                s_axi_rresp   <= 2'b00;
                // 읽기 디코더
                case (s_axi_araddr[15:0])
                    REG_CTRL:        s_axi_rdata <= {30'h0, ctrl_auto, ctrl_en};
                    REG_STATUS:      s_axi_rdata <= {26'h0, viol, (fsm != F_IDLE)};
                    REG_VCCINT:      s_axi_rdata <= {16'h0, rd_vccint, 4'h0};
                    REG_VCCAUX:      s_axi_rdata <= {16'h0, rd_vccaux, 4'h0};
                    REG_VCCBRAM:     s_axi_rdata <= {16'h0, rd_vccbram, 4'h0};
                    REG_THR_INT_LO:  s_axi_rdata <= {20'h0, thr_int_lo};
                    REG_THR_INT_HI:  s_axi_rdata <= {20'h0, thr_int_hi};
                    REG_THR_AUX_LO:  s_axi_rdata <= {20'h0, thr_aux_lo};
                    REG_THR_AUX_HI:  s_axi_rdata <= {20'h0, thr_aux_hi};
                    REG_THR_BRAM_LO: s_axi_rdata <= {20'h0, thr_bram_lo};
                    REG_THR_BRAM_HI: s_axi_rdata <= {20'h0, thr_bram_hi};
                    REG_IP_ID:       s_axi_rdata <= IP_ID;
                    REG_IP_VER:      s_axi_rdata <= IP_VER;
                    REG_IRQ_STATUS:  s_axi_rdata <= {29'h0, irq_status};
                    REG_IRQ_ENABLE:  s_axi_rdata <= {29'h0, irq_enable};
                    default:         s_axi_rdata <= 32'hDEAD_BEEF;
                endcase
            end else
                s_axi_arready <= 1'b0;

            if (s_axi_rvalid && s_axi_rready)
                s_axi_rvalid <= 1'b0;
        end
    end

    // -----------------------------------------------------------------------
    // 쓰기 디코더 (do_wr 펄스)
    // -----------------------------------------------------------------------
    always @(posedge s_axi_aclk) begin
        if (!s_axi_aresetn) begin
            ctrl_en      <= 1'b0;
            ctrl_auto    <= 1'b0;
            thr_int_lo   <= 12'h000; thr_int_hi   <= 12'hFFF;
            thr_aux_lo   <= 12'h000; thr_aux_hi   <= 12'hFFF;
            thr_bram_lo  <= 12'h000; thr_bram_hi  <= 12'hFFF;
            irq_status   <= 3'b000;
            irq_enable   <= 3'b000;
        end else if (do_wr) begin
            case (aw_a)
                REG_CTRL:        {ctrl_auto, ctrl_en} <= wd_d[1:0];
                REG_THR_INT_LO:  thr_int_lo   <= wd_d[11:0];
                REG_THR_INT_HI:  thr_int_hi   <= wd_d[11:0];
                REG_THR_AUX_LO:  thr_aux_lo   <= wd_d[11:0];
                REG_THR_AUX_HI:  thr_aux_hi   <= wd_d[11:0];
                REG_THR_BRAM_LO: thr_bram_lo  <= wd_d[11:0];
                REG_THR_BRAM_HI: thr_bram_hi  <= wd_d[11:0];
                REG_IRQ_STATUS:  irq_status   <= irq_status & ~wd_d[2:0]; // W1C
                REG_IRQ_ENABLE:  irq_enable   <= wd_d[2:0];
                default: ;
            endcase
        end
    end

// tb_xadc_pwr_mon.v — XADC 전원 감시 IP 자체 점검 테스트벤치
// 모의 DRP 모델로 Vccint/Vccaux/Vccbram 판독·임계값 비교·IRQ 발생·W1C 소거 검증.
// 출력: "PASS" 또는 "FAIL <원인>" 로 시작하는 줄.
// Verilog-2001. iverilog 호환.
`timescale 1ns/1ps

// ============================================================
// 모의 DRP 모델: 고정 공급전압 값 반환
// Vccint  = 1.0V → 코드 = 1.0*4096/3 = 1365 = 0x555  → drp_do = 0x5550
// Vccaux  = 1.8V → 코드 = 1.8*4096/3 = 2457 = 0x999  → drp_do = 0x9990
// Vccbram = 1.0V → 코드 = 1.0*4096/3 = 1365 = 0x555  → drp_do = 0x5550
// ============================================================
module xadc_drp_model (
    input  wire        dclk,
    input  wire        den,
    input  wire        dwe,
    input  wire [6:0]  daddr,
    input  wire [15:0] di,
    output reg  [15:0] do_out,
    output reg         drdy
);
    // 외부에서 오버라이드 가능한 채널 값 (12비트 MSB-정렬)
    reg [15:0] val_vccint  = 16'h5550;  // 1.0V
    reg [15:0] val_vccaux  = 16'h9990;  // 1.8V
    reg [15:0] val_vccbram = 16'h5550;  // 1.0V

    always @(posedge dclk) begin
        drdy <= 1'b0;
        if (den && !dwe) begin
            case (daddr)
                7'h01: do_out <= val_vccint;
                7'h02: do_out <= val_vccaux;
                7'h06: do_out <= val_vccbram;
                default: do_out <= 16'h0;
            endcase
            drdy <= 1'b1;  // 1클럭 후 준비
        end
    end
endmodule

module tb_xadc_pwr_mon;

    reg  clk;

    wire [1:0]  bresp;   wire bvalid;  reg  bready;
    reg  [31:0] araddr;  reg  arvalid; wire arready;
    wire [31:0] rdata;   wire [1:0] rresp;  wire rvalid;  reg  rready;

    // DRP
    wire        drp_den, drp_dwe;
    wire [6:0]  drp_daddr;
    wire [15:0] drp_di;
    wire [15:0] drp_do;    // wire: 모의 모델 출력에 연결
    wire        drp_drdy;  // wire: 모의 모델 출력에 연결

    // DUT

        .s_axi_aresetn (resetn),
        .s_axi_awaddr  (awaddr),  .s_axi_awvalid(awvalid), .s_axi_awready(awready),
        .s_axi_wdata   (wdata),   .s_axi_wstrb  (wstrb),   .s_axi_wvalid (wvalid),  .s_axi_wready(wready),
        .s_axi_bresp   (bresp),   .s_axi_bvalid (bvalid),  .s_axi_bready (bready),
        .s_axi_araddr  (araddr),  .s_axi_arvalid(arvalid), .s_axi_arready(arready),
        .s_axi_rdata   (rdata),   .s_axi_rresp  (rresp),   .s_axi_rvalid (rvalid),  .s_axi_rready(rready),
        .drp_den       (drp_den),
        .drp_dwe       (drp_dwe),
        .drp_daddr     (drp_daddr),
        .drp_di        (drp_di),
        .drp_do        (drp_do),
        .drp_drdy      (drp_drdy),
        .irq           (irq)

    // --------------------------------------------------------

    // --------------------------------------------------------

        .dclk   (clk),
        .den    (drp_den),
        .dwe    (drp_dwe),
        .daddr  (drp_daddr),

        .do_out (drp_do),

    initial clk = 0;
    always #5 clk = ~clk;  // 100 MHz

    // --------------------------------------------------------
    // 오류 카운터
    // --------------------------------------------------------
    integer errors;

    task check;
        input [255:0] desc;
        input         cond;
        begin
            if (!cond) begin
                $display("FAIL %0s", desc);
                errors = errors + 1;
            end
        end
    endtask

    // --------------------------------------------------------
    // AXI4-Lite 쓰기 태스크
    // --------------------------------------------------------
    task axil_write;
        input [31:0] addr;
        input [31:0] data;
        begin
            @(posedge clk); #1;
            awaddr = addr; awvalid = 1'b1;
            wdata  = data; wstrb = 4'hF; wvalid = 1'b1;
            // AW+W 동시 핸드셰이크 (RTL이 같은 클럭에 awready/wready를 펄스함)
            // NBA 타이밍: posedge 직후 활성 영역에서는 아직 NBA 미반영.
            // 한 클럭 더 기다리면 두 신호 모두 visible. 분리 루프는 W를 놓침.
            @(posedge clk);
            while (!awready || !wready) @(posedge clk);
            #1;
            awvalid = 1'b0;
            wvalid  = 1'b0;
            // B 핸드셰이크
            bready = 1'b1;
            while (!bvalid) @(posedge clk);
            @(posedge clk); #1;
            bready = 1'b0;
        end
    endtask

    // --------------------------------------------------------
    // AXI4-Lite 읽기 태스크
    // --------------------------------------------------------
    task axil_read;
        input  [31:0] addr;
        output [31:0] data;
        begin
            @(posedge clk); #1;
            araddr = addr; arvalid = 1'b1;
            rready = 1'b1;
            // AR+R 동시 핸드셰이크 (RTL이 같은 클럭에 arready/rvalid를 세트)
            // rready=1이 미리 어서트돼 있어 rvalid는 1클럭 펄스뿐이므로
            // arready/rvalid 를 같은 posedge(NBA 반영 후)에서 확인한다.
            @(posedge clk);
            while (!arready || !rvalid) @(posedge clk);
            data = rdata;    // 활성 영역에서 캡처 (NBA 소거 전)
            #1;
            arvalid = 1'b0;
            @(posedge clk); #1;
            rready = 1'b0;
        end
    endtask

    // --------------------------------------------------------
    // 테스트 시나리오
    // --------------------------------------------------------
    reg [31:0] rd;
    integer    i;

    initial begin
        errors  = 0;
        awaddr  = 0; awvalid = 0;
        wdata   = 0; wstrb = 4'hF; wvalid = 0;
        bready  = 0;
        araddr  = 0; arvalid = 0;
        rready  = 0;
        resetn  = 0;
        repeat(4) @(posedge clk);
        resetn = 1;
        repeat(2) @(posedge clk);

        // ------------------------------------------------
        // TC-1: IP_ID / IP_VER 확인
        // ------------------------------------------------
        axil_read(32'h4016_FFF0, rd);
        check("TC1-IP_ID", rd === 32'h5841_504D);
        axil_read(32'h4016_FFF4, rd);
        check("TC1-IP_VER", rd === 32'h0000_0100);

        // ------------------------------------------------
        // TC-2: 임계값 설정 — 모두 허용 범위 내
        //   Vccint  LO=0x400 HI=0x700  (공칭 0x555 → 정상)
        //   Vccaux  LO=0x800 HI=0xB00  (공칭 0x999 → 정상)
        //   Vccbram LO=0x400 HI=0x700  (공칭 0x555 → 정상)
        // ------------------------------------------------
        axil_write(32'h4016_0014, 32'h400);   // THR_INT_LO
        axil_write(32'h4016_0018, 32'h700);   // THR_INT_HI
        axil_write(32'h4016_001C, 32'h800);   // THR_AUX_LO
        axil_write(32'h4016_0020, 32'hB00);   // THR_AUX_HI
        axil_write(32'h4016_0024, 32'h400);   // THR_BRAM_LO
        axil_write(32'h4016_0028, 32'h700);   // THR_BRAM_HI
        axil_write(32'h4016_FFFC, 32'h7);     // IRQ_ENABLE=[2:0]=1

        // CTRL: en=1, auto=0 (한 번 스캔)
        axil_write(32'h4016_0000, 32'h1);

        // DRP FSM이 세 채널을 읽고 F_CMP까지 완료하기를 기다린다 (~20사이클)
        repeat(30) @(posedge clk);

        // 판독값 확인
        axil_read(32'h4016_0008, rd);
        check("TC2-VCCINT-raw", rd[15:4] === 12'h555);
        axil_read(32'h4016_000C, rd);
        check("TC2-VCCAUX-raw", rd[15:4] === 12'h999);
        axil_read(32'h4016_0010, rd);
        check("TC2-VCCBRAM-raw", rd[15:4] === 12'h555);

        // 임계값 내 → IRQ 없음
        axil_read(32'h4016_FFF8, rd);
        check("TC2-IRQ_STATUS-none", rd[2:0] === 3'b000);
        check("TC2-IRQ-low", irq === 1'b0);

        // ------------------------------------------------
        // TC-3: 저전압 위반 — Vccint를 과낮은 값으로 설정
        //   Vccint 모의 모델을 0x2220 (0x222→0.4V)으로 변경
        //   THR_INT_LO=0x400 이므로 0x222 < 0x400 → 위반
        // ------------------------------------------------
        u_drp.val_vccint = 16'h2220;
        // en=1 다시 토글 (한 번 더 스캔)
        axil_write(32'h4016_0000, 32'h0);
        repeat(2) @(posedge clk);
        axil_write(32'h4016_0000, 32'h1);
        repeat(30) @(posedge clk);

        axil_read(32'h4016_FFF8, rd);
        check("TC3-IRQ_STATUS-vccint", rd[0] === 1'b1);
        check("TC3-IRQ-high", irq === 1'b1);

        // ------------------------------------------------
        // TC-4: IRQ W1C 소거
        // ------------------------------------------------
        axil_write(32'h4016_FFF8, 32'h7);    // 모든 비트 클리어
        repeat(2) @(posedge clk);
        axil_read(32'h4016_FFF8, rd);
        check("TC4-IRQ_STATUS-cleared", rd[2:0] === 3'b000);
        check("TC4-IRQ-low-after-clear", irq === 1'b0);

        // 모의 모델 복원
        u_drp.val_vccint = 16'h5550;

        // ------------------------------------------------
        // TC-5: 고전압 위반 — Vccaux 상한 초과
        //   THR_AUX_HI=0xB00; Vccaux 모의 값을 0xC000 (0xC00=2.5V)
        // ------------------------------------------------
        u_drp.val_vccaux = 16'hC000;
        axil_write(32'h4016_0000, 32'h0);
        repeat(2) @(posedge clk);
        axil_write(32'h4016_0000, 32'h1);
        repeat(30) @(posedge clk);

        axil_read(32'h4016_FFF8, rd);
        check("TC5-IRQ_STATUS-vccaux", rd[1] === 1'b1);

        // 소거 후 최종 상태 클린
        axil_write(32'h4016_FFF8, 32'h7);
        repeat(2) @(posedge clk);

        // ------------------------------------------------
        // 결과 출력
        // ------------------------------------------------
        if (errors === 0)
            $display("PASS: 모든 TC 통과 (TC1~TC5)");
        else
            $display("FAIL: %0d개 체크 실패", errors);

        $finish;
    end

    // 타임아웃
    initial begin
        #500000;
        $display("FAIL: 타임아웃");
        $finish;
    end

endmodule

# Makefile — mmio-xadc-pwr-mon
# 타깃: lint, sim, synth
# 도구: verilator(lint), iverilog+vvp(sim), yosys(synth)

RTL_DIR = rtl
TB_DIR  = tb

TOP     = basys3_top
SIM_TOP = tb_xadc_pwr_mon

RTL_SRCS = \
    $(RTL_DIR)/xadc_pwr_mon_ip.v \
    $(RTL_DIR)/basys3_top.v

SIM_SRCS = \
    $(RTL_DIR)/xadc_pwr_mon_ip.v \
    $(TB_DIR)/tb_xadc_pwr_mon.v

.PHONY: all lint sim synth clean build

all: lint sim synth

# ============================================================
# lint: verilator --lint-only
# XADC 프리미티브 블랙박스로 처리 (-DSIMULATION 없이)
# ============================================================
lint: build
	@echo "=== Lint: verilator ==="
	verilator --lint-only -Wall \
	    --top-module xadc_pwr_mon_ip \
	    $(RTL_DIR)/xadc_pwr_mon_ip.v \
	    && echo "Lint PASS" || echo "Lint FAIL"

# ============================================================
# sim: iverilog + vvp (모의 DRP 모델 포함)
# ============================================================
sim: build
	@echo "=== Sim: $(SIM_TOP) ==="
	iverilog -g2001 -Wall \
	    -DSIMULATION \
	    -o build/$(SIM_TOP).vvp \
	    $(SIM_SRCS) \
	    && vvp build/$(SIM_TOP).vvp \
	    && echo "Sim PASS" || echo "Sim FAIL"

# ============================================================
# synth: yosys (Xilinx 대상, XADC 프리미티브 블랙박스)
# ============================================================
synth: build
	@echo "=== Synth: yosys ==="
	yosys -q -p " \
	    read_verilog $(RTL_DIR)/xadc_pwr_mon_ip.v; \
	    synth_xilinx -top xadc_pwr_mon_ip -flatten; \
	    stat" \
	    && echo "Synth PASS" || echo "Synth FAIL"

# ============================================================
# clean / build
# ============================================================
clean:
	rm -f build/*.vvp build/*.vcd

build:
	mkdir -p build

시뮬레이션 (iverilog)

  1. make lint — xadc_pwr_mon_ip 모듈 verilator 린트
  2. make sim — 모의 DRP 모델로 TC1~TC5 자체 점검; PASS 확인
  3. make synth — yosys로 xadc_pwr_mon_ip 단독 합성·자원 추정

Vivado 연결 (MicroBlaze V SoC)

  1. xadc_pwr_mon_ip.v를 프로젝트에 추가한다.
  2. AXI4-Lite 버스 주소 슬롯 [16] = 0x4016_0000으로 연결한다.
  3. irq를 INTC 채널 13에 연결한다.
  4. XADC 프리미티브를 basys3_top.v 패턴으로 직접 연결하거나 IP Catalog의 XADC Wizard를 사용해 DRP 포트를 노출한다.
  5. 합성 후 LUT/FF/BRAM과 WNS를 기록한다.

위험·한계

  • AMD XADC Wizard IP와 병행 사용 불가: DRP 버스 충돌. 하나만 선택
  • auto 모드(CTRL[1]=1)는 1024클럭마다 스캔: 판독 간격 약 10µs. 빠른 과도 현상은 감지 못할 수 있음
