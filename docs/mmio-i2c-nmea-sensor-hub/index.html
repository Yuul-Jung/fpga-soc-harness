I2C 3장치 순회 + NMEA UART 수신 센서 허브 IP

I2C 마스터가 MPU-6050(0x68), BMP280(0x76), HMC5883L(0x1E) 세 장치를 100 ms 주기로 순회 읽기하고, UART 수신기가 9600 baud NMEA 문장(GGA 계열) 1줄 완료를 인터럽트로 알리며, 모든 표본에 자유 실행 32비트 카운터 타임스탬프를 붙여 깊이 16 FIFO에 쌓는 AXI4-Lite 슬레이브 허브 IP. 보정 계산과 NMEA 필드 추출은 펌웨어 담당.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • I2C 마스터 FSM이 주소 테이블(0x68·0x76·0x1E)을 순회하며 센서 원시값 6 바이트씩 읽기
  • UART 수신기(9600 baud, 100 MHz)가 '\n' 종단 NMEA 문장 한 줄 완료 시 인터럽트 발생
  • 모든 표본(I2C 순회 1회, UART 1문장)에 타임스탬프 붙여 FIFO(깊이 16) 저장
  • AXI4-Lite 레지스터로 샘플 팝·인터럽트 클리어·FIFO 상태 노출
  • iverilog 시뮬레이션 PASS, yosys 합성 성공(S2)

범위 밖

  • 센서 보정·칼만 필터·NMEA 필드 파싱은 펌웨어 담당
  • I2C 슬레이브 모드, 멀티마스터, 클럭 스트레칭 미지원
  • Vivado AXI-IIC IP 미사용(순수 RTL)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-036(§13.7)로 생성됨 (2026-11-05). I2C 비트뱅 마스터가 MPU-6050·BMP280·HMC5883L 세 장치를 100 ms 주기로 순회하고, NMEA UART 수신기가 '\n' 종단 GPS 문장 완료를 인터럽트로 알리며, 모든 샘플에 32비트 타임스탬프를 붙여 깊이 16 FIFO에 저장하는 AXI4-Lite 슬레이브 허브 IP. 기존 mmio-sensor-hub-checker(HC-SR04·DHT11)와 구별: 이 IP는 I2C 3장치 순회 + NMEA UART 수신 조합.

rtl/axil_i2c_nmea_hub.v 300줄
// axil_i2c_nmea_hub.v
// AXI4-Lite slave: I2C 3-device cycling + NMEA UART sensor hub
// Board: Basys3 (100 MHz), Verilog-2001
// Part of capstone mmio-i2c-nmea-sensor-hub

`timescale 1ns/1ps

module axil_i2c_nmea_hub #(
    // AXI4-Lite
    parameter ADDR_W  = 8,
    parameter DATA_W  = 32,
    // defaults
    parameter I2C_PERIOD = 32'd10_000_000,  // 100 ms @ 100 MHz
    parameter BAUD_DIV   = 16'd10417        // 9600 baud @ 100 MHz
) (
    // AXI4-Lite clock & reset
    input  wire        clk,
    input  wire        rst_n,       // synchronous, active-low

    // AXI4-Lite write address channel
    input  wire [ADDR_W-1:0] s_awaddr,
    input  wire              s_awvalid,
    output reg               s_awready,

    // AXI4-Lite write data channel
    input  wire [DATA_W-1:0] s_wdata,
    input  wire [3:0]        s_wstrb,
    input  wire              s_wvalid,
    output reg               s_wready,

    // AXI4-Lite write response channel
    output reg  [1:0]        s_bresp,
    output reg               s_bvalid,
    input  wire              s_bready,

    // AXI4-Lite read address channel
    input  wire [ADDR_W-1:0] s_araddr,
    input  wire              s_arvalid,
    output reg               s_arready,

    // AXI4-Lite read data channel
    output reg  [DATA_W-1:0] s_rdata,
    output reg  [1:0]        s_rresp,
    output reg               s_rrvalid,
    input  wire              s_rready,

    // I2C interface (open-drain, external pull-ups required)
    output wire        i2c_scl_o,
    output wire        i2c_scl_oe,  // 1=drive low
    output wire        i2c_sda_o,
    output wire        i2c_sda_oe,  // 1=drive low
    input  wire        i2c_sda_i,

    // UART RX (9600 baud NMEA)
    input  wire        uart_rx,

    // Interrupt output (active-high, level)
    output wire        irq
);

    // ---------------------------------------------------------------
    // Register addresses
    // ---------------------------------------------------------------
    localparam REG_ID          = 8'h00;
    localparam REG_CTRL        = 8'h04;
    localparam REG_STATUS      = 8'h08;
    localparam REG_IRQ_STATUS  = 8'h0C;
    localparam REG_I2C_CFG     = 8'h10;
    localparam REG_FIFO_HI     = 8'h14;
    localparam REG_FIFO_LO     = 8'h18;
    localparam REG_UART_CFG    = 8'h1C;

    localparam ID_VAL = 32'h4B485542; // 'KHUB'

    // ---------------------------------------------------------------
    // Internal registers
    // ---------------------------------------------------------------
    reg         ctrl_en;
    reg         ctrl_ie;
    reg         ctrl_fifo_clr;

    reg  [31:0] irq_status;   // [0]=FIFO_HF, [1]=UART_LINE_DONE
    reg  [31:0] i2c_cfg;      // period counter threshold
    reg  [15:0] uart_cfg;     // baud divisor

    // Write address capture
    reg  [ADDR_W-1:0] waddr_lat;
    reg               waddr_valid;

    // ---------------------------------------------------------------
    // Sub-module wires
    // ---------------------------------------------------------------
    // I2C FSM
    wire        i2c_busy;
    wire [7:0]  i2c_dev_type;    // which device sample
    /* verilator lint_off UNUSEDSIGNAL */
    wire [47:0] i2c_raw_data;    // 6 bytes from sensor; only [31:0] pushed to FIFO
    /* verilator lint_on UNUSEDSIGNAL */
    wire        i2c_sample_valid;

    // UART NMEA RX
    wire        uart_line_rdy;
    wire        uart_line_done;  // one-cycle pulse when line complete

    // FIFO
    reg         fifo_push;
    reg  [71:0] fifo_din;        // [71:40]=timestamp, [39:32]=type, [31:0]=data_lo
    wire        fifo_pop;
    wire [71:0] fifo_dout;
    wire [4:0]  fifo_cnt;
    wire        fifo_full;
    wire        fifo_empty;

    // Timestamp counter
    reg  [31:0] ts_cnt;

    // ---------------------------------------------------------------
    // Timestamp free-running counter
    // ---------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n) ts_cnt <= 32'd0;
        else        ts_cnt <= ts_cnt + 1'b1;
    end

    // ---------------------------------------------------------------
    // Sub-module instantiations
    // ---------------------------------------------------------------
    i2c_cycle_fsm u_i2c (
        .clk            (clk),
        .rst_n          (rst_n),
        .en             (ctrl_en),
        .period_cfg     (i2c_cfg),
        .scl_o          (i2c_scl_o),
        .scl_oe         (i2c_scl_oe),
        .sda_o          (i2c_sda_o),
        .sda_oe         (i2c_sda_oe),
        .sda_i          (i2c_sda_i),
        .busy           (i2c_busy),
        .dev_type       (i2c_dev_type),
        .raw_data       (i2c_raw_data),
        .sample_valid   (i2c_sample_valid)
    );

    nmea_uart_rx u_uart (
        .clk            (clk),
        .rst_n          (rst_n),
        .en             (ctrl_en),
        .baud_div       (uart_cfg),
        .rx             (uart_rx),
        .line_rdy       (uart_line_rdy),
        .line_done_pulse(uart_line_done)
    );

    ts_fifo #(
        .DEPTH(16),
        .DWIDTH(72)
    ) u_fifo (
        .clk            (clk),
        .rst_n          (rst_n & ~ctrl_fifo_clr),
        .push           (fifo_push),
        .din            (fifo_din),
        .pop            (fifo_pop),
        .dout           (fifo_dout),
        .cnt            (fifo_cnt),
        .full           (fifo_full),
        .empty          (fifo_empty)
    );

    // ---------------------------------------------------------------
    // FIFO push logic
    // ---------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n) begin
            fifo_push <= 1'b0;
            fifo_din  <= 72'd0;
        end else if (i2c_sample_valid && !fifo_full) begin
            fifo_push <= 1'b1;
            fifo_din  <= {ts_cnt, i2c_dev_type, i2c_raw_data[31:0]};
        end else if (uart_line_done && !fifo_full) begin
            fifo_push <= 1'b1;
            fifo_din  <= {ts_cnt, 8'hFF, 24'h0, 8'h01}; // type=0xFF=NMEA
        end else begin
            fifo_push <= 1'b0;
        end
    end

    // ---------------------------------------------------------------
    // IRQ status
    // ---------------------------------------------------------------
    wire fifo_hf = (fifo_cnt >= 5'd8);

    assign irq = ctrl_ie & (|irq_status);

    // ---------------------------------------------------------------
    // AXI4-Lite write channel (simple 1-cycle ready)
    // ---------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n) begin
            s_awready    <= 1'b0;
            s_wready     <= 1'b0;
            s_bvalid     <= 1'b0;
            s_bresp      <= 2'b00;
            waddr_lat    <= {ADDR_W{1'b0}};
            waddr_valid  <= 1'b0;
            ctrl_en      <= 1'b0;
            ctrl_ie      <= 1'b0;
            ctrl_fifo_clr<= 1'b0;
            i2c_cfg      <= I2C_PERIOD;
            uart_cfg     <= BAUD_DIV;
            irq_status   <= 32'd0;
        end else begin
            // auto-clear fifo_clr after one cycle
            ctrl_fifo_clr <= 1'b0;
            // latch write address
            s_awready <= 1'b0;
            if (s_awvalid && !waddr_valid) begin
                s_awready  <= 1'b1;
                waddr_lat  <= s_awaddr;
                waddr_valid<= 1'b1;
            end

            // latch write data and perform write
            s_wready <= 1'b0;
            if (s_wvalid && waddr_valid) begin
                s_wready    <= 1'b1;
                waddr_valid <= 1'b0;
                s_bvalid    <= 1'b1;
                s_bresp     <= 2'b00;
                case (waddr_lat)
                    REG_CTRL: begin
                        ctrl_en       <= s_wdata[0];
                        ctrl_ie       <= s_wdata[1];
                        ctrl_fifo_clr <= s_wdata[2];
                    end
                    REG_IRQ_STATUS: begin
                        // W1C
                        irq_status <= irq_status & ~s_wdata;
                    end
                    REG_I2C_CFG:  i2c_cfg  <= s_wdata;
                    REG_UART_CFG: uart_cfg <= s_wdata[15:0];
                    default: ; // read-only registers ignore write
                endcase
            end

            if (s_bvalid && s_bready) s_bvalid <= 1'b0;

            // Hardware-set sticky IRQ bits (single driver for irq_status)
            if (fifo_hf)        irq_status[0] <= 1'b1;
            if (uart_line_done) irq_status[1] <= 1'b1;
        end
    end

    // ---------------------------------------------------------------
    // AXI4-Lite read channel
    // ---------------------------------------------------------------
    // FIFO pop on FIFO_DATA_LO read
    reg  fifo_pop_r;
    reg  [31:0] fifo_dout_latch;   // stores lower 40 bits of FIFO entry as 32-bit [39:8] dropped; see REG_FIFO_LO
    assign fifo_pop = fifo_pop_r;

    always @(posedge clk) begin
        if (!rst_n) begin
            s_arready    <= 1'b0;
            s_rrvalid    <= 1'b0;
            s_rdata      <= 32'd0;
            s_rresp      <= 2'b00;
            fifo_pop_r   <= 1'b0;
            fifo_dout_latch <= 32'd0;
        end else begin
            s_arready  <= 1'b0;
            fifo_pop_r <= 1'b0;

            if (s_arvalid && !s_rrvalid) begin
                s_arready <= 1'b1;
                s_rrvalid <= 1'b1;
                s_rresp   <= 2'b00;
                case (s_araddr)
                    REG_ID:         s_rdata <= ID_VAL;
                    REG_CTRL:       s_rdata <= {29'd0, ctrl_fifo_clr, ctrl_ie, ctrl_en};
                    REG_STATUS:     s_rdata <= {24'd0, uart_line_rdy, i2c_busy, fifo_full, fifo_cnt};
                    REG_IRQ_STATUS: s_rdata <= irq_status;
                    REG_I2C_CFG:    s_rdata <= i2c_cfg;
                    REG_FIFO_HI: begin
                        fifo_dout_latch <= fifo_dout[31:0];  // latch lower 32 bits for FIFO_LO read
                        s_rdata <= fifo_dout[71:40];         // return timestamp (top 32 bits)
                    end
                    REG_FIFO_LO: begin
                        s_rdata    <= fifo_dout_latch;       // {type[7:0], data_lo[23:0]} from FIFO_HI latch
                        fifo_pop_r <= !fifo_empty;
                    end
                    REG_UART_CFG:   s_rdata <= {16'd0, uart_cfg};
                    default:        s_rdata <= 32'hDEAD_BEEF;
                endcase
            end

            if (s_rrvalid && s_rready) s_rrvalid <= 1'b0;
        end
    end

endmodule

// i2c_cycle_fsm.v
// I2C bit-bang master FSM: cycles through 3 device addresses
// Devices: MPU-6050(0x68), BMP280(0x76), HMC5883L(0x1E)
// Reads 6 bytes from each device register 0x3B (or device default start)
// 100 kHz I2C @ 100 MHz clock (each bit = 1000 cycles, 250 per phase)
// Verilog-2001

`timescale 1ns/1ps

module i2c_cycle_fsm (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        en,
    input  wire [31:0] period_cfg,

    // I2C open-drain interface
    output reg         scl_o,
    output reg         scl_oe,   // 1=drive SCL low
    output reg         sda_o,
    output reg         sda_oe,   // 1=drive SDA low
    input  wire        sda_i,

    // Status
    output reg         busy,

    // Sample output
    output reg  [7:0]  dev_type,      // 0=MPU-6050, 1=BMP280, 2=HMC5883L
    output reg  [47:0] raw_data,      // 6 bytes
    output reg         sample_valid   // one-cycle pulse
);

    // Device address table: 3 devices × 7-bit I2C address
    // MPU-6050: 0x68, BMP280: 0x76, HMC5883L: 0x1E
    // Data register to start read: 0x3B for MPU, 0xF7 for BMP, 0x03 for HMC
    localparam [6:0] ADDR_MPU  = 7'h68;
    localparam [6:0] ADDR_BMP  = 7'h76;
    localparam [6:0] ADDR_HMC  = 7'h1E;
    localparam [7:0] REG_MPU   = 8'h3B;
    localparam [7:0] REG_BMP   = 8'hF7;
    localparam [7:0] REG_HMC   = 8'h03;

    // I2C bit timing: 100 kHz = 1000 cycles/bit at 100 MHz
    // 4 phases per bit: 250 cycles each
    localparam CLK_PER_PHASE = 10'd250;

    // FSM states
    localparam [4:0]
        S_IDLE       = 5'd0,
        S_WAIT       = 5'd1,
        S_START      = 5'd2,
        S_ADDR_W     = 5'd3,  // send addr+W (7 bit + 0)
        S_ACK_AW     = 5'd4,
        S_REG_BYTE   = 5'd5,
        S_ACK_REG    = 5'd6,
        S_RESTART    = 5'd7,
        S_ADDR_R     = 5'd8,  // send addr+R (7 bit + 1)
        S_ACK_AR     = 5'd9,
        S_DATA_BYTE  = 5'd10,
        S_MASTER_ACK = 5'd11,
        S_STOP       = 5'd12,
        S_PUSH       = 5'd13;

    reg [4:0]  state;
    reg [31:0] period_cnt;      // cycle timer
    reg [9:0]  phase_cnt;       // sub-bit phase counter
    reg [1:0]  phase;           // 0=SCL-low, 1=SCL-rise, 2=SCL-high, 3=SCL-fall
    reg [3:0]  bit_cnt;         // bits sent/received in current byte
    reg [2:0]  byte_cnt;        // bytes received (0-5)
    reg [1:0]  dev_idx;         // 0,1,2 device index
    reg [7:0]  tx_byte;         // byte being shifted out
    reg [7:0]  rx_byte;         // byte being shifted in
    reg [47:0] data_buf;        // accumulate 6 bytes

    // Combinational: current device address and register
    reg [6:0] cur_addr;
    reg [7:0] cur_reg;

    always @(*) begin
        case (dev_idx)
            2'd0: begin cur_addr = ADDR_MPU; cur_reg = REG_MPU; end
            2'd1: begin cur_addr = ADDR_BMP; cur_reg = REG_BMP; end
            default: begin cur_addr = ADDR_HMC; cur_reg = REG_HMC; end
        endcase
    end

    // Main FSM (drives SCL/SDA open-drain)
    always @(posedge clk) begin
        if (!rst_n) begin
            state        <= S_IDLE;
            period_cnt   <= 32'd0;
            phase_cnt    <= 10'd0;
            phase        <= 2'd0;
            bit_cnt      <= 4'd0;
            byte_cnt     <= 3'd0;
            dev_idx      <= 2'd0;
            busy         <= 1'b0;
            sample_valid <= 1'b0;
            dev_type     <= 8'd0;
            raw_data     <= 48'd0;
            data_buf     <= 48'd0;
            tx_byte      <= 8'd0;
            rx_byte      <= 8'd0;
            scl_o        <= 1'b1;
            scl_oe       <= 1'b0;
            sda_o        <= 1'b1;
            sda_oe       <= 1'b0;
        end else begin
            sample_valid <= 1'b0;

            case (state)
                S_IDLE: begin
                    busy       <= 1'b0;
                    scl_oe     <= 1'b0;
                    sda_oe     <= 1'b0;
                    period_cnt <= 32'd0;
                    if (en) state <= S_WAIT;
                end

                S_WAIT: begin
                    if (!en) begin
                        state <= S_IDLE;
                    end else if (period_cnt >= period_cfg) begin
                        period_cnt <= 32'd0;
                        state      <= S_START;
                        busy       <= 1'b1;
                        phase_cnt  <= 10'd0;
                        phase      <= 2'd0;
                        byte_cnt   <= 3'd0;
                        data_buf   <= 48'd0;
                    end else begin
                        period_cnt <= period_cnt + 1'b1;
                    end
                end

                // START condition: SDA falls while SCL high
                S_START: begin
                    sda_oe <= 1'b1;  // pull SDA low
                    scl_oe <= 1'b0;  // SCL stays high
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        scl_oe    <= 1'b1; // pull SCL low
                        state     <= S_ADDR_W;
                        bit_cnt   <= 4'd0;
                        tx_byte   <= {cur_addr, 1'b0}; // addr + Write
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                // Send 8-bit byte (MSB first), then read ACK
                S_ADDR_W,
                S_REG_BYTE,
                S_ADDR_R: begin
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        phase     <= phase + 1'b1;
                        case (phase)
                            2'd0: begin // SCL low: set SDA
                                sda_oe <= ~tx_byte[7];
                                tx_byte <= {tx_byte[6:0], 1'b0};
                            end
                            2'd1: scl_oe <= 1'b0; // SCL rise
                            2'd2: ; // SCL high: hold
                            2'd3: begin // SCL fall
                                scl_oe <= 1'b1;
                                bit_cnt <= bit_cnt + 1'b1;
                                if (bit_cnt == 4'd7) begin
                                    // move to ACK state
                                    sda_oe <= 1'b0; // release SDA for ACK
                                    case (state)
                                        S_ADDR_W: state <= S_ACK_AW;
                                        S_REG_BYTE: state <= S_ACK_REG;
                                        S_ADDR_R:   state <= S_ACK_AR;
                                        default: state <= S_IDLE;
                                    endcase
                                    bit_cnt <= 4'd0;
                                end
                            end
                        endcase
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                // ACK: read SDA on SCL high
                S_ACK_AW,
                S_ACK_REG,
                S_ACK_AR: begin
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        phase     <= phase + 1'b1;
                        case (phase)
                            2'd1: scl_oe <= 1'b0; // SCL rise
                            2'd2: ; // read ACK (sda_i should be 0 for ACK)
                            2'd3: begin
                                scl_oe <= 1'b1;
                                // proceed regardless of NACK (best-effort)
                                case (state)
                                    S_ACK_AW: begin
                                        state   <= S_REG_BYTE;
                                        tx_byte <= cur_reg;
                                        bit_cnt <= 4'd0;
                                    end
                                    S_ACK_REG: state <= S_RESTART;
                                    S_ACK_AR: begin
                                        state    <= S_DATA_BYTE;
                                        byte_cnt <= 3'd0;
                                        rx_byte  <= 8'd0;
                                        bit_cnt  <= 4'd0;
                                        sda_oe   <= 1'b0;
                                    end
                                    default: state <= S_STOP;
                                endcase
                            end
                            default: ;
                        endcase
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                // Repeated START
                S_RESTART: begin
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        phase     <= phase + 1'b1;
                        case (phase)
                            2'd0: sda_oe <= 1'b0; // SDA high (release)
                            2'd1: scl_oe <= 1'b0; // SCL high
                            2'd2: sda_oe <= 1'b1; // SDA falls (START)
                            2'd3: begin
                                scl_oe  <= 1'b1;  // SCL low
                                state   <= S_ADDR_R;
                                tx_byte <= {cur_addr, 1'b1}; // addr + Read
                                bit_cnt <= 4'd0;
                            end
                        endcase
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                // Receive 6 data bytes
                S_DATA_BYTE: begin
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        phase     <= phase + 1'b1;
                        case (phase)
                            2'd0: ; // SCL low
                            2'd1: scl_oe <= 1'b0; // SCL rise
                            2'd2: begin // sample SDA
                                rx_byte <= {rx_byte[6:0], sda_i};
                            end
                            2'd3: begin // SCL fall
                                scl_oe  <= 1'b1;
                                bit_cnt <= bit_cnt + 1'b1;
                                if (bit_cnt == 4'd7) begin
                                    // Store byte
                                    data_buf <= {data_buf[39:0], rx_byte};
                                    rx_byte  <= 8'd0;
                                    bit_cnt  <= 4'd0;
                                    state    <= S_MASTER_ACK;
                                end
                            end
                        endcase
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                // Send ACK (after bytes 0-4) or NACK (after byte 5)
                S_MASTER_ACK: begin
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        phase     <= phase + 1'b1;
                        case (phase)
                            2'd0: begin
                                // ACK=0 (pull SDA), NACK=1 (release) for last byte
                                sda_oe <= (byte_cnt < 3'd5) ? 1'b1 : 1'b0;
                            end
                            2'd1: scl_oe <= 1'b0;
                            2'd2: ;
                            2'd3: begin
                                scl_oe   <= 1'b1;
                                byte_cnt <= byte_cnt + 1'b1;
                                if (byte_cnt == 3'd5) begin
                                    state <= S_STOP;
                                end else begin
                                    state <= S_DATA_BYTE;
                                end
                            end
                        endcase
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                // STOP condition: SDA rises while SCL high
                S_STOP: begin
                    if (phase_cnt >= CLK_PER_PHASE) begin
                        phase_cnt <= 10'd0;
                        phase     <= phase + 1'b1;
                        case (phase)
                            2'd0: sda_oe <= 1'b1; // SDA low
                            2'd1: scl_oe <= 1'b0; // SCL high
                            2'd2: sda_oe <= 1'b0; // SDA rises (STOP)
                            2'd3: state  <= S_PUSH;
                        endcase
                    end else begin
                        phase_cnt <= phase_cnt + 1'b1;
                    end
                end

                S_PUSH: begin
                    raw_data     <= data_buf;
                    dev_type     <= {6'd0, dev_idx};  // 6+2=8 bits, matches output reg [7:0]
                    sample_valid <= 1'b1;
                    dev_idx      <= (dev_idx == 2'd2) ? 2'd0 : dev_idx + 1'b1;
                    state        <= S_WAIT;
                    busy         <= 1'b0;
                end

                default: state <= S_IDLE;
            endcase
        end
    end

endmodule

// nmea_uart_rx.v
// UART receiver for NMEA sentences (terminated by '\n' = 0x0A)
// 9600 baud N-8-1, configurable baud divisor
// Signals one-cycle pulse line_done_pulse when '\n' received
// Verilog-2001

`timescale 1ns/1ps

module nmea_uart_rx (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        en,
    input  wire [15:0] baud_div,

    input  wire        rx,

    output reg         line_rdy,        // 1 when last line is complete
    output reg         line_done_pulse  // one-cycle pulse on '\n' detection
);

    // -----------------------------------------------------------------------
    // 2-stage synchronizer for RX input
    // -----------------------------------------------------------------------
    reg rx_s0, rx_s1, rx_s2;
    always @(posedge clk) begin
        if (!rst_n) begin
            rx_s0 <= 1'b1; rx_s1 <= 1'b1; rx_s2 <= 1'b1;
        end else begin
            rx_s0 <= rx;
            rx_s1 <= rx_s0;
            rx_s2 <= rx_s1;
        end
    end

    // -----------------------------------------------------------------------
    // UART receive FSM
    // -----------------------------------------------------------------------
    localparam [2:0]
        RS_IDLE  = 3'd0,
        RS_START = 3'd1,
        RS_DATA  = 3'd2,
        RS_STOP  = 3'd3;

    reg [2:0]  rx_state;
    reg [15:0] rx_baud_cnt;
    reg [3:0]  rx_bit_cnt;
    reg [7:0]  rx_shift;
    reg [7:0]  rx_byte;
    reg        rx_byte_valid;

    always @(posedge clk) begin
        if (!rst_n || !en) begin
            rx_state     <= RS_IDLE;
            rx_baud_cnt  <= 16'd0;
            rx_bit_cnt   <= 4'd0;
            rx_shift     <= 8'd0;
            rx_byte      <= 8'd0;
            rx_byte_valid<= 1'b0;
        end else begin
            rx_byte_valid <= 1'b0;

            case (rx_state)
                RS_IDLE: begin
                    // Detect falling edge (start bit)
                    if (!rx_s2) begin
                        rx_state    <= RS_START;
                        rx_baud_cnt <= 16'd0;
                        rx_bit_cnt  <= 4'd0;
                    end
                end

                RS_START: begin
                    // Wait half-bit to sample in center of start bit
                    rx_baud_cnt <= rx_baud_cnt + 1'b1;
                    if (rx_baud_cnt >= (baud_div >> 1) - 1'b1) begin
                        rx_baud_cnt <= 16'd0;
                        if (!rx_s2) begin
                            rx_state <= RS_DATA;
                        end else begin
                            rx_state <= RS_IDLE; // glitch
                        end
                    end
                end

                RS_DATA: begin
                    rx_baud_cnt <= rx_baud_cnt + 1'b1;
                    if (rx_baud_cnt >= baud_div - 1'b1) begin
                        rx_baud_cnt <= 16'd0;
                        rx_shift    <= {rx_s2, rx_shift[7:1]}; // LSB first
                        rx_bit_cnt  <= rx_bit_cnt + 1'b1;
                        if (rx_bit_cnt == 4'd7) begin
                            rx_state <= RS_STOP;
                        end
                    end
                end

                RS_STOP: begin
                    rx_baud_cnt <= rx_baud_cnt + 1'b1;
                    if (rx_baud_cnt >= baud_div - 1'b1) begin
                        rx_baud_cnt  <= 16'd0;
                        rx_state     <= RS_IDLE;
                        if (rx_s2) begin // valid stop bit
                            rx_byte       <= rx_shift;
                            rx_byte_valid <= 1'b1;
                        end
                    end
                end

                default: rx_state <= RS_IDLE;
            endcase
        end
    end

    // -----------------------------------------------------------------------
    // Line detection: pulse on '\n' (0x0A)
    // -----------------------------------------------------------------------
    always @(posedge clk) begin
        if (!rst_n || !en) begin
            line_rdy        <= 1'b0;
            line_done_pulse <= 1'b0;
        end else begin
            line_done_pulse <= 1'b0;
            if (rx_byte_valid) begin
                if (rx_byte == 8'h0A) begin // '\n'
                    line_rdy        <= 1'b1;
                    line_done_pulse <= 1'b1;
                end
                // Reset line_rdy on next byte (new line starts)
                if (rx_byte == 8'h24) begin // '$' = NMEA sentence start
                    line_rdy <= 1'b0;
                end
            end
        end
    end

endmodule

        axi_read(8'h08, rd_val);

        check(rd_val[4:0] === 5'd0, 2, rd_val, 32'd0);

        // Write I2C_CFG with small period (already set by parameter)
        axi_write(8'h10, I2C_PERIOD_SIM);
        // Enable hub (ctrl_en=1, ie=0)

        repeat (100_000) @(posedge clk);  // i2c_cycle_fsm bit-phase=250 cycles → full txn ≈84k cycles
        // Check FIFO has at least 1 entry
        axi_read(8'h08, rd_val);
        // fifo_cnt should be >= 1 (at least one I2C device was polled)
        check(rd_val[4:0] >= 5'd1, 3, rd_val[4:0], 5'd1);

        // ---- T4: UART NMEA line injection ----
        // Send "$GPGGA,..." terminated by '\n'
        uart_send_byte(8'h24); // '$'
        uart_send_byte(8'h47); // 'G'
        uart_send_byte(8'h50); // 'P'
        uart_send_byte(8'h0A); // '\n'
        // Wait for processing
        repeat (50) @(posedge clk);
        // Check IRQ status bit 1 (UART_LINE_DONE) or FIFO has more entries
        axi_read(8'h0C, rd_val);
        check(rd_val[1] === 1'b1, 4, rd_val, 32'h2);

        // ---- T5: FIFO pop — read FIFO_HI then FIFO_LO ----
        axi_read(8'h08, rd_val);
        // cnt should be >= 1
        check(rd_val[4:0] >= 5'd1, 51, rd_val[4:0], 5'd1);

        axi_read(8'h14, rd_val); // FIFO_HI (timestamp)
        // timestamp should be non-zero (clock has advanced)
        check(rd_val !== 32'd0, 52, rd_val, 32'hXXXXXXXX);

        axi_read(8'h18, rd_val); // FIFO_LO (pops FIFO)
        // just check it doesn't return 0xDEADBEEF
        check(rd_val !== 32'hDEAD_BEEF, 53, rd_val, 32'h0);

        // ---- T6: FIFO CLR ----
        // Fill some entries first (already have some from above)
        // Write CTRL with FIFO_CLR=1
        axi_write(8'h04, 32'h5); // EN=1, FIFO_CLR=1
        repeat (4) @(posedge clk);
        axi_read(8'h08, rd_val);
        // After CLR, FIFO cnt should be 0
        check(rd_val[4:0] === 5'd0, 6, rd_val[4:0], 5'd0);

        // ---- Final result ----
        repeat (10) @(posedge clk);
        if (fail_count === 0)
            $display("PASS: all tests passed");
        else
            $display("FAIL: %0d test(s) failed", fail_count);
        $finish;
    end

    // Timeout watchdog
    initial begin
        #5_000_000;
        $display("FAIL: timeout");
        $finish;
    end

endmodule

## Makefile for mmio-i2c-nmea-sensor-hub
## Targets: lint, sim, synth

RTL_FILES = rtl/axil_i2c_nmea_hub.v rtl/i2c_cycle_fsm.v rtl/nmea_uart_rx.v rtl/ts_fifo.v
TB_FILES  = tb/tb_i2c_nmea_hub.v
TOP_SIM   = tb_i2c_nmea_hub
TOP_SYNTH = axil_i2c_nmea_hub
BUILD_DIR = build

.PHONY: all lint sim synth clean

all: lint sim synth

lint:
	@mkdir -p $(BUILD_DIR)
	verilator --lint-only -Wall --Wno-UNUSEDSIGNAL --top-module $(TOP_SYNTH) $(RTL_FILES) 2>&1 | tee $(BUILD_DIR)/lint.log
	@grep -E "^%Error" $(BUILD_DIR)/lint.log && exit 1 || true

sim:
	@mkdir -p $(BUILD_DIR)
	iverilog -g2001 -o $(BUILD_DIR)/sim_$(TOP_SIM).vvp \
		$(RTL_FILES) $(TB_FILES) -s $(TOP_SIM) 2>&1 | tee $(BUILD_DIR)/sim_compile.log
	vvp $(BUILD_DIR)/sim_$(TOP_SIM).vvp | tee $(BUILD_DIR)/sim_output.log
	@grep -q "^PASS" $(BUILD_DIR)/sim_output.log || (echo "SIMULATION FAILED" && exit 1)

synth:
	@mkdir -p $(BUILD_DIR)
	@echo "read_verilog $(RTL_FILES)" > $(BUILD_DIR)/synth.ys
	@echo "synth -top $(TOP_SYNTH) -flatten" >> $(BUILD_DIR)/synth.ys
	@echo "stat" >> $(BUILD_DIR)/synth.ys
	yosys -Q -l $(BUILD_DIR)/synth.log $(BUILD_DIR)/synth.ys
	@echo "Synth complete. See $(BUILD_DIR)/synth.log for resource usage."

clean:
	rm -rf $(BUILD_DIR)
