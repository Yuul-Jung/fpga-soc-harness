CTRL/DATA형 I2C 마스터 IP

단일 바이트 CTRL/DATA 레지스터 인터페이스로 I2C 마스터 트랜잭션을 제어하는 Basys3 mmio-lite IP. MPU-6050 WHO_AM_I 레지스터 읽기 시뮬레이션(가짜 슬레이브 모델)으로 검증.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • mmio-lite 버스 위에서 I2C 100 kHz 마스터 단일 바이트 쓰기/읽기 트랜잭션 제공
  • CTRL 레지스터(GO 비트 자기 소거, BUSY·NAK 플래그)와 DATA 레지스터로 CPU 없이 FSM에서도 구동 가능
  • IP 표준 Rev.1 관리 레지스터(IP_ID·VER·IRQ_STATUS·IRQ_ENABLE) 준수
  • iverilog 시뮬레이션에서 가짜 MPU-6050 슬레이브와 WHO_AM_I(0x68) 읽기 PASS 확인

범위 밖

  • 멀티 바이트 버스트 트랜잭션 (한 번의 GO로 여러 바이트 전송)
  • 클럭 스트레칭 (SCL 모니터링 없음)
  • 400 kHz / 1 MHz Fast/Fast+ 속도
  • AXI4-Lite 버스 인터페이스 (mmio-lite 전용)
  • 보드 동작 확인 (S3)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. mmio-lite 버스(AXI4-Lite 비호환, 시뮬 전용). 가짜 MPU-6050 슬레이브를 포함한 자체 점검 테스트벤치로 WHO_AM_I 읽기를 검증한다.

5. 외부 부품과 전기 회로도

2절의 계통도는 디지털 모듈 연결도다. 전압, 풀업, 커넥터 배선 같은 전기 사항은 이 절에서 따로 다룬다. 기계가 읽는 사본: bom.csv · connections.json

전기 회로도: 미작성

오픈드레인 버스 + 풀업 2개만 필요. 별도 회로도 미작성.

id부품인터페이스신호 전압구분레벨 변환비고
ext_pullup_sdaSDA 풀업 저항 4.7 kΩI2C3.3 V외부 모듈불필요 (3.3V)오픈드레인 버스 풀업. 시뮬레이션에서는 불필요.
ext_pullup_sclSCL 풀업 저항 4.7 kΩI2C3.3 V외부 모듈불필요 (3.3V)오픈드레인 버스 풀업. 시뮬레이션에서는 불필요.

Basys3 FPGA 핀은 3.3V 신호만 직접 받는다. 5V·12V·모터·릴레이는 직결하지 않는다.

/* basys3_top.v — Basys3 보드 래퍼
 * Origin: capstones/mmio-i2c-master-ip/rtl/basys3_top.v
 * Verilog-2001.
 *
 * I2C 핀 배치 (오픈드레인):
 *   SCL → JA[1] (W2)   SDA → JA[2] (U2)
 * LED[7:0] → 읽은 WHO_AM_I 값 표시
 * BTNC → 동기 리셋
 */
module basys3_top (
    input  wire        clk,         /* 100 MHz XTAL */
    input  wire        btnc,        /* 리셋 버튼 (액티브 하이) */
    /* Pmod JA I2C 핀 (오픈드레인 버퍼) */
    inout  wire        ja1,         /* SCL */
    inout  wire        ja2,         /* SDA */
    /* LED */
    output wire [7:0]  led
);

/* --- 동기 리셋 (2단 플립플롭 디바운스) --- */
reg [1:0] rst_sr;
always @(posedge clk) rst_sr <= {rst_sr[0], btnc};
wire rst_n = ~rst_sr[1];

/* --- I2C 오픈드레인 버퍼 ---
 *   구동 모드: scl_drv=0 → 로우 출력,  scl_drv=1 → Hi-Z (풀업 의존)
 */
wire scl_drv, sda_drv, sda_in;

assign ja1   = scl_drv ? 1'bz : 1'b0;
assign ja2   = sda_drv ? 1'bz : 1'b0;
assign sda_in = ja2;

soc_top #(.CLK_DIV(1000)) u_soc (
    .clk   (clk),
    .rst_n (rst_n),
    .scl_o (scl_drv),
    .sda_o (sda_drv),
    .sda_i (sda_in),
    .led   (led)
);

endmodule

/* mmio_bus.v — 단일 슬레이브 mmio-lite 버스 디코더
 * Origin: capstones/mmio-i2c-master-ip/rtl/mmio_bus.v
 * Verilog-2001.
 *
 * 주소 디코딩: 전체 주소 공간을 단일 슬레이브(I2C IP)에 연결.
 * rdata: 슬레이브가 cs=0일 때 32'd0 반환 가정.
 */
module mmio_bus (
    /* 마스터 포트 (CPU / demo_master) */
    input  wire [31:0] m_addr,
    input  wire [31:0] m_wdata,
    output wire [31:0] m_rdata,
    input  wire        m_we,
    input  wire        m_valid,

    /* 슬레이브 0: I2C Master IP */
    output wire [15:0] s0_addr,
    output wire [31:0] s0_wdata,
    input  wire [31:0] s0_rdata,
    output wire        s0_we,
    output wire        s0_cs
);

/* 슬레이브 선택: 주소 비트[31:16]로 디코딩 (단일 슬레이브이므로 항상 선택) */
assign s0_cs    = m_valid;
assign s0_addr  = m_addr[15:0];
assign s0_wdata = m_wdata;
assign s0_we    = m_we;

assign m_rdata  = s0_cs ? s0_rdata : 32'd0;

endmodule
rtl/i2c_master_ip.v 246줄
/* i2c_master_ip.v — CTRL/DATA I2C Master MMIO IP
 * Origin: capstones/mmio-i2c-master-ip/rtl/i2c_master_ip.v
 * Verilog-2001. iverilog / verilator / yosys 호환.
 *
 * MMIO 레지스터 맵 (32-bit 워드, 바이트 주소):
 *   0x0000  CTRL  [6:0]=addr  [7]=rnw  [8]=go(W,SC)  [9]=stop  [10]=busy(RO)  [11]=nak(RO)
 *   0x0004  DATA  [7:0]=tx_data(W) / rx_data(R)
 *   0xFFF0  IP_ID      = 32'h4932_434D  ("I2CM")
 *   0xFFF4  IP_VER     = 32'h0000_0100
 *   0xFFF8  IRQ_STATUS [0]=done  (W1C)
 *   0xFFFC  IRQ_ENABLE [0]=done_en
 *
 * go 비트(bit8) 쓰기 → 트랜잭션 시작, 자동 소거(self-clearing)
 * 트랜잭션 완료: busy=0, irq (done_en=1일 때)
 * 한 번 GO에 1바이트 송신(rnw=0) 또는 1바이트 수신(rnw=1)
 */
module i2c_master_ip #(
    parameter CLK_DIV = 1000  /* I2C 비트 주기 (클럭 사이클). 100MHz→100kHz=1000 */
)(
    input  wire        clk,
    input  wire        rst_n,
    /* MMIO (mmio-lite: 단일 사이클, word-addr 호환) */
    input  wire [15:0] mmio_addr,
    input  wire [31:0] mmio_wdata,
    output reg  [31:0] mmio_rdata,
    input  wire        mmio_we,
    input  wire        mmio_cs,
    /* I2C 오픈드레인 (0=로우 구동, 1=릴리즈/Hi-Z) */
    output reg         scl_o,
    output reg         sda_o,
    input  wire        sda_i,
    /* 인터럽트 */
    output wire        irq
);

/* ===== 레지스터 ===== */
reg [6:0] r_addr;    /* I2C 슬레이브 주소 */
reg       r_rnw;     /* 0=쓰기, 1=읽기 */
reg       r_stop;    /* STOP 조건 추가 여부 */
reg       r_busy;
reg       r_nak;
reg [7:0] r_wdat;    /* TX 데이터 */
reg [7:0] r_rdat;    /* RX 데이터 */
reg       r_irq_st;  /* IRQ 상태: done */
reg       r_irq_en;  /* IRQ 활성화 */

/* ===== FSM 상태 ===== */
localparam [3:0]
    S_IDLE    = 4'd0, S_START   = 4'd1,
    S_ADDR    = 4'd2, S_ADDR_AK = 4'd3,
    S_TX      = 4'd4, S_TX_AK   = 4'd5,
    S_RX      = 4'd6, S_RX_NAK  = 4'd7,
    S_STOP    = 4'd8, S_DONE    = 4'd9;

reg  [3:0] state;
reg [11:0] pcnt;    /* 비트 내 위상 카운터 (0 .. CLK_DIV-1) */
reg  [2:0] bcnt;    /* 비트 카운터 (7 .. 0) */
reg  [7:0] shift;   /* 송수신 시프트 레지스터 */

/* 위상 타이밍 */
localparam [11:0] P_END = CLK_DIV - 1;
localparam [11:0] P_Q1  = CLK_DIV / 4;        /* SCL 상승 */
localparam [11:0] P_H   = CLK_DIV / 2;        /* 샘플 / SCL 중간 */
localparam [11:0] P_Q3  = (CLK_DIV * 3) / 4;  /* SCL 하강 */

wire p_end = (pcnt == P_END);
wire p_q1  = (pcnt == P_Q1);
wire p_h   = (pcnt == P_H);
wire p_q3  = (pcnt == P_Q3);

/* go 펄스: CTRL 레지스터 bit8 쓰기 (IDLE 상태에서만 유효) */
wire go_pulse = mmio_cs && mmio_we &&
                (mmio_addr[15:0] == 16'h0000) &&
                mmio_wdata[8] && !r_busy;

/* IRQ 출력 */
assign irq = r_irq_st & r_irq_en;

/* ===== 통합 always 블록 ===== */
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        r_addr <= 7'd0; r_rnw <= 1'b0; r_stop <= 1'b0;
        r_busy <= 1'b0; r_nak <= 1'b0;
        r_wdat <= 8'd0; r_rdat <= 8'd0;
        r_irq_st <= 1'b0; r_irq_en <= 1'b0;
        state <= S_IDLE; pcnt <= 12'd0; bcnt <= 3'd7; shift <= 8'd0;
        scl_o <= 1'b1; sda_o <= 1'b1;
    end else begin
        /* --- MMIO 쓰기 --- */
        if (mmio_cs && mmio_we) begin
            case (mmio_addr[15:0])
                16'h0000: begin
                    r_addr <= mmio_wdata[6:0];
                    r_rnw  <= mmio_wdata[7];
                    r_stop <= mmio_wdata[9];
                    /* bit8 (go)는 FSM에서 처리 */
                end
                16'h0004: r_wdat  <= mmio_wdata[7:0];
                16'hFFF8: r_irq_st <= r_irq_st & ~mmio_wdata[0]; /* W1C */
                16'hFFFC: r_irq_en <= mmio_wdata[0];
                default: ;
            endcase
        end

        /* --- FSM --- */
        case (state)
            /* 유휴 */
            S_IDLE: begin
                scl_o <= 1'b1; sda_o <= 1'b1;
                if (go_pulse) begin
                    r_busy <= 1'b1; r_nak <= 1'b0; r_irq_st <= 1'b0;
                    state <= S_START; pcnt <= 12'd0;
                end
            end

            /* START 조건: SDA 하강 (SCL=1) → SCL 하강 */
            S_START: begin
                if (pcnt == 12'd0) begin
                    sda_o <= 1'b0;                   /* SDA 하강 → START */
                    shift <= {r_addr, r_rnw};        /* 주소+RnW 래치 */
                    bcnt  <= 3'd7;
                end else if (p_h) scl_o <= 1'b0;    /* SCL 하강, 첫 비트 준비 */
                if (p_end) begin state <= S_ADDR; pcnt <= 12'd0; end
                else        pcnt <= pcnt + 12'd1;
            end

            /* 주소 + RnW 송신 (8비트) */
            S_ADDR: begin
                if (pcnt == 12'd0) sda_o <= shift[7];   /* SDA 세팅 */
                if (p_q1) scl_o <= 1'b1;                /* SCL 상승 */
                if (p_q3) scl_o <= 1'b0;                /* SCL 하강 */
                if (p_end) begin
                    if (bcnt == 3'd0) begin
                        state <= S_ADDR_AK; pcnt <= 12'd0;
                    end else begin
                        bcnt  <= bcnt - 3'd1;
                        shift <= {shift[6:0], 1'b0};
                        pcnt  <= 12'd0;
                    end
                end else pcnt <= pcnt + 12'd1;
            end

            /* 주소 ACK 수신 */
            S_ADDR_AK: begin
                if (pcnt == 12'd0) sda_o <= 1'b1;   /* SDA 릴리즈 (슬레이브 ACK) */
                if (p_q1) scl_o <= 1'b1;
                if (p_h)  r_nak <= sda_i;            /* 0=ACK, 1=NAK */
                if (p_q3) scl_o <= 1'b0;
                if (p_end) begin
                    pcnt <= 12'd0;
                    if (r_nak)       state <= r_stop ? S_STOP : S_DONE;
                    else if (!r_rnw) begin state <= S_TX; shift <= r_wdat; bcnt <= 3'd7; end
                    else             begin state <= S_RX; bcnt <= 3'd7; end
                end else pcnt <= pcnt + 12'd1;
            end

            /* 데이터 송신 (8비트, 쓰기 모드) */
            S_TX: begin
                if (pcnt == 12'd0) sda_o <= shift[7];
                if (p_q1) scl_o <= 1'b1;
                if (p_q3) scl_o <= 1'b0;
                if (p_end) begin
                    if (bcnt == 3'd0) begin
                        state <= S_TX_AK; pcnt <= 12'd0;
                    end else begin
                        bcnt  <= bcnt - 3'd1;
                        shift <= {shift[6:0], 1'b0};
                        pcnt  <= 12'd0;
                    end
                end else pcnt <= pcnt + 12'd1;
            end

            /* 데이터 ACK 수신 (쓰기 모드) */
            S_TX_AK: begin
                if (pcnt == 12'd0) sda_o <= 1'b1;
                if (p_q1) scl_o <= 1'b1;
                if (p_h)  r_nak <= sda_i;
                if (p_q3) scl_o <= 1'b0;
                if (p_end) begin
                    state <= r_stop ? S_STOP : S_DONE; pcnt <= 12'd0;
                end else pcnt <= pcnt + 12'd1;
            end

            /* 데이터 수신 (8비트, 읽기 모드) */
            S_RX: begin
                if (pcnt == 12'd0) sda_o <= 1'b1;   /* SDA 릴리즈 */
                if (p_q1) scl_o <= 1'b1;
                if (p_h)  shift <= {shift[6:0], sda_i}; /* 샘플 */
                if (p_q3) scl_o <= 1'b0;
                if (p_end) begin
                    if (bcnt == 3'd0) begin
                        r_rdat <= shift;
                        state  <= S_RX_NAK; pcnt <= 12'd0;
                    end else begin
                        bcnt <= bcnt - 3'd1; pcnt <= 12'd0;
                    end
                end else pcnt <= pcnt + 12'd1;
            end

            /* 마스터 NAK 송신 (읽기 1바이트 완료) */
            S_RX_NAK: begin
                if (pcnt == 12'd0) sda_o <= 1'b1;   /* NAK: SDA 하이 */
                if (p_q1) scl_o <= 1'b1;
                if (p_q3) scl_o <= 1'b0;
                if (p_end) begin
                    state <= r_stop ? S_STOP : S_DONE; pcnt <= 12'd0;
                end else pcnt <= pcnt + 12'd1;
            end

            /* STOP 조건: SCL 상승 → SDA 상승 */
            S_STOP: begin
                if (pcnt == 12'd0) sda_o <= 1'b0;   /* SDA 로우 확보 */
                if (p_q1) scl_o <= 1'b1;            /* SCL 상승 */
                if (p_h)  sda_o <= 1'b1;            /* SDA 상승 → STOP */
                if (p_end) begin state <= S_DONE; pcnt <= 12'd0; end
                else        pcnt <= pcnt + 12'd1;
            end

            /* 완료 */
            S_DONE: begin
                r_busy <= 1'b0; r_irq_st <= 1'b1;
                state <= S_IDLE;
            end

            default: state <= S_IDLE;
        endcase
    end
end

/* ===== MMIO 읽기 (조합 논리) ===== */
always @(*) begin
    mmio_rdata = 32'd0;
    if (mmio_cs && !mmio_we) begin
        case (mmio_addr[15:0])
            16'h0000: mmio_rdata = {20'd0, r_nak, r_busy, r_stop, 1'b0, r_rnw, r_addr};
            16'h0004: mmio_rdata = {24'd0, r_rdat};
            16'hFFF0: mmio_rdata = 32'h4932_434D; /* "I2CM" */
            16'hFFF4: mmio_rdata = 32'h0000_0100;
            16'hFFF8: mmio_rdata = {31'd0, r_irq_st};
            16'hFFFC: mmio_rdata = {31'd0, r_irq_en};
            default:  mmio_rdata = 32'd0;
        endcase
    end
end

/* demo_master.v — WHO_AM_I 읽기 시연 FSM
 * Origin: capstones/mmio-i2c-master-ip/rtl/demo_master.v
 * Verilog-2001.
 *
 * 전원 투입 후 MPU-6050(주소 0x68)의 WHO_AM_I(레지스터 0x75)를 읽어
 * LED[7:0]에 표시한다. BUSY 폴링으로 트랜잭션 완료 대기.
 *
 * 사용 순서:
 *   1. CTRL = {addr=0x68, rnw=0, go=1, stop=0}, DATA = 0x75  → 레지스터 주소 쓰기
 *   2. BUSY 폴링 (CTRL[10] = 0 까지 대기)
 *   3. CTRL = {addr=0x68, rnw=1, go=1, stop=1}               → 1바이트 읽기
 *   4. BUSY 폴링
 *   5. DATA[7:0] 읽어 LED 출력
 */
module demo_master (
    input  wire        clk,
    input  wire        rst_n,
    /* MMIO 마스터 포트 */
    output reg  [31:0] m_addr,
    output reg  [31:0] m_wdata,
    input  wire [31:0] m_rdata,
    output reg         m_we,
    output reg         m_valid,
    /* 결과 출력 */
    output reg  [7:0]  led
);

localparam [4:0]
    D_IDLE      = 5'd0,
    D_WR_DATA   = 5'd1,  /* DATA = 0x75 */
    D_WR_CTRL1  = 5'd2,  /* CTRL: go, addr=0x68, rnw=0, stop=0 */
    D_WAIT1_A   = 5'd3,  /* CTRL 읽기 */
    D_WAIT1_B   = 5'd4,  /* BUSY 확인 */
    D_WR_CTRL2  = 5'd5,  /* CTRL: go, addr=0x68, rnw=1, stop=1 */
    D_WAIT2_A   = 5'd6,  /* CTRL 읽기 */
    D_WAIT2_B   = 5'd7,  /* BUSY 확인 */
    D_RD_DATA   = 5'd8,  /* DATA 읽기 */
    D_DONE      = 5'd9;

reg [4:0] state;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state   <= D_IDLE;
        m_addr  <= 32'd0; m_wdata <= 32'd0;
        m_we    <= 1'b0;  m_valid <= 1'b0;
        led     <= 8'hFF; /* 초기: 전체 점등 */
    end else begin
        m_we    <= 1'b0;
        m_valid <= 1'b0;

        case (state)
            D_IDLE: begin
                state <= D_WR_DATA;
            end

            /* Step 1: DATA ← 0x75 (레지스터 주소) */
            D_WR_DATA: begin
                m_addr  <= 32'h0000_0004; /* DATA 오프셋 */
                m_wdata <= 32'h0000_0075;
                m_we    <= 1'b1;
                m_valid <= 1'b1;
                state   <= D_WR_CTRL1;
            end

            /* Step 2: CTRL ← {stop=1, go=1, rnw=0, addr=0x68} */
            D_WR_CTRL1: begin
                /* bit[6:0]=0x68, bit[7]=rnw=0, bit[8]=go=1, bit[9]=stop=1 */
                m_addr  <= 32'h0000_0000;
                m_wdata <= 32'h0000_0368; /* 0x68 | (1<<8) | (1<<9) */
                m_we    <= 1'b1;
                m_valid <= 1'b1;
                state   <= D_WAIT1_A;
            end

            /* Step 3: BUSY 폴링 (CTRL 읽기) */
            D_WAIT1_A: begin
                m_addr  <= 32'h0000_0000;
                m_we    <= 1'b0;
                m_valid <= 1'b1;
                state   <= D_WAIT1_B;
            end

            D_WAIT1_B: begin
                if (m_rdata[10] == 1'b0) /* BUSY = 0 */
                    state <= D_WR_CTRL2;
                else
                    state <= D_WAIT1_A; /* 계속 폴링 */
            end

            /* Step 4: CTRL ← {stop=1, go=1, rnw=1, addr=0x68} */
            D_WR_CTRL2: begin
                /* bit[6:0]=0x68, bit[7]=rnw=1, bit[8]=go=1, bit[9]=stop=1 */
                m_addr  <= 32'h0000_0000;
                m_wdata <= 32'h0000_03E8; /* 0x68|(1<<7)|(1<<8)|(1<<9) = 0x3E8 */
                m_we    <= 1'b1;
                m_valid <= 1'b1;
                state   <= D_WAIT2_A;
            end

            /* Step 5: BUSY 폴링 */
            D_WAIT2_A: begin
                m_addr  <= 32'h0000_0000;
                m_we    <= 1'b0;
                m_valid <= 1'b1;
                state   <= D_WAIT2_B;
            end

            D_WAIT2_B: begin
                if (m_rdata[10] == 1'b0)
                    state <= D_RD_DATA;
                else
                    state <= D_WAIT2_A;
            end

            /* Step 6: DATA 읽기 → LED */
            D_RD_DATA: begin
                m_addr  <= 32'h0000_0004;
                m_we    <= 1'b0;
                m_valid <= 1'b1;
                state   <= D_DONE;
            end

            D_DONE: begin
                led   <= m_rdata[7:0];
                state <= D_DONE; /* 완료: 유휴 */
            end

            default: state <= D_IDLE;
        endcase
    end
end

endmodule

/* tb_i2c_master.v — I2C Master IP 자체 점검 테스트벤치
 * Origin: capstones/mmio-i2c-master-ip/tb/tb_i2c_master.v
 * Verilog-2001.
 *
 * 시나리오:
 *   1. CTRL = {addr=0x68, rnw=0, go=1, stop=1}, DATA = 0x75
 *      → I2C: START + 0xD0(addr+W) + ACK + 0x75(reg) + ACK + STOP
 *   2. CTRL = {addr=0x68, rnw=1, go=1, stop=1}
 *      → I2C: START + 0xD1(addr+R) + ACK + 0x68(data) + NAK + STOP
 *   3. DATA[7:0] 읽기 → 0x68 여야 PASS
 *
 * 가짜 MPU-6050 슬레이브: 주소 0x68, registers[0x75] = 0x68
 * CLK_DIV = 20 (빠른 시뮬레이션; 위상: q1=5, h=10, q3=15, end=19)
 */
`timescale 1ns/1ps

module tb_i2c_master;

localparam CLK_DIV  = 20;
localparam CLK_HALF = 5;  /* 10 ns 주기 클럭 */

/* --- DUT 신호 --- */
reg         clk, rst_n;
reg  [15:0] mmio_addr;
reg  [31:0] mmio_wdata;
wire [31:0] mmio_rdata;
reg         mmio_we, mmio_cs;
wire        scl_o, sda_o;
wire        irq;

/* --- I2C 버스 (오픈드레인 와이어-AND) --- */
reg  sda_slave;        /* 슬레이브 SDA 구동: 0=로우, 1=릴리즈 */
wire SDA = sda_o & sda_slave;  /* 와이어드-AND */
wire SCL = scl_o;

/* --- DUT 인스턴스 --- */
i2c_master_ip #(.CLK_DIV(CLK_DIV)) dut (
    .clk        (clk),
    .rst_n      (rst_n),
    .mmio_addr  (mmio_addr),
    .mmio_wdata (mmio_wdata),
    .mmio_rdata (mmio_rdata),
    .mmio_we    (mmio_we),
    .mmio_cs    (mmio_cs),
    .scl_o      (scl_o),
    .sda_o      (sda_o),
    .sda_i      (SDA),
    .irq        (irq)
);

/* ============================================================
 * 가짜 I2C 슬레이브 — MPU-6050 (주소 0x68)
 * ============================================================ */
localparam SLAVE_ADDR = 7'h68;
integer slave_state;
integer slave_bit;
reg [7:0] slave_rx;
reg [7:0] slave_tx;
integer slave_tx_bit;
integer slave_reg_ptr;
reg [7:0] slave_regs [0:255];

localparam SS_IDLE           = 0,
           SS_RX_ADDR        = 1,
           SS_ACK_ADDR       = 2,
           SS_RX_DATA        = 3,
           SS_ACK_DATA       = 4,
           SS_TX_DATA        = 5,
           SS_RX_NAK         = 6,
           SS_ACK_ADDR_DONE  = 7; /* posedge(ACK) 를 지나고 나서 상태 전환 */

integer k;
initial begin
    for (k = 0; k < 256; k = k + 1) slave_regs[k] = 8'h00;
    slave_regs[8'h75] = 8'h68;  /* WHO_AM_I */
    sda_slave    = 1'b1;
    slave_state  = SS_IDLE;
    slave_bit    = 7;
    slave_rx     = 8'h00;
    slave_tx     = 8'h00;
    slave_tx_bit = 7;
    slave_reg_ptr = 0;
end

/* START 감지: SDA 하강 중 SCL=1 */
always @(negedge SDA) begin
    if (SCL) begin
        slave_state = SS_RX_ADDR;
        slave_bit   = 7;
        slave_rx    = 8'h00;
        sda_slave   = 1'b1;
    end
end

/* STOP 감지: SDA 상승 중 SCL=1 */
always @(posedge SDA) begin
    if (SCL) begin
        slave_state = SS_IDLE;
        sda_slave   = 1'b1;
    end
end

/* 비트 샘플: posedge SCL */
always @(posedge SCL) begin
    case (slave_state)
        SS_RX_ADDR: begin
            slave_rx[slave_bit] = SDA;
            if (slave_bit == 0) begin
                slave_state = SS_ACK_ADDR;
                slave_bit   = 7;
            end else
                slave_bit = slave_bit - 1;
        end
        SS_RX_DATA: begin
            slave_rx[slave_bit] = SDA;
            if (slave_bit == 0) begin
                slave_state = SS_ACK_DATA;
                slave_bit   = 7;
            end else
                slave_bit = slave_bit - 1;
        end
        SS_RX_NAK: begin
            /* 마스터 NAK 수신 → IDLE (SDA 릴리즈는 negedge default에서 처리) */
            slave_state = SS_IDLE;
        end
        default: ;
    endcase
end

/* SDA 구동: negedge SCL */
always @(negedge SCL) begin
    #1; /* 경쟁 조건 방지 */
    case (slave_state)
        SS_ACK_ADDR: begin
            if (slave_rx[7:1] == SLAVE_ADDR) begin
                sda_slave   = 1'b0;             /* ACK */
                slave_state = SS_ACK_ADDR_DONE; /* ACK posedge 를 지나고 나서 전환 */
            end else begin
                sda_slave   = 1'b1; /* NAK */
                slave_state = SS_IDLE;
            end
        end
        SS_ACK_ADDR_DONE: begin
            /* ACK posedge 이후 negedge: 이제 데이터 단계로 전환 */
            if (slave_rx[0] == 1'b0) begin
                /* 쓰기 모드: SDA 릴리즈 후 SS_RX_DATA */
                sda_slave   = 1'b1;
                slave_state = SS_RX_DATA;
            end else begin
                /* 읽기 모드: 첫 비트(bit7)를 지금 구동해 마스터가 p_h에서 샘플 */
                slave_tx     = slave_regs[slave_reg_ptr];
                sda_slave    = slave_regs[slave_reg_ptr][7]; /* bit 7 */
                slave_tx_bit = 6;            /* 이후 bit 6..0은 SS_TX_DATA 에서 */
                slave_state  = SS_TX_DATA;
            end
        end
        SS_ACK_DATA: begin
            slave_reg_ptr = slave_rx; /* 레지스터 포인터 저장 */
            sda_slave     = 1'b0;    /* ACK */
            slave_state   = SS_IDLE; /* STOP 또는 반복 START 대기 */
        end
        SS_TX_DATA: begin
            sda_slave = slave_tx[slave_tx_bit]; /* 데이터 비트 송신 */
            if (slave_tx_bit == 0)
                slave_state = SS_RX_NAK;        /* 마지막 비트 후 NAK 대기 */
            else
                slave_tx_bit = slave_tx_bit - 1;
        end
        default: sda_slave = 1'b1; /* ACK 해제 / 릴리즈 */
    endcase
end

/* ============================================================
 * 클럭 생성
 * ============================================================ */
initial clk = 1'b0;
always #CLK_HALF clk = ~clk;

/* ============================================================
 * MMIO 쓰기 태스크
 * ============================================================ */
task mmio_write;
    input [15:0] addr;
    input [31:0] data;
    begin
        @(posedge clk); #1;
        mmio_addr  = addr;
        mmio_wdata = data;
        mmio_we    = 1'b1;
        mmio_cs    = 1'b1;
        @(posedge clk); #1;
        mmio_we    = 1'b0;
        mmio_cs    = 1'b0;
    end
endtask

/* MMIO 읽기 태스크 */
task mmio_read;
    input  [15:0] addr;
    output [31:0] data;
    begin
        @(posedge clk); #1;
        mmio_addr  = addr;
        mmio_we    = 1'b0;
        mmio_cs    = 1'b1;
        @(posedge clk); #1;
        data       = mmio_rdata;
        mmio_cs    = 1'b0;
    end
endtask

/* BUSY 폴링 태스크 */
task wait_not_busy;
    reg [31:0] ctrl_val;
    integer timeout;
    begin
        timeout = 0;
        ctrl_val = 32'h0000_0400; /* BUSY 비트 세팅 상태 */
        while (ctrl_val[10] == 1'b1 && timeout < 50000) begin
            mmio_read(16'h0000, ctrl_val);
            timeout = timeout + 1;
        end
        if (timeout >= 50000)
            $display("FAIL: BUSY 타임아웃");
    end
endtask

/* ============================================================
 * 메인 테스트
 * ============================================================ */
integer fail_count;
reg [31:0] rd_val;

initial begin
    fail_count = 0;
    rst_n      = 1'b0;
    mmio_addr  = 16'h0000;
    mmio_wdata = 32'd0;
    mmio_we    = 1'b0;
    mmio_cs    = 1'b0;

    repeat(4) @(posedge clk);
    rst_n = 1'b1;
    repeat(2) @(posedge clk);

    /* --- IP_ID 확인 --- */
    mmio_read(16'hFFF0, rd_val);
    if (rd_val !== 32'h4932_434D) begin
        $display("FAIL: IP_ID 기대값 0x4932434D, 실제값 0x%08H", rd_val);
        fail_count = fail_count + 1;
    end

    /* ---- 트랜잭션 1: 레지스터 주소 쓰기 (0x75) ---- */
    /* DATA = 0x75 */
    mmio_write(16'h0004, 32'h0000_0075);
    /* CTRL = {stop=1, go=1, rnw=0, addr=0x68} = 0x0000_0368 */
    mmio_write(16'h0000, 32'h0000_0368);

    wait_not_busy;

    /* NAK 확인 (트랜잭션 1은 ACK 기대) */
    mmio_read(16'h0000, rd_val);
    if (rd_val[11] !== 1'b0) begin
        $display("FAIL: 트랜잭션1 NAK 비트 세팅됨 (ACK 기대)");
        fail_count = fail_count + 1;
    end

    /* ---- 트랜잭션 2: WHO_AM_I 읽기 ---- */
    /* CTRL = {stop=1, go=1, rnw=1, addr=0x68} = 0x0000_03E8 */
    mmio_write(16'h0000, 32'h0000_03E8);

    wait_not_busy;

    /* DATA[7:0] = 0x68 여야 PASS */
    mmio_read(16'h0004, rd_val);
    if (rd_val[7:0] !== 8'h68) begin
        $display("FAIL: WHO_AM_I 기대값 0x68, 실제값 0x%02H", rd_val[7:0]);
        fail_count = fail_count + 1;
    end

    /* IRQ 확인 (IRQ_ENABLE = 1 설정 후 IRQ_STATUS 확인) */
    mmio_write(16'hFFFC, 32'h0000_0001);    /* done_en = 1 */
    /* IRQ_STATUS는 직전 트랜잭션에서 세팅됨 */
    mmio_read(16'hFFF8, rd_val);
    if (rd_val[0] !== 1'b1) begin
        $display("FAIL: IRQ_STATUS done 비트 미세팅");
        fail_count = fail_count + 1;
    end

    /* --- 최종 판정 --- */
    if (fail_count == 0)
        $display("PASS: I2C Master IP 검증 완료 (WHO_AM_I=0x68 확인)");
    else
        $display("FAIL: %0d 항목 실패", fail_count);

    $finish;
end

	    -Wno-STMTDLY \
	    $(RTL_IP) $(RTL_SOC) 2>&1 | tee $(BUILD)/lint.log
	@if grep -q '%Error' $(BUILD)/lint.log; then \
	    echo "LINT FAIL"; exit 1; \
	else \
	    echo "LINT PASS"; \
	fi

## iverilog 시뮬레이션 (Verilog-2001)
sim: $(BUILD)
	iverilog -g2001 -Wall -o $(BUILD)/sim.vvp \
	    $(RTL_IP) $(TB_SRC) 2>&1 | tee $(BUILD)/sim_compile.log
	vvp $(BUILD)/sim.vvp 2>&1 | tee $(BUILD)/sim.log
	@if grep -q '^PASS' $(BUILD)/sim.log; then \
	    echo "SIM PASS"; \
	elif grep -q '^FAIL' $(BUILD)/sim.log; then \
	    echo "SIM FAIL"; exit 1; \
	else \
	    echo "SIM FAIL (no PASS/FAIL marker)"; exit 1; \
	fi

## Yosys 합성 (자원 추정)
synth: $(BUILD)
	yosys -p " \
	    read_verilog -sv $(RTL_IP); \
	    synth_xilinx -top $(TOP_SYNTH) -flatten; \
	    stat; \
	" 2>&1 | tee $(BUILD)/synth.log
	@echo "SYNTH done — 자원은 yosys 추정 (Vivado와 다를 수 있음)"

clean:
	rm -rf $(BUILD)

시뮬레이션 (iverilog)

  1. make lint — RTL 문법 검사 (iverilog -tnull)
  2. make sim — 가짜 MPU-6050 슬레이브와 WHO_AM_I 읽기 검증; PASS 출력 확인
  3. make synth — yosys로 자원 추정 (XC7A35T 기준)

Vivado 합성

  1. Vivado에서 capstone.json의 sources 목록 파일을 프로젝트에 추가한다.
  2. constr/basys3_i2c.xdc를 제약 파일로 추가한다.
  3. top=basys3_top으로 합성·구현한다.
  4. 합성 후 LUT/FF/BRAM 사용량과 WNS를 기록한다.
