AXI4-Lite 워치독 타이머 IP — T1 기본 카운터

MMIO 맵 가능한 32비트 카운트다운 워치독 타이머. AXI4-Lite 슬레이브 인터페이스, 타임아웃 리셋 펄스(wdt_rst), 옵션 IRQ 출력(wdt_irq). I-070 아이디어 T1 패키지.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-verified

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • AXI4-Lite MMIO로 제어 가능한 카운트다운 워치독 타이머 IP 구현
  • KICK(소프트웨어 갱신)으로 타임아웃 방지, 미갱신 시 wdt_rst 펄스 출력
  • IRQ_EN/IRQ_PEND 레지스터로 인터럽트 지원 (IEC 61508 아님, T3에서 확장)
  • iverilog 시뮬레이션 통과, yosys 합성 가능 RTL

범위 밖

  • 창 워치독(MIN/MAX 경계): T2에서 추가
  • 비밀번호 잠금, 에러 카운터, NMI: T3에서 추가
  • 보드에서의 동작 확인(S3): 사용자 직접 수행
  • Vivado IP-XACT 패키지화
  • 소프트웨어 드라이버(C) 포함

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-070 기반(ready 상태). AXI4-Lite 슬레이브. 동결 맵 free_slots[1B]=0x401B_0000 배정 제안(미동결). 시뮬레이션·합성 결과는 CI에서 검증한다.

// wdt_core.v — 워치독 타이머 코어 (기본 카운터 + 타임아웃 FSM)
// I-070 AXI4-Lite 워치독 IP (mmio-wdt-tier1, T1)
// Verilog-2001. iverilog/verilator/yosys 호환.
//
// 포트:
//   clk        — 시스템 클럭
//   rst        — 동기 active-high 리셋
//   enable     — 1=카운터 활성 (0이면 IDLE, 카운터 정지)
//   kick       — 1사이클 펄스: 카운터를 load_val로 재적재 (timeout_lat도 클리어)
//   srst       — 1사이클 펄스: 소프트 리셋 (타임아웃 클리어, 카운터 재적재)
//   load_val   — 카운트다운 초기값 (0이면 즉시 타임아웃)
//   count_out  — 현재 카운터 값
//   timeout    — 1사이클 펄스 (카운터가 0에 도달할 때 1사이클만 high)
//   timeout_lat — 래치된 타임아웃 플래그 (kick/srst로 클리어)
//   running    — 1=카운터 감소 중 (enable=1 이고 타임아웃 미발생)

`default_nettype none

module wdt_core #(
    parameter CNT_W = 32
) (
    input  wire             clk,
    input  wire             rst,
    input  wire             enable,
    input  wire             kick,
    input  wire             srst,
    input  wire [CNT_W-1:0] load_val,
    output reg  [CNT_W-1:0] count_out,
    output reg              timeout,
    output reg              timeout_lat,
    output reg              running
);

    // ── FSM 상태 ─────────────────────────────────────────
    localparam S_IDLE    = 2'd0;  // enable=0
    localparam S_RUNNING = 2'd1;  // 카운터 감소 중
    localparam S_TIMEOUT = 2'd2;  // 카운터 0 도달, 리셋 대기

    reg [1:0] state;

    always @(posedge clk) begin
        if (rst) begin
            state       <= S_IDLE;
            count_out   <= {CNT_W{1'b0}};
            timeout     <= 1'b0;
            timeout_lat <= 1'b0;
            running     <= 1'b0;
        end else begin
            // 1사이클 펄스 자동 클리어
            timeout <= 1'b0;

            case (state)
                S_IDLE: begin
                    running <= 1'b0;
                    if (srst) begin
                        count_out   <= load_val;
                        timeout_lat <= 1'b0;
                    end
                    if (enable && !timeout_lat) begin
                        // enable이 올라오면 카운터를 적재하고 감소 시작
                        state     <= S_RUNNING;
                        count_out <= load_val;
                        running   <= 1'b1;
                    end
                end

                S_RUNNING: begin
                    running <= 1'b1;
                    if (srst || kick) begin
                        // kick 또는 소프트 리셋: 카운터 재적재, 타임아웃 클리어
                        count_out   <= load_val;
                        timeout_lat <= 1'b0;
                        if (!enable) begin
                            state   <= S_IDLE;
                            running <= 1'b0;
                        end
                    end else if (!enable) begin
                        // enable 내려가면 카운터 정지
                        state   <= S_IDLE;
                        running <= 1'b0;
                    end else if (count_out == {CNT_W{1'b0}}) begin
                        // 카운터가 이미 0: 타임아웃 발생
                        state       <= S_TIMEOUT;
                        timeout     <= 1'b1;
                        timeout_lat <= 1'b1;
                        running     <= 1'b0;
                    end else begin
                        // 정상 감소
                        count_out <= count_out - 1'b1;
                    end
                end

                S_TIMEOUT: begin
                    running <= 1'b0;
                    if (srst || kick) begin
                        // kick/소프트 리셋으로 타임아웃 해제
                        count_out   <= load_val;
                        timeout_lat <= 1'b0;
                        if (enable) begin
                            state   <= S_RUNNING;
                            running <= 1'b1;
                        end else begin
                            state   <= S_IDLE;
                        end
                    end else if (!enable) begin
                        // enable이 내려오면 IDLE (timeout_lat은 유지)
                        state <= S_IDLE;
                    end
                end

                default: state <= S_IDLE;
            endcase
        end
    end

endmodule

// axil_wdt_wrap.v — AXI4-Lite 슬레이브 래퍼 (wdt_core 인스턴스)
// I-070 AXI4-Lite 워치독 IP (mmio-wdt-tier1, T1)
// Verilog-2001. iverilog/verilator/yosys 호환.
//
// 레지스터 맵 (AXI araddr[15:0] 기준):
//   0x0000  CTRL      [RW] bit0=ENABLE, bit1=SRST(자동 클리어)
//   0x0004  LOAD      [RW] [31:0]=카운트다운 초기값 (100MHz × 값 = 타임아웃)
//   0x0008  STATUS    [RO] bit0=TIMEOUT_LAT, bit1=RUNNING
//   0x000C  KICK      [WO] 아무 값 쓰면 카운터를 LOAD 값으로 재적재
//   0x0010  COUNT     [RO] 현재 카운터 값
//   0xFFF0  ID_REG    [RO] 0x57445431 ("WDT1")
//   0xFFF4  VER_REG   [RO] 0x00010000 (v1.0.0)
//   0xFFF8  IRQ_EN    [RW] bit0=타임아웃 인터럽트 활성
//   0xFFFC  IRQ_PEND  [RW] bit0=인터럽트 대기 중, 1 쓰면 클리어
//
// 출력:
//   wdt_rst  — 타임아웃 시 1사이클 리셋 펄스 (SoC 시스템 리셋 연결용)
//   wdt_irq  — 인터럽트 레벨 신호 (IRQ_EN=1 && IRQ_PEND=1)

`default_nettype none

module axil_wdt_wrap (
    // AXI 클럭/리셋
    input  wire        aclk,
    input  wire        aresetn,  // active-low

    // AXI4-Lite Write Channel
    input  wire [15:0] s_awaddr,
    input  wire        s_awvalid,
    output reg         s_awready,
    input  wire [31:0] s_wdata,
    input  wire [3:0]  s_wstrb,
    input  wire        s_wvalid,
    output reg         s_wready,
    output reg  [1:0]  s_bresp,
    output reg         s_bvalid,
    input  wire        s_bready,

    // AXI4-Lite Read Channel
    input  wire [15:0] s_araddr,
    input  wire        s_arvalid,
    output reg         s_arready,
    output reg  [31:0] s_rdata,
    output reg  [1:0]  s_rresp,
    output reg         s_rvalid,
    input  wire        s_rready,

    // 기능 출력
    output wire        wdt_rst,  // 타임아웃 리셋 펄스
    output wire        wdt_irq   // 인터럽트 출력 (레벨)
);

    // ── 내부 리셋 (active-high) ───────────────────────────
    wire rst = !aresetn;

    // ── 제어 레지스터 ────────────────────────────────────
    reg        ctrl_enable;
    reg        ctrl_srst;       // 1사이클 펄스 (자동 클리어)
    reg [31:0] ctrl_load;
    reg        ctrl_kick;       // 1사이클 펄스 (자동 클리어)
    reg        irq_en;
    reg        irq_pend;

    // ── wdt_core 출력 ───────────────────────────────────
    wire [31:0] count_out;
    wire        timeout_pulse;
    wire        timeout_lat;
    wire        running;

    // ── wdt_core 인스턴스 ────────────────────────────────
    wdt_core #(
        .CNT_W(32)
    ) u_core (
        .clk        (aclk),
        .rst        (rst),
        .enable     (ctrl_enable),
        .kick       (ctrl_kick),
        .srst       (ctrl_srst),
        .load_val   (ctrl_load),
        .count_out  (count_out),
        .timeout    (timeout_pulse),
        .timeout_lat(timeout_lat),
        .running    (running)
    );

    // ── 인터럽트 / 리셋 출력 ────────────────────────────
    // IRQ_PEND는 AXI 쓰기 도메인에서 갱신; timeout_pulse는 wdt_core 클럭 도메인
    // (같은 aclk를 쓰므로 동기 처리 가능)
    always @(posedge aclk) begin
        if (rst) begin
            irq_pend <= 1'b0;
        end else begin
            if (timeout_pulse && irq_en)
                irq_pend <= 1'b1;
            // IRQ_PEND[0] 쓰기 클리어는 Write FSM에서 처리
        end
    end

    assign wdt_rst = timeout_pulse;              // 1사이클 리셋 펄스
    assign wdt_irq = irq_en && irq_pend;         // 레벨 인터럽트

    // ── Write FSM ─────────────────────────────────────────
    reg       aw_pend;
    reg [15:0] aw_addr_r;
    reg       w_pend;
    reg [31:0] w_data_r;

    always @(posedge aclk) begin
        if (rst) begin
            s_awready  <= 1'b0;
            s_wready   <= 1'b0;
            s_bvalid   <= 1'b0;
            s_bresp    <= 2'b00;
            aw_pend    <= 1'b0;
            w_pend     <= 1'b0;
            ctrl_enable <= 1'b0;
            ctrl_srst  <= 1'b0;
            ctrl_load  <= 32'd0;
            ctrl_kick  <= 1'b0;
            irq_en     <= 1'b0;
        end else begin
            // 1사이클 펄스 자동 클리어
            ctrl_srst <= 1'b0;
            ctrl_kick <= 1'b0;

            // awready 핸드셰이크 (!s_bvalid: re-capture 방지)
            if (s_awvalid && !aw_pend && !s_bvalid) begin
                s_awready <= 1'b1;
                aw_pend   <= 1'b1;
                aw_addr_r <= s_awaddr;
            end else begin
                s_awready <= 1'b0;
            end

            // wready 핸드셰이크 (!s_bvalid: re-capture 방지)
            if (s_wvalid && !w_pend && !s_bvalid) begin
                s_wready <= 1'b1;
                w_pend   <= 1'b1;
                w_data_r <= s_wdata;
            end else begin
                s_wready <= 1'b0;
            end

            // 주소와 데이터 모두 수신 → 레지스터 쓰기 + bresp
            if (aw_pend && w_pend && !s_bvalid) begin
                case (aw_addr_r)
                    16'h0000: begin  // CTRL
                        ctrl_enable <= w_data_r[0];
                        ctrl_srst   <= w_data_r[1];
                    end
                    16'h0004: ctrl_load  <= w_data_r;         // LOAD
                    16'h000C: ctrl_kick  <= 1'b1;             // KICK (write-only)
                    16'hFFF8: irq_en     <= w_data_r[0];      // IRQ_EN
                    16'hFFFC: begin  // IRQ_PEND: W1C (bit0)
                        if (w_data_r[0])
                            irq_pend <= 1'b0;
                    end
                    default: ;  // STATUS, COUNT, ID, VER: 무시, OKAY 반환
                endcase
                s_bvalid <= 1'b1;
                s_bresp  <= 2'b00;  // OKAY
                aw_pend  <= 1'b0;
                w_pend   <= 1'b0;
            end

            if (s_bvalid && s_bready)
                s_bvalid <= 1'b0;
        end
    end

    // ── Read FSM ──────────────────────────────────────────
    reg [15:0] ar_addr_r;

    always @(posedge aclk) begin
        if (rst) begin
            s_arready <= 1'b0;
            s_rvalid  <= 1'b0;
            s_rdata   <= 32'h0;
            s_rresp   <= 2'b00;
        end else begin
            if (s_arvalid && !s_rvalid) begin
                s_arready <= 1'b1;
                ar_addr_r <= s_araddr;
            end else begin
                s_arready <= 1'b0;
            end

            if (s_arready && s_arvalid) begin
                s_rvalid <= 1'b1;
                s_rresp  <= 2'b00;
                case (ar_addr_r)
                    16'h0000: s_rdata <= {30'b0, ctrl_srst, ctrl_enable};    // CTRL
                    16'h0004: s_rdata <= ctrl_load;                           // LOAD
                    16'h0008: s_rdata <= {30'b0, running, timeout_lat};       // STATUS
                    16'h000C: s_rdata <= 32'h0;                               // KICK: 읽으면 0
                    16'h0010: s_rdata <= count_out;                           // COUNT
                    16'hFFF0: s_rdata <= 32'h57445431;                        // ID ("WDT1")
                    16'hFFF4: s_rdata <= 32'h00010000;                        // VER (1.0.0)
                    16'hFFF8: s_rdata <= {31'b0, irq_en};                     // IRQ_EN
                    16'hFFFC: s_rdata <= {31'b0, irq_pend};                   // IRQ_PEND
                    default:  s_rdata <= 32'hDEAD_BEEF;
                endcase
            end

            if (s_rvalid && s_rready)
                s_rvalid <= 1'b0;
        end
    end

endmodule

// tb_axil_wdt_wrap.v — 자체 점검형 테스트벤치
// I-070 AXI4-Lite 워치독 IP (mmio-wdt-tier1, T1)
// Verilog-2001. iverilog 호환.
//
// 통과 → "PASS" 로 시작하는 줄, 실패 → "FAIL" 로 시작하는 줄 출력.
// 비교는 !== 로 X/Z도 실패로 센다.
//
// 테스트 케이스:
//   TC1: LOAD=10 → ENABLE → 10클럭 뒤 STATUS.TIMEOUT_LAT=1
//   TC2: LOAD=20 → ENABLE → 5클럭 뒤 KICK → TIMEOUT_LAT 없음 (킥이 타임아웃 방지)
//   TC3: SRST → 타임아웃 클리어 후 카운터 재적재 확인
//   TC4: IRQ 동작 → TIMEOUT 후 IRQ_PEND=1, W1C 클리어
//   TC5: ENABLE=0 → 카운터 정지, TIMEOUT_LAT 없음

`timescale 1ns/1ps
`default_nettype none

module tb_axil_wdt_wrap;

    // ── 클럭 ──────────────────────────────────────────────
    reg aclk;
    initial aclk = 0;
    always #5 aclk = ~aclk;  // 100 MHz

    // ── DUT 포트 ──────────────────────────────────────────
    reg        aresetn;
    reg [15:0] s_awaddr;
    reg        s_awvalid;
    wire       s_awready;
    reg [31:0] s_wdata;
    reg  [3:0] s_wstrb;
    reg        s_wvalid;
    wire       s_wready;
    wire [1:0] s_bresp;
    wire       s_bvalid;
    reg        s_bready;
    reg [15:0] s_araddr;
    reg        s_arvalid;
    wire       s_arready;
    wire [31:0] s_rdata;
    wire [1:0]  s_rresp;
    wire        s_rvalid;
    reg         s_rready;
    wire        wdt_rst;
    wire        wdt_irq;

    // ── DUT 인스턴스 ─────────────────────────────────────
    axil_wdt_wrap dut (
        .aclk      (aclk),
        .aresetn   (aresetn),
        .s_awaddr  (s_awaddr),
        .s_awvalid (s_awvalid),
        .s_awready (s_awready),
        .s_wdata   (s_wdata),
        .s_wstrb   (s_wstrb),
        .s_wvalid  (s_wvalid),
        .s_wready  (s_wready),
        .s_bresp   (s_bresp),
        .s_bvalid  (s_bvalid),
        .s_bready  (s_bready),
        .s_araddr  (s_araddr),
        .s_arvalid (s_arvalid),
        .s_arready (s_arready),
        .s_rdata   (s_rdata),
        .s_rresp   (s_rresp),
        .s_rvalid  (s_rvalid),
        .s_rready  (s_rready),
        .wdt_rst   (wdt_rst),
        .wdt_irq   (wdt_irq)
    );

    // ── 실패 카운터 ───────────────────────────────────────
    integer fail_cnt;

    // ── AXI4-Lite 헬퍼 태스크 ────────────────────────────
    task axil_write;
        input [15:0] addr;
        input [31:0] data;
        begin
            @(posedge aclk); #1;
            s_awaddr = addr; s_awvalid = 1;
            s_wdata  = data; s_wvalid  = 1; s_wstrb = 4'hF;
            s_bready = 1;
            wait(s_bvalid); @(posedge aclk); #1;
            s_awvalid = 0;
            s_wvalid  = 0;
            s_bready  = 0;
        end
    endtask

    task axil_read;
        input  [15:0] addr;
        output [31:0] data;
        begin
            @(posedge aclk); #1;
            s_araddr = addr; s_arvalid = 1;
            wait(s_arready); @(posedge aclk); #1;
            s_arvalid = 0;
            s_rready  = 1;
            wait(s_rvalid);
            data = s_rdata;
            @(posedge aclk); #1;
            s_rready = 0;
        end
    endtask

    // ── 체크 헬퍼 ────────────────────────────────────────
    task check_eq;
        input [255:0] name;
        input [31:0]  got;
        input [31:0]  exp;
        begin
            if (got !== exp) begin
                $display("FAIL %0s: got=%0h exp=%0h", name, got, exp);
                fail_cnt = fail_cnt + 1;
            end else begin
                $display("pass %0s: %0h", name, got);
            end
        end
    endtask

    // ── 메인 ─────────────────────────────────────────────
    reg [31:0] rd;
    integer i;

    initial begin
        // 초기화
        fail_cnt  = 0;
        aresetn   = 0;
        s_awvalid = 0; s_wvalid = 0; s_bready = 0;
        s_arvalid = 0; s_rready = 0;
        s_awaddr  = 0; s_wdata = 0; s_wstrb = 0;
        s_araddr  = 0;

        repeat(5) @(posedge aclk);
        aresetn = 1;
        repeat(2) @(posedge aclk);

        // ────────────────────────────────────────────────
        // TC1: LOAD=10 → ENABLE → 타임아웃 확인
        //   10클럭 카운트다운 후 TIMEOUT_LAT=1, wdt_rst 펄스
        // ────────────────────────────────────────────────
        $display("=== TC1: LOAD=10, ENABLE → 타임아웃 ===");
        axil_write(16'h0004, 32'd10);    // LOAD=10
        axil_write(16'hFFF8, 32'h0);     // IRQ_EN=0
        axil_write(16'h0000, 32'h1);     // CTRL: ENABLE=1

        // 10클럭 감소 + 여유
        repeat(20) @(posedge aclk);

        axil_read(16'h0008, rd);
        check_eq("TC1:TIMEOUT_LAT", rd & 32'h1, 32'h1);
        check_eq("TC1:RUNNING",     (rd >> 1) & 32'h1, 32'h0);

        // ID/VER 확인
        axil_read(16'hFFF0, rd);
        check_eq("TC1:ID_REG",  rd, 32'h57445431);
        axil_read(16'hFFF4, rd);
        check_eq("TC1:VER_REG", rd, 32'h00010000);

        // ────────────────────────────────────────────────
        // TC2: KICK이 타임아웃을 방지한다
        //   LOAD=20, ENABLE, 10클럭 후 KICK → 추가 20클럭 후 TIMEOUT_LAT=1
        //   단, KICK 직후 20클럭 이내에는 TIMEOUT_LAT=0 이어야 함
        // ────────────────────────────────────────────────
        $display("=== TC2: KICK이 타임아웃을 방지 ===");
        // 먼저 현재 타임아웃 상태 클리어 (SRST)
        axil_write(16'h0000, 32'h2);     // CTRL: SRST=1
        repeat(3) @(posedge aclk);

        axil_write(16'h0004, 32'd20);    // LOAD=20
        axil_write(16'h0000, 32'h1);     // ENABLE=1

        repeat(10) @(posedge aclk);
        axil_write(16'h000C, 32'h1);     // KICK

        repeat(5) @(posedge aclk);
        axil_read(16'h0008, rd);
        check_eq("TC2:TIMEOUT_LAT_after_kick", rd & 32'h1, 32'h0);
        check_eq("TC2:RUNNING_after_kick",     (rd >> 1) & 32'h1, 32'h1);

        // 20클럭 더 대기 → 타임아웃
        repeat(25) @(posedge aclk);
        axil_read(16'h0008, rd);
        check_eq("TC2:TIMEOUT_LAT_eventual", rd & 32'h1, 32'h1);

        // ────────────────────────────────────────────────
        // TC3: SRST가 타임아웃을 클리어하고 카운터를 재적재한다
        // ────────────────────────────────────────────────
        $display("=== TC3: SRST 클리어 ===");
        axil_write(16'h0004, 32'd50);    // LOAD=50
        axil_write(16'h0000, 32'h2);     // SRST

        repeat(3) @(posedge aclk);
        axil_read(16'h0008, rd);
        check_eq("TC3:TIMEOUT_LAT_cleared", rd & 32'h1, 32'h0);

        axil_read(16'h0010, rd);         // COUNT 읽기
        // SRST 후 enable=1이면 RUNNING 상태, count_out≈50
        // (AXI 레이턴시로 정확한 값이 아닐 수 있으나 50 이하여야 함)
        if (rd > 32'd50) begin
            $display("FAIL TC3:COUNT_after_SRST: got=%0d (expected <=50)", rd);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("pass TC3:COUNT_after_SRST: %0d (<=50)", rd);
        end

        // ────────────────────────────────────────────────
        // TC4: IRQ 동작 — TIMEOUT 후 IRQ_PEND=1, W1C 클리어
        // ────────────────────────────────────────────────
        $display("=== TC4: IRQ 동작 ===");
        // 현재 카운터 클리어
        axil_write(16'h0000, 32'h2);     // SRST
        repeat(3) @(posedge aclk);
        axil_write(16'h0000, 32'h0);     // ENABLE=0
        repeat(2) @(posedge aclk);

        axil_write(16'hFFF8, 32'h1);     // IRQ_EN=1
        axil_write(16'h0004, 32'd5);     // LOAD=5
        axil_write(16'h0000, 32'h1);     // ENABLE=1

        repeat(15) @(posedge aclk);     // 타임아웃 대기

        axil_read(16'hFFFC, rd);
        check_eq("TC4:IRQ_PEND", rd & 32'h1, 32'h1);

        // W1C 클리어
        axil_write(16'hFFFC, 32'h1);
        repeat(2) @(posedge aclk);
        axil_read(16'hFFFC, rd);
        check_eq("TC4:IRQ_PEND_cleared", rd & 32'h1, 32'h0);

        // ────────────────────────────────────────────────
        // TC5: ENABLE=0 → 카운터 정지, TIMEOUT_LAT 없음
        // ────────────────────────────────────────────────
        $display("=== TC5: ENABLE=0 카운터 정지 ===");
        axil_write(16'h0000, 32'h2);     // SRST 클리어
        repeat(3) @(posedge aclk);
        axil_write(16'h0000, 32'h0);     // ENABLE=0
        repeat(2) @(posedge aclk);

        axil_write(16'h0004, 32'd5);     // LOAD=5
        // ENABLE=0 상태 유지

        repeat(20) @(posedge aclk);     // 많이 기다려도

        axil_read(16'h0008, rd);
        check_eq("TC5:TIMEOUT_LAT_disabled", rd & 32'h1, 32'h0);
        check_eq("TC5:RUNNING_disabled",     (rd >> 1) & 32'h1, 32'h0);

        // ────────────────────────────────────────────────
        // 최종 판정
        // ────────────────────────────────────────────────
        if (fail_cnt === 0)
            $display("PASS all 11 checks");
        else
            $display("FAIL %0d check(s) failed", fail_cnt);

        $finish;
    end

    // ── 타임아웃 ─────────────────────────────────────────
    initial begin
        #500000;
        $display("FAIL timeout");
        $finish;
    end

endmodule

# Makefile — mmio-wdt-tier1 빌드 규칙
# I-070 AXI4-Lite 워치독 IP (T1)
# 타깃: lint, sim, synth, clean

RTL_DIR := rtl
TB_DIR  := tb
SIM_TOP := tb_axil_wdt_wrap

RTL_SRCS := $(RTL_DIR)/wdt_core.v \
             $(RTL_DIR)/axil_wdt_wrap.v
TB_SRCS  := $(TB_DIR)/tb_axil_wdt_wrap.v

.PHONY: all lint sim synth clean

all: lint sim

# ── 린트 (iverilog -Wall -g2001) ──────────────────────
lint:
	@echo "=== Lint ==="
	iverilog -Wall -g2001 -t null $(RTL_SRCS)
	iverilog -Wall -g2001 -t null $(RTL_SRCS) $(TB_SRCS)
	@echo "Lint OK"

# ── 시뮬레이션 (iverilog + vvp) ──────────────────────
sim:
	@echo "=== Simulation ==="
	iverilog -g2001 -o /tmp/tb_wdt $(RTL_SRCS) $(TB_SRCS)
	vvp /tmp/tb_wdt | tee /tmp/tb_wdt.log
	@if grep -q "^FAIL" /tmp/tb_wdt.log; then \
		echo "SIMULATION FAILED"; exit 1; \
	elif grep -q "^PASS" /tmp/tb_wdt.log; then \
		echo "SIMULATION PASSED"; \
	else \
		echo "SIMULATION: no PASS/FAIL found"; exit 1; \
	fi

# ── 합성 (yosys, 자원 추정) ───────────────────────────
synth:
	@echo "=== Synthesis (yosys, Xilinx 7-series 추정) ==="
	yosys -p "synth_xilinx -flatten -top axil_wdt_wrap; stat" \
	      $(RTL_SRCS) 2>&1 | tee /tmp/synth_wdt.log
	@echo "Synthesis done (see /tmp/synth_wdt.log)"

# ── 정리 ─────────────────────────────────────────────
clean:
	rm -f /tmp/tb_wdt /tmp/tb_wdt.log /tmp/synth_wdt.log

위험·한계

  • LOAD=0 설정 시 ENABLE 직후 즉시 타임아웃 (의도된 동작이나 혼동 가능)
  • 타임아웃 중 ENABLE=0→1 재적재 시 카운터가 LOAD 값으로 돌아옴 — TIMEOUT_LAT는 kick/srst 없이는 클리어 안 됨
  • wdt_rst는 1사이클 펄스 — SoC 리셋 도메인에 맞게 연장 필요 시 외부 로직 추가
  • IRQ_PEND는 소프트웨어 클리어 전에 인터럽트 핸들러가 wdt_rst를 처리해야 경쟁 없음
  • WSTRB 무시: 바이트 쓰기 불가 (AXI4-Lite 단순화)
