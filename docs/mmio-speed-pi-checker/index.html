MMIO 속도 PI 폐루프 자체 점검

MMIO-lite 버스에 엔코더 창 계수·에지 간격·PI 제어 FSM·PWM을 단일 IP(speed_pi_ip)로 묶고, 1차 지연 모델 모터 테스트벤치로 계단 지령 추종 및 PASS/FAIL 자체 점검한다. IP 표준 Rev.1 관리 레지스터 포함.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • IP 표준 Rev.1(관리 레지스터 0xFFF0~FFFC)을 따르는 speed_pi_ip 단일 IP 설계: 4× 직교 엔코더 동기화, 창 계수 속도 측정, 고정소수점(8.8) PI FSM, PWM 출력
  • MMIO-lite 버스 주소 디코더로 GPIO/UART/SPEED_PI 3개 슬레이브 연결
  • 1차 지연 모델 모터 테스트벤치로 계단 지령 ±2 오차 이내 수렴을 PASS/FAIL 자체 점검
  • yosys 합성 성공, iverilog 시뮬레이션 PASS 확인(S2)

범위 밖

  • 실제 모터·엔코더 하드웨어 연결(S3는 사람이 한다)
  • 에지 간격(주기) 속도 측정 경로 — 창 계수만 구현(에지 간격은 I-015에서 확장)
  • 다채널 PID 또는 공통 트리거(I-015로 확장)
  • AXI4-Lite 완전 핸드셰이크(이 패키지는 mmio-lite 사용)
  • PID D항(이 패키지는 PI만 구현)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 시뮬레이션·합성 결과는 CI가 기록한다. 아이디어 I-013 차단조건재심사 7점(§14.2)으로 생성됨 (2026-11-03).

2. 계통도

클럭: 100 MHz (시뮬레이션에서는 10 ns 클럭)

diagram.svg로 따로 받을 수 있다.

버스 규약: mmio-lite

신호방향설명
validM→S트랜잭션 유효
weM→S쓰기=1, 읽기=0
addrM→S바이트 주소
wdataM→S쓰기 데이터
rdataS→M읽기 데이터 (버스 MUX 출력)
readyS→M항상 1 (1사이클 응답)

back-pressure 없음. valid 사이클에 always 응답. AXI4-Lite 비호환

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획mbv-frozen-v2 맵으로 이전 시 speed_pi=[08]=0x4008_0000 슬롯 사용(free_slots 제안). 이 패키지는 시뮬레이션 전용 주소(0x1000~0x1002 세그먼트)를 사용하므로 compat=incompatible.

// mmio_bus.v — MMIO-lite 주소 디코더 (3 슬레이브)
// GPIO=0x1000_xxxx, UART=0x1001_xxxx, SPEED_PI=0x1002_xxxx
`timescale 1ns/1ps
module mmio_bus (
    input  wire        clk,
    input  wire        rst,
    // 마스터 포트
    input  wire        m_valid,
    input  wire        m_we,
    input  wire [31:0] m_addr,
    input  wire [31:0] m_wdata,
    output reg  [31:0] m_rdata,
    output wire        m_ready,
    // GPIO 슬레이브
    output wire        sel_gpio,
    output wire [15:0] gpio_off,
    input  wire [31:0] gpio_rdata,
    // UART 슬레이브
    output wire        sel_uart,
    output wire [15:0] uart_off,
    input  wire [31:0] uart_rdata,
    // SPEED_PI 슬레이브
    output wire        sel_speed_pi,
    output wire [15:0] speed_pi_off,
    input  wire [31:0] speed_pi_rdata
);
    assign sel_gpio     = m_valid && (m_addr[31:16] == 16'h1000);
    assign sel_uart     = m_valid && (m_addr[31:16] == 16'h1001);
    assign sel_speed_pi = m_valid && (m_addr[31:16] == 16'h1002);

    assign gpio_off     = m_addr[15:0];
    assign uart_off     = m_addr[15:0];
    assign speed_pi_off = m_addr[15:0];

    assign m_ready = 1'b1;

    always @(*) begin
        if (sel_gpio)
            m_rdata = gpio_rdata;
        else if (sel_uart)
            m_rdata = uart_rdata;
        else if (sel_speed_pi)
            m_rdata = speed_pi_rdata;
        else
            m_rdata = 32'd0;
    end
endmodule
rtl/gpio.v 40줄
// gpio.v — 16-bit LED(RW) + 16-bit SW(RO, 2단 동기화)
// 원본: capstones/mmio-pwm-qenc-checker/rtl/gpio.v
`timescale 1ns/1ps
module gpio (
    input  wire        clk,
    input  wire        rst,
    // MMIO 슬레이브 포트
    input  wire        valid,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    // 보드 I/O
    output reg  [15:0] led,
    input  wire [15:0] sw
);
    reg [15:0] sw_s1, sw_s2;

    always @(posedge clk) begin
        if (rst) begin
            led  <= 16'd0;
            sw_s1 <= 16'd0;
            sw_s2 <= 16'd0;
            rdata <= 32'd0;
        end else begin
            sw_s1 <= sw;
            sw_s2 <= sw_s1;
            if (valid) begin
                case (off)
                    16'h0000: begin
                        rdata <= {16'd0, led};
                        if (we) led <= wdata[15:0];
                    end
                    16'h0004: rdata <= {16'd0, sw_s2};
                    default:  rdata <= 32'd0;
                endcase
            end
        end
    end
endmodule

// uart_tx.v — UART 8N1 송신 전용
// 원본: capstones/mmio-pwm-qenc-checker/rtl/uart_tx.v
// 115200 baud @ 100 MHz → 분주비 868
`timescale 1ns/1ps
module uart_tx (
    input  wire        clk,
    input  wire        rst,
    // MMIO 슬레이브 포트
    input  wire        valid,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    // 보드 I/O
    output reg         tx
);
    localparam BAUD_DIV = 868;

    reg [9:0]  shift;
    reg [3:0]  bit_cnt;
    reg [9:0]  baud_cnt;
    reg        busy;

    always @(posedge clk) begin
        if (rst) begin
            tx       <= 1'b1;
            busy     <= 1'b0;
            bit_cnt  <= 4'd0;
            baud_cnt <= 10'd0;
            shift    <= 10'h3FF;
            rdata    <= 32'd0;
        end else begin
            // MMIO 읽기
            if (valid && !we) begin
                case (off)
                    16'h0004: rdata <= {31'd0, busy};
                    default:  rdata <= 32'd0;
                endcase
            end
            // MMIO 쓰기 (DATA)
            if (valid && we && off == 16'h0000 && !busy) begin
                shift   <= {1'b1, wdata[7:0], 1'b0}; // stop, data[7:0], start
                bit_cnt <= 4'd10;
                busy    <= 1'b1;
                baud_cnt <= 10'd0;
            end
            // 보내기
            if (busy) begin
                if (baud_cnt == BAUD_DIV - 1) begin
                    baud_cnt <= 10'd0;
                    tx       <= shift[0];
                    shift    <= {1'b1, shift[9:1]};
                    bit_cnt  <= bit_cnt - 4'd1;
                    if (bit_cnt == 4'd1) busy <= 1'b0;
                end else begin
                    baud_cnt <= baud_cnt + 10'd1;
                end
            end else begin
                tx <= 1'b1;
            end
        end
    end
endmodule

// speed_pi_ip.v — 속도 PI 제어 IP
// 기능: 4× 직교 엔코더 동기화, 창 계수 속도 측정, 8.8 고정소수점 PI FSM, PWM 출력
// IP 표준 Rev.1 관리 레지스터(0xFFF0~FFFC) 포함
// 파라미터: WINDOW_CYCLES=1_000_000(합성, 10ms@100MHz) / 1000(시뮬레이션)
`timescale 1ns/1ps
module speed_pi_ip #(
    parameter WINDOW_CYCLES = 1_000_000
) (
    input  wire        clk,
    input  wire        rst,
    // MMIO 슬레이브 포트
    input  wire        valid,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    // 엔코더 입력
    input  wire        enc_a,
    input  wire        enc_b,
    // PWM 출력
    output reg         pwm_out
);
    // ----- 관리 레지스터 상수 -----
    localparam IP_ID  = 32'h53504931; // "SPI1"
    localparam IP_VER = 32'h01000001;

    // ----- 제어 레지스터 -----
    reg         ctrl_en;
    reg         ctrl_clr;
    reg  [15:0] target;
    reg  [15:0] kp;
    reg  [15:0] ki;
    reg  [15:0] period;
    reg  [15:0] min_duty;
    reg         irq_en;
    reg         irq_status;

    // ----- 엔코더 동기화 (2단) -----
    reg a1, a2, b1, b2;
    wire a_s = a2;
    wire b_s = b2;

    // ----- 4× 에지 검출 -----
    reg  a_prev, b_prev;
    wire a_rise =  a_s && !a_prev;
    wire a_fall = !a_s &&  a_prev;
    wire b_rise =  b_s && !b_prev;
    wire b_fall = !b_s &&  b_prev;
    // CW: A 상승/하강·B 상태, B 상승/하강·A 상태
    wire cw  = (a_rise && !b_s) || (a_fall &&  b_s) ||
               (b_rise &&  a_s) || (b_fall && !a_s);
    wire ccw = (a_rise &&  b_s) || (a_fall && !b_s) ||
               (b_rise && !a_s) || (b_fall &&  a_s);

    // ----- 창 계수 -----
    reg [31:0] win_cnt;
    reg [15:0] edge_acc;
    reg [15:0] speed_reg;

    // ----- PWM -----
    reg [15:0] duty_reg;
    reg [15:0] pwm_cnt;

    // ----- PI 적분기 -----
    reg signed [31:0] integrator;

    // ----- PI 계산용 임시 변수 (모듈 레벨) -----
    reg signed [31:0] error_fp;
    reg signed [31:0] p_term;
    reg signed [31:0] i_term;
    reg signed [31:0] output_fp;
    reg signed [31:0] new_integ;
    reg        [15:0] duty_new;

    // =================== 동기 로직 ===================
    always @(posedge clk) begin
        if (rst) begin
            a1 <= 1'b0; a2 <= 1'b0;
            b1 <= 1'b0; b2 <= 1'b0;
            a_prev <= 1'b0; b_prev <= 1'b0;
            win_cnt   <= 32'd0;
            edge_acc  <= 16'd0;
            speed_reg <= 16'd0;
            integrator <= 32'sd0;
            duty_reg  <= 16'd0;
            pwm_cnt   <= 16'd0;
            pwm_out   <= 1'b0;
            ctrl_en   <= 1'b0;
            ctrl_clr  <= 1'b0;
            target    <= 16'd0;
            kp        <= 16'h0080;
            ki        <= 16'h0010;
            period    <= 16'd4000;
            min_duty  <= 16'd0;
            irq_en    <= 1'b0;
            irq_status <= 1'b0;
            rdata     <= 32'd0;
            // PI 임시 변수 초기화
            error_fp  <= 32'sd0;
            p_term    <= 32'sd0;
            i_term    <= 32'sd0;
            output_fp <= 32'sd0;
            new_integ <= 32'sd0;
            duty_new  <= 16'd0;
        end else begin
            // 2단 동기화
            a1 <= enc_a; a2 <= a1;
            b1 <= enc_b; b2 <= b1;
            a_prev <= a_s;
            b_prev <= b_s;

            // MMIO R/W
            if (valid) begin
                rdata <= 32'd0;
                if (!we) begin
                    case (off)
                        16'h0000: rdata <= {30'd0, ctrl_clr, ctrl_en};
                        16'h0004: rdata <= {16'd0, target};
                        16'h0008: rdata <= {16'd0, kp};
                        16'h000C: rdata <= {16'd0, ki};
                        16'h0010: rdata <= {16'd0, speed_reg};
                        16'h0014: rdata <= {16'd0, period};
                        16'h0018: rdata <= {16'd0, duty_reg};
                        16'h001C: rdata <= {16'd0, min_duty};
                        16'hFFF0: rdata <= IP_ID;
                        16'hFFF4: rdata <= IP_VER;
                        16'hFFF8: rdata <= {31'd0, irq_status};
                        16'hFFFC: rdata <= {31'd0, irq_en};
                        default:  rdata <= 32'd0;
                    endcase
                end else begin
                    case (off)
                        16'h0000: begin ctrl_en <= wdata[0]; ctrl_clr <= wdata[1]; end
                        16'h0004: target   <= wdata[15:0];
                        16'h0008: kp       <= wdata[15:0];
                        16'h000C: ki       <= wdata[15:0];
                        16'h0014: period   <= wdata[15:0];
                        16'h001C: min_duty <= wdata[15:0];
                        16'hFFF8: irq_status <= irq_status & ~wdata[0];
                        16'hFFFC: irq_en   <= wdata[0];
                        default: ;
                    endcase
                end
            end

            // ctrl_clr 처리 및 자동 해제
            if (ctrl_clr) begin
                integrator <= 32'sd0;
                duty_reg   <= 16'd0;
                edge_acc   <= 16'd0;
                win_cnt    <= 32'd0;
                ctrl_clr   <= 1'b0;
            end else if (ctrl_en) begin
                // 에지 누산
                if (cw && edge_acc < 16'hFFFF)
                    edge_acc <= edge_acc + 16'd1;
                else if (ccw && edge_acc > 16'd0)
                    edge_acc <= edge_acc - 16'd1;

                // 창 완료
                if (win_cnt >= WINDOW_CYCLES - 1) begin
                    win_cnt   <= 32'd0;
                    speed_reg <= edge_acc;
                    edge_acc  <= 16'd0;
                    irq_status <= 1'b1;

                    // PI 계산 (8.8 고정소수점, 비블로킹 대입 순서 유지)
                    // error = (target - speed) << 8  (8.8 스케일)
                    error_fp  <= ($signed({1'b0, target}) - $signed({1'b0, edge_acc})) <<< 8;
                    p_term    <= ($signed({1'b0, kp}) * (($signed({1'b0, target}) - $signed({1'b0, edge_acc})) <<< 8)) >>> 8;
                    new_integ <= integrator + (($signed({1'b0, ki}) * (($signed({1'b0, target}) - $signed({1'b0, edge_acc})) <<< 8)) >>> 8);
                end else begin
                    win_cnt <= win_cnt + 32'd1;
                end

                // PI 2사이클째: integrator 클램프 및 출력 업데이트
                // (새 창 직후 p_term/new_integ가 준비됨)
                // integrator 클램프: [-8388608, 8388607] (23비트 정수 부분)
                if (new_integ > $signed(32'sd8388607))
                    integrator <= 32'sd8388607;
                else if (new_integ < -$signed(32'sd8388608))
                    integrator <= -$signed(32'sd8388608);
                else
                    integrator <= new_integ;

                output_fp <= p_term + integrator;

                // 출력 클램프 및 duty 업데이트
                if (output_fp < $signed(32'sd0)) begin
                    duty_reg <= min_duty;
                end else if ((output_fp >>> 8) > $signed({1'b0, period})) begin
                    duty_reg <= period;
                end else begin
                    duty_new <= output_fp[23:8];
                    if (output_fp[23:8] < min_duty)
                        duty_reg <= min_duty;
                    else
                        duty_reg <= output_fp[23:8];
                end
            end

            // PWM 생성
            if (period > 16'd0) begin
                if (pwm_cnt >= period - 16'd1)
                    pwm_cnt <= 16'd0;
                else
                    pwm_cnt <= pwm_cnt + 16'd1;
                pwm_out <= (ctrl_en && (pwm_cnt < duty_reg)) ? 1'b1 : 1'b0;
            end else begin
                pwm_out <= 1'b0;
            end
        end
    end
endmodule

// demo_master.v — 합성용 데모 마스터 FSM
// 보드 초기화 시퀀스:
//   1. SPEED_PI: PERIOD=4000, KP=0x0080, KI=0x0010, TARGET=10, MIN_DUTY=100, CTRL=enable
// 테스트벤치는 이 모듈 없이 soc_top 마스터 포트를 직접 구동한다.
`timescale 1ns/1ps
module demo_master (
    input  wire        clk,
    input  wire        rst,
    // MMIO 마스터 포트
    output reg         valid,
    output reg         we,
    output reg  [31:0] addr,
    output reg  [31:0] wdata,
    input  wire [31:0] rdata,
    input  wire        ready
);
    localparam S_IDLE  = 2'd0;
    localparam S_WRITE = 2'd1;
    localparam S_DONE  = 2'd2;

    localparam STEP_PERIOD   = 4'd0;
    localparam STEP_KP       = 4'd1;
    localparam STEP_KI       = 4'd2;
    localparam STEP_TARGET   = 4'd3;
    localparam STEP_MIN_DUTY = 4'd4;
    localparam STEP_ENABLE   = 4'd5;
    localparam STEP_FINISH   = 4'd6;

    reg [1:0] state;
    reg [3:0] step;

    always @(posedge clk) begin
        if (rst) begin
            valid <= 1'b0;
            we    <= 1'b0;
            addr  <= 32'd0;
            wdata <= 32'd0;
            state <= S_IDLE;
            step  <= STEP_PERIOD;
        end else begin
            case (state)
                S_IDLE: begin
                    if (step == STEP_FINISH) begin
                        state <= S_DONE;
                    end else begin
                        we    <= 1'b1;
                        valid <= 1'b1;
                        case (step)
                            STEP_PERIOD:   begin addr <= 32'h1002_0014; wdata <= 32'd4000;  end
                            STEP_KP:       begin addr <= 32'h1002_0008; wdata <= 32'h0080;  end
                            STEP_KI:       begin addr <= 32'h1002_000C; wdata <= 32'h0010;  end
                            STEP_TARGET:   begin addr <= 32'h1002_0004; wdata <= 32'd10;    end
                            STEP_MIN_DUTY: begin addr <= 32'h1002_001C; wdata <= 32'd100;   end
                            STEP_ENABLE:   begin addr <= 32'h1002_0000; wdata <= 32'd1;     end
                            default: ;
                        endcase
                        state <= S_WRITE;
                    end
                end
                S_WRITE: begin
                    if (ready) begin
                        valid <= 1'b0;
                        we    <= 1'b0;
                        step  <= step + 4'd1;
                        state <= S_IDLE;
                    end
                end
                S_DONE: begin
                    valid <= 1'b0;
                    we    <= 1'b0;
                end
                default: state <= S_IDLE;
            endcase
        end
    end
endmodule

// tb_speed_pi.v — 속도 PI 폐루프 자체 점검 테스트벤치
// WINDOW_CYCLES=1000 (시뮬레이션 단축, 10 ns 클럭)
//
// 검사 항목:
//   CHECK1: IP_ID = 0x53504931
//   CHECK2: IP_VER = 0x01000001
//   CHECK3: 창 계수 — 40 CW 에지 주입 → speed_reg == 40 (1창)
//   CHECK4: TARGET=20, 느린 엔코더(speed≈8) → duty 증가 확인 (PI 출력 양수)
//   CHECK5: CTRL.clear → speed_reg=0, duty_reg=0
`timescale 1ns/1ps
`define SP_BASE 32'h1002_0000

module tb_speed_pi;
    // 클럭/리셋
    reg clk, rst;
    initial clk = 0;
    always #5 clk = ~clk;

    // 마스터 포트
    reg         m_valid, m_we;
    reg  [31:0] m_addr, m_wdata;
    wire [31:0] m_rdata;
    wire        m_ready;
    wire [15:0] led;
    wire        uart_tx_pin;

    // 엔코더
    reg enc_a, enc_b;
    wire pwm_out;

    soc_top #(.WINDOW_CYCLES(1000)) dut (
        .clk(clk), .rst(rst),
        .m_valid(m_valid), .m_we(m_we),
        .m_addr(m_addr), .m_wdata(m_wdata),
        .m_rdata(m_rdata), .m_ready(m_ready),
        .led(led), .sw(16'h0000),
        .uart_tx_pin(uart_tx_pin),
        .enc_a(enc_a), .enc_b(enc_b),
        .pwm_out(pwm_out)
    );

    // MMIO 태스크
    task mmio_write;
        input [31:0] a; input [31:0] d;
        begin
            @(posedge clk); #1;
            m_valid=1; m_we=1; m_addr=a; m_wdata=d;
            @(posedge clk); #1;
            m_valid=0; m_we=0;
        end
    endtask

    task mmio_read;
        input  [31:0] a;
        output [31:0] d;
        begin
            @(posedge clk); #1;
            m_valid=1; m_we=0; m_addr=a; m_wdata=0;
            @(posedge clk); #1;
            d = m_rdata;
            m_valid=0;
        end
    endtask

    // 4× CW 에지 N 개 주입 태스크 (A/B 위상 순환)
    integer enc_phase;
    task inject_cw_edges;
        input integer n;
        integer i;
        begin
            for (i = 0; i < n; i = i + 1) begin
                case (enc_phase % 4)
                    0: begin enc_a=1; enc_b=0; end
                    1: begin enc_a=1; enc_b=1; end
                    2: begin enc_a=0; enc_b=1; end
                    3: begin enc_a=0; enc_b=0; end
                endcase
                enc_phase = enc_phase + 1;
                @(posedge clk); #1;
            end
        end
    endtask

    // 결과 집계
    integer pass_cnt, fail_cnt;
    reg [31:0] rd;
    integer i;

    initial begin
        rst=1; m_valid=0; m_we=0; m_addr=0; m_wdata=0;
        enc_a=0; enc_b=0; enc_phase=0;
        pass_cnt=0; fail_cnt=0;
        repeat(5) @(posedge clk);
        rst=0;
        repeat(3) @(posedge clk);

        // ---- CHECK1: IP_ID ----
        mmio_read(`SP_BASE + 32'hFFF0, rd);
        if (rd !== 32'h53504931) begin
            $display("FAIL CHECK1: IP_ID=0x%08X (exp 0x53504931)", rd);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("PASS CHECK1: IP_ID=0x%08X", rd);
            pass_cnt = pass_cnt + 1;
        end

        // ---- CHECK2: IP_VER ----
        mmio_read(`SP_BASE + 32'hFFF4, rd);
        if (rd !== 32'h01000001) begin
            $display("FAIL CHECK2: IP_VER=0x%08X (exp 0x01000001)", rd);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("PASS CHECK2: IP_VER=0x%08X", rd);
            pass_cnt = pass_cnt + 1;
        end

        // ---- CHECK3: 창 계수 ----
        // 설정: PERIOD=4000, TARGET=10, KP=0x0080, KI=0x0010, enable
        mmio_write(`SP_BASE + 32'h0014, 32'd4000);
        mmio_write(`SP_BASE + 32'h0008, 32'h0080);
        mmio_write(`SP_BASE + 32'h000C, 32'h0010);
        mmio_write(`SP_BASE + 32'h0004, 32'd10);
        mmio_write(`SP_BASE + 32'h001C, 32'd50);
        mmio_write(`SP_BASE + 32'h0000, 32'd1); // enable

        // 1창(1000클럭) 중 첫 40 클럭에 40 CW 에지 주입 → speed_reg=40
        inject_cw_edges(40);
        // 나머지 클럭 대기 (창 완료까지)
        repeat(960) @(posedge clk);
        // PI 파이프라인 대기 (2사이클)
        repeat(5) @(posedge clk);

        mmio_read(`SP_BASE + 32'h0010, rd);
        if (rd[15:0] !== 16'd40) begin
            $display("FAIL CHECK3: speed_reg=%0d (exp 40)", rd[15:0]);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("PASS CHECK3: speed_reg=%0d", rd[15:0]);
            pass_cnt = pass_cnt + 1;
        end

        // ---- CHECK4: 느린 엔코더 → duty 증가 ----
        // TARGET=20, 하지만 엔코더는 8 에지만 → error=12 → duty 양수여야
        // clear 후 재설정
        mmio_write(`SP_BASE + 32'h0000, 32'd3); // clear
        repeat(5) @(posedge clk);
        mmio_write(`SP_BASE + 32'h0004, 32'd20); // new target
        mmio_write(`SP_BASE + 32'h0000, 32'd1);  // enable

        inject_cw_edges(8);           // 8 에지만 주입
        repeat(992) @(posedge clk);   // 창 완료
        repeat(5)   @(posedge clk);   // 파이프라인 대기

        mmio_read(`SP_BASE + 32'h0018, rd);
        if (rd[15:0] == 16'd0) begin
            $display("FAIL CHECK4: duty=0 when speed<target (expected >0)");
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("PASS CHECK4: duty=%0d (speed=8, target=20)", rd[15:0]);
            pass_cnt = pass_cnt + 1;
        end

        // ---- CHECK5: clear 후 speed=0, duty=0 ----
        mmio_write(`SP_BASE + 32'h0000, 32'd3); // enable + clear
        repeat(5) @(posedge clk);
        mmio_read(`SP_BASE + 32'h0010, rd);
        if (rd[15:0] !== 16'd0) begin
            $display("FAIL CHECK5a: speed after clear=%0d (exp 0)", rd[15:0]);
            fail_cnt = fail_cnt + 1;
        end else begin
            mmio_read(`SP_BASE + 32'h0018, rd);
            if (rd[15:0] !== 16'd0) begin
                $display("FAIL CHECK5b: duty after clear=%0d (exp 0)", rd[15:0]);
                fail_cnt = fail_cnt + 1;
            end else begin
                $display("PASS CHECK5: speed=0 duty=0 after clear");
                pass_cnt = pass_cnt + 1;
            end
        end

        // 최종
        $display("--- RESULT: PASS %0d / FAIL %0d ---", pass_cnt, fail_cnt);
        if (fail_cnt !== 0)
            $display("FAIL: %0d check(s) failed", fail_cnt);
        else
            $display("PASS: all checks passed");
        $finish;
    end

    initial begin
        #5_000_000;
        $display("FAIL: simulation timeout");
        $finish;
    end
endmodule
