PID 계단 응답 캡처·이득 조정 IP

MMIO-lite 버스에 KP/KI/KD 가변 PID FSM·BRAM 32엔트리 계단 응답 로거·PWM 출력을 단일 IP(pid_cap_ip)로 묶고, 1차 지연 모델 모터 테스트벤치로 계단 지령 추종 및 BRAM 로그 비어있지 않음을 PASS/FAIL 자체 점검한다. IP 표준 Rev.1 관리 레지스터 포함. 아이디어 I-017 (차단조건재심사 9점)으로 생성됨 (2026-11-04).

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • IP 표준 Rev.1(관리 레지스터 0xFFF0~FFFC)을 따르는 pid_cap_ip 단일 IP 설계: 4× 직교 엔코더 동기화, 창 계수 속도 측정, 8.8 고정소수점 PID FSM(D항 포함), PWM 출력
  • BRAM 32엔트리 계단 응답 로거: CAPTURE_CTRL 레지스터로 캡처 시작, CAPTURE_ADDR/DATA로 호스트 읽기
  • MMIO-lite 버스 주소 디코더로 GPIO/UART/PID_CAP 3개 슬레이브 연결
  • 1차 지연 모델 모터 테스트벤치로 계단 지령 ±5 오차 이내 수렴(8.8 고정소수 양자화 고려) 및 BRAM 로그 비어있지 않음을 PASS/FAIL 자체 점검
  • yosys 합성 성공, iverilog 시뮬레이션 PASS 확인(S2)

범위 밖

  • 실제 모터·엔코더 하드웨어 연결(S3는 사람이 한다)
  • AXI4-Lite 완전 핸드셰이크(이 패키지는 mmio-lite 사용)
  • 다채널 PID(이 패키지는 단일 채널)
  • 속도 적분 위치 제어(이 패키지는 속도 루프만)
  • 실시간 튜닝 UI(다음 단계 T5로 연기)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 시뮬레이션·합성 결과는 CI가 기록한다.

2. 계통도

클럭: 100 MHz (시뮬레이션에서는 10 ns 클럭)

diagram.svg로 따로 받을 수 있다.

버스 규약: mmio-lite

신호방향설명
validM→S트랜잭션 유효
weM→S쓰기=1, 읽기=0
addrM→S바이트 주소
wdataM→S쓰기 데이터
rdataS→M읽기 데이터 (버스 MUX 출력)
readyS→M항상 1 (1사이클 응답)

back-pressure 없음. valid 사이클에 always 응답. AXI4-Lite 비호환

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획mbv-frozen-v2 맵으로 이전 시 pid_cap=[16]=0x4016_0000 슬롯 사용(free_slots 첫 빈 칸). INTC 13 사용(free_intc 첫 빈 번호). 이 패키지는 시뮬레이션 전용 주소(0x1000~0x1002 세그먼트)를 사용하므로 compat=incompatible.

// mmio_bus.v — MMIO-lite 주소 디코더 (mmio-pid-step-capture)
// addr[31:16] 기준 3개 슬레이브 선택. ready=1 사이클 고정.
// 슬레이브:
//   0x1000_xxxx : GPIO (LED/SW)
//   0x1001_xxxx : UART TX
//   0x1002_xxxx : PID_CAP IP
// Verilog-2001
`default_nettype none

module mmio_bus (
    input  wire        clk,
    input  wire        rst,
    // 마스터 포트
    input  wire        m_valid,
    input  wire        m_we,
    input  wire [31:0] m_addr,
    input  wire [31:0] m_wdata,
    output reg  [31:0] m_rdata,
    output wire        m_ready,
    // 보드 I/O (GPIO)
    input  wire [15:0] sw,
    output wire [15:0] led,
    // PID_CAP I/O
    input  wire        enc_a,
    input  wire        enc_b,
    output wire        pwm_out,
    // UART
    output wire        uart_txd
);

    // 슬레이브 선택 (상위 16비트 비교)
    wire sel_gpio    = m_valid && (m_addr[31:16] == 16'h1000);
    wire sel_uart    = m_valid && (m_addr[31:16] == 16'h1001);
    wire sel_pid_cap = m_valid && (m_addr[31:16] == 16'h1002);

    // 슬레이브별 버스 신호
    wire [31:0] gpio_rdata;
    wire [31:0] uart_rdata;
    wire [31:0] pid_cap_rdata;

    // GPIO 슬레이브
    gpio u_gpio (
        .clk   (clk),
        .rst   (rst),
        .valid (sel_gpio),
        .we    (m_we),
        .addr  (m_addr[3:0]),
        .wdata (m_wdata),
        .rdata (gpio_rdata),
        .sw    (sw),
        .led   (led)
    );

    // UART TX 슬레이브
    uart_tx u_uart (
        .clk   (clk),
        .rst   (rst),
        .valid (sel_uart),
        .we    (m_we),
        .addr  (m_addr[3:0]),
        .wdata (m_wdata),
        .rdata (uart_rdata),
        .txd   (uart_txd)
    );

    // PID_CAP 슬레이브
    pid_cap_ip u_pid_cap (
        .clk   (clk),
        .rst   (rst),
        .valid (sel_pid_cap),
        .we    (m_we),
        .addr  (m_addr[15:0]),
        .wdata (m_wdata),
        .rdata (pid_cap_rdata),
        .enc_a (enc_a),
        .enc_b (enc_b),
        .pwm   (pwm_out)
    );

    // 읽기 데이터 MUX
    always @(*) begin
        if (sel_gpio)
            m_rdata = gpio_rdata;
        else if (sel_uart)
            m_rdata = uart_rdata;
        else if (sel_pid_cap)
            m_rdata = pid_cap_rdata;
        else
            m_rdata = 32'hDEAD_BEEF;
    end

    // ready는 항상 1 (1사이클 응답)
    assign m_ready = 1'b1;

endmodule
`default_nettype wire
rtl/gpio.v 44줄
// gpio.v — GPIO LED/SW 슬레이브 (mmio-pid-step-capture)
// 0x00: LED[15:0] (RW, 리셋=0x0000)
// 0x04: SW[15:0]  (RO, 2단 동기화)
// Verilog-2001
`default_nettype none

module gpio (
    input  wire        clk,
    input  wire        rst,
    input  wire        valid,
    input  wire        we,
    input  wire [3:0]  addr,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    input  wire [15:0] sw,
    output reg  [15:0] led
);

    // 2단 동기화기
    reg [15:0] sw_meta, sw_sync;
    always @(posedge clk) begin
        sw_meta <= sw;
        sw_sync <= sw_meta;
    end

    // 쓰기
    always @(posedge clk) begin
        if (rst)
            led <= 16'h0000;
        else if (valid && we && addr == 4'h0)
            led <= wdata[15:0];
    end

    // 읽기
    always @(*) begin
        case (addr)
            4'h0: rdata = {16'h0, led};
            4'h4: rdata = {16'h0, sw_sync};
            default: rdata = 32'h0;
        endcase
    end

endmodule

rtl/uart_tx.v 65줄
// uart_tx.v — UART 8N1 송신 전용 슬레이브 (mmio-pid-step-capture)
// 0x00: DATA   (WO) 송신 바이트 [7:0]
// 0x04: STATUS (RO) bit0=busy
// 115200 baud @ 100 MHz → 분주: 100_000_000/115200 ≈ 868
// Verilog-2001
`default_nettype none

module uart_tx (
    input  wire        clk,
    input  wire        rst,
    input  wire        valid,
    input  wire        we,
    input  wire [3:0]  addr,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    output reg         txd
);

    parameter CLK_HZ   = 100_000_000;
    parameter BAUD     = 115_200;
    localparam DIVIDER = CLK_HZ / BAUD;  // 868

    reg [9:0]  shift;   // start + 8 data + stop
    reg [9:0]  baud_cnt;
    reg [3:0]  bit_cnt;
    reg        busy;

    // 쓰기: 바이트 적재
    always @(posedge clk) begin
        if (rst) begin
            busy    <= 1'b0;
            txd     <= 1'b1;
            bit_cnt <= 4'd0;
            baud_cnt<= 10'd0;
            shift   <= 10'h3FF;
        end else if (!busy && valid && we && addr == 4'h0) begin
            // start(0) + data[7:0] + stop(1)
            shift    <= {1'b1, wdata[7:0], 1'b0};
            bit_cnt  <= 4'd10;
            baud_cnt <= 10'd0;
            busy     <= 1'b1;
        end else if (busy) begin
            if (baud_cnt == DIVIDER - 1) begin
                baud_cnt <= 10'd0;
                txd      <= shift[0];
                shift    <= {1'b1, shift[9:1]};
                bit_cnt  <= bit_cnt - 4'd1;
                if (bit_cnt == 4'd1)
                    busy <= 1'b0;
            end else begin
                baud_cnt <= baud_cnt + 10'd1;
            end
        end
    end

    // 읽기
    always @(*) begin
        case (addr)
            4'h4: rdata = {31'h0, busy};
            default: rdata = 32'h0;
        endcase
    end

endmodule
`default_nettype wire
rtl/pid_cap_ip.v 349줄
// pid_cap_ip.v — PID 캡처 IP (mmio-pid-step-capture)
// 기능:
//   - 4× 직교 엔코더 2단 동기화 + 창 계수 속도 측정
//   - 8.8 고정소수점 PID FSM (anti-windup 적분기 클램프)
//   - PWM 출력
//   - BRAM 32엔트리 계단 응답 로거
//   - IP 표준 Rev.1 관리 레지스터 (0xFFF0~FFFC)
//
// 레지스터맵 (오프셋 바이트, 16비트 버스):
//   0x00: CTRL        RW  bit0=enable, bit1=clear_pos, bit2=cap_start
//   0x04: TARGET      RW  [15:0] 목표 속도 (카운트/창)
//   0x08: KP          RW  [15:0] 비례 이득 8.8 고정소수점
//   0x0C: KI          RW  [15:0] 적분 이득 8.8 고정소수점
//   0x10: KD          RW  [15:0] 미분 이득 8.8 고정소수점
//   0x14: SPEED       RO  [15:0] 현재 속도 (카운트/창)
//   0x18: PERIOD      RW  [15:0] PWM 주기 (클럭 사이클)
//   0x1C: DUTY        RO  [15:0] 현재 PWM 듀티
//   0x20: MIN_DUTY    RW  [15:0] 최소 PWM 듀티
//   0x24: CAP_ADDR    RW  [4:0]  BRAM 읽기 주소 (0~31)
//   0x28: CAP_DATA    RO  [15:0] CAP_ADDR 위치 속도 로그
//   0x2C: CAP_STATUS  RO  bit0=cap_full, [12:8]=cap_count
//   0xFFF0: ID        RO  0x50494431 (PID1)
//   0xFFF4: VER       RO  0x01000001
//   0xFFF8: IRQ_STATUS  RW1C  bit0=speed_ready, bit1=cap_full
//   0xFFFC: IRQ_ENABLE  RW    bit0=speed_irq_en, bit1=cap_irq_en
//
// Verilog-2001
`default_nettype none

module pid_cap_ip #(
    parameter WINDOW_CYCLES = 1_000,    // 시뮬: 1_000, 합성: 1_000_000
    parameter CAP_DEPTH     = 32,       // BRAM 로그 엔트리 수
    parameter PWM_PERIOD    = 1_000     // 기본 PWM 주기 (클럭 사이클)
) (
    input  wire        clk,
    input  wire        rst,
    // MMIO-lite 슬레이브 포트
    input  wire        valid,
    input  wire        we,
    input  wire [15:0] addr,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    // 하드웨어 포트
    input  wire        enc_a,
    input  wire        enc_b,
    output wire        pwm
);

    // ─── 레지스터 ──────────────────────────────────────────────────────
    reg        enable;
    reg        clear_pos;
    reg        cap_start_r;
    reg [15:0] target;
    reg [15:0] kp;
    reg [15:0] ki;
    reg [15:0] kd;
    reg [15:0] period;      // PWM 주기
    reg [15:0] min_duty;
    reg [4:0]  cap_rd_addr;
    reg [1:0]  irq_en;

    // ─── 엔코더 동기화 ─────────────────────────────────────────────────
    reg [1:0] enc_a_sr, enc_b_sr;
    wire enc_a_s = enc_a_sr[1];
    wire enc_b_s = enc_b_sr[1];

    always @(posedge clk) begin
        enc_a_sr <= {enc_a_sr[0], enc_a};
        enc_b_sr <= {enc_b_sr[0], enc_b};
    end

    // 4× 직교 디코더
    reg  enc_a_prev, enc_b_prev;
    wire a_rise = enc_a_s & ~enc_a_prev;
    wire a_fall = ~enc_a_s & enc_a_prev;
    wire b_rise = enc_b_s & ~enc_b_prev;
    wire b_fall = ~enc_b_s & enc_b_prev;
    wire fwd = (a_rise & ~enc_b_s) | (b_rise & enc_a_s) |
               (a_fall & enc_b_s)  | (b_fall & ~enc_a_s);
    wire rev = (a_rise & enc_b_s)  | (b_rise & ~enc_a_s) |
               (a_fall & ~enc_b_s) | (b_fall & enc_a_s);

    always @(posedge clk) begin
        enc_a_prev <= enc_a_s;
        enc_b_prev <= enc_b_s;
    end

    // ─── 창 계수 속도 측정 ─────────────────────────────────────────────
    reg [19:0] win_cnt;
    reg [15:0] pulse_acc;
    reg [15:0] speed;       // 현재 속도 (카운트/창)
    reg        speed_ready;

    always @(posedge clk) begin
        if (rst || !enable) begin
            win_cnt    <= 20'd0;
            pulse_acc  <= 16'd0;
            speed      <= 16'd0;
            speed_ready<= 1'b0;
        end else begin
            speed_ready <= 1'b0;
            if (fwd && !rev) pulse_acc <= pulse_acc + 16'd1;
            if (win_cnt == WINDOW_CYCLES - 1) begin
                win_cnt    <= 20'd0;
                speed      <= pulse_acc;
                pulse_acc  <= 16'd0;
                speed_ready<= 1'b1;
            end else begin
                win_cnt <= win_cnt + 20'd1;
            end
        end
    end

    // ─── PID FSM ───────────────────────────────────────────────────────
    // 상태: IDLE → CALC_ERR → CALC_P → CALC_I → CALC_D → CALC_SUM → UPDATE
    localparam S_IDLE     = 3'd0;
    localparam S_CALC_ERR = 3'd1;
    localparam S_CALC_P   = 3'd2;
    localparam S_CALC_I   = 3'd3;
    localparam S_CALC_D   = 3'd4;
    localparam S_CALC_SUM = 3'd5;
    localparam S_UPDATE   = 3'd6;

    reg [2:0]  pid_state;
    reg signed [15:0] err;
    reg signed [15:0] err_prev;
    reg signed [31:0] integrator;   // 클램프: -32768 ~ 32767
    reg signed [31:0] new_intg;     // 적분기 중간값 (S_CALC_I)
    reg signed [15:0] p_term;
    reg signed [15:0] i_term;
    reg signed [15:0] d_term;
    reg signed [23:0] pid_out_raw;
    reg [15:0] duty;

    // 8.8 고정소수점 곱셈 (부호 있음): 결과 하위 8비트를 버려 16비트 정수 반환
    function signed [15:0] fxp_mul;
        input signed [15:0] a;
        input signed [15:0] b;
        reg signed [31:0] tmp;
        begin
            tmp = $signed(a) * $signed(b);
            fxp_mul = tmp[23:8];
        end
    endfunction

    always @(posedge clk) begin
        if (rst || !enable) begin
            pid_state  <= S_IDLE;
            err        <= 16'sd0;
            err_prev   <= 16'sd0;
            integrator <= 32'sd0;
            duty       <= 16'd0;
            p_term     <= 16'sd0;
            i_term     <= 16'sd0;
            d_term     <= 16'sd0;
            pid_out_raw<= 24'sd0;
        end else begin
            case (pid_state)
                S_IDLE: begin
                    if (speed_ready)
                        pid_state <= S_CALC_ERR;
                end
                S_CALC_ERR: begin
                    err       <= $signed({1'b0, target}) - $signed({1'b0, speed});
                    pid_state <= S_CALC_P;
                end
                S_CALC_P: begin
                    p_term    <= fxp_mul(err, $signed(kp));
                    pid_state <= S_CALC_I;
                end
                S_CALC_I: begin
                    // 적분기 클램프 (anti-windup)
                    new_intg = integrator + $signed({{16{err[15]}}, err});
                    if (new_intg > 32'sd32767)
                        integrator <= 32'sd32767;
                    else if (new_intg < -32'sd32768)
                        integrator <= -32'sd32768;
                    else
                        integrator <= new_intg;
                    i_term    <= fxp_mul($signed(integrator[15:0]), $signed(ki));
                    pid_state <= S_CALC_D;
                end
                S_CALC_D: begin
                    d_term    <= fxp_mul((err - err_prev), $signed(kd));
                    err_prev  <= err;
                    pid_state <= S_CALC_SUM;
                end
                S_CALC_SUM: begin
                    pid_out_raw <= $signed({{8{p_term[15]}}, p_term}) +
                                   $signed({{8{i_term[15]}}, i_term}) +
                                   $signed({{8{d_term[15]}}, d_term});
                    pid_state   <= S_UPDATE;
                end
                S_UPDATE: begin
                    // 듀티 클램프: min_duty ~ period
                    if ($signed(pid_out_raw) <= 24'sd0)
                        duty <= min_duty;
                    else if (pid_out_raw[15:0] > period)
                        duty <= period;
                    else if (pid_out_raw[15:0] < min_duty)
                        duty <= min_duty;
                    else
                        duty <= pid_out_raw[15:0];
                    pid_state <= S_IDLE;
                end
                default: pid_state <= S_IDLE;
            endcase
        end
    end

    // ─── PWM 출력 ──────────────────────────────────────────────────────
    reg [15:0] pwm_cnt;
    reg        pwm_r;

    always @(posedge clk) begin
        if (rst || !enable) begin
            pwm_cnt <= 16'd0;
            pwm_r   <= 1'b0;
        end else begin
            if (pwm_cnt >= period - 1)
                pwm_cnt <= 16'd0;
            else
                pwm_cnt <= pwm_cnt + 16'd1;
            pwm_r <= (pwm_cnt < duty) ? 1'b1 : 1'b0;
        end
    end
    assign pwm = pwm_r;

    // ─── BRAM 계단 응답 로거 ──────────────────────────────────────────
    // 32엔트리 × 16비트 동기 BRAM
    reg [15:0] cap_bram [0:CAP_DEPTH-1];
    reg [4:0]  cap_wr_addr;
    reg        cap_active;
    reg        cap_full;
    reg [5:0]  cap_count;   // 0~32
    reg [15:0] cap_rd_data;

    // BRAM 읽기 (동기)
    always @(posedge clk) begin
        cap_rd_data <= cap_bram[cap_rd_addr];
    end

    // BRAM 쓰기 (속도 준비 시 캡처 활성이면 기록)
    always @(posedge clk) begin
        if (rst) begin
            cap_wr_addr <= 5'd0;
            cap_active  <= 1'b0;
            cap_full    <= 1'b0;
            cap_count   <= 6'd0;
        end else begin
            // cap_start_r 엣지로 캡처 시작
            if (cap_start_r && !cap_active && !cap_full) begin
                cap_active  <= 1'b1;
                cap_wr_addr <= 5'd0;
                cap_full    <= 1'b0;
                cap_count   <= 6'd0;
            end
            // 속도 갱신 시 로그
            if (cap_active && speed_ready) begin
                cap_bram[cap_wr_addr] <= speed;
                if (cap_wr_addr == CAP_DEPTH - 1) begin
                    cap_active  <= 1'b0;
                    cap_full    <= 1'b1;
                    cap_count   <= CAP_DEPTH;
                end else begin
                    cap_wr_addr <= cap_wr_addr + 5'd1;
                    cap_count   <= cap_count + 6'd1;
                end
            end
        end
    end

    // ─── IRQ ──────────────────────────────────────────────────────────
    reg [1:0] irq_status;
    always @(posedge clk) begin
        if (rst)
            irq_status <= 2'b0;
        else begin
            if (speed_ready) irq_status[0] <= 1'b1;
            if (cap_full)    irq_status[1] <= 1'b1;
            // RW1C: 쓰기로 클리어
            if (valid && we && addr == 16'hFFF8)
                irq_status <= irq_status & ~wdata[1:0];
        end
    end

    // ─── 레지스터 쓰기 ────────────────────────────────────────────────
    always @(posedge clk) begin
        if (rst) begin
            enable      <= 1'b0;
            clear_pos   <= 1'b0;
            cap_start_r <= 1'b0;
            target      <= 16'd0;
            kp          <= 16'h0100; // 1.0 in 8.8
            ki          <= 16'h0010;
            kd          <= 16'h0008;
            period      <= PWM_PERIOD[15:0];
            min_duty    <= 16'd50;
            cap_rd_addr <= 5'd0;
            irq_en      <= 2'b0;
        end else begin
            cap_start_r <= 1'b0; // 1사이클 펄스
            clear_pos   <= 1'b0;
            if (valid && we) begin
                case (addr[7:0])
                    8'h00: begin
                        enable      <= wdata[0];
                        clear_pos   <= wdata[1];
                        cap_start_r <= wdata[2];
                    end
                    8'h04: target    <= wdata[15:0];
                    8'h08: kp        <= wdata[15:0];
                    8'h0C: ki        <= wdata[15:0];
                    8'h10: kd        <= wdata[15:0];
                    8'h18: period    <= wdata[15:0];
                    8'h20: min_duty  <= wdata[15:0];
                    8'h24: cap_rd_addr <= wdata[4:0];
                    8'hFC: irq_en    <= wdata[1:0];
                    default: ;
                endcase
            end
        end
    end

    // ─── 레지스터 읽기 ────────────────────────────────────────────────
    always @(*) begin
        case (addr)
            16'h0000: rdata = {29'h0, cap_start_r, clear_pos, enable};
            16'h0004: rdata = {16'h0, target};
            16'h0008: rdata = {16'h0, kp};
            16'h000C: rdata = {16'h0, ki};
            16'h0010: rdata = {16'h0, kd};
            16'h0014: rdata = {16'h0, speed};
            16'h0018: rdata = {16'h0, period};
            16'h001C: rdata = {16'h0, duty};
            16'h0020: rdata = {16'h0, min_duty};
            16'h0024: rdata = {27'h0, cap_rd_addr};
            16'h0028: rdata = {16'h0, cap_rd_data};
            16'h002C: rdata = {19'h0, cap_count[4:0], 3'h0, cap_full};
            16'hFFF0: rdata = 32'h50494431; // "PID1"
            16'hFFF4: rdata = 32'h01000001;
            16'hFFF8: rdata = {30'h0, irq_status};
            16'hFFFC: rdata = {30'h0, irq_en};
            default:  rdata = 32'h0;
        endcase
    end

endmodule

// demo_master.v — 합성용 데모 마스터 FSM (mmio-pid-step-capture)
// MMIO 버스에 초기화 시퀀스를 발행하고 PID 제어를 시작한다.
// 시뮬레이션에서는 테스트벤치가 버스를 직접 구동하므로
// 이 모듈은 합성용 래퍼로만 쓰인다.
// Verilog-2001
`default_nettype none

module demo_master (
    input  wire        clk,
    input  wire        rst,
    // MMIO-lite 마스터 포트
    output reg         valid,
    output reg         we,
    output reg  [31:0] addr,
    output reg  [31:0] wdata,
    input  wire [31:0] rdata,
    input  wire        ready
);

    // PID_CAP 슬레이브 베이스 주소
    localparam BASE_PID = 32'h1002_0000;

    // 초기화 FSM
    localparam [4:0]
        S_RESET     = 5'd0,
        S_SET_KP    = 5'd1,
        S_SET_KI    = 5'd2,
        S_SET_KD    = 5'd3,
        S_SET_PRD   = 5'd4,
        S_SET_MIN   = 5'd5,
        S_SET_TGT   = 5'd6,
        S_SET_CAP   = 5'd7,
        S_SET_EN    = 5'd8,
        S_RUNNING   = 5'd9;

    reg [4:0] state;
    reg [19:0] wait_cnt;

    always @(posedge clk) begin
        if (rst) begin
            state    <= S_RESET;
            valid    <= 1'b0;
            we       <= 1'b0;
            addr     <= 32'h0;
            wdata    <= 32'h0;
            wait_cnt <= 20'd0;
        end else begin
            valid <= 1'b0; // 기본: 트랜잭션 없음

            case (state)
                S_RESET: begin
                    // 짧은 대기 후 초기화 시작
                    if (wait_cnt == 20'd10) begin
                        wait_cnt <= 20'd0;
                        state    <= S_SET_KP;
                    end else begin
                        wait_cnt <= wait_cnt + 20'd1;
                    end
                end
                S_SET_KP: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h08;
                    wdata <= 32'h0180;  // KP = 1.5 (8.8)
                    if (ready) state <= S_SET_KI;
                end
                S_SET_KI: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h0C;
                    wdata <= 32'h0020;  // KI = 0.125 (8.8)
                    if (ready) state <= S_SET_KD;
                end
                S_SET_KD: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h10;
                    wdata <= 32'h0010;  // KD = 0.0625 (8.8)
                    if (ready) state <= S_SET_PRD;
                end
                S_SET_PRD: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h18;
                    wdata <= 32'd1000;  // PWM 주기 1000 클럭
                    if (ready) state <= S_SET_MIN;
                end
                S_SET_MIN: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h20;
                    wdata <= 32'd50;    // 최소 듀티 50
                    if (ready) state <= S_SET_TGT;
                end
                S_SET_TGT: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h04;
                    wdata <= 32'd100;   // 목표 속도 100 카운트/창
                    if (ready) state <= S_SET_CAP;
                end
                S_SET_CAP: begin
                    // CTRL: enable=1, cap_start=1
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h00;
                    wdata <= 32'h5;     // bit2=cap_start, bit0=enable
                    if (ready) state <= S_SET_EN;
                end
                S_SET_EN: begin
                    // CTRL: enable=1 (cap_start 클리어)
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= BASE_PID + 32'h00;
                    wdata <= 32'h1;     // bit0=enable
                    if (ready) state <= S_RUNNING;
                end
                S_RUNNING: begin
                    // 동작 중 — 버스 조용히 유지
                    valid <= 1'b0;
                end
                default: state <= S_RESET;
            endcase
        end
    end

endmodule

// tb_pid_cap.v — PID 캡처 IP 자체 점검 테스트벤치 (mmio-pid-step-capture)
//
// 검증 항목:
//   1. KP/KI/KD/TARGET/PERIOD/MIN_DUTY 쓰기·읽기 왕복
//   2. 1차 지연 모터 모델로 계단 지령 추종: 최종 오차 ±5 이내 수렴
//   3. BRAM 캡처 로그에 32엔트리가 채워짐
//   4. CAP_STATUS.cap_full 플래그 설정
//   5. ID/VER 관리 레지스터 값 확인
//
// 통과하면 "PASS: ...", 실패하면 "FAIL: ..." 를 $display로 출력하고 종료.
// !==로 비교해 X/Z도 실패로 처리.
//
// Verilog-2001
`timescale 1ns/1ps
`default_nettype none

module tb_pid_cap;

    // ─── 클럭·리셋 ────────────────────────────────────────────────────
    reg clk, rst;
    initial clk = 0;
    always #5 clk = ~clk; // 100 MHz (10 ns 주기)

    // ─── MMIO-lite 버스 (테스트벤치가 마스터) ─────────────────────────
    reg         valid;
    reg         we;
    reg  [15:0] addr;
    reg  [31:0] wdata;
    wire [31:0] rdata;

    // ─── 엔코더 모델 ──────────────────────────────────────────────────
    reg  enc_a, enc_b;
    wire pwm;

    // ─── DUT 인스턴스 ─────────────────────────────────────────────────
    pid_cap_ip #(
        .WINDOW_CYCLES (1_000),
        .CAP_DEPTH     (32),
        .PWM_PERIOD    (1_000)
    ) dut (
        .clk   (clk),
        .rst   (rst),
        .valid (valid),
        .we    (we),
        .addr  (addr),
        .wdata (wdata),
        .rdata (rdata),
        .enc_a (enc_a),
        .enc_b (enc_b),
        .pwm   (pwm)
    );

    // ─── MMIO 쓰기·읽기 태스크 ────────────────────────────────────────
    task mmio_write;
        input [15:0] a;
        input [31:0] d;
        begin
            @(negedge clk);
            valid <= 1'b1; we <= 1'b1; addr <= a; wdata <= d;
            @(posedge clk); #1;
            valid <= 1'b0; we <= 1'b0;
        end
    endtask

    task mmio_read;
        input  [15:0] a;
        output [31:0] d;
        begin
            @(negedge clk);
            valid <= 1'b1; we <= 1'b0; addr <= a; wdata <= 32'h0;
            @(posedge clk); #1;
            d = rdata;
            valid <= 1'b0;
        end
    endtask

    // ─── 1차 지연 모터 모델 ───────────────────────────────────────────
    // 속도_new = 속도_old + (pwm_duty * gain - 속도_old) / tau
    // 정수 근사: 오버플로 방지를 위해 32비트 사용
    // 파라미터: gain=1 (PWM 듀티가 목표 속도와 같은 단위), tau=20 창 주기
    real motor_speed;
    real motor_tau;
    integer enc_pulse_budget; // 이번 창에서 출력해야 할 펄스 수

    // 엔코더 펄스 생성: enc_pulse_budget 만큼 A/B 펄스 생성
    // 4× 카운트이므로 1 엔코더 사이클 = 4 펄스. 단순화: 1 pulse = 1 엣지쌍
    integer pulse_i;
    integer gen_pulses;
    task gen_encoder_pulses;
        input integer n; // 발생시킬 4× 카운트 수
        integer i;
        begin
            for (i = 0; i < n; i = i + 1) begin
                // A 상승 엣지 (B=0이면 정방향)
                @(negedge clk); enc_a = 1'b1; enc_b = 1'b0;
                @(negedge clk); enc_b = 1'b1;
                @(negedge clk); enc_a = 1'b0;
                @(negedge clk); enc_b = 1'b0;
            end
        end
    endtask

    // ─── 테스트 변수 ──────────────────────────────────────────────────
    integer     fail_cnt;
    reg  [31:0] rd_val;
    integer     window_idx;
    integer     final_speed;
    integer     cap_entry;
    integer     i;
    reg  [15:0] log_data [0:31];
    integer     log_nonzero;

    // ─── 메인 테스트 ──────────────────────────────────────────────────
    initial begin
        fail_cnt = 0;
        enc_a = 0; enc_b = 0;
        valid = 0; we = 0; addr = 0; wdata = 0;
        motor_speed = 0.0;
        motor_tau   = 20.0; // 20 창 주기의 1차 시상수

        // 리셋
        rst = 1;
        repeat(10) @(posedge clk);
        rst = 0;
        @(posedge clk);

        // ── 테스트 1: 관리 레지스터 확인 ─────────────────────────────
        mmio_read(16'hFFF0, rd_val);
        if (rd_val !== 32'h50494431) begin
            $display("FAIL: ID reg = %08h, expected 50494431", rd_val);
            fail_cnt = fail_cnt + 1;
        end

        mmio_read(16'hFFF4, rd_val);
        if (rd_val !== 32'h01000001) begin
            $display("FAIL: VER reg = %08h, expected 01000001", rd_val);
            fail_cnt = fail_cnt + 1;
        end

        // ── 테스트 2: 레지스터 쓰기·읽기 왕복 ───────────────────────
        mmio_write(16'h08, 32'h0180); // KP = 1.5
        mmio_read (16'h08, rd_val);
        if (rd_val[15:0] !== 16'h0180) begin
            $display("FAIL: KP roundtrip = %04h, expected 0180", rd_val[15:0]);
            fail_cnt = fail_cnt + 1;
        end

        mmio_write(16'h0C, 32'h0020); // KI = 0.125
        mmio_write(16'h10, 32'h0010); // KD = 0.0625
        mmio_write(16'h18, 32'd1000); // PWM 주기
        mmio_write(16'h20, 32'd50);   // 최소 듀티
        mmio_write(16'h04, 32'd100);  // 목표 속도 100

        // ── 테스트 3: cap_start + enable 설정 ────────────────────────
        mmio_write(16'h00, 32'h7); // bit2=cap_start, bit1=clear_pos, bit0=enable
        // cap_start는 1사이클 펄스이므로 enable만 남기도록 다시 쓰기
        mmio_write(16'h00, 32'h1); // bit0=enable

        // ── 테스트 4: 모터 모델 시뮬레이션 ──────────────────────────
        // 50 창 주기 동안 엔코더 펄스를 모터 모델에 따라 생성하고
        // PID 제어 루프가 목표 100에 수렴하는지 확인한다.
        for (window_idx = 0; window_idx < 70; window_idx = window_idx + 1) begin
            // 1차 지연 모터 모델: 현재 PWM 듀티를 입력으로 속도 갱신
            // duty는 DUT 내부 레지스터. 테스트벤치에서 직접 읽기
            mmio_read(16'h1C, rd_val);
            // motor_speed 갱신: 1차 지연
            motor_speed = motor_speed + (rd_val * 1.0 / 1000.0 * 100.0 - motor_speed) / motor_tau;
            if (motor_speed < 0.0) motor_speed = 0.0;

            // 이번 창에서 발생할 4× 카운트 수 (정수화)
            gen_pulses = $rtoi(motor_speed);
            if (gen_pulses > 200) gen_pulses = 200; // 상한 클램프

            // 창 주기의 절반(500 클럭) 안에 펄스 생성, 나머지는 대기
            // 간략화: 각 펄스당 4클럭(A/B 에지 1쌍), 나머지는 클럭만 흘린다
            gen_encoder_pulses(gen_pulses);
            // WINDOW_CYCLES=1000이므로 나머지 클럭 흘리기
            repeat(1000 - gen_pulses * 4) @(posedge clk);
        end

        // 최종 속도 확인 (창 70개 후)
        mmio_read(16'h14, rd_val);
        final_speed = rd_val[15:0];
        if ((final_speed < 95) || (final_speed > 105)) begin
            $display("FAIL: final speed = %0d, expected 95~105 (target=100)", final_speed);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("INFO: final speed = %0d (target=100) OK", final_speed);
        end

        // ── 테스트 5: BRAM 캡처 로그 확인 ───────────────────────────
        // 32개 창이 쌓이면 cap_full이 된다. 이미 70 창을 돌렸으므로 체크
        mmio_read(16'h2C, rd_val);
        if (rd_val[0] !== 1'b1) begin
            $display("FAIL: cap_full not set after 50 windows (CAP_STATUS=%08h)", rd_val);
            fail_cnt = fail_cnt + 1;
        end

        // CAP_ADDR를 0~31로 순서대로 읽어 비어있지 않음(최소 1개 non-zero) 확인
        log_nonzero = 0;
        for (i = 0; i < 32; i = i + 1) begin
            mmio_write(16'h24, i);           // CAP_ADDR 설정
            @(posedge clk); @(posedge clk); // BRAM 동기 읽기 대기
            mmio_read(16'h28, rd_val);
            log_data[i] = rd_val[15:0];
            if (rd_val[15:0] !== 16'h0) log_nonzero = log_nonzero + 1;
        end

        if (log_nonzero == 0) begin
            $display("FAIL: all BRAM log entries are zero");
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("INFO: BRAM log has %0d non-zero entries", log_nonzero);
        end

        // ── 최종 판정 ────────────────────────────────────────────────
        if (fail_cnt == 0)
            $display("PASS: all %0d checks passed", 5);
        else
            $display("FAIL: %0d check(s) failed", fail_cnt);

        $finish;
    end

    // 타임아웃: 15 ms (100 MHz 기준 1_500_000 클럭, 70창 여유)
    initial begin
        #15_000_000;
        $display("FAIL: simulation timeout");
        $finish;
    end

endmodule

# Makefile — mmio-pid-step-capture
# 타깃: lint, sim, synth
# 도구: verilator(lint), iverilog+vvp(sim), yosys(synth)

SLUG    := mmio-pid-step-capture
RTL_DIR := rtl
TB_DIR  := tb
BUILD   := build

RTL_SRC := $(RTL_DIR)/gpio.v \
           $(RTL_DIR)/uart_tx.v \
           $(RTL_DIR)/pid_cap_ip.v \
           $(RTL_DIR)/mmio_bus.v \
           $(RTL_DIR)/demo_master.v \
           $(RTL_DIR)/soc_top.v \
           $(RTL_DIR)/basys3_top.v

TB_SRC  := $(TB_DIR)/tb_pid_cap.v \
           $(RTL_DIR)/pid_cap_ip.v

.PHONY: all lint sim synth clean

all: lint sim synth

$(BUILD):
	mkdir -p $(BUILD)

## ── 린트 (verilator --lint-only) ──────────────────────────────────
lint: $(BUILD)
	verilator --lint-only -Wall --Wno-UNUSEDSIGNAL --Wno-UNUSEDPARAM \
	    $(RTL_SRC) 2>&1 | tee $(BUILD)/lint.log
	@echo "lint done"

## ── 시뮬레이션 (iverilog + vvp) ───────────────────────────────────
sim: $(BUILD)
	iverilog -g2001 -Wall -Wno-timescale \
	    -o $(BUILD)/sim.vvp \
	    $(TB_SRC) 2>&1 | tee $(BUILD)/iverilog.log
	vvp $(BUILD)/sim.vvp 2>&1 | tee $(BUILD)/sim.log
	@grep -q "^PASS" $(BUILD)/sim.log && echo "sim: PASS" || \
	    (echo "sim: FAIL — see $(BUILD)/sim.log" && exit 1)

## ── 합성 (yosys, Artix-7 타깃) ────────────────────────────────────
synth: $(BUILD)
	yosys -p " \
	    read_verilog $(RTL_SRC); \
	    synth_xilinx -top basys3_top -flatten; \
	    stat; \
	    write_json $(BUILD)/synth.json \
	" 2>&1 | tee $(BUILD)/synth.log
	@grep -q "^End of script" $(BUILD)/synth.log && echo "synth: OK" || \
	    (echo "synth: FAIL — see $(BUILD)/synth.log" && exit 1)

## ── 정리 ────────────────────────────────────────────────────────────
clean:
	rm -rf $(BUILD)

위험·한계

  • WINDOW_CYCLES=1000(시뮬) vs 1_000_000(합성) — 합성 빌드 시 pid_cap_ip.v 파라미터 확인 필수
  • KD항 미분 노이즈 — 실제 엔코더 신호에 적용 시 D항 필터 추가 필요. 테스트벤치는 노이즈 없는 모델 사용
  • BRAM 추론: yosys가 BRAM으로 추론하지 않을 수 있음. Vivado는 별도 확인
  • 적분기 클램프(-32768~32767)로 와인드업 방지하나, KI가 너무 크면 오버슈트 발생 가능
  • 최소 PWM(MIN_DUTY)이 0이면 저속 구간에서 모터 스톨 가능
