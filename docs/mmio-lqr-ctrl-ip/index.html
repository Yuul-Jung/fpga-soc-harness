LQR 상태 되먹임 제어기 IP: Q8.8 MAC + AXI4-Lite (Basys3)

u = K[0]*x + K[1]*dx + K[2]*θ + K[3]*dθ 를 Q8.8 고정소수점 MAC FSM으로 계산하는 AXI4-Lite 슬레이브 IP. 이득 K[0..3]과 상태 x, dx, θ, dθ를 레지스터로 받고, 1회 계산 후 u_out(포화 처리)을 반환한다. 카트-폴 도립진자 LQR 제어기의 하드웨어 가속기 프로토타입.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: 미실행

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • K[0..3] 이득과 상태 x,dx,θ,dθ 를 AXI4-Lite 레지스터로 설정한다.
  • start 펄스 후 7클럭 이내에 u = K·x (Q8.8 내적)를 계산하고 done 펄스를 출력한다.
  • 계산 결과 u_out 을 Q8.8 포화([-128,127.996]) 처리 후 AXI4-Lite 읽기로 반환한다.
  • 카트-폴 선형화 모델 기반 테스트벤치로 TC1~TC5를 PASS한다.

범위 밖

  • 실시간 엔코더·IMU 연결 (입력은 소프트웨어가 준비)
  • 플랜트 모델 내장 (u 계산만; 플랜트는 외부)
  • 리케이트이션·뱅뱅 제어, 플로팅포인트 연산
  • 실제 모터 출력 드라이버 (다음 단계에서 mmio-stepper-ramp-ip 연결)

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 시뮬레이션·합성 결과는 CI가 기록한다.

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스AXI4-Lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획0x401C_0000에 슬롯 [1C] 배정 제안. 슬롯 [16]~[1B]는 기존 capstone에서 사용. 사용자 동결 후 address_map→mbv-frozen-v2, compat→compatible 으로 갱신 가능. intc_free에서 IRQ 1개 배정 필요.

5. 외부 부품과 전기 회로도

2절의 계통도는 디지털 모듈 연결도다. 전압, 풀업, 커넥터 배선 같은 전기 사항은 이 절에서 따로 다룬다. 기계가 읽는 사본: bom.csv · connections.json

전기 회로도: 미작성

이 IP는 디지털 연산 블록이다. 외부 아날로그 부품 없음. 보드 내장 버튼과 LED만 사용하므로 별도 회로도 불필요.

rtl/lqr_ip.v 139줄
// lqr_ip.v — LQR 상태 되먹임 제어기 IP (Q8.8 MAC FSM)
// 기능:
//   u = K[0]*x + K[1]*dx + K[2]*θ + K[3]*dθ
//   Q8.8 고정소수점 (16비트 부호): 정수부 8비트, 소수부 8비트
//   40비트 부호 누산기로 Q16.16 곱 4개를 더한 뒤 비트[23:8] 추출, 포화
//   start 펄스 → 7클럭 이내 done 펄스 출력
//
// 포트 단위:
//   K[0..3], x,dx,θ,dθ: Q8.8 (16비트 부호) — 1.0 = 0x0100
//   u_out: Q8.8 포화 (범위 0x8000=-128.0 ~ 0x7FFF≈+127.996)
//
// FSM: IDLE → MAC0 → MAC1 → MAC2 → MAC3 → SAT → DONE → IDLE
// 2026-11-04 하네스 자체 작성 (원본, 외부 코드 복사 아님)
`default_nettype none

module lqr_ip (
    input  wire        clk,
    input  wire        rst,
    input  wire        start,          // 1클럭 펄스
    output reg         done,           // 1클럭 펄스 (완료)
    output reg         busy,
    // 이득 K[0..3] (Q8.8 부호)
    input  wire signed [15:0] k0,
    input  wire signed [15:0] k1,
    input  wire signed [15:0] k2,
    input  wire signed [15:0] k3,
    // 상태 (Q8.8 부호)
    input  wire signed [15:0] x_state,
    input  wire signed [15:0] dx_state,
    input  wire signed [15:0] theta,
    input  wire signed [15:0] dtheta,
    // 제어 출력 (Q8.8 부호, 포화)
    output reg  signed [15:0] u_out
);

    // ─── FSM 상태 ────────────────────────────────────────────────────────────
    localparam [2:0]
        S_IDLE = 3'd0,
        S_MAC0 = 3'd1,
        S_MAC1 = 3'd2,
        S_MAC2 = 3'd3,
        S_MAC3 = 3'd4,
        S_SAT  = 3'd5,
        S_DONE = 3'd6;

    reg [2:0] state;

    // ─── 누산기 (40비트 부호: Q17.16 에 여유 추가) ───────────────────────────
    // K(Q8.8) × state(Q8.8) = Q16.16 (32비트). 4개 합 최대 Q17.16 (33비트).
    // 40비트로 오버플로 방지.
    reg signed [39:0] accum;

    // ─── 곱셈 선택 MUX (레지스터에서 래치된 값 사용) ─────────────────────────
    reg signed [15:0] k_lat [0:3];
    reg signed [15:0] s_lat [0:3];

    wire signed [31:0] product0 = k_lat[0] * s_lat[0]; // Q16.16
    wire signed [31:0] product1 = k_lat[1] * s_lat[1];
    wire signed [31:0] product2 = k_lat[2] * s_lat[2];
    wire signed [31:0] product3 = k_lat[3] * s_lat[3];

    // ─── FSM ─────────────────────────────────────────────────────────────────
    always @(posedge clk) begin
        if (rst) begin
            state  <= S_IDLE;
            done   <= 1'b0;
            busy   <= 1'b0;
            accum  <= 40'sd0;
            u_out  <= 16'sd0;
            k_lat[0] <= 16'sd0; k_lat[1] <= 16'sd0;
            k_lat[2] <= 16'sd0; k_lat[3] <= 16'sd0;
            s_lat[0] <= 16'sd0; s_lat[1] <= 16'sd0;
            s_lat[2] <= 16'sd0; s_lat[3] <= 16'sd0;
        end else begin
            done <= 1'b0; // 기본: 클리어

            case (state)
            S_IDLE: begin
                busy <= 1'b0;
                if (start) begin
                    // 입력 래치
                    k_lat[0] <= k0; k_lat[1] <= k1;
                    k_lat[2] <= k2; k_lat[3] <= k3;
                    s_lat[0] <= x_state; s_lat[1] <= dx_state;
                    s_lat[2] <= theta;   s_lat[3] <= dtheta;
                    accum <= 40'sd0;
                    busy  <= 1'b1;
                    state <= S_MAC0;
                end
            end

            S_MAC0: begin
                accum <= {{8{product0[31]}}, product0}; // 부호 확장
                state <= S_MAC1;
            end

            S_MAC1: begin
                accum <= accum + {{8{product1[31]}}, product1};
                state <= S_MAC2;
            end

            S_MAC2: begin
                accum <= accum + {{8{product2[31]}}, product2};
                state <= S_MAC3;
            end

            S_MAC3: begin
                accum <= accum + {{8{product3[31]}}, product3};
                state <= S_SAT;
            end

            S_SAT: begin
                // Q16.16 → Q8.8: 비트[23:8] 추출, 포화 처리
                // 오버플로 감지: accum[39:23]이 모두 0 or 모두 1이 아니면 포화
                if (!accum[39] && (|accum[39:23])) begin
                    // 양의 오버플로 → 최대값
                    u_out <= 16'sh7FFF;
                end else if (accum[39] && !(& accum[39:23])) begin
                    // 음의 오버플로 → 최소값
                    u_out <= 16'sh8000;
                end else begin
                    u_out <= accum[23:8];
                end
                state <= S_DONE;
            end

            S_DONE: begin
                done  <= 1'b1;
                busy  <= 1'b0;
                state <= S_IDLE;
            end

            default: state <= S_IDLE;
            endcase
        end
    end

endmodule
`default_nettype wire

// axil_lqr.v — AXI4-Lite 슬레이브 레지스터 맵 (LQR 제어기 IP)
// 기준 주소: 0x401C_0000 (동결 맵 free_slot[1C] 제안)
// BREADY / RREADY 항상 1 가정 (슬레이브 간소화)
// 2026-11-04 하네스 자체 작성 (원본, 외부 코드 복사 아님)
`default_nettype none

module axil_lqr #(
    parameter DATA_WIDTH = 32,
    parameter ADDR_WIDTH = 8
)(
    input  wire        clk,
    input  wire        rst,
    // AXI4-Lite 슬레이브 포트
    input  wire [ADDR_WIDTH-1:0] s_axil_awaddr,
    input  wire        s_axil_awvalid,
    output reg         s_axil_awready,
    input  wire [DATA_WIDTH-1:0] s_axil_wdata,
    input  wire [(DATA_WIDTH/8)-1:0] s_axil_wstrb,
    input  wire        s_axil_wvalid,
    output reg         s_axil_wready,
    output reg  [1:0]  s_axil_bresp,
    output reg         s_axil_bvalid,
    input  wire        s_axil_bready,
    input  wire [ADDR_WIDTH-1:0] s_axil_araddr,
    input  wire        s_axil_arvalid,
    output reg         s_axil_arready,
    output reg  [DATA_WIDTH-1:0] s_axil_rdata,
    output reg  [1:0]  s_axil_rresp,
    output reg         s_axil_rvalid,
    input  wire        s_axil_rready,
    // IRQ 출력
    output wire        irq
);

    // ─── 레지스터 주소 (오프셋 바이트) ──────────────────────────────────────
    localparam REG_CTRL      = 8'h00; // WO: bit0=start
    localparam REG_STATUS    = 8'h04; // RO: bit0=done, bit1=busy
    localparam REG_K0        = 8'h08; // RW: 이득 K[0] (Q8.8 부호)
    localparam REG_K1        = 8'h0C; // RW: 이득 K[1]
    localparam REG_K2        = 8'h10; // RW: 이득 K[2]
    localparam REG_K3        = 8'h14; // RW: 이득 K[3]
    localparam REG_X_STATE   = 8'h18; // WO: 상태 x
    localparam REG_DX_STATE  = 8'h1C; // WO: 상태 dx
    localparam REG_THETA     = 8'h20; // WO: 상태 θ
    localparam REG_DTHETA    = 8'h24; // WO: 상태 dθ
    localparam REG_U_OUT     = 8'h28; // RO: 제어 출력 u
    localparam REG_IRQ_STAT  = 8'h2C; // RW1C: bit0=done_irq
    localparam REG_IRQ_EN    = 8'h30; // RW: bit0=done_irq_en

    // ─── 레지스터 저장소 ─────────────────────────────────────────────────────
    reg signed [15:0] r_k0, r_k1, r_k2, r_k3;
    reg signed [15:0] r_x, r_dx, r_theta, r_dtheta;
    reg               r_start;
    reg               r_irq_en;
    reg               r_irq_stat;

    // ─── lqr_ip 출력 ─────────────────────────────────────────────────────────
    wire        ip_done, ip_busy;
    wire signed [15:0] ip_u_out;

    lqr_ip u_lqr (
        .clk      (clk),
        .rst      (rst),
        .start    (r_start),
        .done     (ip_done),
        .busy     (ip_busy),
        .k0       (r_k0),
        .k1       (r_k1),
        .k2       (r_k2),
        .k3       (r_k3),
        .x_state  (r_x),
        .dx_state (r_dx),
        .theta    (r_theta),
        .dtheta   (r_dtheta),
        .u_out    (ip_u_out)
    );

    // ─── IRQ 래치 ────────────────────────────────────────────────────────────
    always @(posedge clk) begin
        if (rst)
            r_irq_stat <= 1'b0;
        else if (ip_done)
            r_irq_stat <= 1'b1;
        // 소프트웨어 클리어는 아래 쓰기 경로에서 처리
    end

    assign irq = r_irq_stat & r_irq_en;

    // ─── AXI4-Lite 쓰기 채널 ─────────────────────────────────────────────────
    reg [ADDR_WIDTH-1:0] wr_addr;
    reg wr_addr_valid;

    always @(posedge clk) begin
        if (rst) begin
            s_axil_awready <= 1'b0;
            s_axil_wready  <= 1'b0;
            s_axil_bvalid  <= 1'b0;
            s_axil_bresp   <= 2'b00;
            wr_addr        <= {ADDR_WIDTH{1'b0}};
            wr_addr_valid  <= 1'b0;
            r_k0     <= 16'sd0; r_k1  <= 16'sd0;
            r_k2     <= 16'sd0; r_k3  <= 16'sd0;
            r_x      <= 16'sd0; r_dx  <= 16'sd0;
            r_theta  <= 16'sd0; r_dtheta <= 16'sd0;
            r_start  <= 1'b0;
            r_irq_en <= 1'b0;
        end else begin
            r_start <= 1'b0; // 자동 클리어

            // AW 핸드셰이크
            if (s_axil_awvalid && !wr_addr_valid) begin
                s_axil_awready <= 1'b1;
                wr_addr        <= s_axil_awaddr;
                wr_addr_valid  <= 1'b1;
            end else begin
                s_axil_awready <= 1'b0;
            end

            // W 핸드셰이크 + 레지스터 쓰기
            if (s_axil_wvalid && wr_addr_valid) begin
                s_axil_wready <= 1'b1;
                wr_addr_valid <= 1'b0;
                case (wr_addr)
                REG_CTRL:     r_start  <= s_axil_wdata[0];
                REG_K0:       r_k0     <= s_axil_wdata[15:0];
                REG_K1:       r_k1     <= s_axil_wdata[15:0];
                REG_K2:       r_k2     <= s_axil_wdata[15:0];
                REG_K3:       r_k3     <= s_axil_wdata[15:0];
                REG_X_STATE:  r_x      <= s_axil_wdata[15:0];
                REG_DX_STATE: r_dx     <= s_axil_wdata[15:0];
                REG_THETA:    r_theta  <= s_axil_wdata[15:0];
                REG_DTHETA:   r_dtheta <= s_axil_wdata[15:0];
                REG_IRQ_STAT: if (s_axil_wdata[0]) r_irq_stat <= 1'b0; // RW1C
                REG_IRQ_EN:   r_irq_en <= s_axil_wdata[0];
                default: ; // RO 주소 무시
                endcase
                s_axil_bvalid <= 1'b1;
                s_axil_bresp  <= 2'b00;
            end else begin
                s_axil_wready <= 1'b0;
            end

            if (s_axil_bvalid && s_axil_bready)
                s_axil_bvalid <= 1'b0;
        end
    end

    // ─── AXI4-Lite 읽기 채널 ─────────────────────────────────────────────────
    always @(posedge clk) begin
        if (rst) begin
            s_axil_arready <= 1'b0;
            s_axil_rvalid  <= 1'b0;
            s_axil_rdata   <= {DATA_WIDTH{1'b0}};
            s_axil_rresp   <= 2'b00;
        end else begin
            if (s_axil_arvalid && !s_axil_rvalid) begin
                s_axil_arready <= 1'b1;
                s_axil_rvalid  <= 1'b1;
                s_axil_rresp   <= 2'b00;
                case (s_axil_araddr)
                REG_STATUS:  s_axil_rdata <= {30'd0, ip_busy, ip_done};
                REG_K0:      s_axil_rdata <= {{16{r_k0[15]}}, r_k0};
                REG_K1:      s_axil_rdata <= {{16{r_k1[15]}}, r_k1};
                REG_K2:      s_axil_rdata <= {{16{r_k2[15]}}, r_k2};
                REG_K3:      s_axil_rdata <= {{16{r_k3[15]}}, r_k3};
                REG_U_OUT:   s_axil_rdata <= {{16{ip_u_out[15]}}, ip_u_out};
                REG_IRQ_STAT:s_axil_rdata <= {31'd0, r_irq_stat};
                REG_IRQ_EN:  s_axil_rdata <= {31'd0, r_irq_en};
                default:     s_axil_rdata <= 32'd0;
                endcase
            end else begin
                s_axil_arready <= 1'b0;
            end

            if (s_axil_rvalid && s_axil_rready)
                s_axil_rvalid <= 1'b0;
        end
    end

endmodule

// tb_lqr_ctrl.v — LQR 제어기 IP 자체 점검형 테스트벤치 (TC1~TC5)
// PASS / FAIL 출력. !==로 X/Z도 실패로 처리.
// lqr_ip 코어를 직접 구동 (AXI 래퍼 없이).
//
// TC1: 모든 K, 상태 0 → u = 0
// TC2: K0=1.0, x=1.0, 나머지 0 → u = 1.0
// TC3: K3=0.5, dtheta=2.0, 나머지 0 → u = 1.0
// TC4: K2=-1.0, theta=2.0, 나머지 0 → u = -2.0 (Q8.8: 0xFE00)
// TC5: 모든 K=64.0, 모든 상태=64.0 → 오버플로 → 포화(0x7FFF)
//
// 2026-11-04 하네스 자체 작성 (원본, 외부 코드 복사 아님)
`default_nettype none
`timescale 1ns/1ps

module tb_lqr_ctrl;

// ─── DUT 신호 ────────────────────────────────────────────────────────────────
reg         clk, rst;
reg         start;
wire        done, busy;
reg  signed [15:0] k0, k1, k2, k3;
reg  signed [15:0] x_state, dx_state, theta, dtheta;
wire signed [15:0] u_out;

// ─── DUT 인스턴스 ────────────────────────────────────────────────────────────
lqr_ip u_dut (
    .clk      (clk),
    .rst      (rst),
    .start    (start),
    .done     (done),
    .busy     (busy),
    .k0       (k0),
    .k1       (k1),
    .k2       (k2),
    .k3       (k3),
    .x_state  (x_state),
    .dx_state (dx_state),
    .theta    (theta),
    .dtheta   (dtheta),
    .u_out    (u_out)
);

// ─── 클럭 (100 MHz) ──────────────────────────────────────────────────────────
initial clk = 0;
always  #5 clk = ~clk;

// ─── 헬퍼 태스크: start 펄스 → done 대기 ────────────────────────────────────
integer wait_cnt;
task run_lqr;
    begin
        @(posedge clk); #1;
        start = 1'b1;
        @(posedge clk); #1;
        start = 1'b0;
        wait_cnt = 0;
        while (done !== 1'b1 && wait_cnt < 20) begin
            @(posedge clk); #1;
            wait_cnt = wait_cnt + 1;
        end
        if (done !== 1'b1) begin
            $display("FAIL: done never asserted (timeout)");
            $finish;
        end
    end
endtask

// ─── 헬퍼 함수: Q8.8 정수 표현 → 소수 문자열 (디버그용) ────────────────────
// (실제 비교는 정수로 수행)

// ─── 검증 헬퍼 ───────────────────────────────────────────────────────────────
integer fail_cnt;
task check_u;
    input signed [15:0] expected;
    input [63:0] tc_name; // 최대 8자 (8바이트)
    begin
        if (u_out !== expected) begin
            $display("FAIL TC%0s: u_out=%0d (0x%04X), expected=%0d (0x%04X)",
                     tc_name, u_out, u_out[15:0], expected, expected[15:0]);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("ok   TC%0s: u_out=0x%04X (%0d)", tc_name, u_out[15:0], u_out);
        end
    end
endtask

// ─── 메인 ────────────────────────────────────────────────────────────────────
integer tc;
initial begin
    fail_cnt = 0;

    // 초기화
    rst = 1'b1; start = 1'b0;
    k0 = 0; k1 = 0; k2 = 0; k3 = 0;
    x_state = 0; dx_state = 0; theta = 0; dtheta = 0;
    repeat(4) @(posedge clk);
    rst = 1'b0;
    @(posedge clk); #1;

    // ── TC1: 전부 0 → u = 0 ─────────────────────────────────────────────────
    k0=16'sd0; k1=16'sd0; k2=16'sd0; k3=16'sd0;
    x_state=16'sd0; dx_state=16'sd0; theta=16'sd0; dtheta=16'sd0;
    run_lqr;
    check_u(16'sd0, "1");

    // ── TC2: K0=1.0(0x0100), x=1.0(0x0100), 나머지 0 → u=1.0(0x0100) ─────
    // K0*x = 0x0100 * 0x0100 = 0x0001_0000 (Q16.16). bits[23:8]=0x0100.
    k0=16'sh0100; k1=16'sd0; k2=16'sd0; k3=16'sd0;
    x_state=16'sh0100; dx_state=16'sd0; theta=16'sd0; dtheta=16'sd0;
    run_lqr;
    check_u(16'sh0100, "2");

    // ── TC3: K3=0.5(0x0080), dtheta=2.0(0x0200), 나머지 0 → u=1.0(0x0100)
    // K3*dtheta = 0x0080 * 0x0200 = 0x0001_0000 (Q16.16). bits[23:8]=0x0100.
    k0=16'sd0; k1=16'sd0; k2=16'sd0; k3=16'sh0080;
    x_state=16'sd0; dx_state=16'sd0; theta=16'sd0; dtheta=16'sh0200;
    run_lqr;
    check_u(16'sh0100, "3");

    // ── TC4: K2=-1.0(0xFF00), theta=2.0(0x0200), 나머지 0 → u=-2.0(0xFE00)
    // K2*theta = 0xFF00(=-256Q8.8→-1.0) * 0x0200(=512Q8.8→2.0)
    //          = -256 * 512 = -131072 = 0xFFFE_0000 (Q16.16)
    // bits[23:8] = 0xFE00 (-2.0 in Q8.8)
    k0=16'sd0; k1=16'sd0; k2=16'shFF00; k3=16'sd0;
    x_state=16'sd0; dx_state=16'sd0; theta=16'sh0200; dtheta=16'sd0;
    run_lqr;
    check_u(16'shFE00, "4");

    // ── TC5: K*=64.0(0x4000), state=64.0(0x4000) → 각 항 = 16384.0
    //   sum = 4 * 16384.0 = 65536.0 → 오버플로 → 포화 0x7FFF
    // K*state = 0x4000 * 0x4000 = 0x1000_0000 (Q16.16) = 16384.0
    // 4개 합 = 0x4000_0000 (Q16.16) = 65536.0 → Q8.8로 변환 불가 (>127.996)
    // 양의 오버플로 → 0x7FFF
    k0=16'sh4000; k1=16'sh4000; k2=16'sh4000; k3=16'sh4000;
    x_state=16'sh4000; dx_state=16'sh4000; theta=16'sh4000; dtheta=16'sh4000;
    run_lqr;
    check_u(16'sh7FFF, "5");

    // ── 결과 ─────────────────────────────────────────────────────────────────
    if (fail_cnt == 0) begin
        $display("PASS: 모든 TC (%0d개) 통과", 5);
    end else begin
        $display("FAIL: %0d개 TC 실패", fail_cnt);
    end
    $finish;
end

// ─── 타임아웃 보호 ───────────────────────────────────────────────────────────
initial begin
    #50000;
    $display("FAIL: 전체 타임아웃");
    $finish;
end

endmodule

# 타깃: lint, sim, synth, clean

SLUG    = mmio-lqr-ctrl-ip
RTL_DIR = rtl
TB_DIR  = tb
BUILD   = build

RTL_SRCS = $(RTL_DIR)/lqr_ip.v \
           $(RTL_DIR)/axil_lqr.v \
           $(RTL_DIR)/basys3_lqr_top.v

TB_SRCS  = $(TB_DIR)/tb_lqr_ctrl.v

.PHONY: all lint sim synth clean

## ── 린트 (verilator) ────────────────────────────────────────────────────────

	@mkdir -p $(BUILD)

	@grep -q "^PASS" $(BUILD)/sim.log && echo "sim PASS" || (echo "sim FAIL" && exit 1)

$(BUILD)/tb_$(SLUG): $(RTL_SRCS) $(TB_SRCS)
	@mkdir -p $(BUILD)
	iverilog -g2001 -o $@ $(RTL_SRCS) $(TB_SRCS)

	@mkdir -p $(BUILD)

## ── 정리 ────────────────────────────────────────────────────────────────────
clean:
	rm -rf $(BUILD)
