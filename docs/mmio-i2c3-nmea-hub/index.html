AXI4-Lite I2C 3장치 순회 + NMEA UART 허브 IP — 타임스탬프 FIFO

I2C 마스터가 MPU-6050·BMP280·HMC5883L을 순회하고, UART 수신기가 NMEA 문장(9600 baud)을 처리하며, 모든 표본에 자유실행 타임스탬프를 붙여 16엔트리 FIFO에 쌓는 AXI4-Lite 허브 IP. I-036 아이디어 기반. I2C 순회 주기 ~100 ms, TEST_MODE로 직접 FIFO 주입 가능.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: not-verified

1. 목표와 범위

목표

  • AXI4-Lite MMIO로 제어 가능한 I2C 3장치 순회 + NMEA UART 허브 IP 구현
  • MPU-6050(0x68), BMP280(0x76), HMC5883L(0x1E) 단일 레지스터 읽기 순회, ~100 ms 주기
  • UART 9600 baud NMEA 문장 수신($ 시작, \n 종료), 완료 인터럽트 지원
  • 모든 표본에 32비트 자유실행 타임스탬프를 붙여 16엔트리 FIFO에 저장
  • TEST_MODE(CTRL[8])로 직접 FIFO 주입 — 테스트벤치가 I2C/UART 모델 없이도 검증 가능
  • iverilog 시뮬레이션 통과, yosys 합성 가능 RTL

범위 밖

  • 센서 보정 계산: 펌웨어가 처리
  • NMEA 필드 추출(lat/lon 파싱): 펌웨어가 처리
  • 창 WDT / 센서 연속 스트림: 추가 단계에서
  • 다중 I2C 버스 / I2C 멀티 바이트 버스트: 단일 레지스터 읽기만 구현
  • 보드 동작 확인(S3): 사용자 직접 수행
  • 소프트웨어 드라이버(C) 포함

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-036 기반(ready → used). 기존 mmio-i2c-master-ip의 설계 참고(코드 미복사). AXI4-Lite 슬레이브. 동결 맵 free_slots[1D]=0x401D_0000 배정 제안(미동결). 시뮬레이션·합성 결과는 CI에서 검증한다.

// i2c_rd_byte.v — I2C 싱글 레지스터 읽기 엔진
// 하네스 직접 작성 Verilog-2001 원본. 외부 코드 미포함.
// 동작: START + (dev_addr<<1|0) + reg_addr + RSTART + (dev_addr<<1|1) + data_byte + NACK + STOP
// 비트 타이밍: CLK_DIV 클럭 1비트. Q1=CLK_DIV/4(SDA), Q2=CLK_DIV/2(SCL↑), Q3=3*CLK_DIV/4(샘플), Q4=CLK_DIV-1(SCL↓)

`default_nettype none
module i2c_rd_byte #(
    parameter CLK_DIV = 1000   // 100kHz @ 100MHz
) (
    input  wire       clk,
    input  wire       rst_n,
    // 제어
    input  wire       start,       // 1-pulse: 트랜잭션 시작
    input  wire [6:0] dev_addr,    // 7비트 슬레이브 주소
    input  wire [7:0] reg_addr,    // 읽을 레지스터
    // 결과
    output reg        done,        // 1-pulse: 완료(NACK/timeout 포함)
    output reg  [7:0] data_out,    // 읽은 바이트
    output reg        ack_err,     // 1=ACK 없음(NACK 또는 timeout)
    // I2C 버스 (오픈 드레인 에뮬레이션)
    output reg        scl_out,     // 1=Hi-Z, 0=drive low
    output reg        sda_out,     // 1=Hi-Z, 0=drive low
    input  wire       sda_in       // 버스 SDA 읽기
);

// ── 타이밍 상수 ──────────────────────────────────────────
localparam integer P_Q1  = CLK_DIV / 4;
localparam integer P_Q2  = CLK_DIV / 2;
localparam integer P_Q3  = (CLK_DIV * 3) / 4;
localparam integer P_END = CLK_DIV - 1;

// ── FSM 상태 ─────────────────────────────────────────────
localparam [4:0]
    S_IDLE    = 5'd0,
    S_START   = 5'd1,   // START 조건
    S_WA      = 5'd2,   // 쓰기 주소 (dev_addr<<1|0)
    S_WA_AK   = 5'd3,   // 쓰기 주소 ACK
    S_RA      = 5'd4,   // 레지스터 주소
    S_RA_AK   = 5'd5,   // 레지스터 주소 ACK
    S_RSTART  = 5'd6,   // 반복 START
    S_RD      = 5'd7,   // 읽기 주소 (dev_addr<<1|1)
    S_RD_AK   = 5'd8,   // 읽기 주소 ACK
    S_DATA    = 5'd9,   // 8비트 데이터 수신
    S_NACK    = 5'd10,  // 마스터 NACK 송출
    S_STOP    = 5'd11,  // STOP 조건
    S_DONE    = 5'd12;

reg [4:0] state;
reg [11:0] clk_cnt;   // CLK_DIV≤4095 가정 (최대 10bit면 충분, 여유 12bit)
reg [3:0]  bit_cnt;   // 비트 카운터 (0~7)
reg [7:0]  shift;     // TX/RX 시프트 레지스터

// 현재 클럭 위상 판단 (combinational)
wire phase_q1  = (clk_cnt == P_Q1[11:0]);
wire phase_q2  = (clk_cnt == P_Q2[11:0]);
wire phase_q3  = (clk_cnt == P_Q3[11:0]);
wire phase_end = (clk_cnt == P_END[11:0]);

// ── 메인 FSM ─────────────────────────────────────────────
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state    <= S_IDLE;
        clk_cnt  <= 12'd0;
        bit_cnt  <= 4'd0;
        shift    <= 8'd0;
        scl_out  <= 1'b1;
        sda_out  <= 1'b1;
        done     <= 1'b0;
        data_out <= 8'd0;
        ack_err  <= 1'b0;
    end else begin
        done <= 1'b0;  // 기본 1사이클 펄스

        case (state)

        // ── IDLE ─────────────────────────────────────────
        S_IDLE: begin
            scl_out <= 1'b1; sda_out <= 1'b1;
            if (start) begin
                shift   <= {dev_addr, 1'b0};  // addr + W
                bit_cnt <= 4'd7;
                clk_cnt <= 12'd0;
                state   <= S_START;
            end
        end

        // ── START 조건: SCL Hi인 상태에서 SDA ↓ ─────────
        S_START: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b0;
            if (phase_q2) scl_out <= 1'b0;
            if (phase_end) begin
                clk_cnt <= 12'd0;
                state   <= S_WA;
            end
        end

        // ── 쓰기 주소 비트 송출 ───────────────────────────
        S_WA: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= shift[7];
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) begin  // 샘플 포인트 (필요 없지만 대칭성)
                /* TX: 슬레이브가 샘플 */
            end
            if (phase_end) begin
                scl_out <= 1'b0;
                shift   <= {shift[6:0], 1'b0};
                clk_cnt <= 12'd0;
                if (bit_cnt == 4'd0) begin
                    state <= S_WA_AK;
                end else begin
                    bit_cnt <= bit_cnt - 4'd1;
                end
            end
        end

        // ── 쓰기 주소 ACK 수신 ───────────────────────────
        S_WA_AK: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b1;  // 릴리즈
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) ack_err <= sda_in; // SDA Hi = NACK
            if (phase_end) begin
                scl_out <= 1'b0;
                clk_cnt <= 12'd0;
                if (ack_err) begin
                    state <= S_STOP;
                end else begin
                    shift   <= reg_addr;
                    bit_cnt <= 4'd7;
                    state   <= S_RA;
                end
            end
        end

        // ── 레지스터 주소 비트 송출 ──────────────────────
        S_RA: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= shift[7];
            if (phase_q2) scl_out <= 1'b1;
            if (phase_end) begin
                scl_out <= 1'b0;
                shift   <= {shift[6:0], 1'b0};
                clk_cnt <= 12'd0;
                if (bit_cnt == 4'd0) begin
                    state <= S_RA_AK;
                end else begin
                    bit_cnt <= bit_cnt - 4'd1;
                end
            end
        end

        // ── 레지스터 주소 ACK 수신 ───────────────────────
        S_RA_AK: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b1;
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) ack_err <= sda_in;
            if (phase_end) begin
                scl_out <= 1'b0;
                clk_cnt <= 12'd0;
                if (ack_err) begin
                    state <= S_STOP;
                end else begin
                    state <= S_RSTART;
                end
            end
        end

        // ── 반복 START ───────────────────────────────────
        S_RSTART: begin
            clk_cnt <= clk_cnt + 12'd1;
            // SDA Hi → SCL Hi → SDA Lo
            if (phase_q1) sda_out <= 1'b1;
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) sda_out <= 1'b0;
            if (phase_end) begin
                scl_out <= 1'b0;
                shift   <= {dev_addr, 1'b1};  // addr + R
                bit_cnt <= 4'd7;
                clk_cnt <= 12'd0;
                state   <= S_RD;
            end
        end

        // ── 읽기 주소 비트 송출 ──────────────────────────
        S_RD: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= shift[7];
            if (phase_q2) scl_out <= 1'b1;
            if (phase_end) begin
                scl_out <= 1'b0;
                shift   <= {shift[6:0], 1'b0};
                clk_cnt <= 12'd0;
                if (bit_cnt == 4'd0) begin
                    state <= S_RD_AK;
                end else begin
                    bit_cnt <= bit_cnt - 4'd1;
                end
            end
        end

        // ── 읽기 주소 ACK 수신 ───────────────────────────
        S_RD_AK: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b1;
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) ack_err <= sda_in;
            if (phase_end) begin
                scl_out <= 1'b0;
                clk_cnt <= 12'd0;
                if (ack_err) begin
                    state <= S_STOP;
                end else begin
                    bit_cnt <= 4'd7;
                    state   <= S_DATA;
                end
            end
        end

        // ── 8비트 데이터 수신 ────────────────────────────
        S_DATA: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b1;  // 릴리즈 (입력)
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) shift <= {shift[6:0], sda_in};  // MSB 먼저
            if (phase_end) begin
                scl_out <= 1'b0;
                clk_cnt <= 12'd0;
                if (bit_cnt == 4'd0) begin
                    data_out <= shift;
                    state    <= S_NACK;
                end else begin
                    bit_cnt <= bit_cnt - 4'd1;
                end
            end
        end

        // ── 마스터 NACK 송출 ─────────────────────────────
        S_NACK: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b1;  // NACK = SDA Hi
            if (phase_q2) scl_out <= 1'b1;
            if (phase_end) begin
                scl_out <= 1'b0;
                clk_cnt <= 12'd0;
                state   <= S_STOP;
            end
        end

        // ── STOP 조건: SCL Hi인 상태에서 SDA ↑ ─────────
        S_STOP: begin
            clk_cnt <= clk_cnt + 12'd1;
            if (phase_q1) sda_out <= 1'b0;
            if (phase_q2) scl_out <= 1'b1;
            if (phase_q3) sda_out <= 1'b1;
            if (phase_end) begin
                clk_cnt <= 12'd0;
                state   <= S_DONE;
            end
        end

        // ── DONE ─────────────────────────────────────────
        S_DONE: begin
            done  <= 1'b1;
            state <= S_IDLE;
        end

        default: state <= S_IDLE;
        endcase
    end
end

// 하네스 직접 작성 Verilog-2001 원본. 외부 코드 미포함.
// I2C: MPU-6050(0x68, reg 0x75), BMP280(0x76, reg 0xD0), HMC5883L(0x1E, reg 0x0A) 순회
// UART: 9600baud 8N1, NMEA 문장($ ~ \n) 완료 IRQ
// FIFO: 16엔트리, [31:24]=sensor_id, [23:0]=data, 별도 타임스탬프
// TEST_MODE(CTRL[8]): FIFO_INJ 레지스터로 직접 FIFO 주입 가능

`default_nettype none
module axil_i2c3_nmea_hub #(
    parameter CLK_DIV_I2C  = 1000,   // I2C 비트 분주 (100kHz@100MHz)
    parameter CLK_DIV_UART = 10417   // UART 비트 분주 (9600baud@100MHz)
) (
    input  wire        aclk,
    input  wire        aresetn,

    // AXI4-Lite 슬레이브
    input  wire [15:0] s_awaddr,
    input  wire        s_awvalid,
    output reg         s_awready,
    input  wire [31:0] s_wdata,
    input  wire [3:0]  s_wstrb,
    input  wire        s_wvalid,
    output reg         s_wready,
    output reg  [1:0]  s_bresp,
    output reg         s_bvalid,
    input  wire        s_bready,

    input  wire [15:0] s_araddr,
    input  wire        s_arvalid,
    output reg         s_arready,
    output reg  [31:0] s_rdata,
    output reg  [1:0]  s_rresp,
    output reg         s_rvalid,
    input  wire        s_rready,

    // 외부 I/O
    output wire        scl,       // I2C SCL (오픈드레인 Hi-Z=1, Lo=0)
    inout  wire        sda,       // I2C SDA (오픈드레인)
    input  wire        uart_rx,   // UART RX (NMEA)

    // 인터럽트
    output reg         irq_out
);

// ── 레지스터 주소 ─────────────────────────────────────────
localparam [15:0]
    A_CTRL      = 16'h0000,
    A_STATUS    = 16'h0004,
    A_FIFO_DATA = 16'h0008,
    A_FIFO_TS   = 16'h000C,
    A_FIFO_CNT  = 16'h0010,
    A_I2C_DIV   = 16'h0014,
    A_UART_DIV  = 16'h0018,
    A_FIFO_INJ  = 16'h001C,
    A_ID_REG    = 16'hFFF0,
    A_VER_REG   = 16'hFFF4,
    A_IRQ_EN    = 16'hFFF8,
    A_IRQ_PEND  = 16'hFFFC;

// ── 레지스터 ─────────────────────────────────────────────
reg        r_en;         // CTRL[0]
reg        r_uart_en;    // CTRL[1]
reg        r_fifo_clr;   // CTRL[7] (자동클리어)
reg        r_test_mode;  // CTRL[8]
reg [31:0] r_i2c_div;
reg [31:0] r_uart_div;
reg        r_irq_en;
reg        r_irq_pend;

// ── FIFO 16엔트리 ─────────────────────────────────────────
localparam FIFO_DEPTH = 16;
reg [31:0] fifo_data [0:FIFO_DEPTH-1];
reg [31:0] fifo_ts   [0:FIFO_DEPTH-1];
reg [3:0]  fifo_wptr;
reg [3:0]  fifo_rptr;
reg [4:0]  fifo_cnt;

wire fifo_empty = (fifo_cnt == 5'd0);
wire fifo_full  = (fifo_cnt == 5'd16);

// FIFO 쓰기 (내부에서만)
task fifo_push;
    input [7:0]  sid;
    input [23:0] dat;
    input [31:0] ts;
    begin
        if (!fifo_full) begin
            fifo_data[fifo_wptr] <= {sid, dat};
            fifo_ts  [fifo_wptr] <= ts;
            fifo_wptr <= fifo_wptr + 4'd1;
            fifo_cnt  <= fifo_cnt  + 5'd1;
        end
    end
endtask

// ── 타임스탬프 카운터 ─────────────────────────────────────
reg [31:0] ts_cnt;
always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) ts_cnt <= 32'd0;
    else          ts_cnt <= ts_cnt + 32'd1;
end

// ── i2c_rd_byte 인스턴스 ─────────────────────────────────
reg        i2c_start;
reg  [6:0] i2c_dev_addr;
reg  [7:0] i2c_reg_addr;
wire       i2c_done;
wire [7:0] i2c_data;
wire       i2c_ack_err;
wire       scl_out_w;
wire       sda_out_w;
wire       sda_in_w = sda;

i2c_rd_byte #(
    .CLK_DIV(CLK_DIV_I2C)
) u_i2c (
    .clk      (aclk),
    .rst_n    (aresetn),
    .start    (i2c_start),
    .dev_addr (i2c_dev_addr),
    .reg_addr (i2c_reg_addr),
    .done     (i2c_done),
    .data_out (i2c_data),
    .ack_err  (i2c_ack_err),
    .scl_out  (scl_out_w),
    .sda_out  (sda_out_w),
    .sda_in   (sda_in_w)
);

// 오픈드레인 출력 (0=drive GND, 1=Hi-Z)
assign scl = scl_out_w ? 1'bz : 1'b0;
assign sda = sda_out_w ? 1'bz : 1'b0;

// ── I2C 순회 FSM ─────────────────────────────────────────
// sensor_id: 0=MPU-6050, 1=BMP280, 2=HMC5883L
localparam [1:0] I_MPU = 2'd0, I_BMP = 2'd1, I_HMC = 2'd2;

// 센서 테이블 (파라미터로 쓰면 더 깔끔하지만 Verilog-2001 호환 간단 버전)
function [6:0] sensor_addr;
    input [1:0] id;
    begin
        case (id)
            I_MPU: sensor_addr = 7'h68;
            I_BMP: sensor_addr = 7'h76;
            I_HMC: sensor_addr = 7'h1E;
            default: sensor_addr = 7'h00;
        endcase
    end
endfunction

function [7:0] sensor_reg;
    input [1:0] id;
    begin
        case (id)
            I_MPU: sensor_reg = 8'h75;  // WHO_AM_I
            I_BMP: sensor_reg = 8'hD0;  // id
            I_HMC: sensor_reg = 8'h0A;  // IRA
            default: sensor_reg = 8'h00;
        endcase
    end
endfunction

localparam [1:0]
    IS_WAIT  = 2'd0,  // 대기 (인터벌 카운터)
    IS_START = 2'd1,  // I2C 시작 펄스
    IS_BUSY  = 2'd2,  // I2C 완료 대기
    IS_PUSH  = 2'd3;  // FIFO 푸시

reg [1:0]  i2c_fsm;
reg [1:0]  i2c_sel;    // 현재 센서 (0~2)
reg [26:0] i2c_timer;  // 대기 카운터 (~100ms @ 100MHz)
reg        i2c_timeout_r;

localparam [26:0] I2C_INTERVAL = 27'd10_000_000;  // 100ms @ 100MHz

always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        i2c_fsm      <= IS_WAIT;
        i2c_sel      <= I_MPU;
        i2c_timer    <= 27'd0;
        i2c_start    <= 1'b0;
        i2c_dev_addr <= 7'd0;
        i2c_reg_addr <= 8'd0;
        i2c_timeout_r <= 1'b0;
    end else begin
        i2c_start <= 1'b0;  // 기본 1사이클

        if (!r_en || r_test_mode) begin
            // 비활성 또는 TEST_MODE: IDLE
            i2c_fsm   <= IS_WAIT;
            i2c_timer <= 27'd0;
        end else begin
            case (i2c_fsm)
            IS_WAIT: begin
                i2c_timer <= i2c_timer + 27'd1;
                if (i2c_timer >= I2C_INTERVAL - 27'd1) begin
                    i2c_timer    <= 27'd0;
                    i2c_dev_addr <= sensor_addr(i2c_sel);
                    i2c_reg_addr <= sensor_reg(i2c_sel);
                    i2c_fsm      <= IS_START;
                end
            end
            IS_START: begin
                i2c_start    <= 1'b1;
                i2c_timeout_r <= 1'b0;
                i2c_fsm      <= IS_BUSY;
            end
            IS_BUSY: begin
                if (i2c_done) begin
                    i2c_timeout_r <= i2c_ack_err;
                    i2c_fsm <= IS_PUSH;
                end
            end
            IS_PUSH: begin
                // FIFO 푸시는 별도 always 블록에서 처리 (task는 always 내에서 직접)
                i2c_sel <= (i2c_sel == 2'd2) ? 2'd0 : i2c_sel + 2'd1;
                i2c_fsm <= IS_WAIT;
            end
            endcase
        end
    end
end

// FIFO 푸시 (I2C)
// IS_PUSH 상태에서 1사이클 동안 fifo_push 수행
// 별도 always로 분리하면 task 충돌 없음 → 직접 인라인
reg i2c_push_r;
always @(posedge aclk) i2c_push_r <= (i2c_fsm == IS_PUSH) && r_en && !r_test_mode;

// ── UART 수신기 (NMEA: $ ~ \n) ───────────────────────────
// 완료 조건: '\n'(0x0A) 수신
// sensor_id = 3 (NMEA 완료)
localparam [2:0]
    US_IDLE = 3'd0,   // 시작비트 대기
    US_STX  = 3'd1,   // 시작비트 확인 (중간 샘플)
    US_DATA = 3'd2,   // 8 데이터 비트
    US_STOP = 3'd3,   // 정지비트
    US_DONE = 3'd4;   // 처리

reg [2:0]  uart_fsm;
reg [15:0] uart_timer;
reg [3:0]  uart_bit_cnt;
reg [7:0]  uart_shift;
reg        uart_rx_sync1, uart_rx_sync2, uart_rx_prev;
reg        uart_in_nmea;  // NMEA 문장 수신 중 ('$' 본 적 있음)
reg [7:0]  uart_last_byte;
reg        uart_nmea_done;  // NMEA 완료 1-펄스

always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        uart_rx_sync1  <= 1'b1;
        uart_rx_sync2  <= 1'b1;
        uart_rx_prev   <= 1'b1;
        uart_fsm       <= US_IDLE;
        uart_timer     <= 16'd0;
        uart_bit_cnt   <= 4'd0;
        uart_shift     <= 8'd0;
        uart_in_nmea   <= 1'b0;
        uart_last_byte <= 8'd0;
        uart_nmea_done <= 1'b0;
    end else begin
        uart_rx_sync1 <= uart_rx;
        uart_rx_sync2 <= uart_rx_sync1;
        uart_rx_prev  <= uart_rx_sync2;
        uart_nmea_done <= 1'b0;

        if (!r_uart_en || !r_en) begin
            uart_fsm   <= US_IDLE;
            uart_timer <= 16'd0;
        end else begin
            case (uart_fsm)
            US_IDLE: begin
                if (uart_rx_sync2 == 1'b0 && uart_rx_prev == 1'b1) begin
                    // 하강 엣지 감지 → 시작비트
                    uart_timer <= r_uart_div[15:0] >> 1; // 반 비트 오프셋으로 중간 샘플
                    uart_fsm   <= US_STX;
                end
            end
            US_STX: begin
                if (uart_timer == 16'd0) begin
                    if (uart_rx_sync2 == 1'b0) begin
                        uart_timer   <= r_uart_div[15:0] - 16'd1;
                        uart_bit_cnt <= 4'd7;
                        uart_shift   <= 8'd0;
                        uart_fsm     <= US_DATA;
                    end else begin
                        uart_fsm <= US_IDLE;  // 잘못된 시작비트
                    end
                end else begin
                    uart_timer <= uart_timer - 16'd1;
                end
            end
            US_DATA: begin
                if (uart_timer == 16'd0) begin
                    uart_shift   <= {uart_rx_sync2, uart_shift[7:1]};  // LSB 먼저
                    uart_timer   <= r_uart_div[15:0] - 16'd1;
                    if (uart_bit_cnt == 4'd0)
                        uart_fsm <= US_STOP;
                    else
                        uart_bit_cnt <= uart_bit_cnt - 4'd1;
                end else begin
                    uart_timer <= uart_timer - 16'd1;
                end
            end
            US_STOP: begin
                if (uart_timer == 16'd0) begin
                    uart_last_byte <= uart_shift;
                    uart_fsm <= US_DONE;
                end else begin
                    uart_timer <= uart_timer - 16'd1;
                end
            end
            US_DONE: begin
                if (uart_last_byte == 8'h24) uart_in_nmea <= 1'b1; // '$'
                if (uart_last_byte == 8'h0A && uart_in_nmea) begin  // '\n'
                    uart_nmea_done <= 1'b1;
                    uart_in_nmea   <= 1'b0;
                end
                uart_fsm <= US_IDLE;
            end
            default: uart_fsm <= US_IDLE;
            endcase
        end
    end
end

// ── FIFO 통합 쓰기 ───────────────────────────────────────
// I2C push (IS_PUSH → i2c_push_r) 또는 UART NMEA done 또는 TEST_MODE INJ
// 간단하게 순위: TEST_MODE > UART > I2C

// TEST_MODE 주입 신호 (AXI 쓰기에서 1-사이클 펄스)
reg        inj_valid;
reg [31:0] inj_data;
reg        push_done;    // FIFO push 완료 1-펄스 (FIFO write block만 구동)
reg        w1c_irq_pend; // IRQ_PEND W1C 요청 1-펄스 (AXI write block만 구동)

integer i;  // Verilog-2001: 모듈 레벨 선언 (unnamed block 불가)
always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        fifo_wptr <= 4'd0;
        fifo_cnt  <= 5'd0;
        push_done <= 1'b0;
        for (i = 0; i < FIFO_DEPTH; i = i + 1) begin
            fifo_data[i] <= 32'd0;
            fifo_ts[i]   <= 32'd0;
        end
    end else begin
        push_done <= 1'b0;  // 기본: 펄스 없음 (매 사이클 클리어)
        if (r_fifo_clr) begin
            fifo_wptr <= 4'd0;
            fifo_cnt  <= 5'd0;
        end else if (r_test_mode && inj_valid) begin
            if (!fifo_full) begin
                fifo_data[fifo_wptr] <= inj_data;
                fifo_ts  [fifo_wptr] <= ts_cnt;
                fifo_wptr <= fifo_wptr + 4'd1;
                fifo_cnt  <= fifo_cnt  + 5'd1;
                push_done <= 1'b1;  // IRQ 이벤트 펄스
            end
        end else if (uart_nmea_done && r_uart_en) begin
            if (!fifo_full) begin
                fifo_data[fifo_wptr] <= {8'd3, 24'h000001};  // sensor_id=3, data=1(완료)
                fifo_ts  [fifo_wptr] <= ts_cnt;
                fifo_wptr <= fifo_wptr + 4'd1;
                fifo_cnt  <= fifo_cnt  + 5'd1;
                push_done <= 1'b1;  // IRQ 이벤트 펄스
            end
        end else if (i2c_push_r) begin
            if (!fifo_full) begin
                fifo_data[fifo_wptr] <= {6'd0, i2c_sel, i2c_timeout_r ? 24'hFF_FFFF : {16'd0, i2c_data}};
                fifo_ts  [fifo_wptr] <= ts_cnt;
                fifo_wptr <= fifo_wptr + 4'd1;
                fifo_cnt  <= fifo_cnt  + 5'd1;
                push_done <= 1'b1;  // IRQ 이벤트 펄스
            end
        end else if (fifo_pop && !fifo_empty) begin
            // 팝은 AXI 읽기 블록이 fifo_pop 펄스로 요청, 여기서 cnt 감소
            fifo_cnt <= fifo_cnt - 5'd1;
        end
    end
end

// ── IRQ 로직 ─────────────────────────────────────────────
// r_irq_pend의 유일한 구동자. W1C 클리어(w1c_irq_pend) 우선, 푸시 이벤트(push_done)로 세트.
// 연속 레벨(!fifo_empty)이 아닌 1-펄스 이벤트 기반 → multi-driver 없음
always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        r_irq_pend <= 1'b0;
        irq_out    <= 1'b0;
    end else begin
        if (w1c_irq_pend)
            r_irq_pend <= 1'b0;           // W1C 클리어 (AXI 쓰기 요청 1-펄스)
        else if (push_done || uart_nmea_done)
            r_irq_pend <= 1'b1;           // 이벤트 기반 세트 (레벨이 아닌 펄스)
        irq_out <= r_irq_en & r_irq_pend;
    end
end

// ── AXI4-Lite 슬레이브 ───────────────────────────────────
reg [15:0] wr_addr;
reg [31:0] wr_data;
reg        wr_pending;

// 쓰기 주소 채널
always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        s_awready  <= 1'b0;
        wr_addr    <= 16'd0;
        wr_pending <= 1'b0;
    end else begin
        s_awready <= 1'b0;
        if (s_awvalid && !wr_pending) begin
            s_awready  <= 1'b1;
            wr_addr    <= s_awaddr;
            wr_pending <= 1'b1;
        end
    end
end

// 쓰기 데이터 채널 + 쓰기 처리
always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        s_wready   <= 1'b0;
        s_bvalid   <= 1'b0;
        s_bresp    <= 2'b00;
        wr_data    <= 32'd0;
        r_en       <= 1'b0;
        r_uart_en  <= 1'b0;
        r_fifo_clr <= 1'b0;
        r_test_mode <= 1'b0;
        r_i2c_div  <= CLK_DIV_I2C;
        r_uart_div <= CLK_DIV_UART;
        r_irq_en   <= 1'b0;
        inj_valid     <= 1'b0;
        inj_data      <= 32'd0;
        w1c_irq_pend  <= 1'b0;
    end else begin
        s_wready      <= 1'b0;
        r_fifo_clr    <= 1'b0;  // 자동 클리어
        inj_valid     <= 1'b0;
        w1c_irq_pend  <= 1'b0;  // 자동 클리어 (1-펄스)

        if (s_bvalid && s_bready) s_bvalid <= 1'b0;

        if (s_wvalid && wr_pending) begin
            s_wready   <= 1'b1;
            wr_data    <= s_wdata;
            wr_pending <= 1'b0;
            s_bvalid   <= 1'b1;
            s_bresp    <= 2'b00;

            case (wr_addr)
            A_CTRL: begin
                r_en        <= s_wdata[0];
                r_uart_en   <= s_wdata[1];
                r_fifo_clr  <= s_wdata[7];
                r_test_mode <= s_wdata[8];
            end
            A_I2C_DIV:  r_i2c_div  <= (s_wdata < 32'd8) ? 32'd8 : s_wdata;
            A_UART_DIV: r_uart_div <= s_wdata;
            A_FIFO_INJ: begin
                inj_valid <= 1'b1;
                inj_data  <= s_wdata;
            end
            A_IRQ_EN:   r_irq_en   <= s_wdata[0];
            A_IRQ_PEND: begin
                if (s_wdata[0]) w1c_irq_pend <= 1'b1;  // W1C 요청 → IRQ block이 처리
            end
            default: ;
            endcase
        end
    end
end

// 읽기 채널
// FIFO 팝: FIFO_DATA 읽기 시 팝 (rdata + rptr 갱신)
reg [31:0] popped_data;
reg [31:0] popped_ts;
reg        fifo_pop;

always @(posedge aclk or negedge aresetn) begin
    if (!aresetn) begin
        s_arready  <= 1'b0;
        s_rvalid   <= 1'b0;
        s_rdata    <= 32'd0;
        s_rresp    <= 2'b00;
        fifo_pop   <= 1'b0;
        fifo_rptr  <= 4'd0;
        popped_data <= 32'd0;
        popped_ts   <= 32'd0;
    end else begin
        s_arready <= 1'b0;
        fifo_pop  <= 1'b0;

        if (s_rvalid && s_rready) s_rvalid <= 1'b0;

        if (s_arvalid && !s_rvalid) begin
            s_arready <= 1'b1;
            s_rvalid  <= 1'b1;
            s_rresp   <= 2'b00;

            case (s_araddr)
            A_CTRL:   s_rdata <= {23'd0, r_test_mode, 6'd0, r_fifo_clr, r_uart_en, r_en};
            A_STATUS: s_rdata <= {29'd0, i2c_timeout_r, fifo_full, fifo_empty};
            A_FIFO_DATA: begin
                if (!fifo_empty) begin
                    popped_data <= fifo_data[fifo_rptr];
                    popped_ts   <= fifo_ts[fifo_rptr];
                    fifo_rptr   <= fifo_rptr + 4'd1;
                    fifo_pop    <= 1'b1;
                    s_rdata     <= fifo_data[fifo_rptr];
                end else begin
                    s_rdata <= 32'd0;
                end
            end
            A_FIFO_TS:  s_rdata <= popped_ts;
            A_FIFO_CNT: s_rdata <= {27'd0, fifo_cnt};
            A_I2C_DIV:  s_rdata <= r_i2c_div;
            A_UART_DIV: s_rdata <= r_uart_div;
            A_ID_REG:   s_rdata <= 32'h48554231;  // HUB1
            A_VER_REG:  s_rdata <= 32'h00010000;  // v1.0.0
            A_IRQ_EN:   s_rdata <= {31'd0, r_irq_en};
            A_IRQ_PEND: s_rdata <= {31'd0, r_irq_pend};
            default:    s_rdata <= 32'd0;
            endcase
        end
    end
end

endmodule

// PASS/FAIL 출력 규약: "PASS ..." / "FAIL ..." 로 시작하는 $display
// TEST_MODE(CTRL[8]=1)로 FIFO 직접 주입해 I2C/UART 슬레이브 모델 없이 검증
// TC1: 리셋 후 레지스터 기본값
// TC2: TEST_MODE FIFO 주입 + 팝
// TC3: TEST_MODE FIFO 풀(16엔트리) + 오버플로 무시
// TC4: FIFO_CLR 동작
// TC5: IRQ_EN/IRQ_PEND W1C 동작
// TC6: ID/VER 레지스터 값

`timescale 1ns/1ps
`default_nettype none

module tb_axil_i2c3_nmea_hub;

localparam CLK_PERIOD = 10;  // 100MHz

reg         aclk;
reg         aresetn;

// AXI4-Lite 버스
reg  [15:0] s_awaddr;
reg         s_awvalid;
wire        s_awready;
reg  [31:0] s_wdata;
reg  [3:0]  s_wstrb;
reg         s_wvalid;
wire        s_wready;
wire [1:0]  s_bresp;
wire        s_bvalid;
reg         s_bready;
reg  [15:0] s_araddr;
reg         s_arvalid;
wire        s_arready;
wire [31:0] s_rdata;
wire [1:0]  s_rresp;
wire        s_rvalid;
reg         s_rready;

// I2C (테스트에서 미사용, Hi-Z)
wire scl_w;
wire sda_w;

// UART (테스트에서 미사용, Hi)
reg         uart_rx;
wire        irq_out;

integer fail_cnt;
integer tc;

// ── DUT 인스턴스 ─────────────────────────────────────────
axil_i2c3_nmea_hub #(
    .CLK_DIV_I2C (1000),
    .CLK_DIV_UART(10417)
) dut (
    .aclk      (aclk),
    .aresetn   (aresetn),
    .s_awaddr  (s_awaddr),
    .s_awvalid (s_awvalid),
    .s_awready (s_awready),
    .s_wdata   (s_wdata),
    .s_wstrb   (s_wstrb),
    .s_wvalid  (s_wvalid),
    .s_wready  (s_wready),
    .s_bresp   (s_bresp),
    .s_bvalid  (s_bvalid),
    .s_bready  (s_bready),
    .s_araddr  (s_araddr),
    .s_arvalid (s_arvalid),
    .s_arready (s_arready),
    .s_rdata   (s_rdata),
    .s_rresp   (s_rresp),
    .s_rvalid  (s_rvalid),
    .s_rready  (s_rready),
    .scl       (scl_w),
    .sda       (sda_w),
    .uart_rx   (uart_rx),
    .irq_out   (irq_out)
);

// ── 클럭 ─────────────────────────────────────────────────
initial aclk = 0;
always #(CLK_PERIOD/2) aclk = ~aclk;

// ── AXI4-Lite 헬퍼 태스크 ────────────────────────────────
task axil_write;
    input [15:0] addr;
    input [31:0] data;
    begin
        @(posedge aclk); #1;
        s_awaddr  = addr; s_awvalid = 1;
        s_wdata   = data; s_wstrb = 4'hF; s_wvalid = 1;
        s_bready  = 1;
        // 주소 수락
        @(posedge aclk);
        while (!s_awready) @(posedge aclk);
        s_awvalid = 0;
        // 데이터 수락
        while (!s_wready) @(posedge aclk);
        s_wvalid = 0;
        // 응답 수락
        while (!s_bvalid) @(posedge aclk);
        @(posedge aclk); #1;
        s_bready = 0;
    end
endtask

task axil_read;
    input  [15:0] addr;
    output [31:0] data;
    begin
        @(posedge aclk); #1;
        s_araddr  = addr; s_arvalid = 1;
        s_rready  = 1;
        @(posedge aclk);
        while (!s_arready) @(posedge aclk);
        s_arvalid = 0;
        while (!s_rvalid) @(posedge aclk);
        data = s_rdata;
        @(posedge aclk); #1;
        s_rready = 0;
    end
endtask

// ── 체크 태스크 ──────────────────────────────────────────
task check_eq;
    input [255:0] name;
    input [31:0]  got;
    input [31:0]  expected;
    begin
        if (got !== expected) begin
            $display("FAIL TC%0d: %s got=0x%08x expected=0x%08x", tc, name, got, expected);
            fail_cnt = fail_cnt + 1;
        end
    end
endtask

// ── 테스트 메인 ──────────────────────────────────────────
reg [31:0] rd;
integer i;

initial begin
    fail_cnt   = 0;
    aresetn    = 0;
    uart_rx    = 1;
    s_awvalid  = 0; s_awaddr  = 0;
    s_wvalid   = 0; s_wdata   = 0; s_wstrb = 0;
    s_bready   = 0;
    s_arvalid  = 0; s_araddr  = 0;
    s_rready   = 0;

    repeat(5) @(posedge aclk);
    aresetn = 1;
    repeat(3) @(posedge aclk);

    // ── TC1: 리셋 후 기본값 ──────────────────────────────
    tc = 1;
    axil_read(16'h0000, rd); check_eq("CTRL default",      rd, 32'h0000_0000);
    axil_read(16'h0004, rd); check_eq("STATUS init",       rd, 32'h0000_0001);  // FIFO_EMPTY=1
    axil_read(16'h0010, rd); check_eq("FIFO_CNT init",     rd, 32'h0000_0000);
    axil_read(16'hFFF0, rd); check_eq("ID_REG",            rd, 32'h4855_4231);  // HUB1
    axil_read(16'hFFF4, rd); check_eq("VER_REG",           rd, 32'h0001_0000);
    $display("TC1: basic register defaults checked");

    // ── TC2: TEST_MODE FIFO 주입 + 팝 ──────────────────
    tc = 2;
    // TEST_MODE 활성 (CTRL[8]=1, EN=1)
    axil_write(16'h0000, 32'h0000_0101);  // TEST_MODE=1, EN=1
    // FIFO_INJ에 sensor_id=0, data=0xABCDEF
    axil_write(16'h001C, 32'h00AB_CDEF);
    repeat(2) @(posedge aclk);
    axil_read(16'h0010, rd); check_eq("FIFO_CNT after 1 inj", rd, 32'h0000_0001);
    axil_read(16'h0004, rd); check_eq("STATUS not empty",     rd[0], 1'b0);  // FIFO_EMPTY=0

    axil_read(16'h0008, rd); check_eq("FIFO_DATA pop",        rd, 32'h00AB_CDEF);
    axil_read(16'h0010, rd); check_eq("FIFO_CNT after pop",   rd, 32'h0000_0000);
    axil_read(16'h0004, rd); check_eq("STATUS empty after pop", rd[0], 1'b1);
    $display("TC2: TEST_MODE inject and pop OK");

    // ── TC3: FIFO 풀(16엔트리) + 오버플로 무시 ─────────
    tc = 3;
    for (i = 0; i < 16; i = i + 1) begin
        axil_write(16'h001C, {8'd0, 8'd0, 8'h00, i[7:0]});
    end
    repeat(5) @(posedge aclk);
    axil_read(16'h0010, rd); check_eq("FIFO_CNT full", rd, 32'd16);
    axil_read(16'h0004, rd); check_eq("STATUS FIFO_FULL", rd[1], 1'b1);

    // 17번째 주입: 드롭되어야 함
    axil_write(16'h001C, 32'hFF_FFFF);
    repeat(2) @(posedge aclk);
    axil_read(16'h0010, rd); check_eq("FIFO_CNT overflow ignored", rd, 32'd16);
    $display("TC3: FIFO full and overflow drop OK");

    // ── TC4: FIFO_CLR ───────────────────────────────────
    tc = 4;
    axil_write(16'h0000, 32'h0000_0181);  // FIFO_CLR=1 | TEST_MODE=1 | EN=1
    repeat(2) @(posedge aclk);
    axil_read(16'h0010, rd); check_eq("FIFO_CNT after CLR", rd, 32'h0000_0000);
    axil_read(16'h0004, rd); check_eq("STATUS FIFO_EMPTY after CLR", rd[0], 1'b1);
    $display("TC4: FIFO_CLR OK");

    // ── TC5: IRQ_EN / IRQ_PEND W1C ─────────────────────
    tc = 5;
    // 한 항목 주입으로 IRQ_PEND 세트 유도
    axil_write(16'h001C, 32'h01_000001);  // sensor_id=1, data=1
    repeat(5) @(posedge aclk);
    axil_read(16'hFFFC, rd); check_eq("IRQ_PEND set", rd[0], 1'b1);
    // IRQ_EN 활성
    axil_write(16'hFFF8, 32'h0000_0001);
    @(posedge aclk);
    if (irq_out !== 1'b1)
        $display("FAIL TC5: irq_out expected 1 got %b", irq_out);
    // IRQ_PEND W1C
    axil_write(16'hFFFC, 32'h0000_0001);
    repeat(3) @(posedge aclk);
    axil_read(16'hFFFC, rd); check_eq("IRQ_PEND cleared", rd[0], 1'b0);
    $display("TC5: IRQ_EN / IRQ_PEND W1C OK");

    // ── TC6: I2C_DIV / UART_DIV 레지스터 R/W ───────────
    tc = 6;
    axil_write(16'h0014, 32'd500);
    axil_read (16'h0014, rd); check_eq("I2C_DIV write", rd, 32'd500);
    axil_write(16'h0018, 32'd20833);
    axil_read (16'h0018, rd); check_eq("UART_DIV write", rd, 32'd20833);
    // I2C_DIV < 8이면 클램프
    axil_write(16'h0014, 32'd4);
    axil_read (16'h0014, rd); check_eq("I2C_DIV clamp", rd, 32'd8);
    $display("TC6: I2C_DIV / UART_DIV R/W and clamp OK");

    // ── 최종 결과 ─────────────────────────────────────────
    repeat(5) @(posedge aclk);
    if (fail_cnt == 0)
        $display("PASS all %0d checks", 6);
    else
        $display("FAIL %0d checks failed", fail_cnt);

    $finish;
end

// 타임아웃 가드
initial begin
    #5_000_000;
    $display("FAIL timeout");
    $finish;
end

endmodule

위험·한계

  • I2C 슬레이브 무응답(NACK) 시 TIMEOUT 비트 세트, 순회 계속 — 센서 없는 환경에서는 timeout 로그 다수 발생
  • FIFO_FULL 시 새 샘플 드롭(오버플로 무시) — 펌웨어가 주기적으로 비워야 함
  • NMEA 문장 최대 82바이트이지만 본 IP는 완료 IRQ만 제공, 문장 내용은 별도 버퍼에 저장하지 않음
  • I2C SCL 스트레칭 미지원 — 슬레이브가 클럭 스트레칭하면 ACK 타이밍 오류 가능
  • TEST_MODE는 디버그 전용, 실제 운영 시 반드시 CTRL[8]=0 확인
  • 센서 모듈 5V 버전 사용 시 레벨 변환 필수 (Basys3 I/O 3.3V 한계)
