소프트 논리 분석기 IP (BRAM 샘플 메모리 + 에지 트리거)

16채널 디지털 신호를 BRAM에 캡처하고 AXI4-Lite로 읽어내는 소프트 논리 분석기. 에지 트리거 FSM(상승/하강), 프리트리거 버퍼, 채널 마스크 기능 포함.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: 미실행

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • 16채널 디지털 입력을 BRAM에 연속 캡처
  • 에지 트리거(상승/하강) FSM으로 캡처 시점 제어
  • 프리트리거 버퍼로 트리거 이전 신호 보존
  • AXI4-Lite 슬레이브 인터페이스로 호스트 CPU 접근
  • 채널 마스크(CH_EN)로 관심 신호 선택
  • iverilog/yosys CI 통과 (S1/S2)

범위 밖

  • UART .vcd 덤프 (이 패키지는 BRAM 캡처 코어만 구현)
  • 아날로그 신호 캡처 (디지털 전용)
  • 보드 동작 확인 (S3, 사람이 수행)
  • 트리거 체인 / 복합 트리거

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-067 기반(ready 상태). AXI4-Lite 슬레이브. 동결 맵 free_slots[16]=0x401A_0000 배정 제안(미동결). 시뮬레이션·합성 결과는 CI에서 검증한다.

4. 메모리맵

axil_sla_wrap 0x401A_0000 크기 0x10000

오프셋이름접근설명
0x00CTRLRWbit0=ARM(자동 클리어), bit1=FORCE(자동 클리어), bit2=TRIG_POL, bit3=SRST(자동 클리어)
0x04STATUSRObit0=ARMED, bit1=CAPTURING, bit2=FULL, bit3=TRIG_FOUND, [19:8]=TRIG_ADDR
0x08TRIG_CHRW[3:0]=트리거 채널 번호 (0~15)
0x0CPRETRIGRW[DEPTH_LOG2-1:0]=프리트리거 샘플 수 (기본 0)
0x10CH_ENRW[15:0]=채널 활성 마스크 (기본 0xFFFF)
0x14RD_ADDRRW[DEPTH_LOG2-1:0]=BRAM 읽기 주소
0x18RD_DATARO[15:0]=RD_ADDR 위치의 샘플 데이터 (1사이클 BRAM 레이턴시)

// sla_core.v — 소프트 논리 분석기 코어 (16채널 BRAM 샘플 메모리 + 에지 트리거 FSM)
// I-067 소프트 논리 분석기 IP (mmio-sla-bram-capture, T2)
// Verilog-2001. iverilog/verilator/yosys 호환.
//
// 포트:
//   clk        — 시스템 클럭
//   rst        — 동기 active-high 리셋
//   arm        — 1사이클 펄스: IDLE→ARMED 전이
//   force_trig — 1사이클 펄스: 트리거 강제 발생 (ARMED 상태에서만 유효)
//   srst       — 소프트 리셋 (모든 카운터·상태 초기화)
//   trig_ch    — [3:0] 트리거 채널 번호 (0~15)
//   trig_pol   — 1=하강 에지, 0=상승 에지
//   pretrig    — [DEPTH_LOG2-1:0] 트리거 전 보관 샘플 수
//   ch_en      — [15:0] 채널 활성 마스크
//   ch_in      — [15:0] 디지털 입력 (외부 메타스테이블 방지 2단 FF 적용 후 연결)
//   rd_addr    — [DEPTH_LOG2-1:0] BRAM 읽기 주소
//
//   sample_out — [15:0] rd_addr의 샘플 데이터
//   armed      — 1=ARMED 상태
//   capturing  — 1=캡처 진행 중 (pre-trigger 포함)
//   full       — 1=BRAM 가득 참 (캡처 완료)
//   trig_found — 1=트리거 발생 확인
//   trig_addr  — [DEPTH_LOG2-1:0] 트리거 발생 시점의 쓰기 주소

`default_nettype none

module sla_core #(
    parameter DEPTH_LOG2 = 12   // BRAM 깊이 (2^12 = 4096 샘플). 테스트벤치는 8로 줄임
) (
    input  wire                   clk,
    input  wire                   rst,

    // 제어
    input  wire                   arm,
    input  wire                   force_trig,
    input  wire                   srst,
    input  wire [3:0]             trig_ch,
    input  wire                   trig_pol,
    input  wire [DEPTH_LOG2-1:0]  pretrig,
    input  wire [15:0]            ch_en,

    // 채널 입력
    input  wire [15:0]            ch_in,

    // BRAM 읽기
    input  wire [DEPTH_LOG2-1:0]  rd_addr,
    output reg  [15:0]            sample_out,

    // 상태 출력
    output reg                    armed,
    output reg                    capturing,
    output reg                    full,
    output reg                    trig_found,
    output reg  [DEPTH_LOG2-1:0]  trig_addr
);

    localparam DEPTH = 1 << DEPTH_LOG2;

    // ── 2단 메타스테이블 방지 ──────────────────────────────
    reg [15:0] ch0, ch1;
    always @(posedge clk) begin
        ch0 <= ch_in;
        ch1 <= ch0;
    end

    // ── 이전 사이클 ch1 (에지 검출용) ────────────────────
    reg [15:0] ch1_d;
    always @(posedge clk)
        ch1_d <= ch1;

    // ── 채널 마스크 ──────────────────────────────────────
    wire [15:0] masked = ch1 & ch_en;

    // ── 트리거 에지 검출 ──────────────────────────────────
    wire trig_bit     = ch1[trig_ch];
    wire trig_bit_d   = ch1_d[trig_ch];
    // 상승 에지(pol=0): d=0, now=1 / 하강 에지(pol=1): d=1, now=0
    wire edge_hit = trig_pol ? ( trig_bit_d & ~trig_bit)
                              : (~trig_bit_d &  trig_bit);

    // ── BRAM (추론 BRAM, Xilinx Block RAM 매핑 대상) ──────
    (* ram_style = "block" *)
    reg [15:0] bram [0:DEPTH-1];

    // ── 쓰기 포인터와 샘플 카운터 ─────────────────────────
    reg [DEPTH_LOG2-1:0] wr_addr;
    reg [DEPTH_LOG2:0]   sample_cnt;   // DEPTH_LOG2+1 비트 (DEPTH까지 셀 수 있음)

    // ── pre-trigger 카운터 ───────────────────────────────
    // pretrig 샘플을 쌓을 때까지 트리거 검출을 억제한다.
    reg [DEPTH_LOG2-1:0] pre_cnt;
    reg                  pre_done;     // pre-trigger 채움 완료
    reg                  trig_req;     // 트리거 요청 래치 (1사이클 펄스 유실 방지)

    // ── FSM ───────────────────────────────────────────────
    // IDLE: armed=0, capturing=0, full=0
    // ARMED: armed=1, capturing=0, full=0  — arm 펄스로 진입, pre-trigger 채움 중
    // CAPTURING: armed=1, capturing=1, full=0  — 트리거 발생 후 나머지 채움
    // FULL: full=1
    localparam S_IDLE      = 2'd0;
    localparam S_ARMED     = 2'd1;
    localparam S_CAPTURING = 2'd2;
    localparam S_FULL      = 2'd3;

    reg [1:0] state;

    always @(posedge clk) begin
        if (rst || srst) begin
            state      <= S_IDLE;
            armed      <= 1'b0;
            capturing  <= 1'b0;
            full       <= 1'b0;
            trig_found <= 1'b0;
            trig_addr  <= {DEPTH_LOG2{1'b0}};
            wr_addr    <= {DEPTH_LOG2{1'b0}};
            sample_cnt <= {(DEPTH_LOG2+1){1'b0}};
            pre_cnt    <= {DEPTH_LOG2{1'b0}};
            pre_done   <= 1'b0;
            trig_req   <= 1'b0;
        end else begin
            case (state)
                S_IDLE: begin
                    if (arm) begin
                        state      <= S_ARMED;
                        armed      <= 1'b1;
                        capturing  <= 1'b0;
                        full       <= 1'b0;
                        trig_found <= 1'b0;
                        wr_addr    <= {DEPTH_LOG2{1'b0}};
                        sample_cnt <= {(DEPTH_LOG2+1){1'b0}};
                        pre_cnt    <= {DEPTH_LOG2{1'b0}};
                        pre_done   <= (pretrig == {DEPTH_LOG2{1'b0}});
                        trig_req   <= 1'b0;
                    end
                end

                S_ARMED: begin
                    // 샘플을 계속 써넣는다 (링 버퍼처럼 순환)
                    bram[wr_addr] <= masked;
                    wr_addr    <= wr_addr + 1'b1;

                    // pre-trigger 카운터
                    if (!pre_done) begin
                        if (pre_cnt >= pretrig - 1'b1)
                            pre_done <= 1'b1;
                        else
                            pre_cnt <= pre_cnt + 1'b1;
                    end

                    // 트리거 요청 래치 (pre_done 전 도착한 1사이클 펄스도 보존)
                    if (force_trig || edge_hit)
                        trig_req <= 1'b1;

                    // 트리거 검출 (pre_done 후에만)
                    if (pre_done && (trig_req || force_trig || edge_hit)) begin
                        state      <= S_CAPTURING;
                        capturing  <= 1'b1;
                        trig_found <= 1'b1;
                        trig_addr  <= wr_addr;
                        sample_cnt <= {{DEPTH_LOG2{1'b0}}, 1'b1};
                        trig_req   <= 1'b0;
                    end
                end

                S_CAPTURING: begin
                    // 트리거 이후 나머지 DEPTH 샘플을 채운다
                    bram[wr_addr] <= masked;
                    wr_addr    <= wr_addr + 1'b1;
                    sample_cnt <= sample_cnt + 1'b1;

                    if (sample_cnt >= DEPTH - 1) begin
                        state     <= S_FULL;
                        capturing <= 1'b0;
                        full      <= 1'b1;
                    end
                end

                S_FULL: begin
                    // arm 펄스가 오면 재무장
                    if (arm) begin
                        state      <= S_ARMED;
                        armed      <= 1'b1;
                        capturing  <= 1'b0;
                        full       <= 1'b0;
                        trig_found <= 1'b0;
                        wr_addr    <= {DEPTH_LOG2{1'b0}};
                        sample_cnt <= {(DEPTH_LOG2+1){1'b0}};
                        pre_cnt    <= {DEPTH_LOG2{1'b0}};
                        pre_done   <= (pretrig == {DEPTH_LOG2{1'b0}});
                        trig_req   <= 1'b0;
                    end
                end

                default: state <= S_IDLE;
            endcase
        end
    end

    // ── BRAM 읽기 포트 (1사이클 레이턴시) ────────────────
    always @(posedge clk)
        sample_out <= bram[rd_addr];

endmodule

// axil_sla_wrap.v — AXI4-Lite 슬레이브 래퍼 (sla_core 인스턴스)
// I-067 소프트 논리 분석기 IP (mmio-sla-bram-capture, T2)
// Verilog-2001. iverilog/verilator/yosys 호환.
//
// 레지스터 맵 (AXI araddr[7:0] 기준):
//   0x00  CTRL      [RW] bit0=ARM(자동 클리어), bit1=FORCE(자동 클리어),
//                        bit2=TRIG_POL(0=상승/1=하강), bit3=SRST
//   0x04  STATUS    [RO] bit0=ARMED, bit1=CAPTURING, bit2=FULL, bit3=TRIG_FOUND,
//                        [19:8]=TRIG_ADDR[11:0] (상위 비트 MSB 위치)
//   0x08  TRIG_CH   [RW] [3:0]=트리거 채널 번호 (0~15)
//   0x0C  PRETRIG   [RW] [DEPTH_LOG2-1:0]=프리트리거 샘플 수
//   0x10  CH_EN     [RW] [15:0]=채널 활성 마스크 (기본 0xFFFF)
//   0x14  RD_ADDR   [RW] [DEPTH_LOG2-1:0]=BRAM 읽기 주소
//   0x18  RD_DATA   [RO] [15:0]=RD_ADDR의 샘플 데이터 (1사이클 레이턴시, AXI 읽기 2사이클)

`default_nettype none

module axil_sla_wrap #(
    parameter DEPTH_LOG2 = 12   // sla_core와 맞춤 (테스트벤치는 8로 줄임)
) (
    // AXI 클럭/리셋
    input  wire        aclk,
    input  wire        aresetn,  // active-low

    // AXI4-Lite Write Channel
    input  wire [7:0]  s_awaddr,
    input  wire        s_awvalid,
    output reg         s_awready,
    input  wire [31:0] s_wdata,
    input  wire [3:0]  s_wstrb,
    input  wire        s_wvalid,
    output reg         s_wready,
    output reg  [1:0]  s_bresp,
    output reg         s_bvalid,
    input  wire        s_bready,

    // AXI4-Lite Read Channel
    input  wire [7:0]  s_araddr,
    input  wire        s_arvalid,
    output reg         s_arready,
    output reg  [31:0] s_rdata,
    output reg  [1:0]  s_rresp,
    output reg         s_rvalid,
    input  wire        s_rready,

    // 채널 입력 (메타스테이블 방지는 sla_core 내부)
    input  wire [15:0] ch_in
);

    // ── 내부 리셋 (active-high) ───────────────────────────
    wire rst = !aresetn;

    // ── 제어 레지스터 ────────────────────────────────────
    reg        ctrl_arm;        // 1사이클 펄스 (자동 클리어)
    reg        ctrl_force;      // 1사이클 펄스 (자동 클리어)
    reg        ctrl_trig_pol;
    reg        ctrl_srst;       // 1사이클 펄스 (자동 클리어)
    reg [3:0]  ctrl_trig_ch;
    reg [DEPTH_LOG2-1:0] ctrl_pretrig;
    reg [15:0] ctrl_ch_en;
    reg [DEPTH_LOG2-1:0] ctrl_rd_addr;

    // ── sla_core 출력 ───────────────────────────────────
    wire [15:0]            sample_out;
    wire                   armed;
    wire                   capturing;
    wire                   full;
    wire                   trig_found;
    wire [DEPTH_LOG2-1:0]  trig_addr;

    // STATUS[19:8]=TRIG_ADDR를 항상 12비트 필드에 넣기 위한 zero-extension
    // (DEPTH_LOG2 <= 12 범위에서만 사용)
    wire [11:0]            trig_addr_12 = trig_addr;

    // ── sla_core 인스턴스 ────────────────────────────────
    sla_core #(
        .DEPTH_LOG2(DEPTH_LOG2)
    ) u_core (
        .clk        (aclk),
        .rst        (rst),
        .arm        (ctrl_arm),
        .force_trig (ctrl_force),
        .srst       (ctrl_srst),
        .trig_ch    (ctrl_trig_ch),
        .trig_pol   (ctrl_trig_pol),
        .pretrig    (ctrl_pretrig),
        .ch_en      (ctrl_ch_en),
        .ch_in      (ch_in),
        .rd_addr    (ctrl_rd_addr),
        .sample_out (sample_out),
        .armed      (armed),
        .capturing  (capturing),
        .full       (full),
        .trig_found (trig_found),
        .trig_addr  (trig_addr)
    );

    // ── Write FSM ─────────────────────────────────────────
    reg       aw_pend;
    reg [7:0] aw_addr_r;
    reg       w_pend;
    reg [31:0] w_data_r;

    always @(posedge aclk) begin
        if (rst) begin
            s_awready    <= 1'b0;
            s_wready     <= 1'b0;
            s_bvalid     <= 1'b0;
            s_bresp      <= 2'b00;
            aw_pend      <= 1'b0;
            w_pend       <= 1'b0;
            ctrl_arm     <= 1'b0;
            ctrl_force   <= 1'b0;
            ctrl_trig_pol <= 1'b0;
            ctrl_srst    <= 1'b0;
            ctrl_trig_ch <= 4'd0;
            ctrl_pretrig <= {DEPTH_LOG2{1'b0}};
            ctrl_ch_en   <= 16'hFFFF;
            ctrl_rd_addr <= {DEPTH_LOG2{1'b0}};
        end else begin
            // 1사이클 펄스 자동 클리어
            ctrl_arm   <= 1'b0;
            ctrl_force <= 1'b0;
            ctrl_srst  <= 1'b0;

            // awready 핸드셰이크
            if (s_awvalid && !aw_pend && !s_bvalid) begin
                s_awready <= 1'b1;
                aw_pend   <= 1'b1;
                aw_addr_r <= s_awaddr;
            end else begin
                s_awready <= 1'b0;
            end

            // wready 핸드셰이크
            if (s_wvalid && !w_pend && !s_bvalid) begin
                s_wready <= 1'b1;
                w_pend   <= 1'b1;
                w_data_r <= s_wdata;
            end else begin
                s_wready <= 1'b0;
            end

            // 주소와 데이터 모두 수신 → 레지스터 쓰기 + bresp
            if (aw_pend && w_pend && !s_bvalid) begin
                case (aw_addr_r[7:2])
                    6'h00: begin  // CTRL
                        ctrl_arm     <= w_data_r[0];
                        ctrl_force   <= w_data_r[1];
                        ctrl_trig_pol <= w_data_r[2];
                        ctrl_srst    <= w_data_r[3];
                    end
                    6'h02: ctrl_trig_ch  <= w_data_r[3:0];       // TRIG_CH
                    6'h03: ctrl_pretrig  <= w_data_r[DEPTH_LOG2-1:0]; // PRETRIG
                    6'h04: ctrl_ch_en    <= w_data_r[15:0];      // CH_EN
                    6'h05: ctrl_rd_addr  <= w_data_r[DEPTH_LOG2-1:0]; // RD_ADDR
                    default: ;  // 나머지는 무시, OKAY 반환
                endcase
                s_bvalid <= 1'b1;
                s_bresp  <= 2'b00;  // OKAY
                aw_pend  <= 1'b0;
                w_pend   <= 1'b0;
            end

            if (s_bvalid && s_bready)
                s_bvalid <= 1'b0;
        end
    end

    // ── Read FSM ──────────────────────────────────────────
    // RD_DATA는 sla_core에서 1사이클 레이턴시가 있으므로
    // arready를 assert한 다음 사이클에 rvalid를 올린다.
    reg [7:0] ar_addr_r;

    always @(posedge aclk) begin
        if (rst) begin
            s_arready <= 1'b0;
            s_rvalid  <= 1'b0;
            s_rdata   <= 32'h0;
            s_rresp   <= 2'b00;
        end else begin
            if (s_arvalid && !s_rvalid) begin
                s_arready <= 1'b1;
                ar_addr_r <= s_araddr;
            end else begin
                s_arready <= 1'b0;
            end

            if (s_arready && s_arvalid) begin
                s_rvalid <= 1'b1;
                s_rresp  <= 2'b00;
                case (ar_addr_r[7:2])
                    6'h00: s_rdata <= {28'b0, ctrl_srst, ctrl_trig_pol,
                                       ctrl_force, ctrl_arm};  // CTRL 읽기
                    6'h01: s_rdata <= {12'b0, trig_addr_12,
                                       4'b0, trig_found, full, capturing, armed}; // STATUS
                    6'h02: s_rdata <= {28'b0, ctrl_trig_ch};   // TRIG_CH
                    6'h03: s_rdata <= {{(32-DEPTH_LOG2){1'b0}}, ctrl_pretrig}; // PRETRIG
                    6'h04: s_rdata <= {16'b0, ctrl_ch_en};     // CH_EN
                    6'h05: s_rdata <= {{(32-DEPTH_LOG2){1'b0}}, ctrl_rd_addr}; // RD_ADDR
                    6'h06: s_rdata <= {16'b0, sample_out};     // RD_DATA
                    default: s_rdata <= 32'hDEAD_BEEF;
                endcase
            end

            if (s_rvalid && s_rready)
                s_rvalid <= 1'b0;
        end
    end

endmodule

// tb_axil_sla_wrap.v — 자체 점검형 테스트벤치
// I-067 소프트 논리 분석기 IP (mmio-sla-bram-capture, T2)
// Verilog-2001. iverilog 호환.
//
// 통과 → "PASS" 로 시작하는 줄, 실패 → "FAIL" 로 시작하는 줄 출력.
// 비교는 !== 로 X/Z도 실패로 센다.
//
// 테스트 케이스:
//   TC1: ARM + FORCE 트리거 → STATUS.FULL=1, STATUS.TRIG_FOUND=1, STATUS.ARMED=1
//   TC2: ARM + 실제 상승 에지 (ch0) → TRIG_FOUND=1, RD_DATA[trig_addr] bit0=1
//   TC3: CH_EN 마스킹 → ARM + FORCE → RD_DATA에 비활성 채널 비트 = 0
//
// DEPTH_LOG2=8 (256 샘플)으로 짧게 설정해 빠른 시뮬레이션.

`timescale 1ns/1ps
`default_nettype none

module tb_axil_sla_wrap;

    // ── 클럭 ──────────────────────────────────────────────
    reg aclk;
    initial aclk = 0;
    always #5 aclk = ~aclk;  // 100 MHz (10 ns 주기)

    // ── DUT 포트 ──────────────────────────────────────────
    reg        aresetn;
    reg  [7:0] s_awaddr;
    reg        s_awvalid;
    wire       s_awready;
    reg  [31:0] s_wdata;
    reg  [3:0]  s_wstrb;
    reg         s_wvalid;
    wire        s_wready;
    wire [1:0]  s_bresp;
    wire        s_bvalid;
    reg         s_bready;
    reg  [7:0]  s_araddr;
    reg         s_arvalid;
    wire        s_arready;
    wire [31:0] s_rdata;
    wire [1:0]  s_rresp;
    wire        s_rvalid;
    reg         s_rready;
    reg  [15:0] ch_in;

    // ── DUT 인스턴스 (DEPTH_LOG2=8, 256 샘플) ───────────
    axil_sla_wrap #(
        .DEPTH_LOG2(8)
    ) dut (
        .aclk      (aclk),
        .aresetn   (aresetn),
        .s_awaddr  (s_awaddr),
        .s_awvalid (s_awvalid),
        .s_awready (s_awready),
        .s_wdata   (s_wdata),
        .s_wstrb   (s_wstrb),
        .s_wvalid  (s_wvalid),
        .s_wready  (s_wready),
        .s_bresp   (s_bresp),
        .s_bvalid  (s_bvalid),
        .s_bready  (s_bready),
        .s_araddr  (s_araddr),
        .s_arvalid (s_arvalid),
        .s_arready (s_arready),
        .s_rdata   (s_rdata),
        .s_rresp   (s_rresp),
        .s_rvalid  (s_rvalid),
        .s_rready  (s_rready),
        .ch_in     (ch_in)
    );

    // ── 실패 카운터 ───────────────────────────────────────
    integer fail_cnt;

    // ── AXI4-Lite 헬퍼 태스크 ────────────────────────────
    task axil_write;
        input [7:0]  addr;
        input [31:0] data;
        begin
            @(posedge aclk); #1;
            s_awaddr = addr; s_awvalid = 1;
            s_wdata  = data; s_wvalid  = 1; s_wstrb = 4'hF;
            s_bready = 1;
            wait(s_bvalid); @(posedge aclk); #1;
            s_awvalid = 0;
            s_wvalid  = 0;
            s_bready  = 0;
        end
    endtask

    task axil_read;
        input  [7:0]  addr;
        output [31:0] data;
        begin
            @(posedge aclk); #1;
            s_araddr = addr; s_arvalid = 1;
            wait(s_arready); @(posedge aclk); #1;
            s_arvalid = 0;
            s_rready  = 1;
            wait(s_rvalid);
            data = s_rdata;
            @(posedge aclk); #1;
            s_rready = 0;
        end
    endtask

    // ── 체크 헬퍼 ────────────────────────────────────────
    task check_eq;
        input [255:0] name;
        input [31:0]  got;
        input [31:0]  exp;
        begin
            if (got !== exp) begin
                $display("FAIL %0s: got=%0h exp=%0h", name, got, exp);
                fail_cnt = fail_cnt + 1;
            end else begin
                $display("pass %0s: %0h", name, got);
            end
        end
    endtask

    task check_ne;
        input [255:0] name;
        input [31:0]  got;
        input [31:0]  neq;
        begin
            if (got === neq) begin
                $display("FAIL %0s: got=%0h (expected !==%0h)", name, got, neq);
                fail_cnt = fail_cnt + 1;
            end else begin
                $display("pass %0s: %0h != %0h", name, got, neq);
            end
        end
    endtask

    // ── 메인 ─────────────────────────────────────────────
    reg [31:0] rd;
    reg [7:0]  trig_a;
    integer    i;

    initial begin
        // 초기화
        fail_cnt  = 0;
        aresetn   = 0;
        s_awvalid = 0; s_wvalid = 0; s_bready = 0;
        s_arvalid = 0; s_rready = 0;
        s_awaddr  = 0; s_wdata = 0; s_wstrb = 0;
        s_araddr  = 0;
        ch_in = 16'h0000;

        repeat(5) @(posedge aclk);
        aresetn = 1;
        repeat(2) @(posedge aclk);

        // 공통 설정: TRIG_CH=0, PRETRIG=10, CH_EN=0xFFFF, TRIG_POL=0(상승)
        axil_write(8'h08, 32'h0);        // TRIG_CH=0
        axil_write(8'h0C, 32'd10);       // PRETRIG=10
        axil_write(8'h10, 32'hFFFF);     // CH_EN=0xFFFF

        // ────────────────────────────────────────────────
        // TC1: ARM + FORCE 트리거
        //   PRETRIG=10 → 10클럭 대기 후 FORCE 발생
        //   256 - pretrig(10) + 트리거 위치 ≈ 256 샘플 채움
        //   STATUS: FULL=1, TRIG_FOUND=1, ARMED=1
        // ────────────────────────────────────────────────
        $display("=== TC1: ARM + FORCE 트리거 ===");
        axil_write(8'h00, 32'h1);  // ARM (bit0=1)
        // pre-trigger 채움 대기 (2단 FF + pre_cnt=10 + 여유)
        repeat(20) @(posedge aclk);
        // FORCE 트리거 (bit1=1)
        axil_write(8'h00, 32'h2);
        // 256 샘플 캡처 대기 (256 클럭 + 여유)
        repeat(400) @(posedge aclk);

        axil_read(8'h04, rd);
        check_eq("TC1:ARMED",      rd & 32'h1, 32'h1);
        check_eq("TC1:FULL",       (rd >> 2) & 32'h1, 32'h1);
        check_eq("TC1:TRIG_FOUND", (rd >> 3) & 32'h1, 32'h1);

        // ────────────────────────────────────────────────
        // TC2: ARM + 실제 상승 에지 (ch0=bit0)
        //   PRETRIG=0으로 설정 → 즉시 트리거 검출 가능
        //   ch_in[0] 를 0→1로 올리면 트리거 발생
        //   RD_DATA[trig_addr][0] = 1 이어야 함
        // ────────────────────────────────────────────────
        $display("=== TC2: ARM + 실제 상승 에지 ===");
        axil_write(8'h0C, 32'd0);   // PRETRIG=0
        ch_in = 16'h0000;

        axil_write(8'h00, 32'h1);   // ARM
        // 2단 FF 전파 + 여유
        repeat(5) @(posedge aclk);
        // ch0 상승 에지
        @(posedge aclk); #1;
        ch_in = 16'h0001;
        // 2단 FF 전파 대기
        repeat(5) @(posedge aclk);
        // 256 샘플 채움 대기
        repeat(280) @(posedge aclk);

        axil_read(8'h04, rd);
        check_eq("TC2:TRIG_FOUND", (rd >> 3) & 32'h1, 32'h1);
        check_eq("TC2:FULL",       (rd >> 2) & 32'h1, 32'h1);

        // trig_addr 읽기 (STATUS[19:8])
        trig_a = (rd >> 8) & 8'hFF;
        $display("pass TC2:TRIG_ADDR=%0d", trig_a);

        // trig_addr 위치의 샘플: ch0=1 이어야 함
        axil_write(8'h14, {24'd0, trig_a});  // RD_ADDR=trig_addr
        repeat(3) @(posedge aclk);           // BRAM 1사이클 레이턴시 + 여유
        axil_read(8'h18, rd);
        check_eq("TC2:RD_DATA_bit0_at_trig", rd & 32'h1, 32'h1);

        // ────────────────────────────────────────────────
        // TC3: CH_EN 마스킹
        //   CH_EN=0x00FE (bit0 비활성) → ARM + FORCE
        //   캡처 후 모든 샘플의 bit0 = 0 이어야 함 (ch_in[0]=1 유지)
        // ────────────────────────────────────────────────
        $display("=== TC3: CH_EN 마스킹 ===");
        axil_write(8'h10, 32'h00FE);  // CH_EN: bit0 비활성
        axil_write(8'h0C, 32'd5);     // PRETRIG=5
        ch_in = 16'h0001;              // bit0=1 (하지만 마스킹됨)

        axil_write(8'h00, 32'h1);     // ARM
        repeat(15) @(posedge aclk);
        axil_write(8'h00, 32'h2);     // FORCE
        repeat(400) @(posedge aclk);

        axil_read(8'h04, rd);
        check_eq("TC3:FULL", (rd >> 2) & 32'h1, 32'h1);

        // 주소 0의 샘플 읽기: bit0은 마스킹되어 0이어야 함
        axil_write(8'h14, 32'h0);     // RD_ADDR=0
        repeat(3) @(posedge aclk);
        axil_read(8'h18, rd);
        check_eq("TC3:RD_DATA_bit0_masked", rd & 32'h1, 32'h0);

        // ────────────────────────────────────────────────
        // 최종 판정
        // ────────────────────────────────────────────────
        if (fail_cnt === 0)
            $display("PASS all %0d checks", 7);
        else
            $display("FAIL %0d check(s) failed", fail_cnt);

        $finish;
    end

    // ── 타임아웃 ─────────────────────────────────────────
    initial begin
        #2000000;
        $display("FAIL timeout");
        $finish;
    end

endmodule

9. 참고 카드와 위험

참고 카드

  • mmio-qenc-diag-checker — AXI4-Lite FSM 패턴, BRAM 추론, 2단 메타스테이블 FF 참고 (카드 미작성)
  • mmio-uart-axilite-bridge — UART 덤프 연동 로드맵 참고 (카드 미작성)

위험·한계

  • BRAM 1사이클 읽기 레이턴시: RD_ADDR 쓰기 후 1사이클 뒤 RD_DATA가 유효. AXI 읽기는 2사이클 후 rvalid 반환.
  • DEPTH_LOG2=12에서 BRAM 타일 소비 증가 (4096×16bit = 8KB). Basys3 50개 BRAM 중 1개 소비.
  • pre_done 계산: pretrig=0이면 즉시 pre_done=1. 작은 pretrig 값에서 1사이클 오차 가능.
  • STATUS[19:8]은 DEPTH_LOG2=8 테스트벤치에서 하위 8비트만 유효 (상위 4비트 항상 0).
