PS/2 마우스 호스트 FSM + VGA 커서 오버레이 (프레임 버퍼 없음)

PS/2 마우스를 초기화(BAT 대기 → F4 전송 → 스트리밍)하고 3바이트 패킷을 디코딩해 커서 좌표(X/Y 클램프)를 갱신하는 호스트 FSM. VGA 640×480 60Hz에서 프레임 버퍼 없이 커서만 오버레이한다. IP 표준 rev.1 관리 레지스터(ID=PS2M) 포함.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: pending

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

1. 목표와 범위

목표

  • PS/2 마우스 초기화: BAT 수신(0xAA→0x00) 후 F4 전송, ACK(0xFA) 대기
  • 3바이트 이동 패킷 디코딩: (flags, dx, dy), 부호 확장·오버플로 처리
  • 커서 좌표 클램프: X 0–639, Y 0–479, PS/2 Y 반전(위=Y감소) 처리
  • mmio-lite 슬레이브로 MOUSE_X/Y/BTN/STATUS 레지스터 노출
  • IP 표준 rev.1 관리 레지스터(ID=PS2M / VER / IRQ_STATUS W1C / IRQ_ENABLE) 포함
  • VGA 640×480 60Hz: 프레임 버퍼 없이 5×5픽셀 흰색 커서 오버레이
  • iverilog 시뮬레이션 PASS 확인(S1 목표)

범위 밖

  • 실제 MicroBlaze V CPU 연동 및 드라이버 작성
  • AXI4-Lite 래퍼 RTL(동결 맵 이식 시 별도 작성)
  • 보드에서 직접 검증(S3)은 사용자 담당
  • PS/2 키보드와 동시 사용, USB HID 직접 제어
  • VGA 프레임 버퍼, 배경 그래픽, 스프라이트
  • 마우스 가속·감도 소프트웨어 제어

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-008 기반. Basys3 USB-HID PS/2 에뮬레이션(C17/B17) + 보드 내장 VGA 커넥터 사용. kbd 주변장치([0B])와 동일 PS/2 포트 공유 불가 — 마우스 사용 시 키보드 IP 비활성 필요.

클럭: 100 MHz 단일 클럭 (CLK_100, W5), 비동기 리셋 (BTNC, U18, 액티브 하이)

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite (자체, 단독 실행용)
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획이 패키지는 단독 시뮬레이션과 보드 합성 확인을 위해 mmio-lite 버스(offset[15:0] 직접 비교)를 사용한다. 동결 맵(mbv-frozen-v2) 슬롯 [16]=0x4016_0000 / intc=13 에 이식하려면 (1) IP 표준 래퍼(axil_slave 패턴, 64KB 창)를 추가 작성하고 (2) AXI Interconnect에 연결하며 (3) IRQ는 INTC 입력 13번으로 배선한다. 레지스터 오프셋(0x0000~0x0010, 0xFFF0~0xFFFC)은 그대로 유지된다. VGA와 PS/2 CLK/DATA는 이식 시에도 보드 핀(C17/B17/P19/R19)을 직접 사용한다.

rtl/ps2_mouse_host.v 289줄
// ps2_mouse_host.v
// PS/2 마우스 호스트 수신기 + MMIO-lite 레지스터
// IP 표준 rev.1 관리 레지스터 (ID/VER/IRQ_STATUS W1C/IRQ_ENABLE) 포함
// Verilog-2001. iverilog/verilator/yosys 호환.
// 원작자: 하네스 자체 작성 (2026-11-04)

`default_nettype none

module ps2_mouse_host #(
    parameter INHIBIT_CNT = 10000  // CLK 인히빗 클럭 수 (기본: 100μs @100MHz)
) (
    input  wire        clk,
    input  wire        rst,

    // MMIO-lite 슬레이브
    input  wire        sel,
    input  wire        we,
    input  wire [15:0] offset,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,

    // 커서 출력
    output reg  [9:0]  cursor_x,
    output reg  [9:0]  cursor_y,
    output reg  [2:0]  mouse_btn,
    output wire        irq_out,

    // PS/2 오픈드레인 (oe=1 → 선 로우, oe=0 → Hi-Z)
    input  wire        ps2_clk_in,
    input  wire        ps2_data_in,
    output reg         ps2_clk_oe,
    output reg         ps2_data_oe
);

// ─── MMIO 레지스터 오프셋 ─────────────────────────────────
localparam OFF_CTRL      = 16'h0000;
localparam OFF_MOUSE_X   = 16'h0004;
localparam OFF_MOUSE_Y   = 16'h0008;
localparam OFF_MOUSE_BTN = 16'h000C;
localparam OFF_STATUS    = 16'h0010;
localparam OFF_ID        = 16'hFFF0;
localparam OFF_VER       = 16'hFFF4;
localparam OFF_IRQ_STS   = 16'hFFF8;
localparam OFF_IRQ_EN    = 16'hFFFC;

localparam IP_ID  = 32'h5053_324D; // 'PS2M'
localparam IP_VER = 32'h0100_0100;

// ─── 제어 레지스터 ───────────────────────────────────────
reg ctrl_enable;
reg ctrl_cursor_en;
reg irq_status_r;
reg irq_enable_r;
reg [1:0] status_r;  // [0]=init_done, [1]=streaming

assign irq_out = irq_status_r & irq_enable_r;

// ─── PS/2 CLK 2단 동기화 + 하강 에지 검출 ───────────────
reg ps2_clk_s0, ps2_clk_s1, ps2_clk_prev;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        ps2_clk_s0 <= 1'b1; ps2_clk_s1 <= 1'b1; ps2_clk_prev <= 1'b1;
    end else begin
        ps2_clk_s0 <= ps2_clk_in; ps2_clk_s1 <= ps2_clk_s0;
        ps2_clk_prev <= ps2_clk_s1;
    end
end
wire ps2_clk_fall = ps2_clk_prev & ~ps2_clk_s1;
// TX 중에는 수신기 비활성 (TX_INH, TX_START, TX_BITS 상태)
wire rx_en = (state != S_TX_INH) && (state != S_TX_START) && (state != S_TX_BITS);

// PS/2 DATA 2단 동기화
reg ps2_data_s0, ps2_data_s1;
always @(posedge clk or posedge rst) begin
    if (rst) begin ps2_data_s0 <= 1'b1; ps2_data_s1 <= 1'b1; end
    else     begin ps2_data_s0 <= ps2_data_in; ps2_data_s1 <= ps2_data_s0; end
end

// ─── 11비트 프레임 수신기 ────────────────────────────────
reg [3:0]  rx_bit_cnt;
reg [8:0]  rx_shift;   // D7..D0, parity (9비트 순차 누적)
reg        rx_done;
reg [7:0]  rx_byte;
reg        rx_ok;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        rx_bit_cnt <= 4'd0; rx_shift <= 9'd0;
        rx_done <= 1'b0; rx_byte <= 8'd0; rx_ok <= 1'b0;
    end else begin
        rx_done <= 1'b0;
        if (ps2_clk_fall && rx_en) begin
            if (rx_bit_cnt == 4'd0) begin
                // 스타트 비트: DATA=0 이어야 함
                if (!ps2_data_s1) rx_bit_cnt <= 4'd1;
            end else begin
                rx_shift <= {ps2_data_s1, rx_shift[8:1]};
                rx_bit_cnt <= rx_bit_cnt + 4'd1;
                if (rx_bit_cnt == 4'd10) begin
                    // rx_shift 최종: [8]=stop [7]=parity [6:0]=D7..D1, ps2_data_s1=D0
                    // 실제 비트 순서: D0 먼저 수신 → shift right: shift[0]=D0, ..., shift[7]=D7, shift[8]=parity, 현재bit=stop
                    rx_bit_cnt <= 4'd0;
                    rx_byte    <= rx_shift[7:0];     // D0=shift[0] ... D7=shift[7]
                    rx_ok      <= ^{rx_shift[8], rx_shift[7:0]}; // 홀수패리티: 모두 XOR=1
                    rx_done    <= 1'b1;
                end
            end
        end
    end
end

// ─── TX: 0xF4 Enable Data Reporting ─────────────────────
// 0xF4 = 1111_0100 (1의 개수=5, 홀수) → 패리티 비트=0
// tx_frame[0]=start=0, [8:1]=D0..D7, [9]=parity=0, [10]=stop=1
localparam [10:0] TX_FRAME = 11'b1_0_11110100_0;

reg [19:0] inh_cnt;
reg [3:0]  tx_bit_cnt;

// ─── 메인 FSM ────────────────────────────────────────────
localparam [2:0]
    S_WAIT_AA  = 3'd0,
    S_WAIT_00  = 3'd1,
    S_TX_INH   = 3'd2,
    S_TX_START = 3'd3,
    S_TX_BITS  = 3'd4,
    S_WAIT_ACK = 3'd5,
    S_STREAM   = 3'd6;

reg [2:0] state;
reg [1:0] pkt_cnt;
reg [7:0] pkt0, pkt1, pkt2;
reg       pkt_ready;  // 1클럭 펄스: 3바이트 패킷 완성

always @(posedge clk or posedge rst) begin
    if (rst) begin
        state       <= S_WAIT_AA;
        ps2_clk_oe  <= 1'b0; ps2_data_oe <= 1'b0;
        inh_cnt     <= 20'd0; tx_bit_cnt <= 4'd0;
        pkt_cnt     <= 2'd0;
        pkt0 <= 8'd0; pkt1 <= 8'd0; pkt2 <= 8'd0;
        pkt_ready   <= 1'b0;
        mouse_btn   <= 3'd0;
        irq_status_r   <= 1'b0;
        status_r       <= 2'd0;
        ctrl_enable    <= 1'b0;
        ctrl_cursor_en <= 1'b0;
        irq_enable_r   <= 1'b0;
    end else begin
        pkt_ready <= 1'b0; // 기본: 클리어

        // MMIO 쓰기
        if (sel && we) begin
            case (offset)
                OFF_CTRL:    begin ctrl_enable <= wdata[0]; ctrl_cursor_en <= wdata[1]; end
                OFF_IRQ_STS: irq_status_r <= irq_status_r & ~wdata[0]; // W1C
                OFF_IRQ_EN:  irq_enable_r <= wdata[0];
                default: ;
            endcase
        end

        case (state)
            S_WAIT_AA: begin
                if (rx_done && rx_ok && rx_byte == 8'hAA) state <= S_WAIT_00;
            end
            S_WAIT_00: begin
                if (rx_done && rx_ok && rx_byte == 8'h00) begin
                    ps2_clk_oe <= 1'b1;
                    inh_cnt    <= 20'd0;
                    state      <= S_TX_INH;
                end
            end
            S_TX_INH: begin
                if (inh_cnt >= INHIBIT_CNT - 1) begin
                    ps2_data_oe <= 1'b1; // 스타트 비트 세팅
                    state       <= S_TX_START;
                end else inh_cnt <= inh_cnt + 20'd1;
            end
            S_TX_START: begin
                ps2_clk_oe <= 1'b0; // CLK 개방 → 장치 CLK 생성 시작
                tx_bit_cnt <= 4'd0;
                state      <= S_TX_BITS;
            end
            S_TX_BITS: begin
                if (ps2_clk_fall) begin
                    if (tx_bit_cnt == 4'd10) begin
                        ps2_data_oe <= 1'b0;
                        state       <= S_WAIT_ACK;
                    end else begin
                        // oe=1 → DATA=0; oe=0 → DATA=Hi-Z=1 (풀업)
                        // tx_bit_cnt=0 → D0=TX_FRAME[1], ..., =9 → stop=TX_FRAME[10]
                        ps2_data_oe <= ~TX_FRAME[tx_bit_cnt + 1];
                        tx_bit_cnt  <= tx_bit_cnt + 4'd1;
                    end
                end
            end
            S_WAIT_ACK: begin
                if (rx_done && rx_ok && rx_byte == 8'hFA) begin
                    status_r[0] <= 1'b1; // init_done
                    state       <= S_STREAM;
                end
            end
            S_STREAM: begin
                if (ctrl_enable) begin
                    status_r[1] <= 1'b1;
                    if (rx_done && rx_ok) begin
                        case (pkt_cnt)
                            2'd0: begin pkt0 <= rx_byte; pkt_cnt <= 2'd1; end
                            2'd1: begin pkt1 <= rx_byte; pkt_cnt <= 2'd2; end
                            2'd2: begin
                                pkt2      <= rx_byte;
                                pkt_cnt   <= 2'd0;
                                mouse_btn <= pkt0[2:0];
                                pkt_ready <= 1'b1;
                                irq_status_r <= 1'b1;
                            end
                            default: pkt_cnt <= 2'd0;
                        endcase
                    end
                end
            end
            default: state <= S_WAIT_AA;
        endcase
    end
end

// ─── 좌표 갱신 (pkt_ready 다음 클럭에서 pkt2 유효) ───────
// pkt_ready 직전 클럭에 pkt2 <= rx_byte 가 실행되므로
// pkt_ready 펄스가 나오는 클럭에 pkt0/pkt1 은 이미 등록됨.
// pkt2 도 같은 클럭에 등록 완료 → pkt_ready 다음 클럭에서 사용.

reg pkt_ready_d;  // 1클럭 지연
always @(posedge clk or posedge rst) begin
    if (rst) pkt_ready_d <= 1'b0;
    else     pkt_ready_d <= pkt_ready;
end

// 좌표 계산 조합 논리
wire [8:0]  dx_sgn  = {pkt0[4], pkt1};        // 부호있는 9비트
wire [8:0]  dy_sgn  = {pkt0[5], pkt2};
wire [10:0] new_x_w = {1'b0, cursor_x} + {{2{dx_sgn[8]}}, dx_sgn};
// PS/2 양수 dy = 위로 → cursor_y 감소 → cursor_y - dy_sgn
wire [10:0] neg_dy  = ~{2'b00, dy_sgn} + 11'd1;
wire [10:0] new_y_w = {1'b0, cursor_y} + neg_dy;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        cursor_x <= 10'd320;
        cursor_y <= 10'd240;
    end else if (pkt_ready_d) begin
        // X 갱신 (클램프 0~639)
        if (new_x_w[10])
            cursor_x <= 10'd0;
        else if (new_x_w[9:0] > 10'd639)
            cursor_x <= 10'd639;
        else

module vga_sync_cursor (
    input  wire        clk,         // 100 MHz 시스템 클럭
    input  wire        rst,

    // 커서 위치 및 활성화
    input  wire [9:0]  cursor_x,    // 0~639
    input  wire [9:0]  cursor_y,    // 0~479
    input  wire        cursor_en,

    // VGA 출력
    output reg         hs,          // HSYNC (active-low)
    output reg         vs,          // VSYNC (active-low)
    output reg  [3:0]  red,
    output reg  [3:0]  green,
    output reg  [3:0]  blue

// ─── 픽셀 클럭: 100MHz / 4 = 25MHz ──────────────────────
reg [1:0] clk_div;
wire      pix_clk;

always @(posedge clk or posedge rst) begin
    if (rst) clk_div <= 2'd0;
    else     clk_div <= clk_div + 2'd1;
end
assign pix_clk = (clk_div == 2'd3);

// ─── VGA 타이밍 파라미터 (640×480 @ 60Hz) ────────────────
// 수평: 800 픽셀 클럭 / 라인
//   활성: 0~639 (640)
//   프론트 포치: 640~655 (16)
//   HSYNC: 656~751 (96) [active-low]
//   백 포치: 752~799 (48)
// 수직: 525 라인
//   활성: 0~479 (480)
//   프론트 포치: 480~489 (10)
//   VSYNC: 490~491 (2) [active-low]
//   백 포치: 492~524 (33)

localparam H_TOTAL = 10'd800;
localparam H_SYNC_START = 10'd656;
localparam H_SYNC_END   = 10'd752;

localparam V_TOTAL = 10'd525;
localparam V_SYNC_START = 10'd490;
localparam V_SYNC_END   = 10'd492;

reg [9:0] h_cnt;
reg [9:0] v_cnt;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        h_cnt <= 10'd0;
        v_cnt <= 10'd0;
    end else if (pix_clk) begin
        if (h_cnt == H_TOTAL - 1) begin
            h_cnt <= 10'd0;
            if (v_cnt == V_TOTAL - 1)
                v_cnt <= 10'd0;
            else
                v_cnt <= v_cnt + 10'd1;
        end else begin
            h_cnt <= h_cnt + 10'd1;
        end
    end
end

// ─── 활성 화면 영역 ──────────────────────────────────────
wire h_active = (h_cnt < 10'd640);
wire v_active = (v_cnt < 10'd480);
wire display  = h_active && v_active;

// ─── 커서 범위 판별 (5×5 박스, 중심 = cursor_x/y) ───────
// 가장자리 클램프: 중심이 2 이하이면 왼쪽 상한을 0으로
wire [9:0] cx_lo = (cursor_x >= 10'd2) ? (cursor_x - 10'd2) : 10'd0;
wire [9:0] cx_hi = (cursor_x <= 10'd637) ? (cursor_x + 10'd2) : 10'd639;
wire [9:0] cy_lo = (cursor_y >= 10'd2) ? (cursor_y - 10'd2) : 10'd0;
wire [9:0] cy_hi = (cursor_y <= 10'd477) ? (cursor_y + 10'd2) : 10'd479;

wire in_cursor_x = (h_cnt >= cx_lo) && (h_cnt <= cx_hi);
wire in_cursor_y = (v_cnt >= cy_lo) && (v_cnt <= cy_hi);
wire in_cursor   = in_cursor_x && in_cursor_y && cursor_en && display;

// ─── 픽셀 출력 (레지스터: pix_clk 사이클에 샘플) ─────────
always @(posedge clk or posedge rst) begin
    if (rst) begin
        hs    <= 1'b1;
        vs    <= 1'b1;
        red   <= 4'h0;
        green <= 4'h0;
        blue  <= 4'h0;
    end else if (pix_clk) begin
        // SYNC (active-low)
        hs <= ~((h_cnt >= H_SYNC_START) && (h_cnt < H_SYNC_END));
        vs <= ~((v_cnt >= V_SYNC_START) && (v_cnt < V_SYNC_END));

        // 픽셀 색상
        if (!display) begin
            red   <= 4'h0;
            green <= 4'h0;
            blue  <= 4'h0;
        end else if (in_cursor) begin
            // 커서: 흰색
            red   <= 4'hF;
            green <= 4'hF;
            blue  <= 4'hF;
        end else begin
            // 배경: 어두운 파랑
            red   <= 4'h0;
            green <= 4'h0;
            blue  <= 4'h2;
        end
    end
end

// tb_ps2_mouse_cursor.v
// ps2_mouse_host 자체 점검 테스트벤치
// PASS / FAIL 접두어 줄 출력 (!==/X/Z 감지)
// Verilog-2001. iverilog 호환.
// 원작자: 하네스 자체 작성 (2026-11-04)

`timescale 1ns / 1ps
`default_nettype none

module tb_ps2_mouse_cursor;

// ─── 파라미터 ────────────────────────────────────────────
parameter INHIBIT = 200;     // 인히빗 2μs: ps2_send(0x00) 복귀(~700ns) 보다 충분히 김 (200×10ns=2000ns)
parameter CLK_HALF = 5;      // 100 MHz 클럭 반주기 (5ns)
parameter PS2_HALF = 100;    // PS/2 CLK 반주기 (100ns = 10MHz, 시뮬용)

// ─── DUT 신호 ─────────────────────────────────────────────
reg  clk, rst;
reg  [15:0] offset;
reg  [31:0] wdata;
reg  sel, we;
wire [31:0] rdata;
wire [9:0]  cursor_x, cursor_y;
wire [2:0]  mouse_btn;
wire        irq_out;
wire        ps2_clk_oe, ps2_data_oe;

// ─── 오픈드레인 시뮬레이션 ───────────────────────────────
// TB와 DUT가 각각 선을 로우로 당길 수 있음
// tb_ps2_clk/data: TB 쪽 오픈드레인 드라이버 (0 → 로우, 1 → Hi-Z)
reg tb_ps2_clk_drive;   // 1=TB가 CLK을 로우로
reg tb_ps2_data_drive;  // 1=TB가 DATA를 로우로

// 선 전압: 어느 한쪽이 로우(oe=1)이면 로우
wire ps2_clk_line  = !(tb_ps2_clk_drive  || ps2_clk_oe);
wire ps2_data_line = !(tb_ps2_data_drive || ps2_data_oe);

// ─── DUT 인스턴스 ─────────────────────────────────────────
ps2_mouse_host #(.INHIBIT_CNT(INHIBIT)) dut (
    .clk         (clk),
    .rst         (rst),
    .sel         (sel),
    .we          (we),
    .offset      (offset),
    .wdata       (wdata),
    .rdata       (rdata),
    .cursor_x    (cursor_x),
    .cursor_y    (cursor_y),
    .mouse_btn   (mouse_btn),
    .irq_out     (irq_out),
    .ps2_clk_in  (ps2_clk_line),
    .ps2_data_in (ps2_data_line),
    .ps2_clk_oe  (ps2_clk_oe),
    .ps2_data_oe (ps2_data_oe)
);

// ─── 클럭 ────────────────────────────────────────────────
initial clk = 0;
always #CLK_HALF clk = ~clk;

// ─── MMIO 쓰기 태스크 ────────────────────────────────────
task mmio_write;
    input [15:0] addr;
    input [31:0] data;
    begin
        @(posedge clk); #1;
        sel    = 1'b1;
        we     = 1'b1;
        offset = addr;
        wdata  = data;
        @(posedge clk); #1;
        sel = 1'b0; we = 1'b0;
    end
endtask

// ─── MMIO 읽기 함수 대용 태스크 ──────────────────────────
task mmio_read;
    input  [15:0] addr;
    output [31:0] data;
    begin
        @(posedge clk); #1;
        sel    = 1'b1;
        we     = 1'b0;
        offset = addr;
        @(posedge clk); #1;
        data = rdata;
        sel = 1'b0;
    end
endtask

// ─── PS/2 바이트 송신 태스크 (TB → DUT) ──────────────────
// PS/2 프로토콜: start=0, D0..D7, parity(홀수), stop=1
// TB가 CLK을 생성 (장치 역할)
task ps2_send;
    input [7:0] data;
    reg parity;
    integer i;
    reg [10:0] frame;
    begin
        parity = ^data; // 짝수 XOR → 홀수 패리티 비트 = ~^data
        // parity bit = 1이면 data의 1의 개수가 짝수 → 1을 더해 홀수
        parity = ~(^data);
        frame = {1'b1, parity, data, 1'b0}; // [10]=stop [9]=par [8:1]=D7..D0 [0]=start

        // 스타트 비트 (DATA=0)
        tb_ps2_data_drive = 1'b1; // DATA 로우
        #PS2_HALF;
        tb_ps2_clk_drive = 1'b1;  // CLK 로우
        #PS2_HALF;
        tb_ps2_clk_drive = 1'b0;  // CLK 하이 (상승 에지 — DUT가 여기서 읽음)
        #PS2_HALF;

        // D0 ~ D7, parity, stop (10 비트)
        for (i = 1; i <= 10; i = i + 1) begin
            tb_ps2_data_drive = ~frame[i]; // frame[i]=1 → DATA=Hi-Z=1; frame[i]=0 → DATA=로우
            #PS2_HALF;
            tb_ps2_clk_drive = 1'b1;  // CLK 로우 (하강 에지 — DUT가 여기서 샘플)
            #PS2_HALF;
            tb_ps2_clk_drive = 1'b0;  // CLK 하이
            #PS2_HALF;
        end

        // 종료 간격
        tb_ps2_data_drive = 1'b0; // DATA 릴리즈
        #(PS2_HALF * 4);
    end
endtask

// ─── DUT가 보낸 바이트 수신 태스크 (DUT → TB) ────────────
// DUT가 CLK 인히빗 → DATA 로우 (스타트) → CLK 릴리즈 → TB가 CLK 생성
task ps2_recv_host_tx;
    output [7:0] data;
    integer i;
    reg bit_val;
    reg [7:0] recv_byte;
    integer timeout;
    begin
        recv_byte = 8'h00;

        // DUT가 CLK 인히빗(ps2_clk_oe=1) 대기
        timeout = 0;
        while (!ps2_clk_oe && timeout < 100000) begin
            @(posedge clk); timeout = timeout + 1;
        end
        if (timeout >= 100000) begin
            $display("FAIL tb_recv: timeout waiting for CLK inhibit");
            $finish;
        end

        // 인히빗 완료 대기 (DUT가 DATA 로우 → CLK 릴리즈)
        @(negedge ps2_clk_oe); // CLK 개방 시까지 대기
        #(PS2_HALF * 2);

        // 스타트 비트는 이미 세팅됨 (DATA=0). CLK 1사이클 생성
        tb_ps2_clk_drive = 1'b1; #PS2_HALF;
        tb_ps2_clk_drive = 1'b0; #PS2_HALF;

        // D0 ~ D7, parity, stop = 10 CLK 사이클
        for (i = 0; i < 10; i = i + 1) begin
            #(PS2_HALF/2);
            if (i < 8) recv_byte[i] = ps2_data_line; // 데이터 샘플
            tb_ps2_clk_drive = 1'b1; #PS2_HALF;
            tb_ps2_clk_drive = 1'b0; #PS2_HALF;
        end

        // 0xFA ACK 전송
        // 주의: 하드웨어 ACK 펄스(DATA 로우 1 CLK)는 생략한다.
        // ACK 펄스를 보내면 DUT 수신기의 bit_cnt가 0→1로 증가한 채로
        // ps2_send(0xFA)가 시작되어 11비트 프레임이 오정렬된다.
        // DUT S_WAIT_ACK는 rx_byte==0xFA 전체 프레임만 확인하므로 문제없다.
        #(PS2_HALF * 4);
        ps2_send(8'hFA);

        data = recv_byte;
    end
endtask

// ─── 메인 테스트 시퀀스 ──────────────────────────────────
integer fail_cnt;
reg [31:0] rd;
reg [7:0]  rx_byte_host;

initial begin
    // 초기화
    fail_cnt = 0;
    clk = 0; rst = 1;
    sel = 0; we = 0; offset = 0; wdata = 0;
    tb_ps2_clk_drive = 0; tb_ps2_data_drive = 0;
    repeat(5) @(posedge clk);
    rst = 0;
    repeat(3) @(posedge clk);

    // ── T1: IP ID 확인 ───────────────────────────────────
    mmio_read(16'hFFF0, rd);
    if (rd !== 32'h5053_324D) begin
        $display("FAIL T1: ID expected 0x5053324D got 0x%08X", rd); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T1: IP ID = 0x%08X", rd);
    end

    // ── T2: BAT 시퀀스 ───────────────────────────────────
    ps2_send(8'hAA);  // BAT 완료
    ps2_send(8'h00);  // 장치 ID
    // DUT가 TX 준비 → TB가 수신
    ps2_recv_host_tx(rx_byte_host);
    if (rx_byte_host !== 8'hF4) begin
        $display("FAIL T2: host TX expected 0xF4 got 0x%02X", rx_byte_host);
        fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T2: host TX = 0xF4");
    end

    // ACK(0xFA)는 ps2_recv_host_tx 내부에서 전송됨

    // init_done 대기
    begin : wait_init
        integer t;
        t = 0;
        repeat(200) @(posedge clk);
        mmio_read(16'h0010, rd);
        if (rd[0] !== 1'b1) begin
            $display("FAIL T2b: STATUS init_done=0"); fail_cnt = fail_cnt + 1;
        end else begin
            $display("PASS T2b: init_done=1");
        end
    end

    // ── T3: CTRL 쓰기 (enable + cursor_en) ───────────────
    mmio_write(16'h0000, 32'h0000_0003);
    mmio_read(16'h0000, rd);
    if (rd[1:0] !== 2'b11) begin
        $display("FAIL T3: CTRL expected 3 got %0d", rd[1:0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T3: CTRL = 0x%08X", rd);
    end

    // ── T4: 이동 패킷 (dx=+20, dy=0, L_BTN) ─────────────
    // pkt0=0x09: bit3=always1, bit0=L_BTN, X_OVF=0, Y_OVF=0
    // pkt1=0x14: dx=20
    // pkt2=0x00: dy=0
    ps2_send(8'h09);
    ps2_send(8'h14);
    ps2_send(8'h00);

    // cursor_x 갱신 대기 (pkt_ready_d 2클럭 후)
    repeat(50) @(posedge clk);

    mmio_read(16'h0004, rd);
    if (rd[9:0] !== 10'd340) begin
        $display("FAIL T4: MOUSE_X expected 340 got %0d", rd[9:0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T4: MOUSE_X=%0d (320+20=340)", rd[9:0]);
    end

    mmio_read(16'h0008, rd);
    if (rd[9:0] !== 10'd240) begin
        $display("FAIL T4b: MOUSE_Y expected 240 got %0d", rd[9:0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T4b: MOUSE_Y=%0d (unchanged)", rd[9:0]);
    end

    mmio_read(16'h000C, rd);
    if (rd[0] !== 1'b1) begin
        $display("FAIL T4c: L_BTN expected 1 got %0d", rd[0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T4c: L_BTN=1");
    end

    // ── T5: 위로 이동 패킷 (dy=+10 → cursor_y 감소) ──────
    // pkt0=0x08: bit3=always1, Y_SGN=0, X_SGN=0, 버튼 없음
    // pkt1=0x00: dx=0
    // pkt2=0x0A: dy=+10 (양수 = 위로 = cursor_y 감소)
    ps2_send(8'h08);
    ps2_send(8'h00);
    ps2_send(8'h0A);

    repeat(50) @(posedge clk);

    mmio_read(16'h0008, rd);
    if (rd[9:0] !== 10'd230) begin
        $display("FAIL T5: MOUSE_Y expected 230 got %0d", rd[9:0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T5: MOUSE_Y=%0d (240-10=230)", rd[9:0]);
    end

    // ── T6: IRQ 확인 ─────────────────────────────────────
    mmio_read(16'hFFF8, rd);
    if (rd[0] !== 1'b1) begin
        $display("FAIL T6: IRQ_STATUS expected 1 got %0d", rd[0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T6: IRQ_STATUS[0]=1 (packet IRQ)");
    end

    // W1C 클리어
    mmio_write(16'hFFF8, 32'h0000_0001);
    mmio_read(16'hFFF8, rd);
    if (rd[0] !== 1'b0) begin
        $display("FAIL T6b: IRQ_STATUS W1C failed, got %0d", rd[0]); fail_cnt = fail_cnt + 1;
    end else begin
        $display("PASS T6b: IRQ_STATUS W1C cleared");
    end

    // ── 최종 결과 ────────────────────────────────────────
    if (fail_cnt === 0)
        $display("PASS all tests (%0d failures)", fail_cnt);
    else
        $display("FAIL %0d test(s) failed", fail_cnt);

    #(PS2_HALF * 10);
    $finish;
end

// 타임아웃 보호
initial begin
    #20000000; // 20ms
    $display("FAIL timeout: simulation exceeded 20ms");
    $finish;
end

# Makefile — mmio-ps2-mouse-cursor
# lint, sim, synth 타깃
# 의존: iverilog, yosys (verilator는 선택)

SLUG    = mmio-ps2-mouse-cursor
TOP     = basys3_ps2_mouse_cursor_top
TB      = tb_ps2_mouse_cursor

RTL_SRC = rtl/ps2_mouse_host.v \
          rtl/vga_sync_cursor.v \
          rtl/basys3_ps2_mouse_cursor_top.v

TB_SRC  = tb/tb_ps2_mouse_cursor.v \
          rtl/ps2_mouse_host.v \
          rtl/vga_sync_cursor.v

BUILD   = build
SIM_OUT = $(BUILD)/$(TB).vvp
LINT_OK = $(BUILD)/lint.ok
SIM_LOG = $(BUILD)/sim.log
SYNTH_J = $(BUILD)/synth.json

.PHONY: all lint sim synth clean

all: lint sim synth

## ─── 린트 (iverilog -t null) ───────────────────────────────
lint: $(LINT_OK)

$(LINT_OK): $(RTL_SRC) $(TB_SRC)
	@mkdir -p $(BUILD)
	iverilog -g2001 -Wall -t null $(RTL_SRC) 2>&1 | tee $(BUILD)/lint_rtl.log
	iverilog -g2001 -Wall -t null $(TB_SRC) 2>&1 | tee $(BUILD)/lint_tb.log
	@echo "LINT PASS" > $@

## ─── 시뮬레이션 (iverilog + vvp) ──────────────────────────
sim: $(SIM_LOG)

$(SIM_LOG): $(TB_SRC) $(LINT_OK)
	@mkdir -p $(BUILD)
	iverilog -g2001 -o $(SIM_OUT) $(TB_SRC) -s $(TB)
	vvp $(SIM_OUT) | tee $(SIM_LOG)
	@if grep -q "^FAIL" $(SIM_LOG); then \
	    echo "=== SIMULATION FAILED ==="; exit 1; \
	else \
	    echo "=== SIMULATION PASSED ==="; \
	fi

## ─── 합성 (yosys 추정) ─────────────────────────────────────
synth: $(SYNTH_J)

$(SYNTH_J): $(RTL_SRC)
	@mkdir -p $(BUILD)
	yosys -p "read_verilog -sv $(RTL_SRC); \
	           synth_xilinx -top $(TOP) -flatten; \
	           stat -json -top $(TOP)" \
	      2>&1 | tee $(BUILD)/synth.log
	@if yosys -p "read_verilog -sv $(RTL_SRC); \
	               synth_xilinx -top $(TOP) -flatten; \
	               write_json $(SYNTH_J)" \
	          2>&1 | grep -q "ERROR"; then \
	    echo "SYNTH FAILED"; exit 1; \
	else \
	    echo "SYNTH PASS"; \
	fi

## ─── 정리 ────────────────────────────────────────────────────
clean:
	rm -rf $(BUILD)

위험·한계

  • Basys3 USB-HID 에뮬레이션이 마우스와 키보드 동시 연결을 지원하지 않음(허브 미지원) — kbd IP([0B])와 동시 사용 불가
  • PS/2 마우스가 없거나 HID 에뮬레이터가 마우스 프로토콜을 지원하지 않으면 BAT 수신 실패(init_done 미점등)
  • PS/2 타이밍: 100 MHz 기준 CLK 억제 100 μs(INHIBIT_CNT=10000). 일부 마우스는 더 긴 억제 시간 필요
  • VGA 픽셀 클럭: 100 MHz /4 = 25 MHz (표준 25.175 MHz보다 0.7% 느림). 대부분의 모니터는 허용
  • 커서 오버레이는 프레임 버퍼 없이 온더플라이 계산 — 커서 외 배경 색은 고정(검정·진한 파랑)
  • 동결 맵 이식(T3)에서 AXI4-Lite 래퍼 추가 필요. IRQ 선번(intc=13)은 프로젝트에서 동결 전까지 제안값
