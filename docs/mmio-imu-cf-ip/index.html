IMU 상보 필터 IP (CORDIC atan2 + AXI4-Lite)

MPU-6050 원시 가속도·각속도 값을 AXI4-Lite 레지스터로 받아, CORDIC 벡터링 모드 atan2와 고정소수점 상보 필터로 롤·피치 각도를 계산하는 T4 IP 패키지. 테스트벤치는 알려진 각도의 합성 센서 값을 주입해 수렴을 자체 점검한다.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • CORDIC 벡터링 모드 atan2를 Verilog-2001 고정소수점으로 구현 (16반복, Q7.9 도 출력)
  • 상보 필터 FSM: roll = alpha*(roll_prev + GX_DEL) + (1-alpha)*atan2(ay, az)
  • AXI4-Lite 슬레이브 래퍼로 MicroBlaze V SoC에 플러그인 가능한 IP 작성
  • 테스트벤치가 TC1~TC5 (0°, 90°, 45°, α=0 순간 응답, 자이로 드리프트 보정)를 PASS/FAIL로 검사

범위 밖

  • 실제 I2C 읽기 (기존 IIC IP[05]에서 소프트웨어가 처리)
  • 칼만 필터, 사원수(Quaternion) 출력
  • 보드 실물 검증 (S3는 사용자 담당)
  • 부동소수점 연산

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-019 기반. free_slots [19]=0x4019_0000 제안. 시뮬레이션·합성 결과는 CI에서 검증한다.

4. 메모리맵

axil_imu_cf 0x4019_0000 크기 0x10000

0x00CTRLWObit0: start (자동 클리어). 쓰면 필터 1스텝 계산 시작.
0x04STATUSRObit0: done, bit1: busy
0x08AYWO가속도 Y, 16비트 부호, Q2.14 (16384=1g)
0x0CAZWO가속도 Z, 16비트 부호, Q2.14
0x10AXWO가속도 X, 16비트 부호, Q2.14 (피치용)
0x14GX_DELWO자이로 각도 증분 (degrees*512, 18비트 부호, [17:0]). 소프트: gx_dps/fs*512

0x1CGY_DELWO자이로 피치 증분 (degrees*512, 18비트 부호). gy_dps/fs*512
0x20ROLLRO롤 각도 출력 (degrees*512, 18비트 부호, [17:0])
0x24PITCHRO피치 각도 출력 (degrees*512, 18비트 부호, [17:0])

5. 외부 부품과 전기 회로도

2절의 계통도는 디지털 모듈 연결도다. 전압, 풀업, 커넥터 배선 같은 전기 사항은 이 절에서 따로 다룬다. 기계가 읽는 사본: bom.csv · connections.json

전기 회로도: 미작성

디지털 계통도만 작성. MPU-6050 ↔ Basys3 Pmod 배선은 미작성.

rtl/cordic_atan2.v 130줄
// cordic_atan2.v — CORDIC 벡터링 모드 atan2 (16반복, 순차 처리)
// 입력 : xi, yi — 16비트 부호 (Q2.14: 16384 = 1.0 = 1g)
// 출력 : angle  — 18비트 부호, degrees*512 (Q7.9), 범위 ±180°*512=±92160
// 쿼드런트 보정: xi<=0 → 초기 90° 회전으로 1·4사분면으로 가져온다
// 반복당 1 클럭, 완료까지 18 클럭 (16반복 + 로드 + 완료 출력)
// 2026-11-04 하네스 자체 작성 (원본, 외부 코드 복사 아님)

`default_nettype none
module cordic_atan2 (
    input  wire        clk,
    input  wire        rst,
    input  wire        start,
    input  wire signed [15:0] xi,
    input  wire signed [15:0] yi,
    output reg  signed [17:0] angle,
    output reg         done
);
// ─── CORDIC atan 테이블 (degrees * 512 = Q7.9, i번째 = atan(2^-i)도) ─────
function signed [17:0] atan_lut;
    input [3:0] i;
    begin
        case (i)
            4'd0:  atan_lut = 18'd23040; // atan(1.0)   = 45.0000°
            4'd1:  atan_lut = 18'd13601; // atan(0.5)   = 26.5651°
            4'd2:  atan_lut = 18'd7187;  // atan(0.25)  = 14.0362°
            4'd3:  atan_lut = 18'd3648;  // atan(0.125) =  7.1250°
            4'd4:  atan_lut = 18'd1831;  //              =  3.5763°
            4'd5:  atan_lut = 18'd916;   //              =  1.7899°
            4'd6:  atan_lut = 18'd458;   //              =  0.8952°
            4'd7:  atan_lut = 18'd229;   //              =  0.4476°
            4'd8:  atan_lut = 18'd115;   //              =  0.2238°
            4'd9:  atan_lut = 18'd57;    //              =  0.1119°
            4'd10: atan_lut = 18'd29;    //              =  0.0560°
            4'd11: atan_lut = 18'd14;    //              =  0.0280°
            4'd12: atan_lut = 18'd7;     //              =  0.0140°
            4'd13: atan_lut = 18'd4;     //              =  0.0070°
            4'd14: atan_lut = 18'd2;     //              =  0.0035°
            4'd15: atan_lut = 18'd1;     //              =  0.0018°
            default: atan_lut = 18'd0;
        endcase
    end
endfunction

// ─── 상태 ───────────────────────────────────────────────────────────────────
localparam ST_IDLE  = 2'd0;
localparam ST_INIT  = 2'd1; // 쿼드런트 보정 + 로드
localparam ST_ITER  = 2'd2; // 반복 (iter 0..15)
localparam ST_DONE  = 2'd3;

reg [1:0]  state;
reg [3:0]  iter;
reg signed [31:0] xr, yr, zr; // 32비트 내부 (충분한 정밀도)

// ─── 순차 로직 ──────────────────────────────────────────────────────────────
always @(posedge clk or posedge rst) begin
    if (rst) begin
        state <= ST_IDLE;
        done  <= 1'b0;
        angle <= 18'sd0;
        iter  <= 4'd0;
        xr <= 32'sd0; yr <= 32'sd0; zr <= 32'sd0;
    end else begin
        done <= 1'b0; // 기본 클리어
        case (state)
        // ── 대기 ──────────────────────────────────────────────────────────
        ST_IDLE: begin
            if (start) state <= ST_INIT;
        end
        // ── 초기화: 쿼드런트 보정 ─────────────────────────────────────────
        // xi <= 0 이면 (xi, yi)를 90° 회전: (xi,yi)→(yi,−xi), z_init=+90°*512
        //                 또는 −90° 회전: (xi,yi)→(−yi,xi), z_init=−90°*512
        ST_INIT: begin
            if (xi <= 16'sh0000) begin
                // 2·3사분면: x를 양수로 만드는 90°/-90° 회전
                if (yi >= 16'sh0000) begin
                    // 2사분면(x<0, y≥0): +90° → (x,y)→(y,−x)
                    xr <= {{16{yi[15]}}, yi};        // sign-extend
                    yr <= -{{16{xi[15]}}, xi};
                    zr <= 18'sd46080;                // +90°*512
                end else begin
                    // 3사분면(x<0, y<0): −90° → (x,y)→(−y,x)
                    xr <= -{{16{yi[15]}}, yi};
                    yr <= {{16{xi[15]}}, xi};
                    zr <= -18'sd46080;               // −90°*512
                end
            end else begin
                // 1·4사분면: 그대로
                xr <= {{16{xi[15]}}, xi};
                yr <= {{16{yi[15]}}, yi};
                zr <= 18'sd0;
            end
            iter  <= 4'd0;
            state <= ST_ITER;
        end
        // ── CORDIC 반복 ───────────────────────────────────────────────────
        // 벡터링 모드: yr→0 방향으로 회전, zr에 각도 누산
        // σ = (yr>=0) ? -1 : +1
        // x' = xr - σ*(yr >> iter)
        // y' = yr + σ*(xr >> iter)
        // z' = zr - σ*atan_lut[iter]
        ST_ITER: begin
            if (yr >= 32'sd0) begin
                // σ = -1: x' = xr + (yr>>i), y' = yr - (xr>>i), z' = zr + lut
                xr <= xr + (yr >>> iter);
                yr <= yr - (xr >>> iter);
                zr <= zr + {{14{1'b0}}, atan_lut(iter)};
            end else begin
                // σ = +1: x' = xr - (yr>>i), y' = yr + (xr>>i), z' = zr - lut
                xr <= xr - (yr >>> iter);
                yr <= yr + (xr >>> iter);
                zr <= zr - {{14{1'b0}}, atan_lut(iter)};
            end
            if (iter == 4'd15) begin
                state <= ST_DONE;
            end else begin
                iter <= iter + 4'd1;
            end
        end
        // ── 완료 ──────────────────────────────────────────────────────────
        ST_DONE: begin
            angle <= zr[17:0]; // 하위 18비트 취함 (±92160 범위 내)
            done  <= 1'b1;
            state <= ST_IDLE;
        end
        endcase
    end
end

endmodule
`default_nettype wire

// imu_cf_ip.v — IMU 상보 필터 FSM + CORDIC 시퀀서
// 입력 : ay, az (롤용 accel Q2.14), ax (피치용), gx_del, gy_del (자이로 증분 Q7.9)
//        alpha (상보 필터 계수 *65536, 기본 0.98*65536=64225)
// 출력 : roll, pitch (Q7.9 degrees*512, 18비트 부호)
// 상보 필터:
//   roll  = (alpha*(roll_prev  + gx_del) + (65536-alpha)*atan2(ay,az )) >> 16
//   pitch = (alpha*(pitch_prev + gy_del) + (65536-alpha)*atan2(ax,-az)) >> 16
// 2026-11-04 하네스 자체 작성 (원본, 외부 코드 복사 아님)

`default_nettype none
module imu_cf_ip (
    input  wire        clk,
    input  wire        rst,
    // 제어
    input  wire        start,
    output reg         done,
    output reg         busy,
    // 센서 입력
    input  wire signed [15:0] ay,
    input  wire signed [15:0] az,
    input  wire signed [15:0] ax,
    input  wire signed [17:0] gx_del,  // 롤 자이로 증분 (Q7.9)
    input  wire signed [17:0] gy_del,  // 피치 자이로 증분 (Q7.9)
    input  wire        [15:0] alpha,   // *65536 계수 (0x0000=0.0, 0xFFFF≈1.0)
    // 각도 출력
    output reg  signed [17:0] roll,
    output reg  signed [17:0] pitch
);

// ─── CORDIC 인스턴스 신호 ────────────────────────────────────────────────────
reg         cord_start;
reg  signed [15:0] cord_xi, cord_yi;
wire signed [17:0] cord_angle;
wire               cord_done;

cordic_atan2 u_cordic (
    .clk   (clk),
    .rst   (rst),
    .start (cord_start),
    .xi    (cord_xi),
    .yi    (cord_yi),
    .angle (cord_angle),
    .done  (cord_done)
);

// ─── FSM 상태 ────────────────────────────────────────────────────────────────
localparam ST_IDLE         = 3'd0;
localparam ST_CORD_ROLL    = 3'd1;  // atan2(ay, az) 시작
localparam ST_WAIT_ROLL    = 3'd2;  // CORDIC 완료 대기
localparam ST_CORD_PITCH   = 3'd3;  // atan2(ax, -az) 시작
localparam ST_WAIT_PITCH   = 3'd4;  // CORDIC 완료 대기
localparam ST_BLEND        = 3'd5;  // 상보 필터 혼합
localparam ST_DONE         = 3'd6;
localparam ST_OUTPUT       = 3'd7;  // 혼합 결과 >> 16 → 출력

reg [2:0] state;

// 중간 계산용
reg signed [17:0] accel_roll;   // atan2(ay, az) 결과
reg signed [17:0] accel_pitch;  // atan2(ax, -az) 결과
reg signed [17:0] gyro_roll;    // roll_prev + gx_del
reg signed [17:0] gyro_pitch;   // pitch_prev + gy_del

// 상보 필터 계산 (36비트 중간 — 오버플로 방지)
// roll = (alpha*gyro_roll + (65536-alpha)*accel_roll) >> 16
// 주의: alpha는 16비트 비부호, angle/gyro는 18비트 부호
// 곱 최대: 65536 * 92160 = 6,039,797,760 → 33비트 필요. 36비트로 충분
reg signed [35:0] blend_a;
reg signed [35:0] blend_b;
reg signed [17:0] one_minus_alpha_18;  // 65536 - alpha (16비트 비부호 → 항상 ≤65536)

// ─── 순차 로직 ───────────────────────────────────────────────────────────────
always @(posedge clk or posedge rst) begin
    if (rst) begin
        state       <= ST_IDLE;
        done        <= 1'b0;
        busy        <= 1'b0;
        cord_start  <= 1'b0;
        roll        <= 18'sd0;
        pitch       <= 18'sd0;
        accel_roll  <= 18'sd0;
        accel_pitch <= 18'sd0;
        gyro_roll   <= 18'sd0;
        gyro_pitch  <= 18'sd0;
        blend_a     <= 36'sd0;
        blend_b     <= 36'sd0;
        one_minus_alpha_18 <= 18'sd0;
    end else begin
        done       <= 1'b0;
        cord_start <= 1'b0;

        case (state)
        // ── 대기 ──────────────────────────────────────────────────────────
        ST_IDLE: begin
            if (start) begin
                busy  <= 1'b1;
                state <= ST_CORD_ROLL;
                // 자이로 적분항 미리 계산
                gyro_roll  <= roll  + gx_del;
                gyro_pitch <= pitch + gy_del;
            end
        end

        // ── 롤 CORDIC 시작: atan2(ay, az) ────────────────────────────────
        ST_CORD_ROLL: begin
            cord_xi    <= az;
            cord_yi    <= ay;
            cord_start <= 1'b1;
            state      <= ST_WAIT_ROLL;
        end

        // ── 롤 CORDIC 완료 대기 ───────────────────────────────────────────
        ST_WAIT_ROLL: begin
            if (cord_done) begin
                accel_roll <= cord_angle;
                state      <= ST_CORD_PITCH;
            end
        end

        // ── 피치 CORDIC 시작: atan2(ax, -az) ────────────────────────────
        ST_CORD_PITCH: begin
            cord_xi    <= -az;
            cord_yi    <= ax;
            cord_start <= 1'b1;
            state      <= ST_WAIT_PITCH;
        end

        // ── 피치 CORDIC 완료 대기 ─────────────────────────────────────────
        ST_WAIT_PITCH: begin
            if (cord_done) begin
                accel_pitch <= cord_angle;
                state       <= ST_BLEND;
                // 상보 필터 계수: (65536-alpha), 최대 65536 → 18비트 부호로 충분
                one_minus_alpha_18 <= 18'sd65536 - {{2{1'b0}}, alpha};
            end
        end

        // ── 상보 필터 혼합: 36비트 곱셈으로 오버플로 방지 ────────────────
        // alpha(16비트 비부호)→ 36비트 영확장, 각도(18비트 부호)→ 36비트 부호확장
        // 곱셈 결과 최대: 65536×92160 = 6.04×10⁹ < 2³³ → 36비트에 적합
        ST_BLEND: begin
            blend_a <= $signed({{20{1'b0}}, alpha})
                           * $signed({{18{gyro_roll[17]}}, gyro_roll})
                     + $signed({{18{one_minus_alpha_18[17]}}, one_minus_alpha_18})
                           * $signed({{18{accel_roll[17]}}, accel_roll});
            blend_b <= $signed({{20{1'b0}}, alpha})
                           * $signed({{18{gyro_pitch[17]}}, gyro_pitch})
                     + $signed({{18{one_minus_alpha_18[17]}}, one_minus_alpha_18})
                           * $signed({{18{accel_pitch[17]}}, accel_pitch});
            state <= ST_OUTPUT;
        end

            pitch <= blend_b[33:16];
            state <= ST_DONE;

        ST_DONE: begin
            done  <= 1'b1;
            busy  <= 1'b0;
            state <= ST_IDLE;
        end
        endcase
    end

        end

        end
    end

// uart_tx.v — 디버그 UART 송신기 (8N1, 115200 baud @ 100MHz)
// 2026-11-04 하네스 자체 작성 (원본, 외부 코드 복사 아님)

`default_nettype none
module uart_tx #(
    parameter CLK_HZ   = 100_000_000,
    parameter BAUD     = 115200
)(
    input  wire       clk,
    input  wire       rst,
    input  wire [7:0] data,
    input  wire       valid,
    output reg        ready,
    output reg        tx
);

localparam CLKS_PER_BIT = CLK_HZ / BAUD; // 868

localparam ST_IDLE  = 2'd0;
localparam ST_START = 2'd1;
localparam ST_DATA  = 2'd2;
localparam ST_STOP  = 2'd3;

reg [1:0]  state;
reg [9:0]  clk_cnt;
reg [2:0]  bit_idx;
reg [7:0]  shift;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        state   <= ST_IDLE;
        tx      <= 1'b1;
        ready   <= 1'b1;
        clk_cnt <= 10'd0;
        bit_idx <= 3'd0;
        shift   <= 8'd0;
    end else begin
        case (state)
        ST_IDLE: begin
            tx    <= 1'b1;
            ready <= 1'b1;
            if (valid) begin
                ready   <= 1'b0;
                shift   <= data;
                state   <= ST_START;
                clk_cnt <= 10'd0;
            end
        end
        ST_START: begin
            tx <= 1'b0;
            if (clk_cnt == CLKS_PER_BIT - 1) begin
                clk_cnt <= 10'd0;
                bit_idx <= 3'd0;
                state   <= ST_DATA;
            end else clk_cnt <= clk_cnt + 10'd1;
        end
        ST_DATA: begin
            tx <= shift[0];
            if (clk_cnt == CLKS_PER_BIT - 1) begin
                clk_cnt <= 10'd0;
                shift   <= {1'b0, shift[7:1]};
                if (bit_idx == 3'd7) state <= ST_STOP;
                else bit_idx <= bit_idx + 3'd1;
            end else clk_cnt <= clk_cnt + 10'd1;
        end
        ST_STOP: begin
            tx <= 1'b1;
            if (clk_cnt == CLKS_PER_BIT - 1) begin
                clk_cnt <= 10'd0;
                state   <= ST_IDLE;
            end else clk_cnt <= clk_cnt + 10'd1;
        end
        endcase
    end
end

// PASS / FAIL 출력. !==로 X/Z도 실패로 처리.

// ─── 허용 오차 (degrees*512 단위) ───────────────────────────────────────────
// 1도 오차 허용 = 512 단위. CORDIC 16반복 오차 + 고정소수점 오차 감안.
localparam signed [17:0] TOL = 18'd1024; // ±2도

// ─── DUT 신호 ────────────────────────────────────────────────────────────────
reg         clk, rst;
reg         start;
wire        done, busy;
reg  signed [15:0] ay, az, ax;
reg  signed [17:0] gx_del, gy_del;
reg         [15:0] alpha;
wire signed [17:0] roll, pitch;

// ─── DUT 인스턴스 ────────────────────────────────────────────────────────────
imu_cf_ip u_dut (
    .clk    (clk),
    .rst    (rst),
    .start  (start),
    .done   (done),
    .busy   (busy),
    .ay     (ay),
    .az     (az),
    .ax     (ax),
    .gx_del (gx_del),
    .gy_del (gy_del),
    .alpha  (alpha),
    .roll   (roll),
    .pitch  (pitch)
);

// ─── 클럭 생성 (100 MHz) ──────────────────────────────────────────────────────
initial clk = 0;
always #5 clk = ~clk;

// ─── 헬퍼: 1스텝 실행 후 done 대기 ──────────────────────────────────────────
integer wait_cnt;
task run_step;
    begin
        @(posedge clk); #1;
        start = 1'b1;
        @(posedge clk); #1;
        start = 1'b0;
        wait_cnt = 0;
        while (done !== 1'b1 && wait_cnt < 200) begin
            @(posedge clk); #1;
            wait_cnt = wait_cnt + 1;
        end
        if (done !== 1'b1) begin
            $display("FAIL: done never asserted (timeout)");
            $finish;
        end
    end
endtask

// ─── 헬퍼: 범위 검사 ─────────────────────────────────────────────────────────
integer fail_count;
task check_range;
    input signed [17:0] actual;
    input signed [17:0] expected;
    input [255:0]       name;
    begin
        if (actual < expected - TOL || actual > expected + TOL) begin
            $display("FAIL %0s: got %0d expected %0d (tol ±%0d)",
                     name, actual, expected, TOL);
            fail_count = fail_count + 1;
        end else begin
            $display("  ok  %0s: %0d (expected %0d ±%0d)",
                     name, actual, expected, TOL);
        end
    end
endtask

// ─── 테스트 본체 ──────────────────────────────────────────────────────────────

    fail_count = 0;

    rst    = 1'b1;
    start  = 1'b0;
    ay     = 16'sh0000;
    az     = 16'sh4000; // 1g = 16384
    ax     = 16'sh0000;
    gx_del = 18'sd0;
    gy_del = 18'sd0;
    alpha  = 16'd0;     // alpha=0: 순수 가속도계 (즉시 수렴)

    repeat(5) @(posedge clk);
    rst = 1'b0;
    repeat(2) @(posedge clk);

    // ── TC1: ay=0, az=1g → roll≈0° ─────────────────────────────────────────
    $display("TC1: ay=0, az=1g → roll should be ~0°");
    ay     = 16'sh0000;
    az     = 16'sh4000;
    ax     = 16'sh0000;
    gx_del = 18'sd0;
    gy_del = 18'sd0;
    alpha  = 16'd0;
    run_step;
    check_range(roll, 18'sd0, "TC1 roll");

    // ── TC2: ay=1g, az=0 → roll≈+90° (45056 = 90°*512) ─────────────────────
    $display("TC2: ay=1g, az=0 → roll should be ~90°");
    ay     = 16'sh4000;
    az     = 16'sh0000;
    ax     = 16'sh0000;
    gx_del = 18'sd0;
    gy_del = 18'sd0;
    alpha  = 16'd0;
    run_step;
    check_range(roll, 18'sd46080, "TC2 roll"); // 90°*512=46080

    // ── TC3: ay=az=0.707g → roll≈45° (23040=45°*512) ────────────────────────
    $display("TC3: ay=az=0.707g → roll should be ~45°");
    // 0.707*16384 ≈ 11585
    ay     = 16'sh2D41; // 11585
    az     = 16'sh2D41;
    ax     = 16'sh0000;
    gx_del = 18'sd0;
    gy_del = 18'sd0;
    alpha  = 16'd0;
    run_step;
    check_range(roll, 18'sd23040, "TC3 roll");

    // ── TC4: alpha=0 → 순수 가속도계, 즉시 accel값으로 수렴 ──────────────────
    $display("TC4: alpha=0 (pure accel) → roll immediately from ay/az");
    ay     = 16'sh2D41; // 45°
    az     = 16'sh2D41;
    gx_del = 18'sd5120; // 큰 자이로 오프셋 (10°*512) — 무시돼야 함
    alpha  = 16'd0;      // alpha=0: 자이로 가중치 0
    run_step;
    check_range(roll, 18'sd23040, "TC4 roll(alpha=0, accel only)");

    // ── TC5: 자이로 드리프트 → 상보 필터 보정 (alpha=0.98) ───────────────────
    // TC4가 roll≈45°를 남긴다. alpha=0 초기화 스텝으로 roll을 0°로 맞춘다.
    // 그 후 alpha=0.98, gx_del=+1°를 적용: blend → ≈0.98° ≈ 501 단위
    $display("TC5: gyro drift correction (alpha=64225/65536≈0.98)");
    ay     = 16'sh0000;
    az     = 16'sh4000;
    gx_del = 18'sd0;
    alpha  = 16'd0;      // 순수 가속도계: roll → atan2(0, 1g) ≈ 0°
    run_step;            // 초기화 스텝 (roll ← 0)

    // 이제 alpha=0.98, gx_del=+1°로 상보 필터 테스트
    gx_del = 18'sd512;  // +1° 자이로 증분
    alpha  = 16'd64225; // 0.98*65536
    // 예상: alpha*(0°+1°) + (1-alpha)*0° = 0.98*512 ≈ 501 (≈0.98°)
    run_step;
    // 넓은 허용: 0.5°~1.5° 범위 (256~768)
    if (roll < 18'sd256 || roll > 18'sd768) begin
        $display("FAIL TC5 roll: %0d (expected ~501, range 256~768)", roll);
        fail_count = fail_count + 1;
    end else begin
        $display("  ok  TC5 roll: %0d (blend OK)", roll);
    end

    // ─── 최종 판정 ──────────────────────────────────────────────────────────
    if (fail_count == 0) begin
        $display("PASS: all %0d tests passed", 5);
    end else begin
        $display("FAIL: %0d test(s) failed", fail_count);
    end

    $finish;
end

// ─── 타임아웃 가드 ───────────────────────────────────────────────────────────
initial begin
    #200000; // 200μs
    $display("FAIL: global timeout");
    $finish;
end

# Makefile — IMU 상보 필터 IP (mmio-imu-cf-ip)
# 타깃: lint, sim, synth, clean
# 도구: verilator (lint), iverilog+vvp (sim), yosys (synth)

SLUG    = mmio-imu-cf-ip
RTL_DIR = rtl
TB_DIR  = tb
BUILD   = build

RTL_SRCS = $(RTL_DIR)/cordic_atan2.v \
           $(RTL_DIR)/imu_cf_ip.v \
           $(RTL_DIR)/axil_imu_cf.v \
           $(RTL_DIR)/basys3_imu_cf_top.v

TB_SRCS  = $(TB_DIR)/tb_imu_cf.v

.PHONY: all lint sim synth clean

all: lint sim synth

## ── 린트 (verilator) ────────────────────────────────────────────────────────
lint:
	@mkdir -p $(BUILD)
	verilator --lint-only -g2001 --Wall -Wno-UNUSED -Wno-PINMISSING \
	    $(RTL_SRCS) 2>&1 | tee $(BUILD)/lint.log
	@echo "lint done"

## ── 시뮬레이션 (iverilog + vvp) ─────────────────────────────────────────────
sim: $(BUILD)/tb_$(SLUG)
	vvp $(BUILD)/tb_$(SLUG) | tee $(BUILD)/sim.log
	@grep -q "^PASS" $(BUILD)/sim.log && echo "sim PASS" || (echo "sim FAIL" && exit 1)

$(BUILD)/tb_$(SLUG): $(RTL_SRCS) $(TB_SRCS)
	@mkdir -p $(BUILD)
	iverilog -g2001 -o $@ $(RTL_SRCS) $(TB_SRCS)

## ── 합성 (yosys, 자원 추정) ─────────────────────────────────────────────────
synth:
	@mkdir -p $(BUILD)
	yosys -p "read_verilog -sv $(RTL_SRCS); \
	          synth_xilinx -top basys3_imu_cf_top -flatten; \
	          stat" \
	    2>&1 | tee $(BUILD)/synth.log
	@echo "synth done (yosys 추정 — Vivado 수치와 다를 수 있음)"

## ── 정리 ────────────────────────────────────────────────────────────────────
clean:
	rm -rf $(BUILD)
