PS/2 키보드 수신기 + 스캔코드→ASCII 디코더 (Basys3)

mmio-uart-gpio-skeleton(T1)에 PS/2 키보드 수신기를 추가한 확장 패키지. ps2_rx가 11-bit 프레임을 수신하고, ps2_kbd가 Set 2 스캔코드를 ASCII로 변환한 뒤 MMIO 레지스터로 노출한다. demo_master FSM이 kbd_ready를 폴링해 make 이벤트는 LED에 표시하고 UART로 송신, break 이벤트는 LED를 클리어한다.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: 미실행

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • PS/2 프레임(11-bit: start + D0–D7 + 패리티 + stop)을 하강 에지에서 수신하고 홀수 패리티를 검사한다.
  • Set 2 스캔코드(make/break/extended)를 ASCII로 변환한다.
  • MMIO 레지스터(STATUS 0x00, DATA 0x04)를 FSM이 폴링해 make/break 이벤트를 LED와 UART로 출력한다.
  • 자체 점검 테스트벤치로 T1~T5를 시뮬레이션에서 통과한다.

범위 밖

  • CPU 구현 (마스터는 FSM)
  • 키보드 호스트 송신(대기 명령, 초기화)
  • 대문자·특수문자(Shift/Ctrl 조합)
  • NumLock·CapsLock LED 제어
  • 외부 메모리

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-007 기반. 시뮬레이션·합성 결과는 CI가 기록한다.

2. 계통도

클럭: 100 MHz 단일 클럭 (CLK_100, W5), 비동기 리셋 (BTNC, U18, 액티브 하이)

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite (자체, 단독 실행용)
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획이 패키지는 단독 시뮬레이션·보드 확인용으로 자체 mmio-lite 버스(0x1000_xxxx~0x1002_xxxx)를 쓴다. 동결 맵(mbv-frozen-v2) 이식 계획: (1) GPIO·UART는 동결 맵 [00]/[02] AMD IP를 그대로 쓰고 드라이버를 맞춘다. (2) ps2_kbd는 동결 맵 [0B](0x400B_0000)에 IP 표준(64KB 창, 0xFFF0 ID/VER)으로 감싸서 AXI4-Lite 슬레이브로 등록한다. (3) demo_master 대신 MicroBlaze V + 폴링 드라이버(C) 또는 인터럽트(irq→INTC)로 교체한다.

// ps2_rx.v — PS/2 수신기 (PS/2 Set 2 스캔코드, Verilog-2001)
// 하네스 직접 작성 원본.
//
// PS/2 프레임 구조: [START=0] [D0..D7] [PARITY(홀수)] [STOP=1]
// 하강 에지에서 데이터를 샘플링한다.
// PS/2 CLK과 DATA는 2단 동기화 후 사용한다.
//
// 출력:
//   rx_data[7:0]  — 수신된 스캔코드 바이트 (rx_valid 사이클에만 유효)
//   rx_valid      — 1사이클 펄스, 오류 없이 한 프레임을 완수했을 때
//   rx_err        — 1사이클 펄스, 패리티 또는 정지 비트 오류
`timescale 1ns/1ps
module ps2_rx (
    input  wire       clk,
    input  wire       rst,
    // PS/2 핀 (외부 입력)
    input  wire       ps2_clk_in,
    input  wire       ps2_data_in,
    // 출력
    output reg  [7:0] rx_data,
    output reg        rx_valid,
    output reg        rx_err
);

    // ----------------------------------------------------------------
    // PS/2 CLK 2단 동기화 + 하강 에지 검출
    // ----------------------------------------------------------------
    reg ps2_c1, ps2_c2, ps2_c3;
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            ps2_c1 <= 1'b1;
            ps2_c2 <= 1'b1;
            ps2_c3 <= 1'b1;
        end else begin
            ps2_c1 <= ps2_clk_in;
            ps2_c2 <= ps2_c1;
            ps2_c3 <= ps2_c2;
        end
    end
    // 하강 에지: 직전이 High, 현재가 Low
    wire clk_fall = ps2_c3 & (~ps2_c2);

    // ----------------------------------------------------------------
    // PS/2 DATA 2단 동기화
    // ----------------------------------------------------------------
    reg ps2_d1, ps2_d2;
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            ps2_d1 <= 1'b1;
            ps2_d2 <= 1'b1;
        end else begin
            ps2_d1 <= ps2_data_in;
            ps2_d2 <= ps2_d1;
        end
    end

    // ----------------------------------------------------------------
    // 수신 FSM
    //   bit_cnt=0  : 대기 (START 비트 검출)
    //   bit_cnt=1~8: 데이터 비트 D0~D7 (LSB 먼저)
    //   bit_cnt=9  : 패리티 비트
    //   bit_cnt=10 : 정지 비트 → 출력
    // ----------------------------------------------------------------
    reg [3:0]  bit_cnt;
    reg [7:0]  data_sr;    // D0부터 순서대로 쌓는 시프트 레지스터
    reg        parity_acc; // 수신한 데이터 비트의 XOR 누산

    always @(posedge clk or posedge rst) begin
        rx_valid <= 1'b0;
        rx_err   <= 1'b0;
        if (rst) begin
            bit_cnt    <= 4'd0;
            data_sr    <= 8'h00;
            parity_acc <= 1'b0;
        end else if (clk_fall) begin
            case (bit_cnt)
                4'd0: begin
                    // START 비트 = 0 이면 수신 시작
                    if (~ps2_d2) begin
                        data_sr    <= 8'h00;
                        parity_acc <= 1'b0;
                        bit_cnt    <= 4'd1;
                    end
                    // START = 1 이면 무시 (라인 High 상태)
                end
                4'd1, 4'd2, 4'd3, 4'd4,
                4'd5, 4'd6, 4'd7, 4'd8: begin
                    // 데이터 비트: D0(bit_cnt=1)~D7(bit_cnt=8)
                    // LSB 먼저이므로 MSB 쪽으로 밀어 쌓는다
                    data_sr    <= {ps2_d2, data_sr[7:1]};
                    parity_acc <= parity_acc ^ ps2_d2;
                    bit_cnt    <= bit_cnt + 4'd1;
                end
                4'd9: begin
                    // 패리티 비트 (홀수 패리티 → data XOR parity = 1)
                    // 여기서는 parity_acc에 패리티 비트를 XOR해 나중에 판정
                    data_sr    <= data_sr; // 유지
                    parity_acc <= parity_acc ^ ps2_d2;
                    bit_cnt    <= 4'd10;
                end
                4'd10: begin
                    // 정지 비트 = 1
                    bit_cnt <= 4'd0;
                    if (ps2_d2 && (parity_acc == 1'b1)) begin
                        // 정지=1, 홀수 패리티 OK
                        rx_data  <= data_sr;
                        rx_valid <= 1'b1;
                    end else begin
                        rx_err <= 1'b1;
                    end
                end
                default: bit_cnt <= 4'd0;
            endcase
        end
    end
endmodule

// ps2_kbd.v — PS/2 키보드 디코더 + MMIO 슬레이브 (Verilog-2001)
// 하네스 직접 작성 원본.
//
// 역할:
//   - ps2_rx에서 스캔코드를 받아 make/break 시퀀스를 처리한다.
//   - 스캔코드를 ASCII로 변환한다.
//   - MMIO 레지스터 인터페이스를 제공한다.
//
// PS/2 Set 2 특수 코드:
//   0xF0 : Break prefix (다음 바이트는 뗌 코드)
//   0xE0 : Extended key prefix (여기서는 무시)
//
// MMIO 레지스터 (오프셋 7:0 기준):
//   0x00 STATUS : {31'h0, kbd_ready}   kbd_ready=1 → 새 문자 수신
//   0x04 DATA   : {22'h0, key_up, valid, ascii[7:0]}
//                 key_up=1 → 키 뗌(break), valid=1 → ASCII 유효
//                 DATA를 읽으면 kbd_ready가 클리어된다.
`timescale 1ns/1ps
module ps2_kbd (
    input  wire       clk,
    input  wire       rst,
    // ps2_rx 입력
    input  wire [7:0] rx_data,
    input  wire       rx_valid,
    // verilator lint_off UNUSED
    input  wire       rx_err,   // 오류 카운트용 (현재 버전은 무시)
    // verilator lint_on UNUSED
    // MMIO 슬레이브 인터페이스
    input  wire       sel,
    input  wire       we,
    input  wire [7:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    // 인터럽트 (선택 — 현재 버전은 출력만)
    output wire       irq
);
    // verilator lint_off UNUSED
    wire [31:0] _unused_wdata = wdata;
    // verilator lint_on UNUSED

    // ----------------------------------------------------------------
    // 스캔코드 → ASCII 함수
    // ----------------------------------------------------------------
    function [7:0] scan_to_ascii;
        input [7:0] sc;
        case (sc)
            8'h1C: scan_to_ascii = 8'h61; // a
            8'h32: scan_to_ascii = 8'h62; // b
            8'h21: scan_to_ascii = 8'h63; // c
            8'h23: scan_to_ascii = 8'h64; // d
            8'h24: scan_to_ascii = 8'h65; // e
            8'h2B: scan_to_ascii = 8'h66; // f
            8'h34: scan_to_ascii = 8'h67; // g
            8'h33: scan_to_ascii = 8'h68; // h
            8'h43: scan_to_ascii = 8'h69; // i
            8'h3B: scan_to_ascii = 8'h6A; // j
            8'h42: scan_to_ascii = 8'h6B; // k
            8'h4B: scan_to_ascii = 8'h6C; // l
            8'h3A: scan_to_ascii = 8'h6D; // m
            8'h31: scan_to_ascii = 8'h6E; // n
            8'h44: scan_to_ascii = 8'h6F; // o
            8'h4D: scan_to_ascii = 8'h70; // p
            8'h15: scan_to_ascii = 8'h71; // q
            8'h2D: scan_to_ascii = 8'h72; // r
            8'h1B: scan_to_ascii = 8'h73; // s
            8'h2C: scan_to_ascii = 8'h74; // t
            8'h3C: scan_to_ascii = 8'h75; // u
            8'h2A: scan_to_ascii = 8'h76; // v
            8'h1D: scan_to_ascii = 8'h77; // w
            8'h22: scan_to_ascii = 8'h78; // x
            8'h35: scan_to_ascii = 8'h79; // y
            8'h1A: scan_to_ascii = 8'h7A; // z
            8'h45: scan_to_ascii = 8'h30; // 0
            8'h16: scan_to_ascii = 8'h31; // 1
            8'h1E: scan_to_ascii = 8'h32; // 2
            8'h26: scan_to_ascii = 8'h33; // 3
            8'h25: scan_to_ascii = 8'h34; // 4
            8'h2E: scan_to_ascii = 8'h35; // 5
            8'h36: scan_to_ascii = 8'h36; // 6
            8'h3D: scan_to_ascii = 8'h37; // 7
            8'h3E: scan_to_ascii = 8'h38; // 8
            8'h46: scan_to_ascii = 8'h39; // 9
            8'h29: scan_to_ascii = 8'h20; // space
            8'h5A: scan_to_ascii = 8'h0D; // Enter (CR)
            8'h66: scan_to_ascii = 8'h08; // Backspace
            8'h76: scan_to_ascii = 8'h1B; // Escape
            8'h0D: scan_to_ascii = 8'h09; // Tab
            default: scan_to_ascii = 8'h00; // 미지원
        endcase
    endfunction

    // ----------------------------------------------------------------
    // 디코더 FSM
    //   IDLE     : 일반 make 코드 대기
    //   BREAK    : F0 받음, 다음 바이트는 break 코드
    //   EXTENDED : E0 받음, 다음 바이트를 무시 (simple version)
    // ----------------------------------------------------------------
    localparam S_IDLE = 2'd0, S_BREAK = 2'd1, S_EXTENDED = 2'd2;
    reg [1:0] state;

    // 출력 레지스터
    reg       kbd_ready;
    reg [7:0] ascii_buf;
    reg       key_up_buf;

    assign irq = kbd_ready; // 준비되면 IRQ 발생 (외부에서 연결 선택)

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state     <= S_IDLE;
            kbd_ready <= 1'b0;
            ascii_buf <= 8'h00;
            key_up_buf<= 1'b0;
        end else begin
            if (rx_valid) begin
                case (state)
                    S_IDLE: begin
                        if (rx_data == 8'hF0) begin
                            state <= S_BREAK;
                        end else if (rx_data == 8'hE0) begin
                            state <= S_EXTENDED;
                        end else begin
                            // Make 코드
                            if (scan_to_ascii(rx_data) != 8'h00) begin
                                ascii_buf  <= scan_to_ascii(rx_data);
                                key_up_buf <= 1'b0;
                                kbd_ready  <= 1'b1;
                            end
                        end
                    end
                    S_BREAK: begin
                        // Break 코드: 뗌 이벤트
                        if (scan_to_ascii(rx_data) != 8'h00) begin
                            ascii_buf  <= scan_to_ascii(rx_data);
                            key_up_buf <= 1'b1;
                            kbd_ready  <= 1'b1;
                        end
                        state <= S_IDLE;
                    end
                    S_EXTENDED: begin
                        // E0 접두어: 현재 버전은 다음 바이트를 무시
                        if (rx_data == 8'hF0)
                            state <= S_BREAK; // E0 F0 → extended break
                        else
                            state <= S_IDLE;
                    end
                    default: state <= S_IDLE;
                endcase
            end

            // MMIO 읽기: DATA를 읽으면 kbd_ready 클리어
            if (sel && ~we && (off == 8'h04))
                kbd_ready <= 1'b0;
        end
    end

    // ----------------------------------------------------------------
    // MMIO 읽기 조합 로직
    // ----------------------------------------------------------------
    always @* begin
        rdata = 32'h0000_0000;
        if (sel && ~we) begin
            case (off)
                8'h00: rdata = {31'h0, kbd_ready};
                8'h04: rdata = {22'h0, key_up_buf, (ascii_buf != 8'h00), ascii_buf};
                default: rdata = 32'h0000_0000;
            endcase
        end
    end
endmodule

// mmio_bus.v — MMIO 주소 디코더 (64KB 창, addr[31:16] 기준)
// 하네스 직접 작성 원본.
// addr[31:16] 기준으로 슬레이브를 선택하고, addr[15:0]을 오프셋으로 전달한다.
//
// 슬롯 (mmio-lite 버스, 동결 맵과 별도):
//   0x1000_xxxx : GPIO  (LED, SW)
//   0x1001_xxxx : UART TX
//   0x1002_xxxx : PS2_KBD  (동결 맵: 0x400B_0000)
`timescale 1ns/1ps
module mmio_bus (
    input  wire        valid,
    input  wire        we,
    input  wire [31:0] addr,
    input  wire [31:0] wdata,
    // 슬레이브 선택
    output wire        sel_gpio,
    output wire        sel_uart,
    output wire        sel_kbd,
    // 슬레이브 읽기 데이터
    input  wire [31:0] rdata_gpio,
    input  wire [31:0] rdata_uart,
    input  wire [31:0] rdata_kbd,
    // 마스터 읽기 데이터
    output reg  [31:0] rdata,
    output wire        ready
);
    assign sel_gpio = valid && (addr[31:16] == 16'h1000);
    assign sel_uart = valid && (addr[31:16] == 16'h1001);
    assign sel_kbd  = valid && (addr[31:16] == 16'h1002);
    assign ready    = 1'b1; // 모든 접근 1사이클

    always @* begin
        if      (sel_gpio) rdata = rdata_gpio;
        else if (sel_uart) rdata = rdata_uart;
        else if (sel_kbd)  rdata = rdata_kbd;
        else               rdata = 32'h0000_0000;
    end

    // verilator lint_off UNUSED
    wire [31:0] _unused_wdata = wdata;
    // verilator lint_on UNUSED
endmodule

// gpio.v — GPIO 슬레이브 (Verilog-2001)
// 하네스 직접 작성 원본.
// 원본 경로: capstones/mmio-uart-gpio-skeleton/rtl/gpio.v
// 0x00 LED(RW, 16비트), 0x04 SW(RO, 2단 동기화)
`timescale 1ns/1ps
module gpio (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [7:0]  off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    input  wire [15:0] sw,
    output wire [15:0] led
);
    reg [15:0] led_r;
    reg [15:0] sw_s1, sw_s2;
    assign led = led_r;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            led_r <= 16'h0000;
            sw_s1 <= 16'h0000;
            sw_s2 <= 16'h0000;
        end else begin
            sw_s1 <= sw;
            sw_s2 <= sw_s1;
            if (sel && we && (off == 8'h00))
                led_r <= wdata[15:0];
        end
    end

    always @* begin
        case (off)
            8'h00:   rdata = {16'h0000, led_r};
            8'h04:   rdata = {16'h0000, sw_s2};
            default: rdata = 32'h0000_0000;
        endcase
    end
endmodule

// uart_tx.v — UART 송신기 8N1 (Verilog-2001)
// 하네스 직접 작성 원본.
// 원본 경로: capstones/mmio-uart-gpio-skeleton/rtl/uart_tx.v
// 0x00 DATA(WO, busy 중 무시), 0x04 STATUS(RO, bit0=busy)
`timescale 1ns/1ps
module uart_tx #(
    parameter CLK_HZ = 100_000_000,
    parameter BAUD   = 115_200
) (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [7:0]  off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    output wire        txd
);
    localparam DIV = CLK_HZ / BAUD;

    reg [9:0]  sh;
    reg [3:0]  nbit;
    reg [15:0] cnt;
    reg        busy;
    reg        tx;
    assign txd = tx;

    wire wr_data = sel && we && (off == 8'h00);

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            busy <= 1'b0;
            tx   <= 1'b1;
            cnt  <= 16'd0;
            nbit <= 4'd0;
            sh   <= 10'h3FF;
        end else if (!busy) begin
            tx <= 1'b1;
            if (wr_data) begin
                sh   <= {1'b1, wdata[7:0], 1'b0};
                busy <= 1'b1;
                cnt  <= DIV - 1;
                nbit <= 4'd0;
                tx   <= 1'b0;
            end
        end else begin
            if (cnt == 16'd0) begin
                cnt <= DIV - 1;
                if (nbit == 4'd9) begin
                    busy <= 1'b0;
                    tx   <= 1'b1;
                end else begin
                    nbit <= nbit + 4'd1;
                    tx   <= sh[nbit + 4'd1];
                end
            end else begin
                cnt <= cnt - 16'd1;
            end
        end
    end

    always @* begin
        case (off)
            8'h04:   rdata = {31'd0, busy};
            default: rdata = 32'h0000_0000;
        endcase
    end
endmodule

// demo_master.v — PS/2 키보드 폴링 마스터 FSM (Verilog-2001)
// 하네스 직접 작성 원본.
//
// 역할:
//   - PS/2 키보드 MMIO를 1사이클 단위로 폴링한다.
//   - 새 키(kbd_ready=1)가 오면 DATA 레지스터를 읽어 ASCII를 추출한다.
//   - make 이벤트(key_up=0, valid=1)이면:
//     - LED 하위 8비트에 ASCII를 표시한다.
//     - UART TX가 쉬면 ASCII를 UART로 송신한다.
//   - break 이벤트(key_up=1, valid=1)이면:
//     - LED를 클리어한다.
//
// 버스 주소 (mmio-lite 맵, addr[31:0] 기준):
//   0x1000_0000 : GPIO LED  (WO: wdata[15:0])
//   0x1001_0000 : UART TX DATA (WO: wdata[7:0])
//   0x1001_0004 : UART TX STATUS (RO: bit0=busy)
//   0x1002_0000 : KBD STATUS (RO: bit0=kbd_ready)
//   0x1002_0004 : KBD DATA   (RO: {22'h0, key_up, valid, ascii[7:0]})
`timescale 1ns/1ps
module demo_master (
    input  wire        clk,
    input  wire        rst,
    // 버스 마스터 포트
    output reg         valid,
    output reg         we,
    output reg  [31:0] addr,
    output reg  [31:0] wdata,
    input  wire [31:0] rdata,
    input  wire        ready
);
    // FSM 상태
    localparam
        S_POLL_KBD   = 4'd0,  // KBD STATUS 읽기 요청
        S_WAIT_KBD   = 4'd1,  // STATUS 응답 대기
        S_READ_DATA  = 4'd2,  // KBD DATA 읽기 요청
        S_WAIT_DATA  = 4'd3,  // DATA 응답 대기
        S_CHK_UART   = 4'd4,  // UART STATUS 읽기 요청
        S_WAIT_UART  = 4'd5,  // UART STATUS 대기
        S_SEND_UART  = 4'd6,  // UART DATA 쓰기
        S_LED_MAKE   = 4'd7,  // LED 쓰기 (make)
        S_LED_BREAK  = 4'd8,  // LED 클리어 (break)
        S_IDLE       = 4'd9;  // 짧은 대기 (다음 폴링 전)

    reg [3:0]  state;
    reg [7:0]  ascii_hold;
    reg        key_up_hold;
    reg [7:0]  idle_cnt;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state      <= S_POLL_KBD;
            valid      <= 1'b0;
            we         <= 1'b0;
            addr       <= 32'h0;
            wdata      <= 32'h0;
            ascii_hold <= 8'h0;
            key_up_hold<= 1'b0;
            idle_cnt   <= 8'd0;
        end else begin
            case (state)
                // ---- KBD STATUS 읽기 ----
                S_POLL_KBD: begin
                    valid <= 1'b1;
                    we    <= 1'b0;
                    addr  <= 32'h1002_0000;
                    wdata <= 32'h0;
                    state <= S_WAIT_KBD;
                end
                S_WAIT_KBD: begin
                    if (ready) begin
                        valid <= 1'b0;
                        if (rdata[0]) begin
                            // kbd_ready=1 → DATA 읽기
                            state <= S_READ_DATA;
                        end else begin
                            // 아직 없음 → 짧은 대기 후 다시
                            state    <= S_IDLE;
                            idle_cnt <= 8'd4;
                        end
                    end
                end
                // ---- KBD DATA 읽기 ----
                S_READ_DATA: begin
                    valid <= 1'b1;
                    we    <= 1'b0;
                    addr  <= 32'h1002_0004;
                    wdata <= 32'h0;
                    state <= S_WAIT_DATA;
                end
                S_WAIT_DATA: begin
                    if (ready) begin
                        valid       <= 1'b0;
                        ascii_hold  <= rdata[7:0];
                        key_up_hold <= rdata[9];
                        // valid 비트(bit8)가 세트된 경우에만 처리
                        if (rdata[8]) begin
                            if (rdata[9]) begin
                                // key_up → LED 클리어
                                state <= S_LED_BREAK;
                            end else begin
                                // make → UART 전송 가능 여부 확인
                                state <= S_LED_MAKE;
                            end
                        end else begin
                            // valid=0이면 무효 → 다시 폴링
                            state    <= S_IDLE;
                            idle_cnt <= 8'd4;
                        end
                    end
                end
                // ---- LED 쓰기 (make) ----
                S_LED_MAKE: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= 32'h1000_0000;
                    wdata <= {24'h0, ascii_hold};
                    state <= S_CHK_UART;
                end
                // ---- UART STATUS 읽기 ----
                S_CHK_UART: begin
                    if (ready) begin
                        // LED 쓰기 완료, UART 상태 확인
                        valid <= 1'b1;
                        we    <= 1'b0;
                        addr  <= 32'h1001_0004;
                        wdata <= 32'h0;
                        state <= S_WAIT_UART;
                    end
                end
                S_WAIT_UART: begin
                    if (ready) begin
                        valid <= 1'b0;
                        if (!rdata[0]) begin
                            // busy=0 → 전송 가능
                            state <= S_SEND_UART;
                        end else begin
                            // 바쁨 → 잠시 대기
                            state    <= S_IDLE;
                            idle_cnt <= 8'd4;
                        end
                    end
                end
                // ---- UART DATA 쓰기 ----
                S_SEND_UART: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= 32'h1001_0000;
                    wdata <= {24'h0, ascii_hold};
                    state <= S_POLL_KBD;
                end
                // ---- LED 클리어 (break) ----
                S_LED_BREAK: begin
                    valid <= 1'b1;
                    we    <= 1'b1;
                    addr  <= 32'h1000_0000;
                    wdata <= 32'h0000_0000;
                    state <= S_POLL_KBD;
                end
                // ---- 짧은 대기 ----
                S_IDLE: begin
                    valid <= 1'b0;
                    if (idle_cnt == 8'd0) begin
                        state <= S_POLL_KBD;
                    end else begin
                        idle_cnt <= idle_cnt - 8'd1;
                    end
                end
                default: state <= S_POLL_KBD;
            endcase
        end
    end
endmodule

// tb_ps2_kbd.v — PS/2 키보드 수신기 자체 점검 테스트벤치 (Verilog-2001)
// 하네스 직접 작성 원본.
//
// 테스트 항목:
//   T1  make  'H' (스캔코드 0x33, ASCII 0x68)
//   T2  break 'H' (F0+0x33, key_up=1)
//   T3  make  'I' (스캔코드 0x43, ASCII 0x69)
//   T4  패리티 오류 프레임 → rx_err=1, kbd_ready는 올라오지 않아야 함
//   T5  MMIO 읽기: STATUS/DATA 레지스터 값 확인
//
// PS/2 Set 2: 하강 에지에서 데이터를 샘플링한다 (11-bit 프레임).
// 클럭: 100MHz 시스템 / PS/2 CLK 주기 = 40us (25kHz)
`timescale 1ns/1ps

module tb_ps2_kbd;
    // ----------------------------------------------------------------
    // 파라미터
    // ----------------------------------------------------------------
    localparam CLK_PERIOD  = 10;        // 100MHz = 10ns
    localparam PS2_HALF    = 20_000;    // PS/2 CLK 반주기 20us (50ns × 400 = 20us)
    // PS2_HALF를 짧게 해 시뮬 속도 확보: 실제 20us 대신 2us (200클럭)
    // 200클럭 반주기 → 하강 에지에서 수신, 3-사이클 동기화 후 샘플

    // ----------------------------------------------------------------
    // DUT 신호
    // ----------------------------------------------------------------
    reg         clk;
    reg         rst;
    reg         ps2_clk_in;
    reg         ps2_data_in;
    wire [15:0] led;
    wire        txd;

    // MMIO 직접 접근용
    wire        sel_kbd;
    wire [31:0] rdata_kbd;
    reg         tb_sel;
    reg         tb_we;
    reg  [7:0]  tb_off;
    reg  [31:0] tb_wdata;

    // ----------------------------------------------------------------
    // DUT 인스턴스 (테스트 편의를 위해 soc_top 대신 하위 모듈 직접)
    // ----------------------------------------------------------------
    wire [7:0] rx_data;
    wire       rx_valid;
    wire       rx_err;

    ps2_rx u_rx (
        .clk         (clk),
        .rst         (rst),
        .ps2_clk_in  (ps2_clk_in),
        .ps2_data_in (ps2_data_in),
        .rx_data     (rx_data),
        .rx_valid    (rx_valid),
        .rx_err      (rx_err)
    );

    wire kbd_irq;
    wire [31:0] rdata_out;

    ps2_kbd u_kbd (
        .clk      (clk),
        .rst      (rst),
        .rx_data  (rx_data),
        .rx_valid (rx_valid),
        .rx_err   (rx_err),
        .sel      (tb_sel),
        .we       (tb_we),
        .off      (tb_off),
        .wdata    (tb_wdata),
        .rdata    (rdata_out),
        .irq      (kbd_irq)
    );

    // ----------------------------------------------------------------
    // 클럭 생성
    // ----------------------------------------------------------------
    initial clk = 0;
    always #(CLK_PERIOD/2) clk = ~clk;

    // ----------------------------------------------------------------
    // 카운터와 결과
    // ----------------------------------------------------------------
    integer fail_cnt;
    integer test_num;

    // ----------------------------------------------------------------
    // PS/2 프레임 송신 태스크 (DUT에서는 수신)
    //   PS/2 장치가 호스트로 보내는 방향: CLK은 장치가 생성
    //   Start=0, D0..D7 (LSB first), Parity(odd), Stop=1
    //   하강 에지에서 호스트가 샘플링한다
    // ----------------------------------------------------------------
    task send_ps2_byte;
        input [7:0] data;
        input       bad_parity; // 1이면 패리티를 반전해 오류 유도
        integer i;
        reg parity;
        reg [10:0] frame;
        begin
            // 홀수 패리티: 1의 개수가 홀수
            parity = ^data; // data 비트의 XOR → 짝수면 1, 홀수면 0
            if (!parity) parity = 1'b1; else parity = 1'b0;
            // 오류 유도 시 반전
            if (bad_parity) parity = ~parity;
            // 프레임: [0]=start, [1..8]=data LSB→MSB, [9]=parity, [10]=stop
            frame = {1'b1, parity, data, 1'b0};

            // 라인 초기 상태: CLK=1, DATA=1
            ps2_clk_in  = 1'b1;
            ps2_data_in = 1'b1;
            #(PS2_HALF * CLK_PERIOD);

            for (i = 0; i <= 10; i = i + 1) begin
                // DATA 세팅 (CLK 하이 중)
                ps2_data_in = frame[i];
                #(PS2_HALF * CLK_PERIOD / 2);
                // CLK 하강 에지
                ps2_clk_in = 1'b0;
                #(PS2_HALF * CLK_PERIOD);
                // CLK 상승 에지
                ps2_clk_in = 1'b1;
                #(PS2_HALF * CLK_PERIOD / 2);
            end

            // 프레임 후 라인 복귀
            ps2_data_in = 1'b1;
            ps2_clk_in  = 1'b1;
            #(PS2_HALF * CLK_PERIOD * 2);
        end
    endtask

    // ----------------------------------------------------------------
    // MMIO 읽기 헬퍼
    // ----------------------------------------------------------------
    task mmio_read;
        input [7:0] offset;
        output [31:0] data;
        begin
            @(negedge clk);
            tb_sel  = 1'b1;
            tb_we   = 1'b0;
            tb_off  = offset;
            tb_wdata= 32'h0;
            @(negedge clk);
            data = rdata_out;
            tb_sel  = 1'b0;
            @(posedge clk); // 1사이클 여유
        end
    endtask

    // ----------------------------------------------------------------
    // 검증 헬퍼
    // ----------------------------------------------------------------
    task check_eq;
        input [31:0] got;
        input [31:0] exp;
        input [63:0] label; // 최대 8글자 ASCII 태그
        begin
            if (got !== exp) begin
                $display("FAIL T%0d [%s]: got=0x%08X exp=0x%08X",
                         test_num, label, got, exp);
                fail_cnt = fail_cnt + 1;
            end
        end
    endtask

    // ----------------------------------------------------------------
    // 메인
    // ----------------------------------------------------------------
    reg [31:0] rd;

    initial begin
        // 초기화
        fail_cnt    = 0;
        test_num    = 0;
        rst         = 1'b1;
        ps2_clk_in  = 1'b1;
        ps2_data_in = 1'b1;
        tb_sel      = 1'b0;
        tb_we       = 1'b0;
        tb_off      = 8'h0;
        tb_wdata    = 32'h0;

        repeat(10) @(posedge clk);
        rst = 1'b0;
        repeat(5) @(posedge clk);

        // ============================================================
        // T1: make 'H' (스캔코드 0x33, ASCII 0x68)
        // ============================================================
        test_num = 1;
        send_ps2_byte(8'h33, 1'b0); // make H

        // kbd_ready가 올라올 때까지 최대 50사이클 대기
        begin : wait_t1
            integer w;
            for (w = 0; w < 50; w = w + 1) begin
                @(posedge clk);
                if (kbd_irq) disable wait_t1;
            end
        end

        // STATUS 읽기: kbd_ready=1 기대
        mmio_read(8'h00, rd);
        check_eq(rd, 32'h0000_0001, "T1-STAT");

        // DATA 읽기: key_up=0, valid=1, ascii=0x68
        mmio_read(8'h04, rd);
        check_eq(rd[9],   1'b0,       "T1-KUP "); // key_up=0
        check_eq(rd[8],   1'b1,       "T1-VLD "); // valid=1
        check_eq(rd[7:0], 8'h68,      "T1-ASCII"); // 'h'

        // DATA 읽기 후 kbd_ready가 클리어됐는지 확인
        mmio_read(8'h00, rd);
        check_eq(rd, 32'h0000_0000, "T1-CLR ");

        // ============================================================
        // T2: break 'H' (F0 + 0x33, key_up=1)
        // ============================================================
        test_num = 2;
        send_ps2_byte(8'hF0, 1'b0); // break prefix
        // F0은 ASCII=0이므로 kbd_ready는 올라오지 않음; 다음 바이트를 기다림
        repeat(5) @(posedge clk);

        send_ps2_byte(8'h33, 1'b0); // break 'H'
        begin : wait_t2
            integer w;
            for (w = 0; w < 50; w = w + 1) begin
                @(posedge clk);
                if (kbd_irq) disable wait_t2;
            end
        end

        mmio_read(8'h00, rd);
        check_eq(rd, 32'h0000_0001, "T2-STAT");

        mmio_read(8'h04, rd);
        check_eq(rd[9],   1'b1,  "T2-KUP "); // key_up=1
        check_eq(rd[8],   1'b1,  "T2-VLD "); // valid=1
        check_eq(rd[7:0], 8'h68, "T2-ASCII"); // 'h'

        // ============================================================
        // T3: make 'I' (스캔코드 0x43, ASCII 0x69)
        // ============================================================
        test_num = 3;
        send_ps2_byte(8'h43, 1'b0); // make I

        begin : wait_t3
            integer w;
            for (w = 0; w < 50; w = w + 1) begin
                @(posedge clk);
                if (kbd_irq) disable wait_t3;
            end
        end

        mmio_read(8'h04, rd);
        check_eq(rd[9],   1'b0,  "T3-KUP "); // key_up=0
        check_eq(rd[8],   1'b1,  "T3-VLD "); // valid=1
        check_eq(rd[7:0], 8'h69, "T3-ASCII"); // 'i'

        // ============================================================
        // T4: 패리티 오류 프레임 → rx_err=1, kbd_ready 올라오지 않음
        // ============================================================
        test_num = 4;
        // 먼저 kbd_ready 클리어 확인 (DATA 이미 읽음)
        mmio_read(8'h00, rd);
        check_eq(rd, 32'h0000_0000, "T4-PRE ");

        // 오류 프레임 전송
        send_ps2_byte(8'h1C, 1'b1); // 'A' but bad parity

        // rx_err 펄스 후 kbd_ready는 여전히 0이어야 함
        repeat(10) @(posedge clk);
        // rx_err is a 1-cycle pulse, check irq (=kbd_ready) stayed low
        mmio_read(8'h00, rd);
        check_eq(rd, 32'h0000_0000, "T4-ERR ");

        // ============================================================
        // T5: E0 확장 코드 무시 확인 (E0 + 임의 바이트)
        // ============================================================
        test_num = 5;
        send_ps2_byte(8'hE0, 1'b0); // extended prefix
        repeat(5) @(posedge clk);
        send_ps2_byte(8'h75, 1'b0); // Up arrow (extended, 0x75)
        repeat(10) @(posedge clk);
        // kbd_ready는 올라오지 않아야 함
        mmio_read(8'h00, rd);
        check_eq(rd, 32'h0000_0000, "T5-EXT ");

        // ============================================================
        // 결과
        // ============================================================
        if (fail_cnt == 0)
            $display("PASS: 모든 테스트 통과 (%0d개)", test_num);
        else
            $display("FAIL: %0d개 실패", fail_cnt);

        $finish;
    end

    // 타임아웃
    initial begin
        #(50_000_000); // 50ms (100MHz 기준 5백만 클럭)
        $display("FAIL: 타임아웃");
        $finish;
    end

endmodule

# mmio-ps2-kbd-checker Makefile
# 단계: lint(verilator) → sim(iverilog) → synth(yosys, 자원 추정)
# Vivado 빌드는 capstone.json howto 항목을 참조한다.

RTL_ALL = rtl/ps2_rx.v rtl/ps2_kbd.v rtl/mmio_bus.v rtl/gpio.v rtl/uart_tx.v rtl/demo_master.v rtl/soc_top.v rtl/basys3_top.v
RTL_SOC = rtl/ps2_rx.v rtl/ps2_kbd.v rtl/mmio_bus.v rtl/gpio.v rtl/uart_tx.v rtl/demo_master.v rtl/soc_top.v
TB      = tb/tb_ps2_kbd.v

.PHONY: lint sim synth clean

build:
	mkdir -p build

lint: build
	verilator --lint-only -Wall -Wno-fatal --top-module basys3_top $(RTL_ALL) 2>&1 | tee build/lint.log
	@! grep -q "%Error" build/lint.log

sim: build
	iverilog -g2005 -o build/sim.vvp $(TB) $(RTL_SOC)
	vvp build/sim.vvp | tee build/sim.log
	@grep -q "^PASS" build/sim.log && ! grep -q "^FAIL" build/sim.log

synth: build
	yosys -q -l build/synth.log -p "read_verilog $(RTL_ALL); synth_xilinx -top basys3_top -family xc7; stat"
	@grep -E "LUT|FDRE|FDCE|FDPE|Number of cells" build/synth.log | tail -12

clean:
	rm -rf build

시뮬레이션

  1. iverilog, verilator, yosys 설치
  2. make -C capstones/mmio-ps2-kbd-checker lint sim synth

Vivado 빌드 (Basys3)

  1. 새 RTL 프로젝트, 파트 xc7a35tcpg236-1
  2. rtl/*.v 추가 (tb 제외), constr/basys3.xdc 추가, 최상위 basys3_top
  3. Generate Bitstream 후 Program Device
  4. USB 키보드를 Basys3 HID 포트에 연결
  5. 키를 누르면 LED 하위 8비트에 ASCII 코드가 표시됨
  6. PC에서 115200 8N1 터미널을 열면 눌린 키의 ASCII 문자가 도착함
  7. 키를 떼면 (break 이벤트) LED가 꺼짐

위험·한계

  • PS/2 클럭 하강 에지 검출은 3-사이클 동기화(ps2_c3 & ~ps2_c2)를 쓰므로 타이밍 여유가 있지만, 100MHz보다 느린 클럭에서는 메타스태빌리티를 재검토해야 한다.
  • demo_master의 UART 바쁨 처리는 short idle 후 재시도이므로, 연속 키 입력 시 일부 문자가 손실될 수 있다.
  • Extended key(E0 접두어)는 무시하므로 화살표, Fn 키, Insert, Delete 등은 처리되지 않는다.
  • kbd_ready와 MMIO 읽기 클리어가 같은 클럭 에지에서 경합할 경우(rx_valid와 sel && ~we && off==0x04 동시), 읽기 클리어가 우선한다(always 블록 순서). 이 패키지에서는 demo_master가 단독 마스터이므로 경합이 없다.
  • xdc 핀 배치는 Digilent 마스터 XDC 2024.1을 참조했으나 보드 개정판에 따라 다를 수 있다.
