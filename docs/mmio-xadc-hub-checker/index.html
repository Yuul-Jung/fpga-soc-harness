XADC 4채널 아날로그 입력 허브 IP 자체 점검

Basys3 JXADC 헤더의 보조 채널 4개(VAUXP/N 4,5,12,13)를 순회 스캔하고, 채널별 12비트 원시값과 이동 평균(1/4/8/16샘플)을 AXI4-Lite 레지스터로 노출하는 XADC 허브 IP. DRP 인터페이스를 모의하는 테스트벤치로 채널 읽기·이동 평균·인터럽트를 자체 점검한다.

린트: 미실행시뮬레이션: 미실행합성: 미실행보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • XADC DRP FSM으로 4개 보조 채널을 순차 읽기
  • 채널별 최대 16샘플 이동 평균 계산
  • 변환 완료 인터럽트(irq)를 AXI4-Lite로 노출
  • DRP 모의 모델을 테스트벤치에 포함해 iverilog 시뮬레이션 가능하게 구성

범위 밖

  • XADC 온다이 온도·전압 모니터 채널
  • DMA 또는 연속 스트리밍 출력
  • 보드 실제 가변저항 테스트(S3 검증은 사람이 한다)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 시뮬레이션·합성 결과는 CI가 기록한다. 아이디어 I-032 (§14.2)로 생성됨 (2026-11-04).

2. 계통도

클럭: 100 MHz (sys_clk), 분주 없이 DRP 타이밍 카운터로 처리

버스 규약: mmio-lite

신호방향설명
addrM→S16비트 오프셋 주소
wdataM→S쓰기 데이터
rdataS→M읽기 데이터 (슬레이브 출력)
weM→S쓰기=1, 읽기=0
reqM→S트랜잭션 요청
ackS→M1클럭 응답 펄스

기존 capstone mmio-lite 계약(sensor-hub-checker, ps2-kbd-checker와 동일). AXI4-Lite 래퍼 없이 직접 버스 슬레이브로 연결.

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite
주소맵mbv-frozen-v2 → 동결 맵 mbv-frozen-v2 적합 여부: 부분 적합
이식 계획동결 맵 [09] xadc 슬롯(0x4009_0000)이 이미 예약됨. 이 IP는 같은 주소를 사용하므로 MicroBlaze V AXI4-Lite 래퍼를 붙이면 compatible 선언 가능. 현재 mmio-lite 버스이므로 partial.

5. 외부 부품과 전기 회로도

// Basys3 최상위 래퍼: JXADC 핀 연결, 클럭·리셋, LED 상태 표시
// JXADC 핀: VAUXP/N[4]=JA1/2, [5]=JA3/4, [12]=JB1/2, [13]=JB3/4 (xdc 기준)

module basys3_xadc_hub_top (
    input  wire        clk,        // 100 MHz 온보드 클럭 (W5)
    input  wire        btnC,       // 중앙 버튼 = 리셋 (T17)

    // JXADC 아날로그 입력 (Pmod JA/JB 헤더 아날로그 쌍)
    input  wire        vauxp4,     // JA1 - VAUXP[4]
    input  wire        vauxn4,     // JA2 - VAUXN[4]
    input  wire        vauxp5,     // JA3 - VAUXP[5]
    input  wire        vauxn5,     // JA4 - VAUXN[5]
    input  wire        vauxp12,    // JB1 - VAUXP[12]
    input  wire        vauxn12,    // JB2 - VAUXN[12]
    input  wire        vauxp13,    // JB3 - VAUXP[13]
    input  wire        vauxn13,    // JB4 - VAUXN[13]

    // LED (LD0~LD15)
    output wire [15:0] led
);

    // 리셋: btnC 액티브 하이
    wire rst = btnC;

    // SoC 출력
    wire [3:0] led_pass;
    wire       led_busy;
    wire       led_irq;

    // ============================================================
    // SoC 코어
    // ============================================================
    xadc_hub_soc u_soc (
        .clk     (clk),
        .rst     (rst),
        .vauxp4  (vauxp4),  .vauxn4  (vauxn4),
        .vauxp5  (vauxp5),  .vauxn5  (vauxn5),
        .vauxp12 (vauxp12), .vauxn12 (vauxn12),

// xadc_drp_ctrl.v
// XADC DRP 채널 순회 FSM
// 4개 보조 채널(VAUXP/N 4,5,12,13)을 순서대로 읽어 raw_data, valid를 올린다.
// XADC가 EOC를 올리면 DRP den+daddr로 순서대로 읽는다.
// Verilog-2001 호환 (iverilog/verilator/yosys). 개별 출력 포트 사용.

module xadc_drp_ctrl (
    input  wire        clk,
    input  wire        rst,
    input  wire        scan_en,       // 1=순환 스캔, 0=정지

    // DRP 인터페이스 (XADC 또는 모의 모델)
    input  wire        eoc,           // end-of-conversion
    output reg         den,           // DRP enable
    output reg  [6:0]  daddr,         // DRP 주소
    input  wire        drdy,          // DRP data ready
    input  wire [15:0] dout,          // DRP data out

    // 출력 (채널별 12비트 원시값 — 개별 포트, Verilog-2001 호환)
    output reg  [11:0] raw_ch0,  // VAUXP/N[4]  (DRP 0x14)
    output reg  [11:0] raw_ch1,  // VAUXP/N[5]  (DRP 0x15)
    output reg  [11:0] raw_ch2,  // VAUXP/N[12] (DRP 0x1C)
    output reg  [11:0] raw_ch3,  // VAUXP/N[13] (DRP 0x1D)
    output reg  [3:0]  valid          // 갱신 펄스 (1클럭)
);

    // FSM 상태
    localparam S_IDLE  = 3'd0;
    localparam S_WAIT  = 3'd1; // EOC 대기
    localparam S_READ  = 3'd2; // den 올리기
    localparam S_DRDY  = 3'd3; // drdy 대기
    localparam S_NEXT  = 3'd4; // 다음 채널 또는 종료

    reg [2:0] state;
    reg [1:0] ch_idx; // 현재 채널 인덱스 0..3

    // 채널 인덱스에서 DRP 주소를 반환하는 함수
    // VAUXP/N[4]=0x14, [5]=0x15, [12]=0x1C, [13]=0x1D
    function [6:0] ch_addr;
        input [1:0] idx;
        case (idx)
            2'd0: ch_addr = 7'h14;
            2'd1: ch_addr = 7'h15;
            2'd2: ch_addr = 7'h1C;
            2'd3: ch_addr = 7'h1D;
            default: ch_addr = 7'h00;
        endcase
    endfunction

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            state   <= S_WAIT;
            ch_idx  <= 2'd0;
            den     <= 1'b0;
            daddr   <= 7'h00;
            valid   <= 4'b0;
            raw_ch0 <= 12'd0;
            raw_ch1 <= 12'd0;
            raw_ch2 <= 12'd0;
            raw_ch3 <= 12'd0;
        end else begin
            den   <= 1'b0;
            valid <= 4'b0;

            case (state)
                S_WAIT: begin
                    if (scan_en && eoc) begin
                        ch_idx <= 2'd0;
                        state  <= S_READ;
                    end
                end

                S_READ: begin
                    den   <= 1'b1;
                    daddr <= ch_addr(ch_idx);
                    state <= S_DRDY;
                end

                S_DRDY: begin
                    if (drdy) begin

module xadc_avg (

    input  wire        rst,
    input  wire [1:0]  avg_len,        // 0=1, 1=4, 2=8, 3=16

    function [3:0] nxt;

// mmio-lite 레지스터 디코더
// 오프셋 맵 (capstone.json memory_map 기준):
//  0x00 CTRL     RW  bit[1:0]=avg_len bit2=irq_en bit3=scan_en
//  0x04 STATUS   RO  bit0=busy bit[5:2]=ch_valid[3:0]
//  0x08 CH0_RAW  RO  12비트 원시값
//  0x0C CH1_RAW  RO
//  0x10 CH2_RAW  RO
//  0x14 CH3_RAW  RO
//  0x18 CH0_AVG  RO  12비트 이동 평균
//  0x1C CH1_AVG  RO
//  0x20 CH2_AVG  RO
//  0x24 CH3_AVG  RO
//  0x28 IRQ_MASK RW  bit[3:0] 채널별 인터럽트 마스크
//  0x2C IRQ_FLAG RW  bit[3:0] write-1-to-clear

module xadc_regs (
    input  wire        clk,
    input  wire        rst,

    // mmio-lite 슬레이브
    input  wire [15:0] addr,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    input  wire        we,
    input  wire        req,
    output reg         ack,

    // CTRL 필드 출력
    output reg  [1:0]  avg_len,
    output reg         irq_en,
    output reg         scan_en,

    // STATUS 입력
    input  wire        busy,
    input  wire [3:0]  ch_valid,

    // 채널별 데이터 입력
    input  wire [11:0] ch_raw_0,
    input  wire [11:0] ch_raw_1,
    input  wire [11:0] ch_raw_2,
    input  wire [11:0] ch_raw_3,
    input  wire [11:0] ch_avg_0,
    input  wire [11:0] ch_avg_1,
    input  wire [11:0] ch_avg_2,
    input  wire [11:0] ch_avg_3,

    // IRQ
    output wire        irq
);

    // IRQ_MASK, IRQ_FLAG 레지스터
    reg [3:0] irq_mask;
    reg [3:0] irq_flag;

    // 인터럽트: 마스크 된 플래그가 있고 irq_en이면 올림
    assign irq = irq_en && |(irq_mask & irq_flag);

    // IRQ 플래그: ch_valid가 오면 세트
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            irq_flag <= 4'b0;
        end else begin
            // 채널 변환 완료 시 세트
            irq_flag <= irq_flag | ch_valid;
            // write-1-to-clear (동시 세트·클리어 가능: 세트 우선)
            if (we && req && addr[7:0] == 8'h2C)
                irq_flag <= (irq_flag | ch_valid) & ~wdata[3:0];
        end
    end

    // ack: req 다음 사이클
    always @(posedge clk or posedge rst) begin
        if (rst) ack <= 1'b0;
        else     ack <= req & ~ack; // 1클럭 ack
    end

    // 쓰기
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            avg_len  <= 2'd0;
            irq_en   <= 1'b0;
            scan_en  <= 1'b0;
            irq_mask <= 4'b0;
        end else if (we && req) begin
            case (addr[7:0])
                8'h00: begin
                    avg_len <= wdata[1:0];
                    irq_en  <= wdata[2];
                    scan_en <= wdata[3];
                end
                8'h28: irq_mask <= wdata[3:0];
                default: ;
            endcase
        end
    end

    // 읽기 (조합)
    always @(*) begin
        case (addr[7:0])
            8'h00: rdata = {28'd0, scan_en, irq_en, avg_len};
            8'h04: rdata = {26'd0, ch_valid, 1'b0, busy};
            8'h08: rdata = {20'd0, ch_raw_0};
            8'h0C: rdata = {20'd0, ch_raw_1};
            8'h10: rdata = {20'd0, ch_raw_2};
            8'h14: rdata = {20'd0, ch_raw_3};
            8'h18: rdata = {20'd0, ch_avg_0};
            8'h1C: rdata = {20'd0, ch_avg_1};
            8'h20: rdata = {20'd0, ch_avg_2};
            8'h24: rdata = {20'd0, ch_avg_3};
            8'h28: rdata = {28'd0, irq_mask};
            8'h2C: rdata = {28'd0, irq_flag};
            default: rdata = 32'hDEAD_BEEF;
        endcase
    end

endmodule

// 자체 점검 FSM (버스 마스터)

    output reg  [31:0] wdata,

    input  wire        ack,

    input  wire        irq,

    output reg  [3:0]  led_pass,
    output reg         led_busy,
    output reg         led_irq

    localparam CTRL     = 16'h0000;
    localparam STATUS   = 16'h0004;
    localparam CH0_RAW  = 16'h0008;
    localparam CH1_RAW  = 16'h000C;
    localparam CH2_RAW  = 16'h0010;
    localparam CH3_RAW  = 16'h0014;
    localparam CH0_AVG  = 16'h0018;
    localparam CH1_AVG  = 16'h001C;
    localparam CH2_AVG  = 16'h0020;
    localparam CH3_AVG  = 16'h0024;
    localparam IRQ_MASK = 16'h0028;
    localparam IRQ_FLAG = 16'h002C;

    // FSM 상태
    localparam S_RESET      = 5'd0;
    localparam S_WR_CTRL    = 5'd1;  // CTRL 쓰기
    localparam S_WAIT_ACK   = 5'd2;  // ack 대기
    localparam S_WAIT_IRQ   = 5'd3;  // IRQ 대기
    localparam S_RD_STATUS  = 5'd4;  // STATUS 읽기

    localparam S_RD_RAW1    = 5'd7;
    localparam S_RD_RAW2    = 5'd8;

    localparam S_RD_AVG0    = 5'd10;

    localparam S_RD_AVG3    = 5'd13;

    localparam S_WR_MASK    = 5'd17; // IRQ_MASK 설정

            addr        <= 16'd0;

                    state <= S_WAIT_ACK;
                    next_state <= S_RD_IFLAG;

                    state <= S_CLR_IFLAG;

                    do_write(IRQ_FLAG, 32'h0000_000F); // write-1-to-clear

                    next_state <= S_WAIT_IRQ2;

                        state <= S_DONE;

                    end
                end

                S_DONE: begin
                    led_pass <= pass_flags;
                    led_busy <= 1'b0;
                    state    <= S_IDLE;
                end

                S_IDLE: begin
                    // 완료. LED 유지
                end

                default: state <= S_RESET;
            endcase
        end
    end

// ==============================================================
// xadc_drp_model: DRP 인터페이스 모의 모델
// - 32클럭마다 EOC 발생
// - den → 2클럭 후 drdy + 채널별 고정 데이터 반환
// ==============================================================
module xadc_drp_model (
    input  wire        dclk,
    input  wire        rst,
    input  wire        den,
    input  wire [6:0]  daddr,
    output reg         drdy,
    output reg  [15:0] do,
    output reg         eoc
);
    reg [5:0]  eoc_cnt;
    reg        den_d1, den_d2;

    // EOC: 32클럭마다 1클럭 펄스
    always @(posedge dclk or posedge rst) begin
        if (rst) begin
            eoc_cnt <= 6'd0;
            eoc     <= 1'b0;
        end else begin
            eoc_cnt <= eoc_cnt + 1'b1;
            eoc     <= (eoc_cnt == 6'd31);
        end
    end

    // drdy: den 후 2클럭 지연
    always @(posedge dclk or posedge rst) begin
        if (rst) begin
            den_d1 <= 1'b0;
            den_d2 <= 1'b0;
            drdy   <= 1'b0;
            do     <= 16'd0;
        end else begin
            den_d1 <= den;
            den_d2 <= den_d1;
            drdy   <= den_d2;
            if (den_d2) begin
                // 채널별 고정 테스트 데이터 (상위 12비트 유효)
                // daddr: 0x14→ch4, 0x15→ch5, 0x1C→ch12, 0x1D→ch13
                case (daddr)
                    7'h14: do <= 16'hABC0; // ch0 raw = 0xABC
                    7'h15: do <= 16'hDEF0; // ch1 raw = 0xDEF
                    7'h1C: do <= 16'h1230; // ch2 raw = 0x123
                    7'h1D: do <= 16'h4560; // ch3 raw = 0x456
                    default: do <= 16'h0000;
                endcase
            end
        end
    end
endmodule

// tb_xadc_hub.v
// xadc_hub 통합 테스트
// - xadc_drp_model(시뮬레이션용)이 xadc_hub.v 안에 ifdef SIMULATION으로 삽입
// - mmio-lite 버스를 직접 구동해 레지스터 R/W 검증
// 통과: "PASS: xadc_hub" / 실패: "FAIL: xadc_hub"
//
// 테스트 순서:
//  T1. CTRL 쓰기(scan_en=1, avg_len=1)
//  T2. IRQ 대기 (최대 2000클럭)
//  T3. CH0_RAW 읽기 → 0xABC 확인
//  T4. CH1_RAW 읽기 → 0xDEF 확인
//  T5. CH2_RAW 읽기 → 0x123 확인
//  T6. CH3_RAW 읽기 → 0x456 확인
//  T7. IRQ_FLAG 읽기 → 0xF 확인
//  T8. IRQ_FLAG 클리어 → 0x0 확인
//  T9. AVG 읽기 (1샘플 모드이므로 raw와 동일 허용)

`timescale 1ns/1ps
`define SIMULATION 1

module tb_xadc_hub;

    reg         clk, rst;
    reg  [15:0] addr;
    reg  [31:0] wdata;
    wire [31:0] rdata;
    reg         we, req;
    wire        ack;
    wire        irq;

    // --- DUT (SIMULATION 정의 → 내부에 xadc_drp_model 사용) ---
    xadc_hub u_dut (
        .clk     (clk),
        .rst     (rst),
        .addr    (addr),
        .wdata   (wdata),
        .rdata   (rdata),
        .we      (we),
        .req     (req),
        .ack     (ack),
        .vauxp4  (1'b0), .vauxn4  (1'b0),
        .vauxp5  (1'b0), .vauxn5  (1'b0),
        .vauxp12 (1'b0), .vauxn12 (1'b0),
        .vauxp13 (1'b0), .vauxn13 (1'b0),
        .irq     (irq)
    );

    // 클럭 10ns
    initial clk = 0;
    always #5 clk = ~clk;

    integer fail_cnt;
    integer wait_cnt;
    reg [31:0] rd_val;

    // mmio-lite 쓰기 태스크
    task bus_write;
        input [15:0] a;
        input [31:0] d;
        begin
            @(posedge clk);
            addr  = a; wdata = d; we = 1; req = 1;
            @(posedge clk);
            while (!ack) @(posedge clk);
            we = 0; req = 0;
            @(posedge clk);
        end
    endtask

    // mmio-lite 읽기 태스크
    task bus_read;
        input [15:0] a;
        output [31:0] d;
        begin
            @(posedge clk);
            addr = a; wdata = 32'd0; we = 0; req = 1;
            @(posedge clk);
            while (!ack) @(posedge clk);
            d   = rdata;
            req = 0;
            @(posedge clk);
        end
    endtask

    // 체크 매크로
    task check_eq;
        input [31:0] got;
        input [31:0] exp;
        input [63:0] label;
        begin
            if (got !== exp) begin
                $display("FAIL: %s got=0x%08X exp=0x%08X", label, got, exp);
                fail_cnt = fail_cnt + 1;
            end else begin
                $display("PASS: %s = 0x%08X", label, got);
            end
        end
    endtask

    initial begin
        $dumpfile("tb_xadc_hub.vcd");
        $dumpvars(0, tb_xadc_hub);

        rst = 1; we = 0; req = 0; addr = 0; wdata = 0; fail_cnt = 0;
        repeat (4) @(posedge clk);
        rst = 0;
        repeat (2) @(posedge clk);

        // T1: CTRL 쓰기: scan_en=1(bit3), irq_en=1(bit2), avg_len=0(bypass)
        $display("--- T1: CTRL 쓰기 ---");
        bus_write(16'h0000, 32'h0000_000C); // bit3=scan_en bit2=irq_en avg_len=0

        // T2: IRQ 대기
        $display("--- T2: IRQ 대기 ---");
        wait_cnt = 0;
        while (!irq && wait_cnt < 3000) begin
            @(posedge clk);
            wait_cnt = wait_cnt + 1;
        end
        if (!irq) begin
            $display("FAIL: IRQ timeout after %0d clocks", wait_cnt);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("PASS: IRQ asserted at clock %0d", wait_cnt);
        end

        // T3~T6: CH_RAW 읽기
        $display("--- T3: CH0_RAW 읽기 ---");
        bus_read(16'h0008, rd_val);
        check_eq(rd_val, 32'h0000_0ABC, "CH0_RAW");

        $display("--- T4: CH1_RAW 읽기 ---");
        bus_read(16'h000C, rd_val);
        check_eq(rd_val, 32'h0000_0DEF, "CH1_RAW");

        $display("--- T5: CH2_RAW 읽기 ---");
        bus_read(16'h0010, rd_val);
        check_eq(rd_val, 32'h0000_0123, "CH2_RAW");

        $display("--- T6: CH3_RAW 읽기 ---");
        bus_read(16'h0014, rd_val);
        check_eq(rd_val, 32'h0000_0456, "CH3_RAW");

        // T7: IRQ_FLAG 읽기
        $display("--- T7: IRQ_FLAG 읽기 ---");
        bus_read(16'h002C, rd_val);
        check_eq(rd_val & 32'hF, 32'h0000_000F, "IRQ_FLAG");

        // T8: IRQ_FLAG 클리어
        $display("--- T8: IRQ_FLAG 클리어 ---");
        bus_write(16'h002C, 32'h0000_000F);
        bus_read(16'h002C, rd_val);
        check_eq(rd_val & 32'hF, 32'h0000_0000, "IRQ_FLAG_clear");

        // T9: CH0_AVG 읽기 (bypass 모드이므로 raw와 동일해야 함)
        $display("--- T9: CH0_AVG 읽기 (bypass) ---");
        bus_read(16'h0018, rd_val);
        // avg_len=0 bypass: avg_out = 최신 sample_data = raw
        // 주의: avg는 sample_valid 엣지에서 갱신되므로 직전 raw와 같아야 함
        if (rd_val[11:0] !== 12'hABC) begin
            $display("WARN: CH0_AVG=0x%03X (exp=0xABC, 웜업 중일 수 있음)", rd_val[11:0]);
        end else begin
            $display("PASS: CH0_AVG = 0x%03X", rd_val[11:0]);
        end

        // 결과 요약
        repeat (4) @(posedge clk);
        if (fail_cnt == 0)
            $display("PASS: xadc_hub");
        else
            $display("FAIL: xadc_hub (%0d errors)", fail_cnt);

        $finish;
    end

endmodule
