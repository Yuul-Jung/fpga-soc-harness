센서 허브 IP 자체 점검: HC-SR04 초음파 + DHT11 온습도 + I2C(MPU-6050) 모델 테스트벤치

세 종류 센서 프로토콜(HC-SR04 에코 타이밍, DHT11 단선 40비트, I2C 레지스터 읽기)을 하나의 MMIO 창에 묶은 센서 허브 IP. 초음파·온습도·가속도계 모델 테스트벤치가 각 센서의 프로토콜을 흉내 내어 허브 레지스터 값을 PASS/FAIL로 검사한다. 하드웨어 없이 시뮬레이션만으로 세 FSM의 타이밍 정확성을 확인하는 T2 패키지.

1. 목표와 범위

목표

  • HC-SR04 초음파 거리 측정 FSM(트리거 10 μs, 에코 타임 → 거리 cm)
  • DHT11 단선 프로토콜 FSM(시작 신호 18 ms, 40비트 수신, 체크섬 검증)
  • I2C 마스터 FSM으로 MPU-6050 단일 레지스터 읽기
  • 세 센서를 단일 MMIO 창(0x1002_0000)에 통합하고 레지스터로 제어·조회
  • 각 프로토콜 모델 테스트벤치로 PASS/FAIL 자체 점검

범위 밖

  • 실제 하드웨어 센서 구동(S3 보드 검증은 사용자가 수행)
  • 다중 I2C 슬레이브 중재
  • DHT22 호환성
  • DMA 또는 인터럽트 기반 판독

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite (자체, 단독 실행용)
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획이 패키지는 단독 시뮬레이션·보드 확인용으로 자체 mmio-lite 버스(0x1000_xxxx~0x1002_xxxx)를 쓴다. 동결 맵(mbv-frozen-v2) 이식 계획: (1) GPIO·UART는 동결 맵 [00]/[02] AMD IP를 그대로 쓰고 드라이버를 맞춘다. (2) sensor_hub는 동결 맵 [16](0x4016_0000)에 IP 표준(64KB 창, 0xFFF0 ID/VER)으로 감싸서 AXI4-Lite 슬레이브로 등록한다. (3) demo_master 대신 MicroBlaze V + 폴링 드라이버(C) 또는 인터럽트(irq→INTC)로 교체한다. HC-SR04 ECHO 5V→3.3V 레벨 변환 필수. DHT11·SDA·SCL 풀업 4.7 kΩ 필요.

// hcsr04_ctrl.v — HC-SR04 초음파 거리 센서 FSM
// 출처: mmio-sensor-hub-checker 캡스톤 자체 RTL (Verilog-2001)
// 100 MHz 기준: TRIG 10 μs = 1000 cycles, 거리(cm) = echo_cycles / 5800
// ECHO 핀 5V 출력 → Basys3 Pmod 3.3V 레벨 변환 필수

`timescale 1ns/1ps
module hcsr04_ctrl #(
    parameter CLK_HZ     = 100_000_000,
    parameter TRIG_CYC   = 1000,          // 10 μs
    parameter TIMEOUT_CYC= 3_800_000      // 38 ms max echo
)(
    input  wire       clk,
    input  wire       rst_n,
    // MMIO 제어
    input  wire       start,              // 1-cycle 펄스
    // 센서 핀
    output reg        trig_o,
    input  wire       echo_i,             // 레벨 변환 후 입력
    // 결과
    output reg  [15:0] dist_cm,
    output reg        busy,
    output reg        done,
    output reg        timeout
);

    // 에코 카운터 비트 폭: log2(TIMEOUT_CYC) ≈ 22비트
    reg [22:0] cnt;
    reg [22:0] echo_cnt;

    // FSM 상태
    localparam S_IDLE    = 3'd0,
               S_TRIG    = 3'd1,
               S_WAIT_HI = 3'd2,   // 에코 High 대기
               S_MEASURE = 3'd3,   // 에코 High 구간 계측
               S_DONE    = 3'd4;

    reg [2:0] state;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state    <= S_IDLE;
            trig_o   <= 1'b0;
            busy     <= 1'b0;
            done     <= 1'b0;
            timeout  <= 1'b0;
            dist_cm  <= 16'd0;
            cnt      <= 23'd0;
            echo_cnt <= 23'd0;
        end else begin
            done <= 1'b0; // 1-cycle 펄스

            case (state)
                S_IDLE: begin
                    trig_o  <= 1'b0;
                    busy    <= 1'b0;
                    if (start) begin
                        trig_o  <= 1'b1;
                        busy    <= 1'b1;
                        timeout <= 1'b0;
                        cnt     <= 23'd0;
                        state   <= S_TRIG;
                    end
                end

                S_TRIG: begin
                    if (cnt == TRIG_CYC - 1) begin
                        trig_o <= 1'b0;
                        cnt    <= 23'd0;
                        state  <= S_WAIT_HI;
                    end else begin
                        cnt <= cnt + 1'b1;
                    end
                end

                S_WAIT_HI: begin
                    if (echo_i) begin
                        echo_cnt <= 23'd0;
                        state    <= S_MEASURE;
                    end else if (cnt == TIMEOUT_CYC - 1) begin
                        timeout <= 1'b1;
                        state   <= S_DONE;
                    end else begin
                        cnt <= cnt + 1'b1;
                    end
                end

                S_MEASURE: begin
                    if (!echo_i) begin
                        // 에코 종료 → 거리 계산
                        dist_cm <= echo_cnt[22:0] / 5800;
                        state   <= S_DONE;
                    end else if (echo_cnt == TIMEOUT_CYC - 1) begin
                        timeout <= 1'b1;
                        state   <= S_DONE;
                    end else begin
                        echo_cnt <= echo_cnt + 1'b1;
                    end
                end

                S_DONE: begin
                    done  <= 1'b1;
                    busy  <= 1'b0;
                    state <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end

endmodule

// dht11_ctrl.v — DHT11 온습도 센서 단선 FSM
// 출처: mmio-sensor-hub-checker 캡스톤 자체 RTL (Verilog-2001)
// 100 MHz 기준: 18 ms start = 1_800_000 cycles
// 비트 판별: 50 μs Low 후 High 구간 < 40 μs → 0, > 40 μs → 1
// (100 MHz: 40 μs = 4000 cycles 임계값)

`timescale 1ns/1ps
module dht11_ctrl #(
    parameter START_CYC    = 1_800_000,   // 18 ms
    parameter RELEASE_CYC  =    50_000,   // 0.5 ms pull-up wait
    parameter BIT_THRESH   =     4_000    // 40 μs threshold
)(
    input  wire       clk,
    input  wire       rst_n,
    input  wire       start,
    // 단선 IO (tri-state: oe=1이면 출력 0, oe=0이면 입력)
    output wire       dht_oe,   // 1: drive low
    input  wire       dht_in,   // 풀업 후 입력
    // 결과
    output reg  [7:0] humidity,
    output reg  [7:0] temperature,
    output reg        busy,
    output reg        done,
    output reg        chk_ok
);

    reg [20:0] cnt;
    reg [5:0]  bit_idx;   // 0..39
    reg [39:0] shift_reg;
    reg [15:0] hi_cnt;
    reg [8:0]  chksum;

    // FSM
    localparam S_IDLE      = 4'd0,
               S_START_LOW = 4'd1,
               S_RELEASE   = 4'd2,
               S_RESP_LOW  = 4'd3,  // DHT11 응답 Low(80 μs)
               S_RESP_HIGH = 4'd4,  // DHT11 응답 High(80 μs)
               S_BIT_LOW   = 4'd5,  // 각 비트 시작 Low(50 μs)
               S_BIT_HIGH  = 4'd6,  // 비트 High 구간 계측
               S_DONE      = 4'd7,
               S_ERROR     = 4'd8;

    reg [3:0] state;
    reg       driving_low;  // oe 레지스터

    assign dht_oe = driving_low;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state       <= S_IDLE;
            driving_low <= 1'b0;
            busy        <= 1'b0;
            done        <= 1'b0;
            chk_ok      <= 1'b0;
            humidity    <= 8'd0;
            temperature <= 8'd0;
            cnt         <= 21'd0;
            bit_idx     <= 6'd0;
            shift_reg   <= 40'd0;
            hi_cnt      <= 16'd0;
        end else begin
            done <= 1'b0;

            case (state)
                S_IDLE: begin
                    driving_low <= 1'b0;
                    busy        <= 1'b0;
                    if (start) begin
                        driving_low <= 1'b1;
                        busy        <= 1'b1;
                        cnt         <= 21'd0;
                        state       <= S_START_LOW;
                    end
                end

                S_START_LOW: begin
                    if (cnt == START_CYC - 1) begin
                        driving_low <= 1'b0;
                        cnt         <= 21'd0;
                        state       <= S_RELEASE;
                    end else
                        cnt <= cnt + 1'b1;
                end

                S_RELEASE: begin
                    if (cnt == RELEASE_CYC - 1) begin
                        cnt   <= 21'd0;
                        state <= S_RESP_LOW;
                    end else
                        cnt <= cnt + 1'b1;
                end

                S_RESP_LOW: begin
                    // DHT11이 Low를 내림: 최대 100 μs 기다림
                    if (!dht_in) begin
                        cnt   <= 21'd0;
                        state <= S_RESP_HIGH;
                    end else if (cnt == 10_000) begin
                        state <= S_ERROR;
                    end else
                        cnt <= cnt + 1'b1;
                end

                S_RESP_HIGH: begin
                    // DHT11이 High: 80 μs 후 떨어짐
                    if (dht_in) begin
                        cnt   <= 21'd0;
                        state <= S_BIT_LOW;
                        bit_idx <= 6'd0;
                    end else
                        cnt <= cnt + 1'b1;
                end

                S_BIT_LOW: begin
                    // 비트 시작 Low(50 μs)
                    if (dht_in) begin
                        hi_cnt <= 16'd0;
                        state  <= S_BIT_HIGH;
                    end else if (cnt == 10_000) begin
                        state <= S_ERROR;
                    end else
                        cnt <= cnt + 1'b1;
                end

                S_BIT_HIGH: begin
                    if (!dht_in) begin
                        // High 구간 종료
                        shift_reg <= {shift_reg[38:0],
                                      (hi_cnt > BIT_THRESH) ? 1'b1 : 1'b0};
                        bit_idx   <= bit_idx + 1'b1;
                        cnt       <= 21'd0;
                        if (bit_idx == 6'd39)
                            state <= S_DONE;
                        else
                            state <= S_BIT_LOW;
                    end else begin
                        hi_cnt <= hi_cnt + 1'b1;
                        if (hi_cnt == 16'hFFFF)
                            state <= S_ERROR;
                    end
                end

                S_DONE: begin
                    // 체크섬: 하위 8비트 합
                    chksum = shift_reg[39:32] + shift_reg[31:24] +
                             shift_reg[23:16] + shift_reg[15:8];
                    chk_ok <= (chksum[7:0] == shift_reg[7:0]);
                    humidity    <= shift_reg[39:32];
                    temperature <= shift_reg[23:16];
                    done        <= 1'b1;
                    busy        <= 1'b0;
                    state       <= S_IDLE;
                end

                S_ERROR: begin
                    chk_ok  <= 1'b0;
                    done    <= 1'b1;
                    busy    <= 1'b0;
                    state   <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end

endmodule

// i2c_master.v — I2C 마스터 FSM (단일 레지스터 읽기)
// 출처: mmio-sensor-hub-checker 캡스톤 자체 RTL (Verilog-2001)
// 100 MHz, 400 kHz = 250 cycles / 비트. half = 125 cycles
// 동작: START → ADDR(W)+ACK → REG_ADDR+ACK → RESTART → ADDR(R)+ACK → DATA+NACK → STOP

`timescale 1ns/1ps
module i2c_master #(
    parameter CLK_DIV = 125    // 100 MHz / 400 kHz / 2
)(
    input  wire       clk,
    input  wire       rst_n,
    // MMIO 제어
    input  wire       start,       // 1-cycle 펄스
    input  wire [6:0] dev_addr,    // 7비트 슬레이브 주소
    input  wire [7:0] reg_addr,    // 읽을 레지스터 주소
    // I2C 핀 (오픈 드레인: oe=1 → SDA/SCL = 0)
    output reg        scl_oe,
    output reg        sda_oe,
    input  wire       sda_in,
    // 결과
    output reg  [7:0] rdata,
    output reg        busy,
    output reg        done,
    output reg        error
);

    // SCL 반주기 카운터
    reg [7:0]  clk_cnt;
    reg        scl_phase;    // 0=low, 1=high

    // 비트 시프트 레지스터와 비트 카운터
    reg [7:0]  tx_byte;
    reg [2:0]  bit_cnt;
    reg        ack_bit;

    // FSM 상태
    localparam S_IDLE      = 5'd0,
               S_START1    = 5'd1,  // SDA low
               S_START2    = 5'd2,  // SCL low
               S_ADDR_W    = 5'd3,  // ADDR(W) 8비트 (addr<<1 | 0)
               S_ACK1      = 5'd4,
               S_REG       = 5'd5,  // REG_ADDR 8비트
               S_ACK2      = 5'd6,
               S_RESTART1  = 5'd7,  // SCL high
               S_RESTART2  = 5'd8,  // SDA high
               S_RESTART3  = 5'd9,  // SDA low (START)
               S_RESTART4  = 5'd10, // SCL low
               S_ADDR_R    = 5'd11, // ADDR(R) 8비트 (addr<<1 | 1)
               S_ACK3      = 5'd12,
               S_DATA      = 5'd13, // DATA 8비트 읽기
               S_NACK      = 5'd14, // 마스터 NACK
               S_STOP1     = 5'd15, // SCL high
               S_STOP2     = 5'd16, // SDA high
               S_DONE      = 5'd17,
               S_ERROR     = 5'd18;

    reg [4:0] state;

    // SCL 토글 도우미: clk_cnt가 CLK_DIV에 도달하면 1 반주기 완료
    wire half_done = (clk_cnt == CLK_DIV - 1);

    task next_bit;
        input [4:0] next_state;
        begin
            bit_cnt <= bit_cnt - 1'b1;
            if (bit_cnt == 3'd0)
                state <= next_state;
        end
    endtask

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state     <= S_IDLE;
            scl_oe    <= 1'b0;  // SCL high (open drain released)
            sda_oe    <= 1'b0;  // SDA high
            busy      <= 1'b0;
            done      <= 1'b0;
            error     <= 1'b0;
            rdata     <= 8'd0;
            clk_cnt   <= 8'd0;
            scl_phase <= 1'b0;
            bit_cnt   <= 3'd7;
            tx_byte   <= 8'd0;
            ack_bit   <= 1'b0;
        end else begin
            done <= 1'b0;

            case (state)
                S_IDLE: begin
                    scl_oe <= 1'b0; sda_oe <= 1'b0;
                    busy   <= 1'b0;
                    if (start) begin
                        busy    <= 1'b1;
                        error   <= 1'b0;
                        clk_cnt <= 8'd0;
                        state   <= S_START1;
                    end
                end

                // START: SDA low while SCL high
                S_START1: begin
                    sda_oe <= 1'b1;  // SDA = 0
                    if (half_done) begin clk_cnt <= 8'd0; state <= S_START2; end
                    else clk_cnt <= clk_cnt + 1'b1;
                end
                S_START2: begin
                    scl_oe <= 1'b1;  // SCL = 0
                    if (half_done) begin
                        clk_cnt <= 8'd0;
                        tx_byte <= {dev_addr, 1'b0};  // WRITE
                        bit_cnt <= 3'd7;
                        state   <= S_ADDR_W;
                    end else clk_cnt <= clk_cnt + 1'b1;
                end

                // 8비트 전송 (MSB first): SCL low→high→low 1 cycle = 2 half_done
                S_ADDR_W, S_REG, S_ADDR_R: begin
                    if (!scl_phase) begin
                        // SCL low 구간: SDA 세팅
                        sda_oe <= ~tx_byte[7];
                        if (half_done) begin
                            scl_oe    <= 1'b0;  // SCL high
                            scl_phase <= 1'b1;
                            clk_cnt   <= 8'd0;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end else begin
                        // SCL high 구간: SDA 유지
                        if (half_done) begin
                            scl_oe    <= 1'b1;  // SCL low
                            scl_phase <= 1'b0;
                            tx_byte   <= {tx_byte[6:0], 1'b0};
                            if (bit_cnt == 3'd0) begin
                                sda_oe  <= 1'b0;  // SDA 해제 (ACK 수신용)
                                bit_cnt <= 3'd7;
                                state   <= (state == S_ADDR_W) ? S_ACK1 :
                                           (state == S_REG   ) ? S_ACK2 : S_ACK3;
                            end else
                                bit_cnt <= bit_cnt - 1'b1;
                            clk_cnt <= 8'd0;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end
                end

                // ACK 수신: SCL high 중에 SDA 읽기
                S_ACK1, S_ACK2, S_ACK3: begin
                    if (!scl_phase) begin
                        if (half_done) begin
                            scl_oe    <= 1'b0;
                            scl_phase <= 1'b1;
                            clk_cnt   <= 8'd0;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end else begin
                        ack_bit <= sda_in;  // 0=ACK, 1=NACK
                        if (half_done) begin
                            scl_oe    <= 1'b1;
                            scl_phase <= 1'b0;
                            clk_cnt   <= 8'd0;
                            if (sda_in) begin
                                state <= S_ERROR;  // NACK
                            end else begin
                                case (state)
                                    S_ACK1: begin
                                        tx_byte <= reg_addr;
                                        state   <= S_REG;
                                    end
                                    S_ACK2: state <= S_RESTART1;
                                    S_ACK3: begin
                                        rdata   <= 8'd0;
                                        bit_cnt <= 3'd7;
                                        sda_oe  <= 1'b0;  // 입력 모드
                                        state   <= S_DATA;
                                    end
                                    default: state <= S_ERROR;
                                endcase
                            end
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end
                end

                // RESTART: SCL high → SDA high → SDA low → SCL low
                S_RESTART1: begin
                    scl_oe <= 1'b0;
                    if (half_done) begin clk_cnt <= 8'd0; state <= S_RESTART2; end
                    else clk_cnt <= clk_cnt + 1'b1;
                end
                S_RESTART2: begin
                    sda_oe <= 1'b0;
                    if (half_done) begin clk_cnt <= 8'd0; state <= S_RESTART3; end
                    else clk_cnt <= clk_cnt + 1'b1;
                end
                S_RESTART3: begin
                    sda_oe <= 1'b1;  // SDA low
                    if (half_done) begin clk_cnt <= 8'd0; state <= S_RESTART4; end
                    else clk_cnt <= clk_cnt + 1'b1;
                end
                S_RESTART4: begin
                    scl_oe <= 1'b1;  // SCL low
                    if (half_done) begin
                        clk_cnt <= 8'd0;
                        tx_byte <= {dev_addr, 1'b1};  // READ
                        bit_cnt <= 3'd7;
                        state   <= S_ADDR_R;
                    end else clk_cnt <= clk_cnt + 1'b1;
                end

                // DATA 읽기 8비트 MSB first
                S_DATA: begin
                    if (!scl_phase) begin
                        if (half_done) begin
                            scl_oe    <= 1'b0;
                            scl_phase <= 1'b1;
                            clk_cnt   <= 8'd0;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end else begin
                        rdata <= {rdata[6:0], sda_in};  // MSB first shift
                        if (half_done) begin
                            scl_oe    <= 1'b1;
                            scl_phase <= 1'b0;
                            clk_cnt   <= 8'd0;
                            if (bit_cnt == 3'd0) begin
                                sda_oe  <= 1'b1;  // NACK: SDA high
                                state   <= S_NACK;
                            end else
                                bit_cnt <= bit_cnt - 1'b1;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end
                end

                // NACK: SCL high 한 사이클
                S_NACK: begin
                    if (!scl_phase) begin
                        if (half_done) begin
                            scl_oe    <= 1'b0;
                            scl_phase <= 1'b1;
                            clk_cnt   <= 8'd0;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end else begin
                        if (half_done) begin
                            scl_oe    <= 1'b1;
                            scl_phase <= 1'b0;
                            sda_oe    <= 1'b1;  // SDA low for STOP prep
                            clk_cnt   <= 8'd0;
                            state     <= S_STOP1;
                        end else clk_cnt <= clk_cnt + 1'b1;
                    end
                end

                // STOP: SCL high → SDA high
                S_STOP1: begin
                    scl_oe <= 1'b0;  // SCL high
                    if (half_done) begin clk_cnt <= 8'd0; state <= S_STOP2; end
                    else clk_cnt <= clk_cnt + 1'b1;
                end
                S_STOP2: begin
                    sda_oe <= 1'b0;  // SDA high
                    if (half_done) begin clk_cnt <= 8'd0; state <= S_DONE; end
                    else clk_cnt <= clk_cnt + 1'b1;
                end

                S_DONE: begin
                    done  <= 1'b1;
                    busy  <= 1'b0;
                    state <= S_IDLE;
                end

                S_ERROR: begin
                    // STOP 시도 후 오류 반환
                    scl_oe <= 1'b0;
                    sda_oe <= 1'b0;
                    error  <= 1'b1;
                    done   <= 1'b1;
                    busy   <= 1'b0;
                    state  <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end

    reg [7:0] reg_addr;

        input [31:0] exp;
        input [127:0] tag;

            if (got !== exp) begin
                $display("FAIL %s: got=%0d exp=%0d", tag, got, exp);
                fail = fail + 1;
            end
        end
    endtask

    // 슬레이브 모델: I2C 비트 스트림을 ACK/NACK 패턴으로 처리
    // 간단 모델: SCL 상승 엣지를 세어 ACK 위치에서 SDA를 내린다
    // T1: 모든 ACK=0 (ACK), 데이터 바이트 0xAB

    reg        slave_nack;   // T2용

        .clk     (clk),
        .rst_n   (rst_n),
        .addr    (addr),
        .wdata   (wdata),
        .we      (we),
        .sel     (sel),
        .rdata   (rdata),
        .trig_o  (trig_o),
        .echo_i  (echo_i),
        .dht_oe  (dht_oe),
        .dht_in  (dht_in),
        .scl_oe  (scl_oe),
        .sda_oe  (sda_oe),
        .sda_in  (sda_in)
    );

    initial clk = 0;
    always #5 clk = ~clk;

    integer fail = 0;

    // MMIO 쓰기
    task mmio_write;
        input [31:0] a, d;
        begin
            @(posedge clk);
            addr = a; wdata = d; we = 1'b1; sel = 1'b1;
            @(posedge clk);
            we = 1'b0; sel = 1'b0;
        end
    endtask

    // MMIO 읽기
    task mmio_read;
        input [31:0] a;
        output [31:0] d;
        begin
            @(posedge clk);
            addr = a; we = 1'b0; sel = 1'b1;
            @(posedge clk);
            d = rdata;
            sel = 1'b0;
        end
    endtask

    task check_eq;
        input [31:0] got;
        input [31:0] exp;
        input [127:0] tag;
        begin
            if (got !== exp) begin
                $display("FAIL %s: got=0x%08X exp=0x%08X", tag, got, exp);
                fail = fail + 1;
            end
        end
    endtask

    reg [31:0] rdval;

    // ---- HC-SR04 모델 ----
    // start 직후 trig_o 상승 감지, 1200 cycles 뒤 에코 2000 cycles 보냄
    // dist = 2000000/5800 ≈ 344 (10 μs 트리거 후 20 ms 에코 = 344 cm)
    // 단, TRIG_CYC=1000이므로 trig_o 뒤 1200 cycles 대기 후 echo_i=1
    // 실제: 2000 cycles echo → dist = 2000/5800 = 0 (너무 짧음)
    // 의미 있는 테스트: echo 175000 cycles → 30 cm
    task hcsr04_model;
        begin
            wait(trig_o === 1'b1);
            wait(trig_o === 1'b0);
            repeat(50) @(posedge clk);  // 대기
            echo_model = 1'b1;
            repeat(175000) @(posedge clk);
            echo_model = 1'b0;
        end
    endtask

    // ---- DHT11 모델 ----
    // humid=60(0x3C), temp=25(0x19), chk=0x55
    task dht11_model;
        reg [39:0] data;
        integer i;
        begin
            data = 40'h3C_00_19_00_55;
            wait(dht_oe === 1'b0);  // DUT가 release
            repeat(100) @(posedge clk);
            dht_model = 1'b0;
            repeat(8000) @(posedge clk);  // 응답 Low 80 μs
            dht_model = 1'b1;
            repeat(8000) @(posedge clk);  // 응답 High 80 μs
            for (i = 39; i >= 0; i = i - 1) begin
                dht_model = 1'b0;
                repeat(5000) @(posedge clk);
                dht_model = 1'b1;
                if (data[i])
                    repeat(7000) @(posedge clk);
                else
                    repeat(2800) @(posedge clk);
            end
            dht_model = 1'b0;
        end
    endtask

    // ---- I2C 슬레이브 모델 (간단 ACK + 0xAB) ----
    reg prev_scl_b;
    reg [3:0] bit_b;
    reg [1:0] bphase_b;
    reg       in_data_b;

    wire scl_bus = ~scl_oe;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            prev_scl_b <= 1'b1; bit_b <= 0; bphase_b <= 0;
            sda_model  <= 1'b1; in_data_b <= 0;
        end else begin
            prev_scl_b <= scl_bus;
            if (!prev_scl_b && scl_bus) begin
                if (bit_b == 4'd8) begin
                    if (in_data_b) sda_model <= 1'b1;
                    else           sda_model <= 1'b0;
                    bit_b <= 0;
                    if (!in_data_b) bphase_b <= bphase_b + 1'b1;
                    if (bphase_b == 2'd2) in_data_b <= 1'b1;
                end else begin
                    if (in_data_b)
                        sda_model <= (8'hAB >> (7 - bit_b)) & 1'b1;
                    else
                        sda_model <= 1'b1;
                    bit_b <= bit_b + 1'b1;
                end
            end else if (prev_scl_b && !scl_bus) begin
                if (bit_b == 0 && !in_data_b) sda_model <= 1'b1;
            end
        end
    end

    // ---- 메인 테스트 ----
    initial begin
        rst_n      = 0; we = 0; sel = 0;
        addr       = 0; wdata = 0;
        echo_model = 0; dht_model = 1; sda_model = 1;
        @(posedge clk); @(posedge clk);
        rst_n = 1;

        // ---- T1: HC-SR04 ----
        fork
            hcsr04_model;
            mmio_write(32'h1002_0000, 32'h1); // HCSR04_CTRL start
        join

        // 완료 대기
        begin : wait_hc
            integer i;
            for (i = 0; i < 50; i = i + 1) begin
                mmio_read(32'h1002_0004, rdval);
                if (rdval[1]) disable wait_hc;
                repeat(100) @(posedge clk);
            end
        end

        mmio_read(32'h1002_0008, rdval);
        check_eq(rdval[15:0], 16'd30, "T1_hcsr04_30cm");

        // ---- T2: DHT11 ----
        fork
            dht11_model;
            mmio_write(32'h1002_0010, 32'h1); // DHT11_CTRL start
        join

        begin : wait_dht
            integer j;
            for (j = 0; j < 100; j = j + 1) begin
                mmio_read(32'h1002_0014, rdval);
                if (rdval[1]) disable wait_dht;
                repeat(10000) @(posedge clk);
            end
        end

        mmio_read(32'h1002_0018, rdval);
        check_eq(rdval[7:0], 8'd60, "T2_dht11_humid");
        mmio_read(32'h1002_001C, rdval);
        check_eq(rdval[7:0], 8'd25, "T2_dht11_temp");
        mmio_read(32'h1002_0014, rdval);
        check_eq(rdval[2], 1'b1,   "T2_dht11_chk_ok");

        // ---- T3: I2C ----
        // I2C_CMD: [0]=start, [7:1]=0x68=0b110_1000 → wdata[7:1]=0x68
        // wdata = {reg_addr(0x3B)[15:8], dev_addr(0x68)[7:1], start[0]}
        //       = 0x00_3B_D1 → wdata = 32'h003B_D1
        // 실제: wdata[7:1] = 7'h68, wdata[0]=1, wdata[15:8]=8'h3B
        mmio_write(32'h1002_0020, 32'h003B_D1); // I2C_CMD

        begin : wait_i2c
            integer k;
            for (k = 0; k < 200; k = k + 1) begin
                mmio_read(32'h1002_0024, rdval);
                if (rdval[1]) disable wait_i2c;
                repeat(1000) @(posedge clk);
            end
        end

        mmio_read(32'h1002_0028, rdval);
        check_eq(rdval[7:0], 8'hAB, "T3_i2c_rdata");
        mmio_read(32'h1002_0024, rdval);
        check_eq(rdval[2], 1'b0,   "T3_i2c_no_error");

        if (fail == 0)
            $display("PASS tb_sensor_hub: T1(HC-SR04) T2(DHT11) T3(I2C) all checks OK");
        else
            $display("FAIL tb_sensor_hub: %0d check(s) failed", fail);

        $finish;
    end

    initial begin
        #800_000_000;
        $display("FAIL tb_sensor_hub: simulation timeout");
        $finish;
    end

endmodule

  • HC-SR04 ECHO는 5V 출력이므로 레벨 변환 없이 Basys3 Pmod에 직결 시 FPGA I/O 손상 위험

  • DHT11 측정 최소 간격 1초. demo_master가 너무 빠르게 재트리거하면 비트 오류 발생
