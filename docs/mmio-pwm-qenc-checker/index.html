MMIO PWM + 직교 엔코더 자체 점검

MMIO-lite 버스에 2채널 PWM IP와 4× 직교 엔코더 카운터 IP를 연결하고, 시뮬레이션 마스터 FSM으로 주요 기능(듀티비, 위치 카운팅)을 자체 점검한다. IP 표준 Rev.1 관리 레지스터(ID/VER/IRQ) 포함.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: not-tested

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

목표계통도모듈 계층메모리맵외부 부품·전기기술 트리실행 소스검증참고·위험

1. 목표와 범위

목표

  • IP 표준 Rev.1(관리 레지스터 0xFFF0~FFFC)을 따르는 2채널 PWM IP 설계
  • 4× 직교 엔코더 카운터 IP(2단 동기화, Gray 코드 FSM) 설계
  • MMIO-lite 버스 주소 디코더로 GPIO/UART/PWM/QDEC 4개 슬레이브 연결
  • 시뮬레이션 마스터 FSM으로 PASS/FAIL 자체 점검 테스트벤치 작성
  • yosys 합성 성공, iverilog 시뮬레이션 PASS 확인(S2)

범위 밖

  • 실제 보드에서 모터 연결 또는 엔코더 하드웨어 연결(S3는 사람이 한다)
  • AXI4-Lite 완전 핸드셰이크(이 패키지는 mmio-lite 사용)
  • PID 제어 루프 구현(다음 단계 패키지로 확장)

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다. 시뮬레이션·합성 결과는 CI가 기록한다.

2. 계통도

클럭: 100 MHz (시뮬레이션에서는 10 ns 클럭)

diagram.svg로 따로 받을 수 있다.

버스 규약: mmio-lite

신호방향설명
validM→S트랜잭션 유효
weM→S쓰기=1, 읽기=0
addrM→S바이트 주소
wdataM→S쓰기 데이터
rdataS→M읽기 데이터 (버스 MUX 출력)
readyS→M항상 1 (1사이클 응답)

back-pressure 없음. valid 사이클에 always 응답. AXI4-Lite 비호환

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획mbv-frozen-v2 맵으로 이전 시 pwm=[07]=0x4007_0000, qdec=[0F]=0x400F_0000 슬롯 사용. 이 패키지는 시뮬레이션 전용 주소(0x1000~0x1004 세그먼트)를 사용하므로 compat=incompatible.

// mmio_bus.v — MMIO 주소 디코더 (4개 슬레이브: GPIO/UART/PWM/QDEC)
// addr[31:16] 기준으로 슬레이브 선택
// off = addr[15:0] 그대로 슬레이브에 전달
// ready = 1 사이클 고정 (back-pressure 없음)
`timescale 1ns/1ps
module mmio_bus (
    input  wire        valid,
    input  wire        we,
    input  wire [31:0] addr,
    input  wire [31:0] wdata,
    // 슬레이브 선택
    output wire        sel_gpio,
    output wire        sel_uart,
    output wire        sel_pwm,
    output wire        sel_qdec,
    // 슬레이브 읽기 데이터
    input  wire [31:0] rdata_gpio,
    input  wire [31:0] rdata_uart,
    input  wire [31:0] rdata_pwm,
    input  wire [31:0] rdata_qdec,
    // 마스터 읽기 데이터
    output reg  [31:0] rdata,
    output wire        ready
);
    assign sel_gpio = valid && (addr[31:16] == 16'h1000);
    assign sel_uart = valid && (addr[31:16] == 16'h1001);
    assign sel_pwm  = valid && (addr[31:16] == 16'h1003);
    assign sel_qdec = valid && (addr[31:16] == 16'h1004);
    assign ready    = 1'b1;

    always @* begin
        if      (sel_gpio) rdata = rdata_gpio;
        else if (sel_uart) rdata = rdata_uart;
        else if (sel_pwm)  rdata = rdata_pwm;
        else if (sel_qdec) rdata = rdata_qdec;
        else               rdata = 32'h0000_0000;
    end
endmodule
rtl/gpio.v 40줄
// gpio.v — GPIO LED/SW (mmio-lite 슬레이브)
// 0x00 LED(RW, [15:0]), 0x04 SW(RO, 2단 동기화)
`timescale 1ns/1ps
module gpio (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    input  wire [15:0] sw,
    output wire [15:0] led
);
    reg  [15:0] led_r;
    reg  [15:0] sw_s0, sw_s1; // 2단 동기화

    assign led = led_r;

    always @(posedge clk) begin
        if (rst) begin
            led_r <= 16'd0;
            sw_s0 <= 16'd0;
            sw_s1 <= 16'd0;
        end else begin
            sw_s0 <= sw;
            sw_s1 <= sw_s0;
            if (sel && we && (off == 16'h0000))
                led_r <= wdata[15:0];
        end
    end

    always @* begin
        case (off)
            16'h0000: rdata = {16'd0, led_r};
            16'h0004: rdata = {16'd0, sw_s1};
            default:  rdata = 32'h0000_0000;
        endcase
    end
endmodule
rtl/uart_tx.v 67줄
// uart_tx.v — UART 8N1 송신 전용 (mmio-lite 슬레이브)
// 0x00 DATA(WO, busy일 때 쓰기 무시), 0x04 STATUS(RO, bit0=busy)
`timescale 1ns/1ps
module uart_tx #(
    parameter CLK_HZ = 100_000_000,
    parameter BAUD   = 115200
) (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    output wire        txd
);
    localparam DIV = CLK_HZ / BAUD;

    reg [9:0]  sh;    // {stop, data[7:0], start}
    reg [3:0]  nbit;
    reg [15:0] cnt;
    reg        busy;
    reg        tx;

    assign txd = tx;

    wire wr_data = sel && we && (off == 16'h0000);

    always @(posedge clk) begin
        if (rst) begin
            busy <= 1'b0;
            tx   <= 1'b1;
            cnt  <= 16'd0;
            nbit <= 4'd0;
            sh   <= 10'h3FF;
        end else if (!busy) begin
            tx <= 1'b1;
            if (wr_data) begin
                sh   <= {1'b1, wdata[7:0], 1'b0};
                busy <= 1'b1;
                cnt  <= DIV - 1;
                nbit <= 4'd0;
                tx   <= 1'b0;
            end
        end else begin
            if (cnt == 16'd0) begin
                cnt <= DIV - 1;
                if (nbit == 4'd9) begin
                    busy <= 1'b0;
                    tx   <= 1'b1;
                end else begin
                    nbit <= nbit + 4'd1;
                    tx   <= sh[nbit + 4'd1];
                end
            end else begin
                cnt <= cnt - 16'd1;
            end
        end
    end

    always @* begin
        case (off)
            16'h0004: rdata = {31'd0, busy};
            default:  rdata = 32'h0000_0000;
        endcase
    end
endmodule

// pwm_ip.v — 2채널 PWM IP (IP 표준 Rev.1 관리 레지스터 포함)
// 레지스터 맵:
//   0x0000  CTRL    [0]=enable
//   0x0004  PERIOD  [15:0] 클럭 사이클 수 (0이면 출력 OFF)
//   0x0008  DUTY0   [15:0] 채널0 비교값 (0 ≤ DUTY < PERIOD이면 HIGH)
//   0x000C  DUTY1   [15:0] 채널1 비교값
//   0xFFF0  ID      RO = 0x50574D31 ("PWM1")
//   0xFFF4  VER     RO = {8'h01,8'h00,8'h00,8'h01}
//   0xFFF8  IRQ_STATUS  [0]=period_end W1C
//   0xFFFC  IRQ_ENABLE  [0]=period_end IRQ 허용
// pwm0, pwm1: cnt < duty이면 HIGH
// IRQ: 카운터가 0으로 돌아올 때 period_end 펄스 발생
`timescale 1ns/1ps
module pwm_ip (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    output wire        pwm0,
    output wire        pwm1,
    output wire        irq
);
    // 제어 레지스터
    reg         enable;
    reg  [15:0] period;
    reg  [15:0] duty0;
    reg  [15:0] duty1;
    reg         irq_status;
    reg         irq_enable;

    // 카운터
    reg  [15:0] cnt;

    // 쓰기
    wire wr = sel && we;

    always @(posedge clk) begin
        if (rst) begin
            enable     <= 1'b0;
            period     <= 16'd0;
            duty0      <= 16'd0;
            duty1      <= 16'd0;
            irq_enable <= 1'b0;
            irq_status <= 1'b0;
            cnt        <= 16'd0;
        end else begin
            // IRQ_STATUS W1C 클리어
            if (wr && (off == 16'hFFF8))
                irq_status <= irq_status & ~wdata[0];

            // 일반 레지스터 쓰기
            if (wr) begin
                case (off)
                    16'h0000: enable     <= wdata[0];
                    16'h0004: period     <= wdata[15:0];
                    16'h0008: duty0      <= wdata[15:0];
                    16'h000C: duty1      <= wdata[15:0];
                    16'hFFFC: irq_enable <= wdata[0];
                    default: ;
                endcase
            end

            // 카운터 동작
            if (enable && (period != 16'd0)) begin
                if (cnt >= period - 16'd1) begin
                    cnt        <= 16'd0;
                    irq_status <= irq_status | 1'b1; // period_end 펄스
                end else begin
                    cnt <= cnt + 16'd1;
                end
            end else begin
                cnt <= 16'd0;
            end
        end
    end

    // PWM 출력: cnt < duty이면 HIGH (duty=0이면 항상 LOW)
    assign pwm0 = enable && (period != 16'd0) && (duty0 != 16'd0) && (cnt < duty0);
    assign pwm1 = enable && (period != 16'd0) && (duty1 != 16'd0) && (cnt < duty1);
    assign irq  = irq_status & irq_enable;

    // 읽기
    always @* begin
        case (off)
            16'h0000: rdata = {31'd0, enable};
            16'h0004: rdata = {16'd0, period};
            16'h0008: rdata = {16'd0, duty0};
            16'h000C: rdata = {16'd0, duty1};
            16'hFFF0: rdata = 32'h5057_4D31; // "PWM1"
            16'hFFF4: rdata = 32'h0100_0001; // rev=1, major=0, minor=1
            16'hFFF8: rdata = {31'd0, irq_status};
            16'hFFFC: rdata = {31'd0, irq_enable};
            default:  rdata = 32'h0000_0000;
        endcase
    end

// qenc_ip.v — 4× 직교 엔코더 카운터 IP (IP 표준 Rev.1 관리 레지스터 포함)
// 레지스터 맵:
//   0x0000  CTRL    [0]=enable, [1]=clear_pos (SC: 쓰면 클리어, 자동으로 0으로 돌아옴)
//   0x0004  POS     RO, signed 32-bit 위치
//   0x0008  DIR     RO, bit0=1(CW), 0(CCW) — 마지막 이동 방향
//   0x000C  ERR     RO, [15:0] 비정상 전이 횟수
//   0xFFF0  ID      RO = 0x51454E43 ("QENC")
//   0xFFF4  VER     RO = {8'h01,8'h00,8'h00,8'h01}
//   0xFFF8  IRQ_STATUS  [0]=overflow W1C
//   0xFFFC  IRQ_ENABLE  [0]=overflow IRQ 허용
//
// 4× 디코딩: Gray 코드 2비트 FSM으로 모든 에지에서 ±1 카운트
//   CW  순서: 2'b00 → 2'b01 → 2'b11 → 2'b10 → 2'b00 (+1 each step)
//   CCW 순서: 2'b00 → 2'b10 → 2'b11 → 2'b01 → 2'b00 (-1 each step)
// 비정상 전이(00→11, 11→00, 01→10, 10→01): ERR 카운터 증가
// 2단 동기화: a_in/b_in → 2단 FF → {a_s, b_s}
`timescale 1ns/1ps
module qenc_ip (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [15:0] off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    input  wire        a_in,
    input  wire        b_in,
    output wire        irq
);
    // 제어 레지스터
    reg         enable;
    reg         irq_enable;
    reg         irq_status;

    // 엔코더 상태
    reg  [31:0] pos;
    reg         dir;        // 1=CW, 0=CCW
    reg  [15:0] err;

    // 2단 동기화
    reg  [1:0]  a_ff, b_ff;
    wire        a_s = a_ff[1];
    wire        b_s = b_ff[1];

    // 직전 상태 저장
    reg  [1:0]  prev;

    wire wr = sel && we;

    always @(posedge clk) begin
        if (rst) begin
            enable     <= 1'b0;
            irq_enable <= 1'b0;
            irq_status <= 1'b0;
            pos        <= 32'd0;
            dir        <= 1'b1;
            err        <= 16'd0;
            a_ff       <= 2'b0;
            b_ff       <= 2'b0;
            prev       <= 2'b0;
        end else begin
            // 2단 동기화
            a_ff <= {a_ff[0], a_in};
            b_ff <= {b_ff[0], b_in};

            // IRQ_STATUS W1C
            if (wr && (off == 16'hFFF8))
                irq_status <= irq_status & ~wdata[0];

            // 레지스터 쓰기
            if (wr) begin
                case (off)
                    16'h0000: begin
                        enable <= wdata[0];
                        if (wdata[1]) begin
                            pos <= 32'd0;
                            err <= 16'd0;
                        end
                    end
                    16'hFFFC: irq_enable <= wdata[0];
                    default: ;
                endcase
            end

            // 직교 디코딩 (enable=1일 때만)
            if (enable) begin
                prev <= {a_s, b_s};
                // 현재 상태와 이전 상태로 전이 판정
                case ({prev, a_s, b_s})
                    // CW 전이: +1
                    4'b00_01, 4'b01_11, 4'b11_10, 4'b10_00: begin
                        pos <= pos + 32'd1;
                        dir <= 1'b1;
                        // pos가 최댓값에서 overflow 검출
                        if (pos == 32'h7FFF_FFFF)
                            irq_status <= 1'b1;
                    end
                    // CCW 전이: -1
                    4'b00_10, 4'b10_11, 4'b11_01, 4'b01_00: begin
                        pos <= pos - 32'd1;
                        dir <= 1'b0;
                        if (pos == 32'h8000_0000)
                            irq_status <= 1'b1;
                    end
                    // 비정상 전이 (두 비트 동시 변환)
                    4'b00_11, 4'b11_00, 4'b01_10, 4'b10_01: begin
                        err <= err + 16'd1;
                    end
                    // 변화 없음
                    default: ;
                endcase
            end
        end
    end

    assign irq = irq_status & irq_enable;

    // 읽기
    always @* begin
        case (off)
            16'h0000: rdata = {30'd0, 1'b0, enable};
            16'h0004: rdata = pos;
            16'h0008: rdata = {31'd0, dir};
            16'h000C: rdata = {16'd0, err};
            16'hFFF0: rdata = 32'h5145_4E43; // "QENC"
            16'hFFF4: rdata = 32'h0100_0001;
            16'hFFF8: rdata = {31'd0, irq_status};
            16'hFFFC: rdata = {31'd0, irq_enable};
            default:  rdata = 32'h0000_0000;
        endcase
    end
endmodule

// demo_master.v — 합성용 데모 마스터 FSM
// 보드에서 동작하는 간단한 시퀀스:
//   1. PWM PERIOD=50000, DUTY0=25000(50%), DUTY1=37500(75%), enable=1
//   2. 버튼 입력(btn)으로 DUTY0을 증가시켜 LED 밝기 조절 데모
// 테스트벤치는 이 모듈을 쓰지 않고 soc_top 마스터 포트를 직접 구동한다.
`timescale 1ns/1ps
module demo_master (
    input  wire        clk,
    input  wire        rst,
    // MMIO 마스터 포트
    output reg         valid,
    output reg         we,
    output reg  [31:0] addr,
    output reg  [31:0] wdata,
    input  wire [31:0] rdata,
    input  wire        ready
);
    localparam S_IDLE  = 3'd0;
    localparam S_WRITE = 3'd1;
    localparam S_DONE  = 3'd2;

    // 초기화 단계
    localparam STEP_PERIOD  = 3'd0;
    localparam STEP_DUTY0   = 3'd1;
    localparam STEP_DUTY1   = 3'd2;
    localparam STEP_ENABLE  = 3'd3;
    localparam STEP_FINISH  = 3'd4;

    reg [2:0] state;
    reg [2:0] step;

    always @(posedge clk) begin
        if (rst) begin
            valid <= 1'b0;
            we    <= 1'b0;
            addr  <= 32'd0;
            wdata <= 32'd0;
            state <= S_IDLE;
            step  <= STEP_PERIOD;
        end else begin
            case (state)
                S_IDLE: begin
                    if (step == STEP_FINISH) begin
                        state <= S_DONE;
                    end else begin
                        we    <= 1'b1;
                        valid <= 1'b1;
                        case (step)
                            STEP_PERIOD: begin
                                addr  <= 32'h1003_0004; // PWM PERIOD
                                wdata <= 32'd50000;
                            end
                            STEP_DUTY0: begin
                                addr  <= 32'h1003_0008; // PWM DUTY0 = 50%
                                wdata <= 32'd25000;
                            end
                            STEP_DUTY1: begin
                                addr  <= 32'h1003_000C; // PWM DUTY1 = 75%
                                wdata <= 32'd37500;
                            end
                            STEP_ENABLE: begin
                                addr  <= 32'h1003_0000; // PWM CTRL enable
                                wdata <= 32'd1;
                            end
                            default: ;
                        endcase
                        state <= S_WRITE;
                    end
                end
                S_WRITE: begin
                    if (ready) begin
                        valid <= 1'b0;
                        we    <= 1'b0;
                        step  <= step + 3'd1;
                        state <= S_IDLE;
                    end
                end
                S_DONE: begin
                    // 초기화 완료. 버스를 놓아둔다.
                    valid <= 1'b0;
                    we    <= 1'b0;
                end
                default: state <= S_IDLE;
            endcase
        end
    end
endmodule

// tb_pwm_qenc.v — PASS/FAIL 자체 점검형 테스트벤치
// 테스트 항목:
//   CHECK1: PWM PERIOD=100, DUTY0=25 → pwm0 HIGH 사이클 == 25
//   CHECK2: DUTY1=75              → pwm1 HIGH 사이클 == 75
//   CHECK3: CW 4 에지 주입       → POS 레지스터 == 4
//   CHECK4: CCW 3 에지 추가 주입 → POS 레지스터 == 1
// 최종: PASS 또는 FAIL로 시작하는 줄 출력 후 $finish
`timescale 1ns/1ps
module tb_pwm_qenc;
    // 클럭 (10 ns = 100 MHz)
    reg clk;
    initial clk = 0;
    always #5 clk = ~clk;

    reg rst;

    // 마스터 버스 신호
    reg         m_valid;
    reg         m_we;
    reg  [31:0] m_addr;
    reg  [31:0] m_wdata;
    wire [31:0] m_rdata;
    wire        m_ready;

    // GPIO
    reg  [15:0] sw;
    wire [15:0] led;

    // UART
    wire txd;

    // PWM 출력 관찰
    wire pwm0, pwm1, pwm_irq;

    // 직교 엔코더 입력
    reg  enc_a, enc_b;
    wire qdec_irq;

    // DUT
    soc_top dut (
        .clk      (clk),
        .rst      (rst),
        .m_valid  (m_valid),
        .m_we     (m_we),
        .m_addr   (m_addr),
        .m_wdata  (m_wdata),
        .m_rdata  (m_rdata),
        .m_ready  (m_ready),
        .sw       (sw),
        .led      (led),
        .txd      (txd),
        .pwm0     (pwm0),
        .pwm1     (pwm1),
        .pwm_irq  (pwm_irq),
        .enc_a    (enc_a),
        .enc_b    (enc_b),
        .qdec_irq (qdec_irq)
    );

    // 오류 카운터
    integer fail_cnt;
    integer i;

    // 측정 카운터
    integer ch0_hi_cnt;
    integer ch1_hi_cnt;

    // MMIO 쓰기 태스크
    task mmio_write;
        input [31:0] a;
        input [31:0] d;
        begin
            @(posedge clk);
            m_valid <= 1'b1;
            m_we    <= 1'b1;
            m_addr  <= a;
            m_wdata <= d;
            @(posedge clk);
            // ready는 항상 1이므로 1사이클
            m_valid <= 1'b0;
            m_we    <= 1'b0;
        end
    endtask

    // MMIO 읽기 태스크
    task mmio_read;
        input  [31:0] a;
        output [31:0] d;
        begin
            @(posedge clk);
            m_valid <= 1'b1;
            m_we    <= 1'b0;
            m_addr  <= a;
            @(posedge clk);
            d       = m_rdata;
            m_valid <= 1'b0;
        end
    endtask

    reg [31:0] rd_val;

    // CW 직교 시퀀스: 00→01→11→10→00
    task inject_cw_edge;
        input [1:0] next_ab;
        begin
            enc_a <= next_ab[1];
            enc_b <= next_ab[0];
            repeat(4) @(posedge clk); // 2단 동기화 + 여유
        end
    endtask

    initial begin
        // 초기화
        rst     = 1'b1;
        m_valid = 1'b0;
        m_we    = 1'b0;
        m_addr  = 32'd0;
        m_wdata = 32'd0;
        sw      = 16'd0;
        enc_a   = 1'b0;
        enc_b   = 1'b0;
        fail_cnt = 0;

        repeat(4) @(posedge clk);
        rst = 1'b0;
        repeat(2) @(posedge clk);

        // ── CHECK1: pwm0 HIGH 사이클 == 25 ──────────────────────────────
        $display("CHECK1: PWM DUTY0=25%% (period=100)");
        // PWM 설정
        mmio_write(32'h1003_0004, 32'd100);  // PERIOD = 100
        mmio_write(32'h1003_0008, 32'd25);   // DUTY0 = 25
        mmio_write(32'h1003_000C, 32'd75);   // DUTY1 = 75 (CHECK2용으로 미리 설정)
        mmio_write(32'h1003_0000, 32'd1);    // enable = 1

        // 첫 period_end IRQ 대기 후 카운팅 시작 (IRQ_ENABLE은 0이므로 IRQ_STATUS 폴링)
        // 먼저 현재 카운터가 한 번 돌 때까지 최대 150사이클 대기
        // (enable 직후 cnt=0에서 시작하므로 ~100사이클이면 충분)
        begin : blk_check1
            integer wait_lim;
            wait_lim = 0;
            while (wait_lim < 150) begin
                @(posedge clk);
                wait_lim = wait_lim + 1;
            end
        end

        // 정확히 1 period (100사이클) 동안 pwm0 HIGH 카운트
        ch0_hi_cnt = 0;
        repeat(100) begin
            @(posedge clk);
            if (pwm0 === 1'b1) ch0_hi_cnt = ch0_hi_cnt + 1;
        end

        if (ch0_hi_cnt !== 25) begin
            $display("FAIL CHECK1: pwm0 HIGH=%0d expected 25", ch0_hi_cnt);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("  PASS CHECK1: pwm0 HIGH=%0d", ch0_hi_cnt);
        end

        // ── CHECK2: pwm1 HIGH 사이클 == 75 ──────────────────────────────
        $display("CHECK2: PWM DUTY1=75%% (period=100)");
        // DUTY1=75은 이미 설정됨. 1 period 동안 ch1 HIGH 카운트
        ch1_hi_cnt = 0;
        repeat(100) begin
            @(posedge clk);
            if (pwm1 === 1'b1) ch1_hi_cnt = ch1_hi_cnt + 1;
        end

        if (ch1_hi_cnt !== 75) begin
            $display("FAIL CHECK2: pwm1 HIGH=%0d expected 75", ch1_hi_cnt);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("  PASS CHECK2: pwm1 HIGH=%0d", ch1_hi_cnt);
        end

        // ── CHECK3: CW 4 에지 → POS == 4 ───────────────────────────────
        $display("CHECK3: QDEC CW 4 edges → POS==4");
        // QDEC 활성화 & 초기화 (enable=1, clear=1)
        mmio_write(32'h1004_0000, 32'h0000_0003);
        @(posedge clk); // clear 적용
        // clear 비트 해제 (enable만 유지)
        mmio_write(32'h1004_0000, 32'h0000_0001);

        // 초기 상태: enc_a=0, enc_b=0 (2'b00)
        enc_a = 1'b0; enc_b = 1'b0;
        repeat(4) @(posedge clk);

        // CW 4 에지: 00→01→11→10→00
        inject_cw_edge(2'b01); // step 1
        inject_cw_edge(2'b11); // step 2
        inject_cw_edge(2'b10); // step 3
        inject_cw_edge(2'b00); // step 4

        // 동기화 여유 사이클 추가
        repeat(6) @(posedge clk);

        mmio_read(32'h1004_0004, rd_val); // POS 읽기
        if (rd_val !== 32'd4) begin
            $display("FAIL CHECK3: POS=%0d expected 4", rd_val);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("  PASS CHECK3: POS=%0d", rd_val);
        end

        // ── CHECK4: CCW 3 에지 → POS == 1 ──────────────────────────────
        $display("CHECK4: QDEC CCW 3 edges → POS==1 (was 4)");
        // 현재 enc_a=0, enc_b=0 (2'b00에서 시작)
        // CCW 순서: 00→10→11→01
        enc_a = 1'b1; enc_b = 1'b0; repeat(4) @(posedge clk); // 00→10
        enc_a = 1'b1; enc_b = 1'b1; repeat(4) @(posedge clk); // 10→11
        enc_a = 1'b0; enc_b = 1'b1; repeat(4) @(posedge clk); // 11→01
        repeat(6) @(posedge clk);

        mmio_read(32'h1004_0004, rd_val); // POS 읽기
        if (rd_val !== 32'd1) begin
            $display("FAIL CHECK4: POS=%0d expected 1", rd_val);
            fail_cnt = fail_cnt + 1;
        end else begin
            $display("  PASS CHECK4: POS=%0d", rd_val);
        end

        // ── 최종 결과 출력 ───────────────────────────────────────────────
        @(posedge clk);
        if (fail_cnt === 0)
            $display("PASS all checks (4/4)");
        else
            $display("FAIL %0d check(s) failed", fail_cnt);

        $finish;
    end

    // 타임아웃 (5000사이클)
    initial begin
        #50000;
        $display("FAIL timeout");
        $finish;
    end
endmodule

	vvp $(BUILD)/sim.vvp | tee $(BUILD)/sim.log
	@grep -q "^PASS" $(BUILD)/sim.log || \
	  (echo "ERROR: simulation did not output PASS"; exit 1)
	@echo "sim done"

synth:
	mkdir -p $(BUILD)
	yosys -q -p "read_verilog $(RTL_SRC); synth_xilinx -top basys3_top -family xc7; stat;" \
	  2>&1 | tee $(BUILD)/synth.log
	@echo "synth done"

clean:
	rm -rf $(BUILD)

시뮬레이션

  1. iverilog, verilator, yosys 설치
  2. make -C capstones/mmio-pwm-qenc-checker lint sim synth
  3. build/sim.log에서 PASS 확인

Vivado 빌드 (Basys3)

  1. 새 RTL 프로젝트, 파트 xc7a35tcpg236-1
  2. rtl/*.v 추가 (tb/, demo_master.v 제외), constr/basys3.xdc 추가, 최상위 basys3_top
  3. Generate Bitstream 후 Program Device
  4. UART 터미널(115200 8N1) 열어 PWM/QDEC 동작 확인
