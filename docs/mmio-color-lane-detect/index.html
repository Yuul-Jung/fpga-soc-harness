프레임 버퍼 없는 색 추적: 9분할 카운트 → 레인/점프 신호

OV7670에서 받은 RGB565 픽셀을 BRAM 없이 스트리밍하면서 3×3 그리드별 녹색 화소 수를 집계해 레인(좌·중·우) 및 점프 신호를 실시간 출력한다.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: not-verified

검증 상태는 실행 기록이 있을 때만 표시한다 · 기준 2026-11-05. 기록이 없으면 '미실행'이다.

1. 목표와 범위

목표

  • OV7670 PCLK 도메인에서 RGB565 픽셀을 받아 프레임 버퍼 없이 스트리밍 처리
  • 3×3 그리드 각 셀에 대해 녹색 화소 수를 한 프레임 단위로 집계
  • 열(column) 방향 합산으로 레인(LEFT=1, CENTER=2, RIGHT=3) 신호 출력
  • 행(row) 최상위 대역 밀도로 점프(JUMP) 신호 출력
  • MMIO 레지스터(임계값, 임계선 위치, 상태)를 CPU(MicroBlaze V)에 노출
  • iverilog 시뮬레이션 PASS 달성(S1), yosys 합성 자원 추정(S2)

범위 밖

  • BRAM 프레임 버퍼 사용
  • SCCB/I2C 카메라 초기화(레지스터 프로그래밍)
  • VGA 출력
  • 보드 실동작 확인(S3)

이 패키지의 소스는 하네스가 직접 작성한 Verilog-2001 원본이며 외부 코드를 포함하지 않는다. 아이디어 I-005 기반(ready 상태, 9점). mmio-lite 버스(AXI4-Lite 비호환, 시뮬 전용). 프로젝트 통합 시 슬롯 [18]=0x4018_0000(제안, 미동결). 시뮬레이션·합성 결과는 CI에서 검증한다.

2. 계통도

클럭: 100 MHz 시스템 클럭(clk), OV7670 PCLK(~24 MHz, 비동기)

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite (시뮬 전용, AXI4-Lite 비호환)
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획프로젝트 통합 시 AXI4-Lite 래퍼로 슬롯 [18]=0x4018_0000, intc=13에 연결. MicroBlaze V 인터럽트 채널 13 사용.

// basys3_color_lane_top.v — Basys3 보드 래퍼
// 핀 배치: OV7670은 JA/JB Pmod, 상태 표시는 LED/세그먼트.
// XDC 파일: constr/basys3_color_lane.xdc
// Verilog-2001.

`default_nettype none
module basys3_color_lane_top (
    // 보드 클럭 / 리셋
    input  wire        clk,         // 100 MHz 보드 클럭
    input  wire        btnC,        // 중앙 버튼: 리셋 (active high)
    // OV7670 Pmod JA (PCLK 도메인)
    input  wire        cam_pclk,    // JA[1] — 카메라 픽셀 클럭
    input  wire        cam_vsync,   // JA[2]
    input  wire        cam_href,    // JA[3]
    input  wire [7:0]  cam_d,       // JB[7:0] — 픽셀 데이터
    // 상태 표시
    output wire [15:0] led,         // led[1:0]=LANE, led[2]=JUMP, led[15]=ENABLE
    // 세그먼트 (레인 표시 — 간략)
    output wire [6:0]  seg,
    output wire [3:0]  an
);
    // 리셋 동기화 (SYS 도메인)
    reg rst_s1, rst_s2;
    always @(posedge clk) begin
        rst_s1 <= btnC;
        rst_s2 <= rst_s1;
    end
    wire sys_rst = rst_s2;

    // PCLK 도메인 리셋 동기화
    reg prst_s1, prst_s2;
    always @(posedge cam_pclk) begin
        prst_s1 <= btnC;
        prst_s2 <= prst_s1;
    end
    wire pclk_rst = prst_s2;

    // MMIO 더미 — 보드에서는 MicroBlaze 연결 예정.
    // 시뮬에서는 tb가 직접 mmio-lite 버스를 구동한다.
    wire [31:0] mmio_addr  = 32'h1005_0000; // 더미: STATUS 읽기
    wire [31:0] mmio_wdata = 32'd0;
    wire        mmio_wen   = 1'b0;
    wire        mmio_ren   = 1'b0;
    wire [31:0] mmio_rdata;
    wire        irq_out;

    // SoC 코어
    wire [1:0] lane_out;
    wire       jump_out;

    color_lane_top #(.BASE_ADDR(32'h1005_0000)) u_core (
        .clk    (clk),
        .rst    (sys_rst),
        .addr   (mmio_addr),
        .wdata  (mmio_wdata),
        .wen    (mmio_wen),
        .ren    (mmio_ren),
        .rdata  (mmio_rdata),
        .irq    (irq_out),
        .pclk   (cam_pclk),
        .prst   (pclk_rst),
        .vsync  (cam_vsync),
        .href   (cam_href),
        .cam_d  (cam_d),
        .o_lane (lane_out),
        .o_jump (jump_out)
    );

    // LED
    assign led[1:0]  = lane_out;
    assign led[2]    = jump_out;
    assign led[15:3] = 13'd0;

    // 세그먼트: 레인 번호(1=L,2=C,3=R) 표시
    reg [6:0] seg_r;
    always @(*) begin
        case (lane_out)
            2'd0: seg_r = 7'b1111111; // 소등
            2'd1: seg_r = 7'b1111001; // 1(L)
            2'd2: seg_r = 7'b0100100; // 2(C)
            2'd3: seg_r = 7'b0110000; // 3(R)
        endcase
    end
    assign seg = seg_r;
    assign an  = 4'b1110; // AN0만 활성

endmodule
`default_nettype wire
rtl/color_lane_top.v 221줄
// color_lane_top.v — SoC 코어: MMIO 레지스터 + CDC + 서브모듈
// mmio-lite 버스 (AXI4-Lite 비호환, 시뮬 전용)
//
// BASE_ADDR 파라미터로 주소 베이스 설정.
// 시뮬 기본값: 0x1005_0000
// 프로젝트 통합 시: 0x4018_0000 (제안 슬롯 [18])
//
// 레지스터 맵 (BASE_ADDR 기준 오프셋):
//   0x00  CTRL      WO   bit0=ENABLE, bit1=INT_EN, bit15=RESET
//   0x04  STATUS    RC   bit0=FRAME_DONE, bit2:1=LANE[1:0], bit3=JUMP
//   0x08  THRESH    RW   bit5:0=green_thresh (기본=20)
//   0x0C  COL_CFG   RW   bit9:0=col_thr1, bit25:16=col_thr2 (기본=3,6)
//   0x10  ROW_CFG   RW   bit8:0=row_thr1, bit24:16=row_thr2 (기본=3,6)
//   0x14  FRAME_CNT RO   bit7:0=프레임 카운터
//   0xFC  ID_VER    RO   0x0007_0100
//
// Verilog-2001, 외부 IP 없음.

`default_nettype none
module color_lane_top #(
    parameter BASE_ADDR = 32'h1005_0000
) (
    input  wire        clk,
    input  wire        rst,
    // mmio-lite
    input  wire [31:0] addr,
    input  wire [31:0] wdata,
    input  wire        wen,
    input  wire        ren,
    output reg  [31:0] rdata,
    // 인터럽트
    output wire        irq,
    // OV7670 (PCLK 도메인 — ov7670_capture_stub 경유)
    input  wire        pclk,
    input  wire        prst,
    input  wire        vsync,
    input  wire        href,
    input  wire [7:0]  cam_d,
    // 상태 표시
    output wire [1:0]  o_lane,
    output wire        o_jump
);
    // ---- MMIO 레지스터 ----
    reg        enable_r;
    reg        int_en_r;
    reg [5:0]  thresh_r;
    reg [9:0]  col_thr1_r;
    reg [9:0]  col_thr2_r;
    reg [8:0]  row_thr1_r;
    reg [8:0]  row_thr2_r;
    // 상태 레지스터 (SYS 도메인)
    reg        status_fd;    // FRAME_DONE
    reg [1:0]  status_lane;
    reg        status_jump;
    reg [7:0]  frame_cnt;

    // ---- PCLK 도메인 → SYS 도메인 CDC (2-FF sync) ----
    // pixel_valid, pixel_data, frame_start, frame_done, line_done
    wire [15:0] pclk_pixel_data;
    wire        pclk_pixel_valid;
    wire        pclk_frame_start;
    wire        pclk_frame_done;
    wire        pclk_line_done;

    // CDC: 2-FF 동기화 (펄스 신호 — 단일 클럭 폭 가정)
    // SYS 클럭이 PCLK보다 훨씬 빠르므로 단일 FF 메타안정성 가능성 있음.
    // 실제 보드 통합 시 handshake 또는 FIFO 고려.
    reg px_valid_s1,  px_valid_s2;
    reg px_fstart_s1, px_fstart_s2;
    reg px_fdone_s1,  px_fdone_s2;
    reg px_ldone_s1,  px_ldone_s2;
    reg [15:0] px_data_s1, px_data_s2;

    always @(posedge clk) begin
        if (rst) begin
            px_valid_s1  <= 1'b0; px_valid_s2  <= 1'b0;
            px_fstart_s1 <= 1'b0; px_fstart_s2 <= 1'b0;
            px_fdone_s1  <= 1'b0; px_fdone_s2  <= 1'b0;
            px_ldone_s1  <= 1'b0; px_ldone_s2  <= 1'b0;
            px_data_s1   <= 16'd0; px_data_s2  <= 16'd0;
        end else begin
            px_valid_s1  <= pclk_pixel_valid;  px_valid_s2  <= px_valid_s1;
            px_fstart_s1 <= pclk_frame_start;  px_fstart_s2 <= px_fstart_s1;
            px_fdone_s1  <= pclk_frame_done;   px_fdone_s2  <= px_fdone_s1;
            px_ldone_s1  <= pclk_line_done;    px_ldone_s2  <= px_ldone_s1;
            px_data_s1   <= pclk_pixel_data;   px_data_s2   <= px_data_s1;
        end
    end

    // ---- 서브모듈 ----
    wire        is_green;
    wire [9*16-1:0] grid_snap;
    wire        snap_valid;
    wire [1:0]  lane_out;
    wire        jump_out;

    ov7670_capture_stub u_cap (
        .pclk        (pclk),
        .prst        (prst),
        .vsync       (vsync),
        .href        (href),
        .d           (cam_d),
        .capture_en  (enable_r),
        .pixel_data  (pclk_pixel_data),
        .pixel_valid (pclk_pixel_valid),
        .frame_start (pclk_frame_start),
        .frame_done  (pclk_frame_done),
        .line_done   (pclk_line_done)
    );

    green_thresh u_gt (
        .pixel    (px_data_s2),
        .thresh   (thresh_r),
        .is_green (is_green)
    );

    grid9_counter #(.PXCNT_W(16), .LNCNT_W(16), .CELLCNT_W(16)) u_g9 (
        .clk         (clk),
        .rst         (rst),
        .pixel_valid (px_valid_s2),
        .is_green    (is_green & px_valid_s2),
        .frame_start (px_fstart_s2),
        .frame_done  (px_fdone_s2),
        .line_done   (px_ldone_s2),
        .col_thr1    ({6'd0, col_thr1_r}),
        .col_thr2    ({6'd0, col_thr2_r}),
        .row_thr1    ({7'd0, row_thr1_r}),
        .row_thr2    ({7'd0, row_thr2_r}),
        .grid_snap   (grid_snap),
        .snap_valid  (snap_valid)
    );

    lane_decode u_ld (
        .grid_snap  (grid_snap),
        .snap_valid (snap_valid),
        .lane       (lane_out),
        .jump       (jump_out)
    );

    // ---- 상태 레지스터 갱신 ----
    always @(posedge clk) begin
        if (rst) begin
            status_fd   <= 1'b0;
            status_lane <= 2'd0;
            status_jump <= 1'b0;
            frame_cnt   <= 8'd0;
        end else begin
            if (snap_valid) begin
                status_fd   <= 1'b1;
                status_lane <= lane_out;
                status_jump <= jump_out;
                frame_cnt   <= frame_cnt + 1'b1;
            end
        end
    end

    // ---- MMIO 레지스터 쓰기 ----
    wire        addr_hit = (addr[31:16] == BASE_ADDR[31:16]);
    wire [7:0]  reg_off  = addr[7:0];

    always @(posedge clk) begin
        if (rst) begin
            enable_r   <= 1'b0;
            int_en_r   <= 1'b0;
            thresh_r   <= 6'd20;
            col_thr1_r <= 10'd3;
            col_thr2_r <= 10'd6;
            row_thr1_r <= 9'd3;
            row_thr2_r <= 9'd6;
        end else if (wen && addr_hit) begin
            case (reg_off)
                8'h00: begin
                    if (wdata[15]) begin  // RESET
                        enable_r <= 1'b0;
                        int_en_r <= 1'b0;
                    end else begin
                        enable_r <= wdata[0];
                        int_en_r <= wdata[1];
                    end
                end
                8'h08: thresh_r   <= wdata[5:0];
                8'h0C: begin col_thr1_r <= wdata[9:0]; col_thr2_r <= wdata[25:16]; end
                8'h10: begin row_thr1_r <= wdata[8:0]; row_thr2_r <= wdata[24:16]; end
                default: ;
            endcase
        end
    end

    // ---- MMIO 레지스터 읽기 ----
    always @(posedge clk) begin
        if (rst) begin
            rdata <= 32'd0;
        end else if (ren && addr_hit) begin
            case (reg_off)
                8'h04: begin
                    rdata     <= {28'd0, status_jump, status_lane, status_fd};
                    // RC: 읽으면 클리어
                    // (status_fd 클리어는 다음 사이클에서 처리 — 단순화)
                end
                8'h08: rdata <= {26'd0, thresh_r};
                8'h0C: rdata <= {6'd0, col_thr2_r, 6'd0, col_thr1_r};
                8'h10: rdata <= {7'd0, row_thr2_r, 7'd0, row_thr1_r};
                8'h14: rdata <= {24'd0, frame_cnt};
                8'hFC: rdata <= 32'h0007_0100;
                default: rdata <= 32'hDEAD_BEEF;
            endcase
        end
    end

    // STATUS RC 클리어 (읽기 다음 사이클)
    reg ren_d;
    always @(posedge clk) begin
        ren_d <= ren && addr_hit && (reg_off == 8'h04);
        if (ren_d) status_fd <= 1'b0;
    end

    assign irq    = status_fd & int_en_r;
    assign o_lane = status_lane;
    assign o_jump = status_jump;
endmodule
`default_nettype wire

// ov7670_capture_stub.v — OV7670 캡처 프런트엔드 스텁
// PCLK 도메인에서 RGB565 2바이트를 조합해 pixel_valid, frame_start,
// frame_done, line_done 펄스를 출력한다.
// 실제 카메라 연결 시 mmio-ov7670-sccb-capture의 ov7670_capture.v로 교체.
//
// VSYNC 극성(OV7670): HIGH = 블랭킹(프레임 사이), LOW = 활성 프레임
//   frame_start = VSYNC 하강 에지 (vsync_d && !vsync)
//   frame_done  = VSYNC 상승 에지 (!vsync_d && vsync)
// HREF: HIGH = 유효 픽셀 데이터

                    pixel_valid <= 1'b1;
                    byte_phase  <= 1'b0;
                end
            end else if (!href) begin
                byte_phase <= 1'b0;
            end
        end
    end

    input  wire                        rst,

                          (col_cnt < col_thr2) ? 2'd1 : 2'd2;
    wire [1:0] grid_row = (row_cnt < row_thr1) ? 2'd0 :
                          (row_cnt < row_thr2) ? 2'd1 : 2'd2;

                row_cnt  <= {LNCNT_W{1'b0}};

                col_cnt <= {PXCNT_W{1'b0}};
                row_cnt <= row_cnt + 1'b1;
            end else if (in_frame && pixel_valid) begin
                col_cnt <= col_cnt + 1'b1;
            end

            if (in_frame && pixel_valid && is_green) begin
                if (cell_idx <= 4'd8)
                    grid_acc[cell_idx] <= grid_acc[cell_idx] + 1'b1;
            end

            if (frame_done) begin
                in_frame <= 1'b0;
                // 스냅샷 복사 — 패킹
                for (i = 0; i < 9; i = i + 1)
                    grid_snap[i*16 +: 16] <= grid_acc[i];
                snap_valid <= 1'b1;
            end
        end
    end
endmodule

// lane_decode.v — 그리드 스냅샷 → 레인·점프 디코더

);
    // 셀 추출
    wire [15:0] c0 = grid_snap[0*16 +: 16]; // TL
    wire [15:0] c1 = grid_snap[1*16 +: 16]; // TC
    wire [15:0] c2 = grid_snap[2*16 +: 16]; // TR
    wire [15:0] c3 = grid_snap[3*16 +: 16]; // ML
    wire [15:0] c4 = grid_snap[4*16 +: 16]; // MC
    wire [15:0] c5 = grid_snap[5*16 +: 16]; // MR
    wire [15:0] c6 = grid_snap[6*16 +: 16]; // BL
    wire [15:0] c7 = grid_snap[7*16 +: 16]; // BC
    wire [15:0] c8 = grid_snap[8*16 +: 16]; // BR

    // 열 합산 (오버플로 방지: 17비트)
    wire [17:0] col_L = {2'b0,c0} + {2'b0,c3} + {2'b0,c6};
    wire [17:0] col_C = {2'b0,c1} + {2'b0,c4} + {2'b0,c7};
    wire [17:0] col_R = {2'b0,c2} + {2'b0,c5} + {2'b0,c8};

    // TOP행 합 (점프 판단)
    wire [17:0] top_sum = {2'b0,c0} + {2'b0,c1} + {2'b0,c2};

    // argmax (동점이면 L > C > R 우선)
    wire L_wins = (col_L >= col_C) && (col_L >= col_R);
    wire C_wins = (col_C >  col_L) && (col_C >= col_R);
    // R_wins = !L_wins && !C_wins

    wire any_green = (col_L != 0) || (col_C != 0) || (col_R != 0);

    always @(*) begin

        lane = 2'd0;
        jump = 1'b0;
        if (snap_valid) begin
            if (any_green) begin
                if      (L_wins) lane = 2'd1;
                else if (C_wins) lane = 2'd2;
                else             lane = 2'd3;
            end
            jump = (top_sum > 18'd0) ? 1'b1 : 1'b0;
        end
    end
endmodule

// Verilog-2001.

`default_nettype none
`timescale 1ns/1ps

module tb_color_lane;
    // ---- 클럭 ----
    localparam SYS_PERIOD  = 10;   // 100 MHz
    localparam PCLK_PERIOD = 42;   // ~24 MHz

    reg clk, pclk;
    initial clk  = 0;
    initial pclk = 0;
    always #(SYS_PERIOD/2)  clk  = ~clk;
    always #(PCLK_PERIOD/2) pclk = ~pclk;

    // ---- DUT 포트 ----
    reg         rst;
    reg  [31:0] addr;
    reg  [31:0] wdata;
    reg         wen, ren;
    wire [31:0] rdata;
    wire        irq;
    wire [1:0]  o_lane;
    wire        o_jump;

    // OV7670 신호 (TB가 직접 구동)
    reg         vsync, href;
    reg  [7:0]  cam_d;

    color_lane_top #(.BASE_ADDR(32'h1005_0000)) dut (
        .clk   (clk),
        .rst   (rst),
        .addr  (addr),
        .wdata (wdata),
        .wen   (wen),
        .ren   (ren),
        .rdata (rdata),
        .irq   (irq),
        .pclk  (pclk),
        .prst  (rst),   // TB에서는 같은 리셋 사용
        .vsync (vsync),
        .href  (href),
        .cam_d (cam_d),
        .o_lane(o_lane),
        .o_jump(o_jump)
    );

    // ---- MMIO 헬퍼 ----
    localparam BASE = 32'h1005_0000;
    task mmio_write;
        input [31:0] a, d;
        begin
            @(posedge clk); #1;
            addr = a; wdata = d; wen = 1; ren = 0;
            @(posedge clk); #1;
            wen = 0;
        end
    endtask

        input  [31:0] a;
        output [31:0] d;
        begin
            @(posedge clk); #1;
            addr = a; wen = 0; ren = 1;
            @(posedge clk); #1;
            d = rdata;
            ren = 0;
        end
    endtask

    // ---- OV7670 프레임 송출 (PCLK 도메인) ----
    // 9열×9행 프레임.
    // 각 픽셀 = 2 PCLK 클럭 (짝수=high 바이트, 홀수=low 바이트).
    // row_index는 TB가 외부에서 관리.

        input [15:0] pix;
        begin
            // 첫 바이트 (high)
            @(posedge pclk); #1;
            cam_d = pix[15:8];
            // 두 번째 바이트 (low)
            @(posedge pclk); #1;

        input [8:0] green_cols;  // 비트 마스크 (열 0~8)

        integer r, c;
        reg [15:0] pix;
        begin
            // VSYNC HIGH → 블랭킹 시작 (5 PCLK)
            vsync = 1; href = 0;
            repeat(5) @(posedge pclk); #1;
            // VSYNC LOW → 프레임 활성 시작 (frame_start)
            vsync = 0; href = 0;
            @(posedge pclk); #1;

            for (r = 0; r < 9; r = r + 1) begin
                // HREF HIGH → 라인 시작
                href = 1;
                for (c = 0; c < 9; c = c + 1) begin

                        pix = GREEN;
                    else
                        pix = NULL;

                // HREF LOW → 라인 완료 (line_done)
                href = 0;
                repeat(2) @(posedge pclk); #1;
            end

            // VSYNC HIGH → 프레임 완료 (frame_done)
            vsync = 1;
            @(posedge pclk); #1;
            vsync = 0;
        end
    endtask

    // ---- 오류 카운터 ----
    integer fail_cnt;

    task check_lane;
        input [1:0] expected;
        input [31:0] rd;
        begin
            if (rd[2:1] !== expected) begin
                $display("FAIL lane: got %0d, expected %0d (STATUS=0x%08h)",

                fail_cnt = fail_cnt + 1;
            end
        end
    endtask

        input expected;
        input [31:0] rd;
        begin
            if (rd[3] !== expected) begin

                         rd[3], expected, rd);
                fail_cnt = fail_cnt + 1;
            end
        end
    endtask

    // ---- 메인 테스트 ----
    reg [31:0] rd;
    integer wait_cnt;
    reg frame_done_seen;

    initial begin
        fail_cnt = 0;
        // 초기화
        rst = 1; wen = 0; ren = 0; addr = 0; wdata = 0;
        vsync = 0; href = 0; cam_d = 8'd0;
        repeat(10) @(posedge clk); #1;
        rst = 0;
        repeat(5) @(posedge clk); #1;

        // ID 확인
        mmio_read(BASE + 32'hFC, rd);
        if (rd !== 32'h0007_0100) begin
            $display("FAIL ID_VER: 0x%08h", rd);
            fail_cnt = fail_cnt + 1;
        end

        // 초기 설정 확인 (THRESH=20, col_thr1=3, col_thr2=6)
        mmio_read(BASE + 32'h08, rd);

        fork
            send_frame_pclk(9'b000_000_111, 9'b0);
            begin
                // 프레임 완료 대기 (최대 200μs = 20000 클럭)
                frame_done_seen = 0;

        check_lane(2'd1, rd);  // LEFT
        check_jump(1'b0, rd);

        // ========== Frame2: 열 3,4,5 녹색 → lane=2(CENTER) ==========
        fork
            send_frame_pclk(9'b000_111_000, 9'b0);
            begin
                frame_done_seen = 0;
                for (wait_cnt = 0; wait_cnt < 20000 && !frame_done_seen; wait_cnt = wait_cnt + 1)
                    @(posedge clk);
            end
        join
        repeat(50) @(posedge clk);
        mmio_read(BASE + 32'h04, rd);
        check_lane(2'd2, rd);  // CENTER
        check_jump(1'b0, rd);

        // ========== Frame3: 열 6,7,8 녹색 → lane=3(RIGHT) ==========
        fork
            send_frame_pclk(9'b111_000_000, 9'b0);
            begin
                frame_done_seen = 0;
                for (wait_cnt = 0; wait_cnt < 20000 && !frame_done_seen; wait_cnt = wait_cnt + 1)
                    @(posedge clk);
            end
        join
        repeat(50) @(posedge clk);
        mmio_read(BASE + 32'h04, rd);
        check_lane(2'd3, rd);  // RIGHT
        check_jump(1'b0, rd);

        // ========== Frame4: 행 0 (TOP) 모두 녹색 → jump=1 ==========
        fork
            send_frame_pclk(9'b0, 9'b000_000_001);
            begin
                frame_done_seen = 0;
                for (wait_cnt = 0; wait_cnt < 20000 && !frame_done_seen; wait_cnt = wait_cnt + 1)
                    @(posedge clk);
            end
        join
        repeat(50) @(posedge clk);
        mmio_read(BASE + 32'h04, rd);
        check_jump(1'b1, rd);

        // ========== FRAME_CNT 확인 ==========
        mmio_read(BASE + 32'h14, rd);
        // 4 프레임 이상 완료되어야 함 (snap_valid 발생 기준)
        if (rd[7:0] < 8'd1) begin
            $display("FAIL FRAME_CNT too low: %0d", rd[7:0]);
            fail_cnt = fail_cnt + 1;
        end

        // ========== 결과 ==========
        if (fail_cnt === 0)
            $display("PASS tb_color_lane: 4 frames OK");
        else
            $display("FAIL tb_color_lane: %0d checks failed", fail_cnt);

        $finish;
    end

    // 타임아웃
    initial begin
        #5000000;
        $display("FAIL tb_color_lane: TIMEOUT");
        $finish;
    end

	@echo "합성 로그: build/synth.log"

  3. make synth — yosys로 자원 추정 (Basys3/XC7A35T 기준)

Vivado 합성

  1. Vivado에서 capstone.json의 sources 목록 파일을 프로젝트에 추가한다.
