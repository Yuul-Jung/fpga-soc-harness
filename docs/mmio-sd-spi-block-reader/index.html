SD SPI 블록 리더 SoC: 초기화·CMD17 읽기 + 카드 모델 자체 점검 (Basys3)

skeleton(T1)의 MMIO 버스에 SD 카드 SPI 컨트롤러를 더한 T2 패키지. 초기화(CMD0/8/55+41/58)와 512바이트 단일 블록 읽기를 직접 쓴 FSM으로 만들고, 테스트벤치의 간이 SD 카드 모델이 명령 순서·클럭 속도·주소를 PASS/FAIL로 검사한다. 데모 마스터는 블록 0의 첫 16바이트를 UART로 내보낸다.

린트: 건너뜀 (verilator 없음)시뮬레이션: 건너뜀 (iverilog 없음)합성: 건너뜀 (yosys 없음)보드: 미실행

1. 목표와 범위

목표

  • SD 카드(SDHC/SDXC, SPI 모드)를 400 kHz로 초기화하고 12.5 MHz로 전환한다.
  • CMD17로 512바이트 블록을 읽어 내부 버퍼에 넣고, 레지스터(IDX/DATA)로 바이트를 읽는다.
  • 카드 모델로 초기화 순서, 저속 클럭, 80클럭 이상 더미, LBA→CMD17 인자 변환을 자동 점검한다.
  • 읽기 전 요청과 카드 없음(R1 응답 없음)이 오류 비트로 나타나는지 점검한다.

범위 밖

  • SD v1/MMC 카드(CMD8 거부) 지원
  • 쓰기(CMD24), 다중 블록 읽기, CRC 검증
  • FAT 파일시스템
  • CPU 구현, 인터럽트, DMA
  • 카드 감지(CD) 핀 사용

이 패키지의 소스는 하네스가 직접 작성한 원본이며 외부 코드를 포함하지 않는다(gpio.v, uart_tx.v는 이 저장소의 mmio-uart-gpio-skeleton에서 가져온 자체 원본). 근거 카드 중 GPL-3.0 코드는 읽지 않고 구조 개념만 참고했다. 시뮬레이션·합성 결과는 CI가 기록한다.

통합 선언 (SoC 적합성)

이 패키지는 MicroBlaze V 동결 주소맵과 호환되지 않는다 (파일럿). 아래 이식 계획을 따라야 SoC에 합칠 수 있다.

버스mmio-lite
주소맵custom → 동결 맵 mbv-frozen-v2 적합 여부: 적합하지 않음
이식 계획이 패키지는 CPU 없이 시뮬레이션하려고 자체 버스 mmio-lite(0x1000_xxxx)를 쓴 파일럿이다. 동결 맵(mbv-frozen-v2)으로 옮기는 것은 주소만 바꾸는 일이 아니다. (1) gpio·uart는 동결 맵에 이미 AMD IP가 있다([00] AXI GPIO, [02] UART Lite). 레지스터 배치가 달라(AXI GPIO는 DATA/TRI, UART Lite는 RX/TX FIFO·STAT·CTRL) 우리 모듈을 그대로 꽂을 수 없고, 이식 때는 AMD IP를 쓰고 드라이버를 맞춘다. (2) 자체 주변장치는 프로젝트의 IP 표준(64KB 창, 0xFFF0 ID·VER·IRQ_STATUS·IRQ_ENABLE)에 맞춰 새로 감싼다. (3) mmio-lite→AXI4-Lite 마스터 다리는 아직 없다(gap). sd_spi는 동결 맵에 대응 블록이 없으므로 IP 표준(ip_std 관리 영역) 모양으로 감싼 새 IP로 만들어 빈 슬롯 [16]부터 배정한다(제안, 동결 아님). 프로젝트의 C02·C03는 같은 목적을 AMD Quad SPI [0D] + 소프트웨어 SD 스택으로 풀므로, 이 하드웨어 컨트롤러는 그 대안(속도·CPU 부하 비교)이 된다.

6. 기술 트리 (이 캡스톤의 확장 경로)

  • T1MMIO 버스 + GPIO + UART, 마스터는 FSM (mmio-uart-gpio-skeleton)
  • T2SD SPI 컨트롤러 추가: 초기화·블록 읽기, 카드 모델 검증 참고: element14-security-camera-2-sd-card-interfacing, wangxuan95-fpga-sdcard-file, zipcpu-sdspi
  • T2마스터를 PicoRV32로 교체, 부트 ROM(BRAM)과 SRAM 추가 후 SD 레지스터를 C로 구동 참고: fanguoc2len-risc-v-computer, tmahlburg-picosoc-basys3
  • T2FAT16/32 계층 추가(SD 블록 → 파일 이름 읽기) 또는 VGA 슬라이드쇼(I-004) 참고: wangxuan95-fpga-sdcard-file

rtl/spi_byte.v 49줄
// SPI 모드 0 바이트 송수신기: start 펄스로 1바이트(MSB 먼저) 교환. div = SCLK 반주기 클럭 수.
module spi_byte (
    input  wire        clk,
    input  wire        rst,
    input  wire        start,
    input  wire [7:0]  tx,
    input  wire [15:0] div,
    output reg         busy,
    output reg         done,
    output reg  [7:0]  rx,
    output wire        sclk,
    output wire        mosi,
    input  wire        miso
);
    reg        sclk_r;
    reg [7:0]  tsh, rsh;
    reg [2:0]  bn;
    reg [15:0] cnt;
    assign sclk = sclk_r;
    assign mosi = busy ? tsh[7] : 1'b1;

    always @(posedge clk) begin
        done <= 1'b0;
        if (rst) begin
            busy <= 1'b0; sclk_r <= 1'b0; cnt <= 16'd0; bn <= 3'd0;
            tsh <= 8'hFF; rsh <= 8'h00; rx <= 8'h00;
        end else if (!busy) begin
            if (start) begin
                busy <= 1'b1; tsh <= tx; bn <= 3'd0; cnt <= div - 16'd1; sclk_r <= 1'b0;
            end
        end else if (cnt != 16'd0) begin
            cnt <= cnt - 16'd1;
        end else begin
            cnt <= div - 16'd1;
            if (!sclk_r) begin                       // 상승 에지: MISO 표본
                sclk_r <= 1'b1;
                rsh    <= {rsh[6:0], miso};
            end else begin                           // 하강 에지: 다음 비트로 이동
                sclk_r <= 1'b0;
                if (bn == 3'd7) begin
                    busy <= 1'b0; done <= 1'b1; rx <= rsh;
                end else begin
                    bn  <= bn + 3'd1;
                    tsh <= {tsh[6:0], 1'b0};
                end
            end
        end
    end
endmodule
rtl/sd_spi.v 164줄
// SD 카드 SPI 모드 컨트롤러(원본 작성): 초기화(CMD0/8/55+41/58)와 CMD17 단일 블록 읽기.
// 레지스터: 00 CTRL(WO, bit0=초기화, bit1=읽기) 04 STATUS(RO, {sdhc,err,init_done,busy})
//           08 LBA(RW) 0C IDX(RW, 9bit) 10 DATA(RO, 버퍼[IDX]). v2 카드(SDHC/SDXC)만 대상.
module sd_spi #(
    parameter CLK_HZ  = 100_000_000,
    parameter SLOW_HZ = 400_000,
    parameter FAST_HZ = 12_500_000
) (
    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [7:0]  off,
    input  wire [31:0] wdata,
    output reg  [31:0] rdata,
    output reg         sd_cs_n,
    output wire        sd_sclk,
    output wire        sd_mosi,
    input  wire        sd_miso
);
    localparam [15:0] SLOW_DIV = CLK_HZ / (2 * SLOW_HZ);
    localparam [15:0] FAST_DIV = CLK_HZ / (2 * FAST_HZ);
    localparam [3:0] ST_IDLE = 4'd0, ST_DUMMY = 4'd1, ST_PREP = 4'd2, ST_CMD = 4'd3, ST_R1 = 4'd4,
                     ST_EXT = 4'd5, ST_EVAL = 4'd6, ST_GAP = 4'd7, ST_TOKEN = 4'd8, ST_DATA = 4'd9, ST_CRC = 4'd10;
    localparam [2:0] S_CMD0 = 3'd0, S_CMD8 = 3'd1, S_CMD55 = 3'd2, S_ACMD41 = 3'd3, S_CMD58 = 3'd4, S_RD = 3'd5;

    reg [3:0]  st;
    reg [2:0]  step, bidx, ecnt;
    reg [3:0]  dcnt, pcnt;
    reg [9:0]  retry;
    reg [15:0] tcnt;
    reg [8:0]  bcnt, idx;
    reg        iss, busy, init_done, err, sdhc, fast, fin, ext, sp_start;
    reg [5:0]  cmd;
    reg [7:0]  crc, r1, sp_tx;
    reg [31:0] arg, r32, lba;
    reg [7:0]  mem [0:511];

    wire       sp_busy, sp_done;
    wire [7:0] mem_q = mem[idx];
    wire [7:0] sp_rx;
    wire       can_issue = !iss && !sp_busy;
    wire       wr_ctrl   = sel && we && (off == 8'h00) && !busy;
    reg  [7:0] cmd_byte;

    spi_byte u_spi (
        .clk(clk), .rst(rst), .start(sp_start), .tx(sp_tx), .div(fast ? FAST_DIV : SLOW_DIV),
        .busy(sp_busy), .done(sp_done), .rx(sp_rx), .sclk(sd_sclk), .mosi(sd_mosi), .miso(sd_miso)
    );

    always @* begin
        case (bidx)
            3'd0:    cmd_byte = {2'b01, cmd};
            3'd1:    cmd_byte = arg[31:24];
            3'd2:    cmd_byte = arg[23:16];
            3'd3:    cmd_byte = arg[15:8];
            3'd4:    cmd_byte = arg[7:0];
            default: cmd_byte = crc;
        endcase
    end

    always @(posedge clk) begin
        sp_start <= 1'b0;
        if (rst) begin
            st <= ST_IDLE; busy <= 1'b0; init_done <= 1'b0; err <= 1'b0; sdhc <= 1'b0; fast <= 1'b0;
            sd_cs_n <= 1'b1; iss <= 1'b0; lba <= 32'd0; idx <= 9'd0; fin <= 1'b0; step <= S_CMD0;
            bidx <= 3'd0; ecnt <= 3'd0; dcnt <= 4'd0; pcnt <= 4'd0; retry <= 10'd0; tcnt <= 16'd0;
            bcnt <= 9'd0; cmd <= 6'd0; crc <= 8'hFF; r1 <= 8'hFF; sp_tx <= 8'hFF; arg <= 32'd0; r32 <= 32'd0; ext <= 1'b0;
        end else begin
            if (sel && we && off == 8'h08) lba <= wdata;
            if (sel && we && off == 8'h0C) idx <= wdata[8:0];
            if (can_issue && st != ST_IDLE && st != ST_PREP && st != ST_EVAL) begin
                sp_start <= 1'b1; iss <= 1'b1;
                sp_tx <= (st == ST_CMD) ? cmd_byte : 8'hFF;
            end
            case (st)
                ST_IDLE: if (wr_ctrl) begin
                    if (wdata[0]) begin
                        busy <= 1'b1; init_done <= 1'b0; err <= 1'b0; sdhc <= 1'b0; fast <= 1'b0;
                        step <= S_CMD0; retry <= 10'd0; fin <= 1'b0; dcnt <= 4'd0; sd_cs_n <= 1'b1; st <= ST_DUMMY;
                    end else if (wdata[1]) begin
                        if (init_done) begin busy <= 1'b1; err <= 1'b0; fin <= 1'b0; step <= S_RD; st <= ST_PREP; end
                        else err <= 1'b1;
                    end
                end
                ST_DUMMY: if (iss && sp_done) begin               // CS=1에서 80클럭
                    iss <= 1'b0;
                    if (dcnt == 4'd9) st <= ST_PREP; else dcnt <= dcnt + 4'd1;
                end
                ST_PREP: begin
                    sd_cs_n <= 1'b0; bidx <= 3'd0; pcnt <= 4'd0; ecnt <= 3'd0; ext <= 1'b0; st <= ST_CMD;
                    case (step)
                        S_CMD0:   begin cmd <= 6'd0;  arg <= 32'h0;                crc <= 8'h95; end
                        S_CMD8:   begin cmd <= 6'd8;  arg <= 32'h0000_01AA;        crc <= 8'h87; ext <= 1'b1; end
                        S_CMD55:  begin cmd <= 6'd55; arg <= 32'h0;                crc <= 8'h65; end
                        S_ACMD41: begin cmd <= 6'd41; arg <= 32'h4000_0000;        crc <= 8'h77; end
                        S_CMD58:  begin cmd <= 6'd58; arg <= 32'h0;                crc <= 8'hFD; ext <= 1'b1; end
                        default:  begin cmd <= 6'd17; arg <= sdhc ? lba : {lba[22:0], 9'd0}; crc <= 8'hFF; end
                    endcase
                end
                ST_CMD: if (iss && sp_done) begin
                    iss <= 1'b0;
                    if (bidx == 3'd5) st <= ST_R1; else bidx <= bidx + 3'd1;
                end
                ST_R1: if (iss && sp_done) begin                  // R1 응답: 최상위 비트가 0인 첫 바이트
                    iss <= 1'b0;
                    if (!sp_rx[7]) begin r1 <= sp_rx; st <= ext ? ST_EXT : ST_EVAL; end
                    else if (pcnt == 4'd7) begin err <= 1'b1; sd_cs_n <= 1'b1; st <= ST_GAP; end
                    else pcnt <= pcnt + 4'd1;
                end
                ST_EXT: if (iss && sp_done) begin                 // R7/R3 뒤의 4바이트
                    iss <= 1'b0; r32 <= {r32[23:0], sp_rx};
                    if (ecnt == 3'd3) st <= ST_EVAL; else ecnt <= ecnt + 3'd1;
                end
                ST_EVAL: if (step == S_RD && r1 == 8'h00) begin   // CS를 올렸다 내리는 글리치를 피해 분기로 나눈다
                    tcnt <= 16'd0; st <= ST_TOKEN;
                end else begin
                    st <= ST_GAP; sd_cs_n <= 1'b1;
                    case (step)
                        S_CMD0:   if (r1 == 8'h01) step <= S_CMD8; else err <= 1'b1;
                        S_CMD8:   if (r1 == 8'h01 && r32[11:0] == 12'h1AA) step <= S_CMD55; else err <= 1'b1;
                        S_CMD55:  if ((r1 & 8'hFE) == 8'h00) step <= S_ACMD41; else err <= 1'b1;
                        S_ACMD41: if (r1 == 8'h00) step <= S_CMD58;
                                  else if (r1 == 8'h01 && retry != 10'h3FF) begin retry <= retry + 10'd1; step <= S_CMD55; end
                                  else err <= 1'b1;
                        S_CMD58:  if (r1 == 8'h00) begin sdhc <= r32[30]; init_done <= 1'b1; fast <= 1'b1; fin <= 1'b1; end
                                  else err <= 1'b1;
                        default:  err <= 1'b1;
                    endcase
                end
                ST_GAP: if (iss && sp_done) begin                 // CS=1에서 추가 8클럭
                    iss <= 1'b0;
                    if (fin || err) begin busy <= 1'b0; st <= ST_IDLE; end else st <= ST_PREP;
                end
                ST_TOKEN: if (iss && sp_done) begin               // 시작 토큰 0xFE 대기
                    iss <= 1'b0;
                    if (sp_rx == 8'hFE) begin bcnt <= 9'd0; st <= ST_DATA; end

                    if (bcnt == 9'd511) begin ecnt <= 3'd0; st <= ST_CRC; end

                    else ecnt <= ecnt + 3'd1;
                end
                default: st <= ST_IDLE;

            8'h10:   rdata = {24'd0, mem_q};

    parameter CLK_HZ = 100_000_000,
    parameter BAUD   = 115200

    input  wire        clk,
    input  wire        rst,
    input  wire        sel,
    input  wire        we,
    input  wire [7:0]  off,
    input  wire [31:0] wdata,

    reg [3:0]  nbit;
    reg [15:0] cnt;
    reg        busy;

            tx   <= 1'b1;
            cnt  <= 16'd0;

            sh   <= 10'h3FF;
        end else if (!busy) begin
            tx <= 1'b1;

                cnt <= DIV - 1;
                if (nbit == 4'd9) begin
                    busy <= 1'b0;
                    tx   <= 1'b1;
                end else begin
                    nbit <= nbit + 4'd1;
                    tx   <= sh[nbit + 4'd1];
                end
            end else begin
                cnt <= cnt - 16'd1;
            end
        end
    end

        case (off)
            8'h04:   rdata = {31'd0, busy};
            default: rdata = 32'h0000_0000;
        endcase

// 테스트벤치용 간이 SD 카드 모델(SPI 모드 0, SDHC/SDSC 선택). 명령 순서·클럭 속도·주소를 스스로 점검한다.

module sd_card_model #(
    parameter SDHC = 1
) (
    input  wire cs_n,
    input  wire sclk,
    input  wire mosi,
    output wire miso
);
    reg        present   = 1'b1;     // 0이면 카드 없음(MISO 항상 1)
    reg        ready     = 1'b0;
    reg        app       = 1'b0;
    reg        order_err = 1'b0;     // 명령 순서·인자·더미 클럭 위반
    reg        slow_viol = 1'b0;     // 초기화 중 SCLK가 400 kHz를 넘음
    reg        fast_seen = 1'b0;     // 초기화 뒤 1 MHz 이상 클럭을 봄
    integer    n17       = 0;
    reg [31:0] last17    = 32'h0;
    reg [31:0] dummy_clks = 0;
    reg [1:0]  stage     = 2'd0;
    integer    polls     = 0;

    reg [7:0]  out_sh = 8'hFF, in_sh = 8'h00;
    reg [2:0]  bitc = 3'd0;
    reg        col = 1'b0;
    integer    ncol = 0;
    reg [5:0]  cmd = 6'd0;
    reg [31:0] arg = 32'h0;
    reg [7:0]  q [0:599];
    integer    qlen = 0, qpos = 0, i;
    realtime   t_prev = 0.0;
    realtime   dt = 0.0;
    reg        have_prev = 1'b0;

    assign miso = (cs_n || !present) ? 1'b1 : out_sh[7];

    function [7:0] exp_byte(input [31:0] blk, input [8:0] k);
        begin
            exp_byte = blk[7:0] * 7 + blk[15:8] * 11 + k[7:0] * 5 + (k[8] ? 85 : 0);
        end
    endfunction

    task push(input [7:0] b);
        begin q[qlen] = b; qlen = qlen + 1; end
    endtask

    task process_cmd;
        reg [31:0] blk;
        begin
            qlen = 0; qpos = 0;
            push(8'hFF);
            case (cmd)
                0: begin
                    if (dummy_clks < 74) order_err = 1'b1;
                    ready = 1'b0; app = 1'b0; stage = 2'd1; push(8'h01);
                end
                8: begin
                    if (stage != 2'd1 || arg != 32'h0000_01AA) begin order_err = 1'b1; push(8'h05); end
                    else begin stage = 2'd2; push(8'h01); push(8'h00); push(8'h00); push(8'h01); push(8'hAA); end
                end
                55: begin
                    if (stage < 2'd2) begin order_err = 1'b1; push(8'h05); end
                    else begin app = 1'b1; push(ready ? 8'h00 : 8'h01); end
                end
                41: begin
                    if (!app || arg != 32'h4000_0000) begin order_err = 1'b1; push(8'h05); end
                    else begin
                        app = 1'b0;
                        if (polls < 3) begin polls = polls + 1; push(8'h01); end
                        else begin ready = 1'b1; stage = 2'd3; push(8'h00); end
                    end
                end
                58: begin
                    if (!ready) begin order_err = 1'b1; push(8'h05); end
                    else begin push(8'h00); push(SDHC ? 8'hC0 : 8'h80); push(8'hFF); push(8'h80); push(8'h00); end
                end
                17: begin
                    if (!ready) begin order_err = 1'b1; push(8'h05); end
                    else begin
                        n17 = n17 + 1; last17 = arg;
                        blk = SDHC ? arg : (arg >> 9);
                        push(8'h00); push(8'hFF); push(8'hFF); push(8'hFE);
                        for (i = 0; i < 512; i = i + 1) push(exp_byte(blk, i));
                        push(8'hAB); push(8'hCD);
                    end
                end
                default: begin order_err = 1'b1; push(8'h05); end
            endcase
        end
    endtask

    task handle_byte(input [7:0] b);
        begin
            if (col) begin
                case (ncol)
                    0: arg[31:24] = b;
                    1: arg[23:16] = b;
                    2: arg[15:8]  = b;
                    3: arg[7:0]   = b;
                    default: begin col = 1'b0; process_cmd; end
                endcase
                ncol = ncol + 1;
            end else if (qpos >= qlen && b[7:6] == 2'b01) begin
                col = 1'b1; ncol = 0; cmd = b[5:0];
            end
            if (qpos < qlen) begin out_sh = q[qpos]; qpos = qpos + 1; end
            else out_sh = 8'hFF;
        end
    endtask

    always @(posedge sclk) begin
        if (cs_n) dummy_clks = dummy_clks + 1;
        else begin
            in_sh = {in_sh[6:0], mosi};
            dt = $realtime - t_prev;
            if (have_prev) begin
                if (!ready && dt < 2400.0) slow_viol = 1'b1;
                if (ready && dt < 1000.0) fast_seen = 1'b1;
            end
            t_prev = $realtime; have_prev = 1'b1;
        end
    end

    always @(negedge sclk) begin
        if (!cs_n) begin
            if (bitc == 3'd7) begin bitc = 3'd0; handle_byte(in_sh); end
            else begin bitc = bitc + 3'd1; out_sh = {out_sh[6:0], 1'b1}; end
        end
    end

    always @(posedge cs_n) begin
        bitc = 3'd0; col = 1'b0; ncol = 0; qlen = 0; qpos = 0; out_sh = 8'hFF; have_prev = 1'b0;
    end
endmodule

// 자체 점검 테스트벤치(시스템 수준): 데모 마스터가 버튼 한 번에 SD 블록 0의 첫 16바이트를 UART로 내보내는지 확인한다.
module tb_demo;
    localparam BIT_NS = 8680;                       // 115200 baud 근사 (100 MHz에서 868 클럭)

    reg         clk = 1'b0;
    reg         rst = 1'b1;
    reg         btn = 1'b0;
    wire        m_valid, m_we, m_ready;
    wire [31:0] m_addr, m_wdata, m_rdata;
    wire [15:0] led;
    wire        txd, cs_n, sclk, mosi, miso;
    integer     errors = 0;
    integer     got = 0;
    integer     j;
    integer     jj;
    reg  [7:0]  rxb [0:15];
    reg  [7:0]  sh;

    always #5 clk = ~clk;

    demo_master u_master (
        .clk(clk), .rst(rst), .btn(btn),
        .m_valid(m_valid), .m_we(m_we), .m_addr(m_addr), .m_wdata(m_wdata),
        .m_rdata(m_rdata), .m_ready(m_ready)
    );
    soc_top #(.CLK_HZ(100_000_000), .BAUD(115200)) dut (
        .clk(clk), .rst(rst),
        .m_valid(m_valid), .m_we(m_we), .m_addr(m_addr), .m_wdata(m_wdata),
        .m_rdata(m_rdata), .m_ready(m_ready),
        .sw(16'h0000), .led(led), .uart_txd(txd),
        .sd_cs_n(cs_n), .sd_sclk(sclk), .sd_mosi(mosi), .sd_miso(miso)
    );
    sd_card_model #(.SDHC(1)) card (.cs_n(cs_n), .sclk(sclk), .mosi(mosi), .miso(miso));

    always begin : uart_rx                          // 시작 비트 에지 → 각 비트 중앙에서 표본
        @(negedge txd);
        #(BIT_NS / 2);
        for (jj = 0; jj < 8; jj = jj + 1) begin
            #(BIT_NS);
            sh[jj] = txd;
        end
        #(BIT_NS);
        if (txd !== 1'b1) begin errors = errors + 1; $display("FAIL stop bit"); end
        if (got < 16) rxb[got] = sh;
        got = got + 1;
    end

    initial begin
        #100_000_000;
        $display("FAIL watchdog (got %0d bytes) st=%0d sdst=%0d step=%0d r1=%h tcnt=%0d bcnt=%0d", got, u_master.st, dut.u_sd.st, dut.u_sd.step, dut.u_sd.r1, dut.u_sd.tcnt, dut.u_sd.bcnt);
        $finish;
    end

    initial begin
        repeat (5) @(posedge clk);
        rst <= 1'b0;
        repeat (10) @(posedge clk);
        btn <= 1'b1;
        repeat (10) @(posedge clk);
        btn <= 1'b0;
        wait (got == 16);
        repeat (2000) @(posedge clk);
        for (j = 0; j < 16; j = j + 1)
            if (rxb[j] !== card.exp_byte(0, j)) begin
                errors = errors + 1;
                $display("FAIL byte %0d: got %h expected %h", j, rxb[j], card.exp_byte(0, j));
            end
        if (got !== 16) begin errors = errors + 1; $display("FAIL byte count %0d", got); end
        if (led !== 16'h010A) begin errors = errors + 1; $display("FAIL led %h expected 010a", led); end
        if (card.order_err || card.slow_viol) begin errors = errors + 1; $display("FAIL card protocol flags"); end
        if (errors == 0) $display("PASS tb_demo");
        else             $display("FAIL: %0d errors", errors);
        $finish;
    end
endmodule

위험·한계

  • xdc 핀 배치(JA1~JA4 = J1, L2, J2, G2)는 Digilent 마스터 XDC와 대조하지 않았다.
  • 카드 모델은 하네스가 쓴 간이 모델이라 실제 카드의 타이밍·응답 변형(긴 Ncr, busy 토큰 등)을 모두 재현하지 않는다. 실제 카드에서의 동작은 S3에서 확인해야 한다.
  • 12.5 MHz SCLK는 Pmod MicroSD 매뉴얼에 최대 클럭 명시가 없어 근거가 약하다. 불안정하면 FAST_HZ 파라미터를 낮춘다.
  • v1/MMC 카드는 CMD8에서 거부되어 err가 선다. CRC16은 검증하지 않고 버린다.
  • 시작 토큰 대기는 최대 65536바이트 후 오류 처리한다. 버튼 디바운스는 없다.
  • SPI 모드 카드는 초기화 전 MISO 풀업이 필요하다. xdc에 PULLUP을 걸었으나 Pmod 보드의 풀업과 함께 쓰는 값은 확인하지 않았다.
  • RTL 합계가 약 490줄(gpio·uart_tx 97줄은 skeleton 재사용)로 HARNESS §13.7의 400줄 기준을 넘는다.
